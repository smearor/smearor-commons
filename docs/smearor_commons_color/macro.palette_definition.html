Macro palette_definition 

Source 
macro_rules! palette_definition {
    (
        $(#[$meta:meta])*
        $vis:vis enum $name:ident {
            $(
                $(#[$vmeta:meta])*
                $variant:ident => $slot_name:literal
            ),* $(,)?
        }
    ) => { ... };
}
Expand description

Defines a fieldless enum and its PaletteDefinition implementation from a Variant => "name" list.

The generated enum derives Debug, Clone, Copy, PartialEq, Eq, Hash, serde::Serialize and serde::Deserialize - serde must be a dependency of the calling crate. Attributes in front of enum are forwarded, so #[serde(rename_all = "snake_case")] can adapt the wire names.

use smearor_commons_color::PaletteDefinition;
use smearor_commons_color::palette_definition;

palette_definition! {
    /// Colors of a chess board theme.
    pub enum BoardSlot {
        /// Light squares.
        Light => "light",
        /// Dark squares.
        Dark => "dark",
        /// Frame around the board.
        Frame => "frame"
    }
}

assert_eq!(BoardSlot::slots(), &[BoardSlot::Light, BoardSlot::Dark, BoardSlot::Frame]);
assert_eq!(BoardSlot::Frame.name(), "frame");
