40/// A configured effect - one variant per implemented effect.
41///
42/// Serializes as `{ "effect": "crossfade", ... }`: each variant's
43/// payload **is** the effect struct (params as fields, serde derives
44/// on the struct itself), so one definition serves runtime, JSON and
45/// codegen.
46#[derive(Debug, Clone, Serialize, Deserialize)]
47#[serde(tag = "effect", rename_all = "kebab-case")]
48pub enum EffectSpec {
49    /// The reference effect - plain cross-fade.
50    Crossfade(Crossfade),
51    /// UV-transform glide - scale, squish, tilt, shift + dissolve.
52    Glide(Glide),
53    /// Pixelation + angular spoke wipe.
54    PixelWheel(PixelWheel),
55    /// Fractal-noise burn with a 5-stop flame ramp.
56    Fire(Fire),
57    /// Glyph rain + trail composite (Matrix effect).
58    Matrix(Matrix),
59    /// Radial blur + cos-palette glow + superellipse crop.
60    AuraGlow(AuraGlow),
61    /// Pixelated column melt (DOOM screen melt).
62    Doom(Doom),
63    /// Shake + suction + whirl dematerialization.
64    Apparition(Apparition),
65    /// Portal swirl + window travel.
66    Portal(Portal),
67    /// CRT collapse to the center line.
68    Tv(Tv),
69    /// CRT collapse overlaid with glitch rows.
70    TvGlitch(TvGlitch),
71    /// Row shear + interference + grain + RGB split.
72    Glitch(Glitch),
73    /// Turning/folding horizontal slats.
74    VenetianBlinds(VenetianBlinds),
75    /// Falling paper strips with torn edges.
76    Shredder(Shredder),
77    /// Quantized panel growth + tint reveal.
78    Materialize(Materialize),
79    /// Radial atom burst dissolve.
80    EnergizeA(EnergizeA),
81    /// Top-down shower + streak + atom dissolve.
82    EnergizeB(EnergizeB),
83    /// Vertical channel-split peel + brightness pulse. BMW's nick
84    /// is the odd unhyphenated `rgbwarp`.
85    #[serde(rename = "rgbwarp")]
86    RgbWarp(RgbWarp),
87    /// 8-bit scale + sparks + rays + star rings.
88    Mushroom(Mushroom),
89    /// Cell-dropout pixel dissolve.
90    Pixelate(Pixelate),
91    /// Radial pixel wipe from the pointer.
92    PixelWipe(PixelWipe),
93    /// Shrinking hexagonal tiles + glow lines.
94    Hexagon(Hexagon),
95    /// Fiery-edge burn with smoke + embers.
96    Incinerate(Incinerate),

98    Wisps(Wisps),
99    /// Sparkle flare + off-center warp.
100    TeamRocket(TeamRocket),

160    /// Finds + returns the default spec named by `nick` - `Err` on
161    /// unknown `"effect"` tags, giving consumers a typed error
162    /// instead of serde's raw one.
163    pub fn by_nick(nick: &str) -> Result<EffectSpec, FxError> {
164        Self::all()
165            .into_iter()
166            .find(|spec| spec.nick() == nick)
167            .ok_or_else(|| FxError::UnknownEffect(nick.to_string()))
168    }

288    /// Default spec named by `nick` with `values` applied - what a
289    /// parameter panel produces for serialization.
290    pub fn from_params(nick: &str, values: &[ParamValue]) -> Result<EffectSpec, FxError> {
291        let mut spec = Self::by_nick(nick)?;
292        spec.set_values(values)?;
293        Ok(spec)

302    fn spec_roundtrip_is_the_identity() {
303        let spec = EffectSpec::by_nick("crossfade").unwrap();
304        let json = serde_json::to_string(&spec).unwrap();
305        assert_eq!(json, r#"{"effect":"crossfade"}"#);
306        let back: EffectSpec = serde_json::from_str(&json).unwrap();
307        assert_eq!(back.nick(), "crossfade");
308    }

326    #[test]
327    fn params_match_the_serde_payload() {
328        // The param-drift guard: every ParamSpec::key must be a field
329        // name in the effect's serialized form.
330        for spec in EffectSpec::all() {
331            let effect = spec.build();
332            let json = serde_json::to_value(&spec).unwrap();
333            for param in effect.params() {
334                // Dotted keys address nested objects - `colors.ember`
335                // resolves `{"colors": {"ember": ...}}`.
336                let mut node = &json;
337                let mut found = true;
338                for segment in param.key.split('.') {
339                    match node.get(segment) {
340                        Some(child) => node = child,
341                        None => {
342                            found = false;
343                            break;
344                        }
345                    }
346                }
347                assert!(found, "{}: param `{}` does not resolve in the serde payload", spec.nick(), param.key);
348            }
349            let _ = effect.values(); // every effect reports values
350        }
351    }

399    /// `values()` ↔ `params()` length parity + `set_values(values())`
400    /// is a no-op for every shipped spec.
401    #[test]
402    fn values_roundtrip_through_set_values() {
403        for spec in EffectSpec::all() {
404            let effect = spec.build();
405            let values = effect.values();
406            assert_eq!(values.len(), effect.params().len(), "{}: values/params length mismatch", spec.nick());
407            let mut restored = spec.clone();
408            restored
409                .set_values(&values)
410                .unwrap_or_else(|err| panic!("{}: set_values failed: {err}", spec.nick()));
411            let after = serde_json::to_value(&restored).unwrap();
412            let before = serde_json::to_value(&spec).unwrap();
413            assert_eq!(before, after, "{}: set_values(values()) drifted", spec.nick());
414        }
415    }
