1//! `render` - the shared per-pixel driver behind every `apply`.

3use smearor_commons_buffer::BufferViewMut;
4use smearor_commons_buffer::PixelFormat;
5use smearor_commons_buffer::Rgba;

7use smearor_commons_geometry::Pixels;
8use smearor_commons_geometry::Size;

10/// Below this pixel count the thread fan-out costs more than it saves.
11const MIN_PARALLEL_PIXELS: usize = 32 * 1024;
12
13/// Worker count override for benchmarking (`FX_THREADS=1` forces the
14/// sequential path; `0`/unset = `available_parallelism`). Read once.
15static FX_THREADS: std::sync::LazyLock<usize> = std::sync::LazyLock::new(|| {
16    std::env::var("FX_THREADS")
17        .ok()
18        .and_then(|v| v.parse::<usize>().ok())
19        .unwrap_or_else(|| std::thread::available_parallelism().map_or(1, |n| n.get()))
20});
21
22/// Backend override for A/B measurement: `FX_BACKEND=scoped` keeps the
23/// `std::thread::scope` path inside a rayon-enabled build.
24#[cfg(feature = "rayon")]
25static FX_BACKEND_RAYON: std::sync::LazyLock<bool> = std::sync::LazyLock::new(|| std::env::var("FX_BACKEND").map(|v| v != "scoped").unwrap_or(true));
26
27/// Fills `out` pixel-by-pixel with `shade(x, y)` - parallel over row
28/// bands when the view is packed and large enough; strided or tiny
29/// views take the sequential path. With `feature = "rayon"` the bands
30/// run on rayon's work-stealing pool (persistent threads, no per-frame
31/// spawn cost); otherwise on [`std::thread::scope`]. Every row is
32/// written by exactly one worker, so the output is deterministic and
33/// identical to the sequential render.
34pub(crate) fn fill<F>(out: &mut BufferViewMut<'_, Rgba>, shade: F)
35where
36    F: Fn(u32, u32) -> Color<f64> + Sync,
37{
38    let size = out.size();
39    let workers = if out.is_packed() && size.x as usize * size.y as usize >= MIN_PARALLEL_PIXELS {
40        (*FX_THREADS).min(size.y as usize)
41    } else {
42        1
43    };
44    if workers <= 1 {
45        fill_sequential(out, &shade);
46        return;
47    }
48    #[cfg(feature = "rayon")]
49    if *FX_BACKEND_RAYON {
50        return fill_parallel_rayon(out, &shade, workers);
51    }
52    fill_parallel(out, &shade, workers);
53}
54
55/// The sequential row loop - the reference implementation the
56/// parallel path is pinned against.
57fn fill_sequential<F>(out: &mut BufferViewMut<'_, Rgba>, shade: &F)
58where
59    F: Fn(u32, u32) -> Color<f64> + Sync,
60{
61    let size = out.size();
62    for y in 0..size.y {
63        for x in 0..size.x {
64            out.put_color(Pixels::new(x), Pixels::new(y), shade(x, y));
65        }
66    }
67}
68
69/// Row-band fan-out: `out`'s packed rows are split into `workers`
70/// contiguous bands, one scoped thread per band. Bands are disjoint,
71/// so the result is byte-identical to [`fill_sequential`].
72fn fill_parallel<F>(out: &mut BufferViewMut<'_, Rgba>, shade: &F, workers: usize)
73where
74    F: Fn(u32, u32) -> Color<f64> + Sync,
75{
76    let size = out.size();
77    let workers = workers.min(size.y as usize);
78    let row_bytes = size.x as usize * Rgba::BYTES_PER_PIXEL;
79    let band_rows = (size.y as usize).div_ceil(workers) as u32;
80    std::thread::scope(|scope| {
81        for (band, chunk) in out.data_mut().chunks_mut(band_rows as usize * row_bytes).enumerate() {
82            scope.spawn(move || {
83                let band_height = (chunk.len() / row_bytes) as u32;
84                let Ok(mut view) = BufferViewMut::<Rgba>::from_packed(chunk, Size::new(size.x, band_height)) else {
85                    return;
86                };
87                let y_offset = band as u32 * band_rows;
88                for ly in 0..band_height {
89                    for x in 0..size.x {
90                        view.put_color(Pixels::new(x), Pixels::new(ly), shade(x, y_offset + ly));
91                    }
92                }
93            });
94        }
95    });
96}

98/// Same banding on rayon's global pool - no per-frame thread spawns
99/// and work stealing absorbs uneven row costs (wormhole's bisection
100/// rows). `workers` bounds the fan-out via the band count; the pool
101/// itself always runs.
102#[cfg(feature = "rayon")]
103fn fill_parallel_rayon<F>(out: &mut BufferViewMut<'_, Rgba>, shade: &F, workers: usize)
104where
105    F: Fn(u32, u32) -> Color<f64> + Sync,
106{
107    use rayon::prelude::IndexedParallelIterator;
108    use rayon::prelude::ParallelIterator;
109    use rayon::prelude::ParallelSliceMut;
110    let size = out.size();
111    let workers = workers.min(size.y as usize);
112    let row_bytes = size.x as usize * Rgba::BYTES_PER_PIXEL;
113    let band_rows = (size.y as usize).div_ceil(workers) as u32;
114    out.data_mut()
115        .par_chunks_mut(band_rows as usize * row_bytes)
116        .enumerate()
117        .for_each(|(band, chunk)| {
118            let band_height = (chunk.len() / row_bytes) as u32;
119            if let Ok(mut view) = BufferViewMut::<Rgba>::from_packed(chunk, Size::new(size.x, band_height)) {
120                let y_offset = band as u32 * band_rows;
121                for ly in 0..band_height {
122                    for x in 0..size.x {
123                        view.put_color(Pixels::new(x), Pixels::new(ly), shade(x, y_offset + ly));
124                    }
125                }
126            }
127        });
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    use smearor_commons_buffer::PixelBuffer;
135
136    /// `rayon_parity` from the phase-6 test plan, without the rayon
137    /// dependency: the banded path must produce byte-identical output
138    /// to the sequential reference.
139    #[test]
140    fn parallel_output_is_byte_identical() {
141        let size = Size::new(96, 48);
142        let shade =
143            |x: u32, y: u32| Color::<f64>::new_rgba(f64::from(x % 7) / 7.0, f64::from(y % 5) / 5.0, f64::from((x + y) % 3) / 3.0, f64::from((x ^ y) % 9) / 9.0);
144        let mut a = PixelBuffer::<Rgba>::new(size);
145        let mut b = PixelBuffer::<Rgba>::new(size);
146        fill_sequential(&mut a.view_mut(), &shade);
147        fill_parallel(&mut b.view_mut(), &shade, 7);
148        assert_eq!(a.pixels(), b.pixels());
149    }
150
151    /// Same pin for the rayon backend - work stealing may reorder
152    /// *when* a band runs, never *what* it writes.
153    #[cfg(feature = "rayon")]
154    #[test]
155    fn rayon_output_is_byte_identical() {
156        let size = Size::new(96, 48);
157        let shade =
158            |x: u32, y: u32| Color::<f64>::new_rgba(f64::from(x % 7) / 7.0, f64::from(y % 5) / 5.0, f64::from((x + y) % 3) / 3.0, f64::from((x ^ y) % 9) / 9.0);
159        let mut a = PixelBuffer::<Rgba>::new(size);
160        let mut b = PixelBuffer::<Rgba>::new(size);
161        fill_sequential(&mut a.view_mut(), &shade);
162        fill_parallel_rayon(&mut b.view_mut(), &shade, 7);
163        assert_eq!(a.pixels(), b.pixels());
164    }
