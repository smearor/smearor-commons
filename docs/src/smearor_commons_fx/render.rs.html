1//! `render` - the shared per-pixel driver behind every `apply`.
2
3use smearor_commons_buffer::BufferViewMut;
4use smearor_commons_buffer::PixelFormat;
5use smearor_commons_buffer::Rgba;
6use smearor_commons_color::Color;
7use smearor_commons_geometry::Pixels;
8use smearor_commons_geometry::Size;
9
10/// Below this pixel count the thread fan-out costs more than it saves.
11const MIN_PARALLEL_PIXELS: usize = 32 * 1024;
12
13/// Fills `out` pixel-by-pixel with `shade(x, y)` - parallel over row
14/// bands via [`std::thread::scope`] when the view is packed and large
15/// enough; strided or tiny views take the sequential path. Every row
16/// is written by exactly one worker, so the output is deterministic
17/// and identical to the sequential render.
18pub(crate) fn fill<F>(out: &mut BufferViewMut<'_, Rgba>, shade: F)
19where
20    F: Fn(u32, u32) -> Color<f64> + Sync,
21{
22    let size = out.size();
23    let workers = if out.is_packed() && size.x as usize * size.y as usize >= MIN_PARALLEL_PIXELS {
24        std::thread::available_parallelism().map_or(1, |n| n.get()).min(size.y as usize)
25    } else {
26        1
27    };
28    if workers <= 1 {
29        for y in 0..size.y {
30            for x in 0..size.x {
31                out.put_color(Pixels::new(x), Pixels::new(y), shade(x, y));
32            }
33        }
34        return;
35    }
36    let row_bytes = size.x as usize * Rgba::BYTES_PER_PIXEL;
37    let band_rows = (size.y as usize).div_ceil(workers) as u32;
38    let shade = &shade;
39    std::thread::scope(|scope| {
40        for (band, chunk) in out.data_mut().chunks_mut(band_rows as usize * row_bytes).enumerate() {
41            scope.spawn(move || {
42                let band_height = (chunk.len() / row_bytes) as u32;
43                let Ok(mut view) = BufferViewMut::<Rgba>::from_packed(chunk, Size::new(size.x, band_height)) else {
44                    return;
45                };
46                let y_offset = band as u32 * band_rows;
47                for ly in 0..band_height {
48                    for x in 0..size.x {
49                        view.put_color(Pixels::new(x), Pixels::new(ly), shade(x, y_offset + ly));
50                    }
51                }
52            });
53        }
54    });
55}
