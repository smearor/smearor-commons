1//! `assets` - the optional texture channel for effects which sample
2//! auxiliary imagery (BMW's `*.png` resources).
3//!
4//! Every asset effect has a deterministic **procedural fallback** -
5//! the same role `ProceduralGlyphAtlas` plays for `Matrix`: no file
6//! dependency, no loader, reproducible output. A compositor can pass
7//! real textures through [`FxContext::assets`] to override them.
8
9use smearor_commons_buffer::BufferView;
10use smearor_commons_buffer::Rgba;
11use smearor_commons_geometry::Vec2;
12use smearor_commons_noise::Field;
13use smearor_commons_noise::Hash;
14use smearor_commons_noise::Simplex2DFractal;

16/// Optional auxiliary textures an effect may sample - BMW's image
17/// resources as a typed channel. `None` selects the effect's
18/// procedural fallback.
19#[derive(Debug, Clone, Copy, Default)]
20pub struct FxAssets<'a> {
21    /// Claw-scratch map for `trex` (BMW `claws.png`) - `r` is the
22    /// scratch's local time field.
23    pub claw: Option<BufferView<'a, Rgba>>,
24    /// Dust particle map for `snap` (BMW `dust.png`) - `r` dissolve
25    /// threshold, `g` per-particle layer id.
26    pub dust: Option<BufferView<'a, Rgba>>,
27    /// Brush sweep for `paint-brush` (BMW `brush.png`) - one channel
28    /// per brush size, value = sweep time.
29    pub brush: Option<BufferView<'a, Rgba>>,
30    /// Shard map for `broken-glass` (BMW `shards.png`) - `r` per-shard
31    /// random, `g` per-shard layer id.
32    pub shard: Option<BufferView<'a, Rgba>>,
33}
34
35/// The procedural texture substitutes - one function per asset,
36/// returning the sampled channels the shader semantics expect.
37pub(crate) mod procedural {
38    use super::*;
39
40    /// Voronoi F1 helper: nearest jittered site wins, `r`/`g` are two
41    /// independent site hashes - the shared base of BMW's `dust.png`
42    /// and `shards.png`.
43    fn voronoi_rg(uv: Vec2<()>, cells: f64) -> (f64, f64) {
44        let p = Vec2::<()>::new(uv.x * cells, uv.y * cells);
45        let base = Vec2::<()>::new(p.x.floor(), p.y.floor());
46        let mut best = f64::INFINITY;
47        let mut best_id = Vec2::<()>::new(0.0, 0.0);
48        for j in -1..=1 {
49            for i in -1..=1 {
50                let cell = Vec2::<()>::new(base.x + i as f64, base.y + j as f64);
51                let site = cell + Hash::<Vec2<()>>::hash(&cell);
52                let dx = p.x - site.x;
53                let dy = p.y - site.y;
54                let d = dx * dx + dy * dy;
55                if d < best {
56                    best = d;
57                    best_id = cell;
58                }
59            }
60        }
61        (Hash::<f64>::hash(&best_id), Hash::<f64>::hash(&(best_id + Vec2::<()>::new(19.19, 47.47))))
62    }
63
64    /// `dust.png` substitute - `(dissolve threshold, layer id)`.
65    pub fn dust(uv: Vec2<()>) -> (f64, f64) {
66        voronoi_rg(uv, 6.0)
67    }
68
69    /// `shards.png` substitute - `(per-shard random, layer id)`.
70    pub fn shard(uv: Vec2<()>) -> (f64, f64) {
71        voronoi_rg(uv, 8.0)
72    }
73
74    /// `claws.png` substitute - three curved scratch stripes; `r`
75    /// runs 0->1 along the stripe so the scratch "draws" over time
76    /// (`1.0` = untouched).
77    pub fn claw(uv: Vec2<()>) -> f64 {
78        let mut value: f64 = 1.0;
79        for k in 0..3 {
80            let kf = k as f64;
81            // A diagonal stripe, curved by a low-frequency bend.
82            let center = uv.y + 0.08 * (uv.y * std::f64::consts::PI).sin() + 0.16 + kf * 0.18;
83            let dist = (uv.x - center).abs();
84            let width = 0.028 - 0.012 * uv.y;
85            if dist < width {
86                // Along the stripe: earlier parts scratch first.
87                let t = (1.0 - uv.y).clamp(0.0, 1.0);
88                value = value.min(t.max(0.02));
89            }
90        }
91        value
92    }
93
94    /// `brush.png` substitute - a left-to-right sweep with a ragged
95    /// noise edge; `channel` 0..3 mimics the PNG's per-size planes
96    /// (rougher/larger strokes cover earlier).
97    pub fn brush(uv: Vec2<()>, channel: usize) -> f64 {
98        let channel_f = channel as f64;
99        let edge = Simplex2DFractal.at(Vec2::<()>::new(uv.y * 3.0 + channel_f * 17.0, channel_f * 5.0));
100        let ragged = Simplex2DFractal.at(Vec2::<()>::new(uv.x * 12.0, uv.y * 12.0 + channel_f * 29.0)) - 0.5;
101        // Larger brushes (higher channel) sweep faster + rougher.
102        let t = uv.x * (1.05 - 0.08 * channel_f) + edge * 0.12 + ragged * 0.08;
103        t.clamp(0.0, 1.0)
104    }
