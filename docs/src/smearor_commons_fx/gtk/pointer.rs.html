1//! Pointer wiring - `EventController` → `ctx.pointer` in uv space.
2
3use gtk4::prelude::Cast;
4use gtk4::prelude::WidgetExt;
5use smearor_commons_geometry::Vec2;
6use std::cell::Cell;
7use std::rc::Rc;
8
9/// Wires motion and click events on `widget` into `target`.
10///
11/// Stored positions are widget-relative uv (`x / width`,
12/// `y / height`) - the coordinate space [`crate::FxContext::pointer`]
13/// expects (BMW's `uStartPos`/`uEpicenter` are actor-local uvs).
14/// Leaving the widget clears the pointer to `None`; clicks update the
15/// position too, so press-only gestures (the broken-glass epicenter)
16/// work without preceding motion.
17pub(crate) fn wire_pointer(widget: &impl gtk4::prelude::IsA<gtk4::Widget>, target: Rc<Cell<Option<Vec2<()>>>>) {
18    let widget = widget.upcast_ref::<gtk4::Widget>().clone();
19    let to_uv = |widget: &gtk4::Widget, x: f64, y: f64| {
20        let w = f64::from(widget.width()).max(1.0);
21        let h = f64::from(widget.height()).max(1.0);
22        Vec2::<()>::new(x / w, y / h)
23    };
24
25    let motion = gtk4::EventControllerMotion::new();
26    motion.connect_motion({
27        let target = Rc::clone(&target);
28        let widget = widget.clone();
29        move |_, x, y| target.set(Some(to_uv(&widget, x, y)))
30    });
31    motion.connect_leave({
32        let target = Rc::clone(&target);
33        move |_| target.set(None)
34    });
35    widget.add_controller(motion);
36
37    let click = gtk4::GestureClick::new();
38    click.connect_pressed({
39        let widget = widget.clone();
40        move |_, _n, x, y| target.set(Some(to_uv(&widget, x, y)))
41    });
42    widget.add_controller(click);
43}
