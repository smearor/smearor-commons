1//! [`FxPlayer`] - a `DrawingArea` subclass that owns the animation
2//! clock and frame buffers and plays any [`EffectSpec`].
3//!
4//! The player is the *mechanism*; choreography stays with the
5//! consumer. The floating-board recipe:
6//!
7//! 1. `card-added`: [`crate::gtk::snapshot`] the card, place an
8//!    `FxPlayer` at the same coordinates, hide (not destroy) the card,
9//!    `play` an `Open`-direction spec, on `animation-ended` remove
10//!    the player and show the card.
11//! 2. `card-removed`: same in `Close` direction, destroy the card on
12//!    end.
13//! 3. Board-level transitions: snapshot card + board, run a layered
14//!    spec with a [`crate::gtk::SurfaceProvider`], hand back to live
15//!    widgets.

48/// Play-time options for [`FxPlayer::play`] - everything that is *not*
49/// part of the serializable [`EffectSpec`] payload.
50#[derive(Debug, Clone, Copy)]
51pub struct FxPlayOptions {
52    /// Animation duration (the `animation-time` setting in BMW).
53    pub duration: Duration,
54    /// `Open` (window materializes) or `Close` (window dissolves).
55    pub direction: Direction,
56    /// Deterministic frame seed - same seed + progress replays
57    /// bitwise.
58    pub seed: Seed,
59    /// `ctx.fullscreen` - forwarded to effects that branch on it.
60    pub fullscreen: bool,
61    /// Transparent margin around the window rect inside the input
62    /// texture (e.g. where a shadow is drawn).
63    pub padding: Pixels<f64>,
64    /// The time easing applied by [`TimedAnimation`] - `Linear` is
65    /// honest: BMW eases inside the shader, our effects do the same.
66    pub easing: EasingFunction,
67    /// The window rect inside the input texture - `None` means the
68    /// full buffer (the common player == window case).
69    pub window_rect: Option<Rect<Pixels, f64>>,
70}
71
72impl Default for FxPlayOptions {
73    fn default() -> Self {
74        Self {
75            duration: Duration::from_millis(600),
76            direction: Direction::Open,
77            seed: Seed(0),
78            fullscreen: false,
79            padding: Pixels::new(0.0),
80            easing: EasingFunction::Linear,
81            window_rect: None,
82        }
83    }
84}

146    impl gtk4::subclass::prelude::WidgetImpl for FxPlayerImp {
147        /// The frozen input texture is the player's natural size -
148        /// device pixels divided back by the surface scale.
149        fn measure(&self, orientation: gtk4::Orientation, for_size: i32) -> (i32, i32, i32, i32) {
150            use gtk4::glib::subclass::prelude::ObjectSubclassExt;
151            use gtk4::subclass::prelude::WidgetImplExt;
152            if let Some(input) = self.input.borrow().as_ref() {
153                let scale = self.obj().scale_factor().max(1);
154                let extent = match orientation {
155                    gtk4::Orientation::Horizontal => input.size().x as i32 / scale,
156                    _ => input.size().y as i32 / scale,
157                };
158                return (extent, extent, -1, -1);
159            }
160            self.parent_measure(orientation, for_size)
161        }
162
163        /// `size-allocate` rebuilds the frame buffers at
164        /// `allocation * scale_factor` - only while no input is
165        /// installed; with an input the buffers stay bound to the
166        /// frozen texture size (see `set_effect`).
167        fn size_allocate(&self, width: i32, height: i32, _baseline: i32) {
168            use gtk4::glib::subclass::prelude::ObjectSubclassExt;
169            use gtk4::subclass::prelude::WidgetImplExt;
170            self.parent_size_allocate(width, height, _baseline);
171            if self.input.borrow().is_some() {
172                return;
173            }
174            let scale = self.obj().scale_factor().max(1);
175            let size = Size::new((width * scale).max(1) as u32, (height * scale).max(1) as u32);
176
177            *self.out_surface.borrow_mut() = PixelBuffer::<Rgba>::new(size);
178            *self.out_window.borrow_mut() = PixelBuffer::<Rgba>::new(size);
179            *self.display.borrow_mut() = PixelBuffer::<Rgba>::new(size);
180        }
181    }
182

187    /// A `DrawingArea` that plays an [`EffectSpec`] on a frozen input
188    /// texture.
189    ///
190    /// `set_effect` installs spec + input, `play` runs the clock. The
191    /// `animation-ended` signal fires once when the duration elapses.
192    /// Layered specs pull their backdrop through
193    /// [`FxPlayer::set_surface_provider`].
194    pub struct FxPlayer(ObjectSubclass<imp::FxPlayerImp>)
195        @extends gtk4::DrawingArea, gtk4::Widget,
196        @implements gtk4::Accessible, gtk4::Buildable, gtk4::ConstraintTarget;
197}
198
199impl Default for FxPlayer {
200    fn default() -> Self {
201        Self::new()
202    }
203}

205impl FxPlayer {
206    /// An empty player - install an effect via [`Self::set_effect`],
207    /// then [`Self::play`].
208    pub fn new() -> Self {
209        let player: Self = glib::Object::new();
210        wire_pointer(&player, Rc::clone(&player.imp().pointer));
211        player
212    }
213
214    /// Installs the effect and its frozen window texture. Does not
215    /// start the clock - call [`Self::play`].
216    ///
217    /// The input texture is the player's natural render size: the
218    /// frame buffers are resized to it and `measure` reports it as the
219    /// natural widget size, so a parent (`Fixed`, overlay, ...) gives
220    /// the player exactly the snapshotted widget's extent.
221    pub fn set_effect(&self, spec: EffectSpec, input: PixelBuffer<Rgba>) {
222        let size = input.size();
223        let imp = self.imp();
224        *imp.out_surface.borrow_mut() = PixelBuffer::<Rgba>::new(size);
225        *imp.out_window.borrow_mut() = PixelBuffer::<Rgba>::new(size);
226        *imp.display.borrow_mut() = PixelBuffer::<Rgba>::new(size);
227        *imp.spec.borrow_mut() = Some(spec);
228        *imp.input.borrow_mut() = Some(input);
229        self.queue_resize();
230    }
231
232    /// Installs the surface provider for
233    /// [`Layers::SurfaceAndWindow`] specs - called once per frame.
234    /// `None` degrades layered passes to transparent surface reads
235    /// (the [`FxError::MissingSurface`] contract, graceful).
236    pub fn set_surface_provider(&self, provider: SurfaceProvider) {
237        *self.imp().provider.borrow_mut() = Some(provider);
238    }
239
240    /// Starts (or restarts) the animation with `options`.
241    pub fn play(&self, options: FxPlayOptions) {
242        let imp = self.imp();
243        imp.direction.set(options.direction);
244        imp.seed.set(options.seed);
245        imp.fullscreen.set(options.fullscreen);
246        imp.padding.set(options.padding.0);
247        imp.window_rect.set(options.window_rect.map(|r| (r.min.x, r.min.y, r.size.x, r.size.y)));
248        imp.ended_emitted.set(false);
249        imp.paused.set(None);
250        {
251            let mut animation = imp.animation.borrow_mut();
252            animation.duration = options.duration;
253            animation.easing_function = options.easing;
254            animation.start();
255        }
256        imp.playing.set(true);
257
258        self.set_draw_func(|area, cr, width, height| {
259            let area = area.downcast_ref::<FxPlayer>().cloned();
260            if let Some(player) = area {
261                player.paint(cr, width, height);
262            }
263        });
264
265        let player = self.clone();
266        self.add_tick_callback(move |_, _clock| {
267            player.render_frame();
268            player.queue_draw();
269            let imp = player.imp();
270            if imp.playing.get() && player.is_complete() {
271                imp.playing.set(false);
272                if !imp.ended_emitted.replace(true) {
273                    player.emit_by_name::<()>("animation-ended", &[]);
274                    return glib::ControlFlow::Break;
275                }
276            }
277            glib::ControlFlow::Continue
278        });
279    }
280
281    /// Pins the clock at the current progress - the animation stays
282    /// visible but stops advancing until [`Self::resume`].
283    pub fn pause(&self) {
284        let imp = self.imp();
285        let progress = imp.animation.borrow().progress().map(|p| p.as_f64().min(1.0)).unwrap_or(0.0);
286        imp.paused.set(Some(progress));
287    }
288
289    /// Unpins the clock after [`Self::pause`].
290    pub fn resume(&self) {
291        self.imp().paused.set(None);
292    }
293
294    /// Whether the animation ran through.
295    pub fn is_complete(&self) -> bool {
296        self.imp().animation.borrow().is_complete()
297    }
298
299    /// The frame buffer size - the input texture size after
300    /// [`Self::set_effect`], `allocation * scale_factor` before that.
301    pub fn frame_size(&self) -> Size<u32> {
302        self.imp().display.borrow().size()
303    }
304
305    /// Stops the clock without emitting `animation-ended` - the
306    /// consumer's way out when it takes the widget back mid-animation.
307    pub fn stop(&self) {
308        self.imp().playing.set(false);
309        self.imp().paused.set(None);
310    }
311
312    /// The `animation-ended` signal - fires once at the end of a
313    /// [`Self::play`] run.
314    pub fn connect_animation_ended<F: Fn(&Self) + 'static>(&self, f: F) -> glib::SignalHandlerId {
315        self.connect_local("animation-ended", true, move |values| {
316            if let Some(player) = values.first().and_then(|v| v.get::<FxPlayer>().ok()) {
317                f(&player);
318            }
319            None
320        })
321    }

323    /// Renders one frame into the `display` buffer - the two-pass
324    /// pipeline shared with the `fx.rs` experimenter: surface pass
325    /// (layered only), window pass, composite window over surface.
326    fn render_frame(&self) {
327        let imp = self.imp();
328        let Some(spec) = imp.spec.borrow().clone() else {
329            return;
330        };
331        let Some(input) = imp.input.borrow().clone() else {
332            return;
333        };
334        let effect = spec.build();
335        let layered = effect.layers() == Layers::SurfaceAndWindow;
336
337        let progress = if let Some(pinned) = imp.paused.get() {
338            Fraction::new(pinned)
339        } else {
340            imp.animation.borrow().progress().unwrap_or(Fraction::ONE)
341        };
342
343        let surface = imp.provider.borrow().as_ref().and_then(|provider| provider());
344
345        let size = imp.display.borrow().size();
346        let window_rect = imp
347            .window_rect
348            .get()
349            .map(|(x, y, w, h)| Rect::from_xywh(x, y, w, h))
350            .unwrap_or_else(|| Rect::from_xywh(0.0, 0.0, f64::from(size.x), f64::from(size.y)));
351
352        let ctx = FxContext {
353            progress,
354            duration: imp.animation.borrow().duration,
355            seed: imp.seed.get(),
356            direction: imp.direction.get(),
357            layer: if layered { Layer::Surface } else { Layer::Window },
358            window: input.view(),
359            surface: surface.as_ref().map(|s| s.view()),
360            assets: crate::FxAssets::default(),
361            pointer: imp.pointer.get(),
362            fullscreen: imp.fullscreen.get(),
363            padding: Pixels::new(imp.padding.get()),
364            window_rect,
365        };
366
367        // Surface pass only when the layered contract holds.
368        if layered && effect.check(&ctx).is_ok() {
369            effect.apply(&ctx, &mut imp.out_surface.borrow_mut().view_mut());
370        }
371
372        let ctx = FxContext { layer: Layer::Window, ..ctx };
373        effect.apply(&ctx, &mut imp.out_window.borrow_mut().view_mut());
374
375        // Composite: window over surface - `display` keeps the
376        // transparent regions for whatever sits behind the player.
377        let w = f64::from(size.x).max(1.0);
378        let h = f64::from(size.y).max(1.0);
379        {
380            let window_out = imp.out_window.borrow();
381            let surface_out = imp.out_surface.borrow();
382            let mut display = imp.display.borrow_mut();
383            let mut view = display.view_mut();
384            for y in 0..size.y {
385                for x in 0..size.x {
386                    let uv = Vec2::<()>::new((f64::from(x) + 0.5) / w, (f64::from(y) + 0.5) / h);
387                    let over = window_out.view().sample_nearest(uv);
388                    let under = surface_out.view().sample_nearest(uv);
389                    view.put_color(Pixels::new(x), Pixels::new(y), over.alpha_over(&under));
390                }
391            }
392        }
393    }

395    /// Paints `display` as a texture node onto the draw context.
396    fn paint(&self, cr: &cairo::Context, width: i32, height: i32) {
397        let texture = self.imp().display.borrow().to_memory_texture();
398        let bounds = gtk4::graphene::Rect::new(0.0, 0.0, width as f32, height as f32);
399        let node = gtk4::gsk::TextureNode::new(&texture, &bounds).upcast();
400        node.draw(cr);
401    }
