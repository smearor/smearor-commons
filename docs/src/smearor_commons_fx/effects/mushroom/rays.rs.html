8/// BMW `getRays` - diagonal light streaks: `uv` stretched to
9/// `(10, 0.5)`, scrolling down over progress, seeded horizontally;
10/// attenuated to zero at all edges (`zeroStartEnd` in both axes)
11/// and at the animation's ends, remapped `[-5, 1]` and tinted.
12pub(super) fn rays(uv: Vec2<()>, progress: f64, seed_y: f64, color: Color<f64>) -> Color<f64> {
13    let ray_uv = Vec2::<()>::new(uv.x * 10.0 + seed_y, uv.y * 0.5 - progress);
14    let mut ray = Simplex2D.at(ray_uv);
15    ray *= zero_start_end(uv.y, 1.0, 8.0) * zero_start_end(uv.x, 1.0, 8.0) * zero_start_end(progress, 1.0, 8.0);
16    ray = crate::remap(ray * 1.10, 0.0, 1.0, -5.0, 1.0);
17    Color::<f64>::new_rgba(color.r, color.g, color.b, (color.a * ray).clamp(0.0, 1.0))
18}
