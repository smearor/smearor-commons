6/// BMW `getStar` - a five-point star SDF-ish mask: `uv` relative
7/// pixel position, `center` the star center, `size` the radius, all
8/// in the same (aspect-corrected) space. Returns 0/1 coverage via
9/// the cross-product edge test.
10fn get_star(uv: Vec2<()>, center: Vec2<()>, npoints: f64, radius_ratio: f64, size: f64, rotation: f64) -> f64 {
11    let starangle = std::f64::consts::TAU / npoints;
12    // One point always up at rotation = 0.
13    let rotation = rotation + std::f64::consts::FRAC_PI_2 - starangle;
14    let p0 = Vec2::<()>::new(size * rotation.cos(), size * rotation.sin());
15    let p1 = Vec2::<()>::new(size * radius_ratio * (starangle + rotation).cos(), size * radius_ratio * (starangle + rotation).sin());
16    let cur = Vec2::<()>::new(uv.x - center.x, uv.y - center.y);
17    let cur_radius = cur.x.mul_add(cur.x, cur.y * cur.y).sqrt();
18    let cur_angle = cur.y.atan2(cur.x) - rotation;
19    let mut a = crate::fract(cur_angle / starangle);
20    if a >= 0.5 {
21        a = 1.0 - a;
22    }
23    let a = a * starangle + rotation;
24    let cur_pos = Vec2::<()>::new(cur_radius * a.cos(), cur_radius * a.sin());
25    // Cross product of (p1 - p0) x (curPos - p0).
26    let dir0 = Vec2::<()>::new(p1.x - p0.x, p1.y - p0.y);
27    let dir1 = Vec2::<()>::new(cur_pos.x - p0.x, cur_pos.y - p0.y);
28    if dir0.x * dir1.y - dir0.y * dir1.x >= 0.0 { 1.0 } else { 0.0 }
29}

31/// BMW `getStarColor` - interpolates `v` over the six-stop ramp
32/// (`STEPS`), scaling each stop's alpha by `alpha`.
33fn star_color(colors: &[Color<f64>; 6], v: f64, alpha: f64) -> Color<f64> {
34    const STEPS: [f64; 6] = [0.0, 0.1666, 0.3332, 0.4998, 0.6664, 0.8330];
35    let at = |i: usize| colors[i];
36    let v = v.clamp(0.0, 1.0);
37    if v >= STEPS[5] {
38        let c = at(5);
39        return Color::<f64>::new_rgba(c.r, c.g, c.b, alpha * c.a);
40    }
41    for i in 0..5 {
42        if v <= STEPS[i + 1] {
43            let t = (v - STEPS[i]) / (STEPS[i + 1] - STEPS[i]);
44            let (a, b) = (at(i), at(i + 1));
45            return Color::<f64>::new_rgba(crate::mix(a.r, b.r, t), crate::mix(a.g, b.g, t), crate::mix(a.b, b.b, t), alpha * crate::mix(a.a, b.a, t));
46        }
47    }
48    Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0)
49}

51/// BMW `getStars` - `ring_count` rings of `star_count` stars
52/// circling the window. Ring `r` lags by `spread = r/ring_count`;
53/// each star orbits `sin/cos(rotation*τ + s/star_count*τ)` on the
54/// `aspect*0.33` ellipse. `depth < 0` puts a star behind the window
55/// (composited under, scaled by `1 - oColorAlpha`).
56#[allow(clippy::too_many_arguments)]
57pub(super) fn stars(
58    star_uv: Vec2<()>,
59    aspect: f64,
60    progress: f64,
61    window_alpha: f64,
62    ring_count: u32,
63    ring_rotation: f64,
64    star_count: u32,
65    colors: &[Color<f64>; 6],
66) -> Color<f64> {
67    let mut result = Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0);
68    for r in 0..ring_count {
69        let spread = r as f64 / ring_count.max(1) as f64;
70        let y = crate::mix(0.0 - spread, 1.0 + (1.0 - spread), 1.0 - progress).clamp(0.00001, 0.99999);
71        for s in 0..star_count {
72            let phase = progress * ring_rotation * std::f64::consts::TAU + s as f64 * (std::f64::consts::TAU / star_count.max(1) as f64);
73            let center = Vec2::<()>::new(phase.sin() * aspect * 0.33, y);
74            let mask = get_star(star_uv, center, 5.0, 0.5, zero_start_end(y, 0.1, 2.0), 0.0).clamp(0.0, 1.0);
75            let star = star_color(colors, y, mask);
76            // Behind vs in front of the window.
77            if phase.cos() < 0.0 {
78                result = crate::alpha_over(result, star);
79            } else {
80                result = crate::alpha_over(
81                    Color::<f64>::new_rgba(
82                        star.r * (1.0 - window_alpha),
83                        star.g * (1.0 - window_alpha),
84                        star.b * (1.0 - window_alpha),
85                        star.a * (1.0 - window_alpha),
86                    ),
87                    result,
88                );
89            }
90        }
91    }
92    result
