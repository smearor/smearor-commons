18/// The window shrinks with a jerky 8-bit staircase while a mushroom
19/// of effects erupts: four-point sparkles rise, diagonal light rays
20/// streak down, and rings of five-point stars circle the window
21/// (behind and in front).
22///
23/// Behavioral spec (source shader: `mushroom.frag`, re-implemented
24/// from semantics): `progress` inverts for `Open`. The window scales
25/// around its center by `mix(eightBitScale, (easeInOutSine,
26/// easeInQuad), scaleStyle)` - the staircase gives the "8-bit"
27/// growth. `spark_count` sparkles rise on `hash21` offsets with an
28/// `xEdge` envelope and shared `progress*spark_rotation` spin; `rays`
29/// streak on a `(10, 0.5)`-stretched simplex scrolling down; `ring_
30/// count` rings of `star_count` stars orbit the `aspect*0.33`
31/// ellipse, each composited behind (`depth < 0`) or in front of the
32/// window and colored by the six-stop `star_colors` ramp at their
33/// ring offset.
34#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
35#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
36pub struct Mushroom {
37    /// 0 = pure 8-bit staircase, 1 = smooth sine/quad growth - schema
38    /// `mushroom-scale_style`, default `1.0`.
39    #[serde(default = "Mushroom::default_scale_style")]
40    pub scale_style: Factor,
41    /// Rising four-point sparkles - schema `mushroom-spark_count`, default `4`.
42    #[serde(default = "Mushroom::default_spark_count")]
43    pub spark_count: u32,
44    /// Sparkle tint - schema `mushroom-spark_color`, default `rgba(255, 255, 255, 1.0)`.
45    #[serde(default = "Mushroom::default_spark_color")]
46    pub spark_color: Color<f64>,
47    /// Sparkle spin - schema `mushroom-spark_rotation`, default `0.3`.
48    #[serde(default = "Mushroom::default_spark_rotation")]
49    pub spark_rotation: Factor,
50    /// Light-ray tint (`a = 0` disables rays) - schema `mushroom-rays_color`, default `rgba(255, 255, 255, 1.0)`.
51    #[serde(default = "Mushroom::default_rays_color")]
52    pub rays_color: Color<f64>,
53    /// Star ring count - schema `mushroom-ring_count`, default `3`.
54    #[serde(default = "Mushroom::default_ring_count")]
55    pub ring_count: u32,
56    /// Ring orbit speed - schema `mushroom-ring_rotation`, default `1.33`.
57    #[serde(default = "Mushroom::default_ring_rotation")]
58    pub ring_rotation: Factor,
59    /// Stars per ring - schema `mushroom-star_count`, default `5`.
60    #[serde(default = "Mushroom::default_star_count")]
61    pub star_count: u32,
62    /// The six-stop star ramp - the named [`Palette`] equivalent of
63    /// BMW's `mushroom-star-color-0` .. `mushroom-star-color-5` keys.
64    /// Slots are [`StarColor`] in interpolation order; missing slots
65    /// fall back to the default ramp.
66    #[serde(default = "Mushroom::default_star_colors")]
67    #[cfg_attr(feature = "schemars", schemars(with = "std::collections::HashMap<String, Color<f64>>"))]
68    pub star_colors: Palette<StarColor, f64>,
69}

113    /// The six ramp colors in slot order, slot-missing falls back.

115        DEFAULT_RAMP.map(|(slot, fallback)| self.star_colors.get_or(&slot, fallback))

227impl WindowEffect for Mushroom {
228    fn nick(&self) -> &'static str {
229        "mushroom"
230    }
231
232    fn label(&self) -> &'static str {
233        "Mushroom"
234    }
235
236    fn params(&self) -> &'static [ParamSpec] {
237        PARAMS
238    }
239
240    fn values(&self) -> Vec<ParamValue> {
241        let mut values = vec![
242            ParamValue::Float(self.scale_style.0),
243            ParamValue::Int(self.spark_count as i64),
244            ParamValue::Color(self.spark_color),
245            ParamValue::Float(self.spark_rotation.0),
246            ParamValue::Color(self.rays_color),
247            ParamValue::Int(self.ring_count as i64),
248            ParamValue::Float(self.ring_rotation.0),
249            ParamValue::Int(self.star_count as i64),
250        ];
251        values.extend(DEFAULT_RAMP.map(|(slot, fallback)| ParamValue::Color(self.star_colors.get_or(&slot, fallback))));
252        values
253    }
254
255    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
256        let mismatch = |index: usize| FxError::ParamMismatch(format!("mushroom param {index}"));
257        if values.len() != 14 {
258            return Err(FxError::ParamMismatch(format!("mushroom takes 14 parameters, got {}", values.len())));
259        }
260        let float = |index: usize| values[index].as_f64().ok_or(mismatch(index));
261        let int = |index: usize| values[index].as_i64().ok_or(mismatch(index));
262        let color = |index: usize| values[index].as_color().ok_or(mismatch(index));
263        self.scale_style = Factor::new(float(0)?);
264        self.spark_count = int(1)?.max(0) as u32;
265        self.spark_color = color(2)?;
266        self.spark_rotation = Factor::new(float(3)?);
267        self.rays_color = color(4)?;
268        self.ring_count = int(5)?.max(0) as u32;
269        self.ring_rotation = Factor::new(float(6)?);
270        self.star_count = int(7)?.max(0) as u32;
271        for (offset, (slot, _)) in DEFAULT_RAMP.iter().enumerate() {
272            let c = color(8 + offset)?;
273            self.star_colors.insert(*slot, c);
274        }
275        Ok(())
276    }
277
278    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
279        let size = ctx.window.size();
280        let (w, h) = (size.x as f64, size.y as f64);
281        let progress = match ctx.direction {
282            crate::Direction::Open => 1.0 - ctx.progress.as_f64(),
283            crate::Direction::Close => ctx.progress.as_f64(),
284        };
285        // Frame constants.
286        let aspect = w / h.max(f64::EPSILON);
287        let fscale = super::scale::window_scale(progress, self.scale_style.0);
288        let seed = ctx.seed.offset_2d();
289        let ramp = self.ramp_colors();
290        crate::render::fill(out, |x, y| {
291            let uv = Vec2::<()>::new((x as f64 + 0.5) / w, (y as f64 + 0.5) / h);
292            let mut c = ctx.window.sample_bilinear(super::scale::scale_uv(uv, fscale));
293            let window_alpha = c.a;
294            let star_uv = Vec2::<()>::new((uv.x - 0.5) * aspect, 1.0 - uv.y);
295            if self.spark_count > 0 {
296                c = crate::alpha_over(
297                    c,
298                    super::spark::sparks(
299                        Vec2::<()>::new(uv.x * aspect, uv.y),
300                        aspect,
301                        progress,
302                        self.spark_count,
303                        self.spark_rotation.0,
304                        seed.x,
305                        self.spark_color,
306                    ),
307                );
308            }
309            if self.rays_color.a > 0.0 {
310                c = crate::alpha_over(c, super::rays::rays(uv, progress, seed.y, self.rays_color));
311            }
312            if self.ring_count > 0 && self.star_count > 0 {
313                c = crate::alpha_over(
314                    c,
315                    super::star::stars(star_uv, aspect, progress, window_alpha, self.ring_count, self.ring_rotation.0, self.star_count, &ramp),
316                );
317            }
318            c
319        });
320    }
