11/// `sdRoundBox` - signed distance to a rounded box centred on the
12/// origin (negative inside).
13fn sd_round_box(p: Vec2<()>, b: Vec2<()>, r: f64) -> f64 {
14    let d = Vec2::<()>::new(p.x.abs() - b.x + r, p.y.abs() - b.y + r);
15    let mx = d.x.max(0.0);
16    let my = d.y.max(0.0);
17    (mx * mx + my * my).sqrt() + d.x.max(d.y).min(0.0) - r
18}

20/// `drawCake` - the cake in straight alpha. `p` is centred on the
21/// cake (y up), `aa` is one pixel in cake units, `t` is seconds.
22pub(crate) fn draw_cake(p: Vec2<()>, aa: f64, t: f64) -> Color<f64> {
23    let mut col = (0.0, 0.0, 0.0);
24    let mut a = 0.0;
25    let layer = |d: f64, c: (f64, f64, f64), col: &mut (f64, f64, f64), a: &mut f64| {
26        let m = crate::smoothstep(aa, -aa, d);
27        col.0 = crate::mix(col.0, c.0, m);
28        col.1 = crate::mix(col.1, c.1, m);
29        col.2 = crate::mix(col.2, c.2, m);
30        *a = a.max(m);
31    };
32
33    let sponge = (0.26, 0.13, 0.07);
34    let filling = (0.45, 0.26, 0.15);
35    let icing = (0.17, 0.08, 0.04);
36    let cream = (0.98, 0.95, 0.90);
37    layer(
38        sd_round_box(p - Vec2::<()>::new(0.0, -0.58), Vec2::<()>::new(0.78, 0.04), 0.04),
39        (0.78, 0.80, 0.82),
40        &mut col,
41        &mut a,
42    );
43    layer(sd_round_box(p - Vec2::<()>::new(0.0, -0.26), Vec2::<()>::new(0.56, 0.29), 0.06), sponge, &mut col, &mut a);
44    layer(sd_round_box(p - Vec2::<()>::new(0.0, -0.26), Vec2::<()>::new(0.56, 0.03), 0.02), filling, &mut col, &mut a);
45    layer(sd_round_box(p - Vec2::<()>::new(0.0, 0.06), Vec2::<()>::new(0.58, 0.07), 0.07), icing, &mut col, &mut a);
46    for i in 0..6 {
47        let fi = i as f64;
48        let x = -0.45 + fi * 0.18;
49        let len = 0.04 + 0.05 * crate::fract((fi * 12.9898).sin() * 43758.5453);
50        layer(
51            sd_round_box(p - Vec2::<()>::new(x, -0.02 - len * 0.5), Vec2::<()>::new(0.035, len), 0.035),
52            icing,
53            &mut col,
54            &mut a,
55        );
56    }
57    layer(
58        sd_round_box(p - Vec2::<()>::new(-0.12, 0.10), Vec2::<()>::new(0.30, 0.012), 0.012),
59        (0.42, 0.25, 0.15),
60        &mut col,
61        &mut a,
62    );
63    for i in 0..4 {
64        let x = if i < 2 { -0.42 + i as f64 * 0.2 } else { 0.22 + (i - 2) as f64 * 0.2 };
65        layer((p - Vec2::<()>::new(x, 0.15)).magnitude_f64() - 0.06, cream, &mut col, &mut a);
66        layer((p - Vec2::<()>::new(x, 0.21)).magnitude_f64() - 0.055, (0.70, 0.05, 0.08), &mut col, &mut a);
67    }
68    let stripe_k = if crate::fract((p.y + p.x) * 14.0) >= 0.5 {
69        (0.85, 0.15, 0.2)
70    } else {
71        (1.0, 1.0, 1.0)
72    };
73    layer(sd_round_box(p - Vec2::<()>::new(0.0, 0.30), Vec2::<()>::new(0.03, 0.17), 0.01), stripe_k, &mut col, &mut a);
74    let fp = p - Vec2::<()>::new(0.01 * (t * 31.0).sin(), 0.56);
75    let stretch = if fp.y > 0.0 { 0.70 } else { 0.55 };
76    let flame = (fp.x * fp.x + fp.y * fp.y * stretch * stretch).sqrt() - 0.045;
77    // Flame colour: mix(orange, cream) by flame depth.
78    let k = crate::smoothstep(0.0, -0.04, flame);
79    layer(flame, (crate::mix(1.0, 0.95, k), crate::mix(0.55, 0.95, k), crate::mix(0.1, 0.6, k)), &mut col, &mut a);
80
81    Color::<f64>::new_rgba(col.0, col.1, col.2, a)
82}
