5/// Panel-wave constants (the shader's tunables).
6pub(crate) const SPREAD: f64 = 0.6;
7pub(crate) const WAVE_AMP: f64 = 0.08;
8pub(crate) const WAVE_FREQ: f64 = 2.0;
9pub(crate) const WAVE_NOISE_PX: f64 = 300.0;
10pub(crate) const BORDER_LEAD: f64 = 0.25;
11pub(crate) const BORDER_TRAIL: f64 = 0.15;
12pub(crate) const BORDER_PX: f64 = 1.0;
13pub(crate) const EDGE_SHADE: f64 = 0.4;
14pub(crate) const PERSPECTIVE: f64 = 2.5;

16/// `wavePosition` - where a panel sits on the wave's leading edge
17/// (`0..1`), by `wave_shape`.
18pub(crate) fn wave_position(cell: Vec2<()>, ucell: Vec2<()>, frame_size: Vec2<()>, shape: u32) -> f64 {
19    let pos = (cell + Vec2::<()>::new(0.5, 0.5)).component_mul(&ucell);
20    let diag = (pos.x + pos.y) / (frame_size.x + frame_size.y);
21    match shape {
22        1 => (pos.x * pos.x + pos.y * pos.y).sqrt() / (frame_size.x * frame_size.x + frame_size.y * frame_size.y).sqrt(),
23        2 => {
24            let along = (pos.x - pos.y) / (frame_size.x + frame_size.y);
25            (diag + WAVE_AMP * (along * WAVE_FREQ * std::f64::consts::TAU).sin() + WAVE_AMP) / (1.0 + 2.0 * WAVE_AMP)
26        }
27        3 => {
28            let bent = diag + WAVE_AMP * (Simplex2D.at(Vec2::<()>::new(pos.x / WAVE_NOISE_PX + 17.3, pos.y / WAVE_NOISE_PX + 5.9)) * 2.0 - 1.0);
29            (bent + WAVE_AMP) / (1.0 + 2.0 * WAVE_AMP)
30        }
31        4 => {
32            let dx = pos.x - frame_size.x * 0.5;
33            let dy = pos.y - frame_size.y * 0.5;
34            (dx * dx + dy * dy).sqrt() / (0.5 * (frame_size.x * frame_size.x + frame_size.y * frame_size.y).sqrt())
35        }
36        _ => diag,
37    }
38}

40/// `cellTiming` - flip start/end of a panel within `[border_lead,
41/// 1 - border_trail]` of the animation.
42pub(crate) fn cell_timing(cell: Vec2<()>, ucell: Vec2<()>, frame_size: Vec2<()>, shape: u32, lead: f64, trail: f64) -> (f64, f64) {
43    let wave = wave_position(cell, ucell, frame_size, shape).clamp(0.0, 1.0);
44    let span = 1.0 - lead - trail;
45    let start = lead + wave * SPREAD * span;
46    (start, start + (1.0 - SPREAD) * span)
47}

49/// `cellProgress` - flip progress (`0..1`) of a panel.
50pub(crate) fn cell_progress(cell: Vec2<()>, ucell: Vec2<()>, frame_size: Vec2<()>, shape: u32, lead: f64, trail: f64, progress: f64) -> f64 {
51    let (t0, t1) = cell_timing(cell, ucell, frame_size, shape, lead, trail);
52    ((progress - t0) / (t1 - t0)).clamp(0.0, 1.0)
53}

55/// `cellBorder` - border strength (`0..1`) of a panel right now.
56pub(crate) fn cell_border(cell: Vec2<()>, ucell: Vec2<()>, frame_size: Vec2<()>, shape: u32, lead: f64, trail: f64, progress: f64) -> f64 {
57    let (t0, t1) = cell_timing(cell, ucell, frame_size, shape, lead, trail);
58    let fade_in = crate::smoothstep(t0 - lead.max(0.001), t0, progress);
59    let fade_out = crate::smoothstep(t1, t1 + trail.max(0.001), progress);
60    fade_in * (1.0 - fade_out)
61}

63/// The inverse perspective projection: from screen position to the
64/// `panelUV` on the turning panel, or `None` when the fragment falls
65/// in the gap (the shader's `onPanel` test).
66pub(crate) fn panel_uv(local: f64, ucell: Vec2<()>, centre: Vec2<()>, screen: Vec2<()>, dist: f64, vertical: bool) -> Option<Vec2<()>> {
67    let theta = local * std::f64::consts::PI;
68    let c_t = theta.cos();
69    let s_t = theta.sin();
70    let den = dist * c_t + screen.x * s_t;
71    if den.abs() <= 0.001 {
72        return None;
73    }
74    let mut u = dist * (screen.x - centre.x) / den;
75    let s = dist / (dist - u * s_t);
76    let v = screen.y / s - centre.y;
77    if local > 0.5 {
78        u = -u;
79    }
80    let m = if vertical { Vec2::<()>::new(u, v) } else { Vec2::<()>::new(v, u) };
81    let panel = m.component_mul(&Vec2::<()>::new(1.0 / ucell.x, 1.0 / ucell.y)) + Vec2::<()>::new(0.5, 0.5);
82    const EPS: f64 = 0.002;
83    let on = dist - u * s_t > 0.0 && panel.x >= -EPS && panel.y >= -EPS && panel.x <= 1.0 + EPS && panel.y <= 1.0 + EPS;
84    on.then(|| Vec2::<()>::new(panel.x.clamp(0.0, 1.0), panel.y.clamp(0.0, 1.0)))
85}
