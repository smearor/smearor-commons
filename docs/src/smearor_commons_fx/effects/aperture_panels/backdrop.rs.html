6/// "Star Field" backdrop constants.
7const STAR_LAYERS: usize = 3;
8const STAR_CELL_PX: f64 = 22.0;
9const STAR_CELL_STEP: f64 = 28.0;
10const STAR_DENSITY: f64 = 0.45;
11const STAR_DENSITY_STEP: f64 = 0.1;
12const STAR_TWINKLE: f64 = 0.4;
13const STAR_GLOW: f64 = 0.15;
14
15/// "Nebula" backdrop constants.
16const NEBULA_SCALE: f64 = 520.0;
17const NEBULA_WARP: f64 = 1.6;
18const NEBULA_BRIGHT: f64 = 1.5;
19const NEBULA_DRIFT: f64 = 0.04;
20const NEBULA_COL_A: (f64, f64, f64) = (0.16, 0.06, 0.38);
21const NEBULA_COL_B: (f64, f64, f64) = (0.62, 0.14, 0.52);
22const NEBULA_COL_C: (f64, f64, f64) = (0.10, 0.50, 0.66);
23const NEBULA_CORE: (f64, f64, f64) = (1.00, 0.86, 0.80);

25/// `starHash2` - the shader's sin-hash, two components.
26fn star_hash2(p: Vec2<()>) -> Vec2<()> {
27    let a = (p.x * 127.1 + p.y * 311.7).sin() * 43758.5453;
28    let b = (p.x * 269.5 + p.y * 183.3).sin() * 43758.5453;
29    Vec2::<()>::new(crate::fract(a), crate::fract(b))
30}

32/// `starField` - stars at layer pixel `p`, fixed to the screen via
33/// `screen_origin`; `seed` reshuffles the sky per animation.
34pub(crate) fn star_field(p: Vec2<()>, screen_origin: Vec2<()>, seed: f64, t: f64) -> (f64, f64, f64) {
35    let p = p + screen_origin;
36    let mut col = (0.0, 0.0, 0.0);
37    for i in 0..STAR_LAYERS {
38        let fi = i as f64;
39        let cell_px = STAR_CELL_PX + fi * STAR_CELL_STEP;
40        let sp = Vec2::<()>::new(p.x / cell_px + seed * 97.3 + fi * 17.7, p.y / cell_px + seed * 53.1 - fi * 11.3);
41        let cell = Vec2::<()>::new(sp.x.floor(), sp.y.floor());
42        let h1 = star_hash2(cell);
43        let h2 = star_hash2(cell + Vec2::<()>::new(5.17, 9.73));
44        if h2.x < STAR_DENSITY - fi * STAR_DENSITY_STEP {
45            let star = cell + Vec2::<()>::new(0.15 + 0.7 * h1.x, 0.15 + 0.7 * h1.y);
46            let d = ((sp.x - star.x) * (sp.x - star.x) + (sp.y - star.y) * (sp.y - star.y)).sqrt() * cell_px;
47            let radius = crate::mix(0.6, 0.9 + fi * 0.5, h2.y);
48            let bright = crate::mix(0.5, 1.0, crate::fract(h2.y * 7.31)) * (0.75 + 0.15 * fi);
49            let twinkle = 1.0 - STAR_TWINKLE * (0.5 + 0.5 * (t * (2.0 + 5.0 * h1.x) + std::f64::consts::TAU * h1.y).sin());
50            let tint = {
51                let f = crate::fract(h1.x * 13.7);
52                (crate::mix(0.72, 1.0, f), crate::mix(0.82, 0.88, f), crate::mix(1.0, 0.72, f))
53            };
54            let core = (-(d * d) / (radius * radius)).exp();
55            let halo = if i == STAR_LAYERS - 1 { STAR_GLOW * (-d / (radius * 3.0)).exp() } else { 0.0 };
56            let k = bright * twinkle * (core + halo);
57            col.0 += tint.0 * k;
58            col.1 += tint.1 * k;
59            col.2 += tint.2 * k;
60        }
61    }
62    (col.0.min(1.0), col.1.min(1.0), col.2.min(1.0))
63}

65/// `nebula` - rgb plus density in alpha, at layer pixel `p`.
66pub(crate) fn nebula(p: Vec2<()>, screen_origin: Vec2<()>, seed: f64, t: f64) -> Color<f64> {
67    let p = p + screen_origin;
68    let u = Vec2::<()>::new(p.x / NEBULA_SCALE + seed * 41.7, p.y / NEBULA_SCALE - seed * 23.3);
69
70    let big = Simplex2DFractal.at(Vec2::<()>::new(u.x * 0.45 + 11.3, u.y * 0.45 - 4.1));
71    let mask = crate::smoothstep(0.33, 0.60, big);
72
73    let q = Vec2::<()>::new(Simplex2DFractal.at(u), Simplex2DFractal.at(u + Vec2::<()>::new(5.2, 1.3)));
74    let f = Simplex2DFractal.at(u + q * NEBULA_WARP + Vec2::<()>::new(NEBULA_DRIFT * t, 0.0));
75    let dens = mask * crate::smoothstep(0.25, 0.75, f);
76
77    let k1 = crate::smoothstep(0.35, 0.65, q.x);
78    let mut col = (
79        crate::mix(NEBULA_COL_A.0, NEBULA_COL_B.0, k1),
80        crate::mix(NEBULA_COL_A.1, NEBULA_COL_B.1, k1),
81        crate::mix(NEBULA_COL_A.2, NEBULA_COL_B.2, k1),
82    );
83    let k2 = crate::smoothstep(0.40, 0.65, big) * crate::smoothstep(0.45, 0.60, q.y);
84    col = (
85        crate::mix(col.0, NEBULA_COL_C.0, k2),
86        crate::mix(col.1, NEBULA_COL_C.1, k2),
87        crate::mix(col.2, NEBULA_COL_C.2, k2),
88    );
89    let core = crate::smoothstep(0.6, 0.95, dens);
90    let k3 = core * core * 0.5;
91    col = (
92        crate::mix(col.0, NEBULA_CORE.0, k3),
93        crate::mix(col.1, NEBULA_CORE.1, k3),
94        crate::mix(col.2, NEBULA_CORE.2, k3),
95    );
96
97    let dust = crate::smoothstep(0.45, 0.60, Simplex2DFractal.at(Vec2::<()>::new(u.x * 2.1 + 3.1, u.y * 2.1 - 7.7)));
98    let lum = dens.powf(1.05) * NEBULA_BRIGHT * crate::mix(0.45, 1.0, dust);
99
100    Color::<f64>::new_rgba(col.0 * lum, col.1 * lum, col.2 * lum, dens)
101}
