20/// An expanding ring of glowing "atom" particles swallows the window
21/// - the content dissolves into the effect color while simplex
22///   molecules condense around it, then a bright heart fades last.
23///
24/// Behavioral spec (source shader: `energize-a.frag`, re-implemented
25/// from semantics): `progress = easeOutQuad(uProgress)` (only the
26/// window mask reads the direction). `atomMask` expands from the
27/// center over `FADE_IN_TIME`, dies over `FADE_OUT_TIME`, bounded by
28/// a 50-px absolute edge fade; `heartMask` is the
29/// `relativeEdgeMask(0.5)⁵ * 3` center flash over the last
30/// `HEART_FADE_TIME`. The window fades by `(1 - fadeOut)²` (inverted
31/// for `Open`) and tints toward `color`. Particles: one static
32/// `simplex3D` octave cubed, plus three octaves of
33/// `1/(1-atoms)-1`-sharped atoms scrolling up at `0.1 * progress *
34/// duration`, all scaled by `scale` and masked by `atomMask`.
35#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
36#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
37pub struct EnergizeA {
38    /// Particle/tint color - schema `energize-a-color`, default `rgb(255, 200, 50)`.
39    #[serde(default = "EnergizeA::default_color")]
40    pub color: Color<f64>,
41    /// Noise scale - schema `energize-a-scale`, default `1.0`.
42    #[serde(default = "EnergizeA::default_scale")]
43    pub scale: ScaleFactor,
44}
45

64/// Particle fade-in - `FADE_IN_TIME`.
65const FADE_IN_TIME: f64 = 0.3;
66/// Particle fade-out - `FADE_OUT_TIME`.
67const FADE_OUT_TIME: f64 = 0.6;
68/// Center-flash duration - `HEART_FADE_TIME`.
69const HEART_FADE_TIME: f64 = 0.3;
70/// Edge fade in px - `EDGE_FADE_WIDTH`.
71const EDGE_FADE_WIDTH: f64 = 50.0;
72
73const PARAMS: &[ParamSpec] = &[
74    ParamSpec {
75        key: "color",
76        label: "Effect Color",
77        kind: ParamKind::Color {
78            default: EnergizeA::default_color(),
79        },
80    },
81    ParamSpec {
82        key: "scale",
83        label: "Scale",
84        kind: ParamKind::Float {
85            min: 0.1,
86            max: 5.0,
87            step: 0.1,
88            default: 1.0,
89        },
90    },
91];

93impl WindowEffect for EnergizeA {
94    fn nick(&self) -> &'static str {
95        "energize-a"
96    }
97
98    fn label(&self) -> &'static str {
99        "Energize A"
100    }
101
102    fn params(&self) -> &'static [ParamSpec] {
103        PARAMS
104    }
105
106    fn values(&self) -> Vec<ParamValue> {
107        vec![ParamValue::Color(self.color), ParamValue::Float(self.scale.0)]
108    }
109
110    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
111        match values {
112            [ParamValue::Color(color), ParamValue::Float(scale)] => {
113                self.color = *color;
114                self.scale = ScaleFactor::new(*scale);
115                Ok(())
116            }
117            _ => Err(FxError::ParamMismatch(format!("energize-a takes color + float, got {values:?}"))),
118        }
119    }
120
121    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
122        let size = ctx.window.size();
123        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
124        let progress = EasingFunction::EaseOut.apply(ctx.progress).as_f64();
125        // Frame constants (getMasks' time schedule).
126        let fade_in = (progress / FADE_IN_TIME).clamp(0.0, 1.0);
127        let fade_out = ((progress - FADE_IN_TIME) / FADE_OUT_TIME).clamp(0.0, 1.0);
128        let heart = ((progress - (1.0 - HEART_FADE_TIME)) / HEART_FADE_TIME).clamp(0.0, 1.0);
129        let atom_grow = fade_in * 2.0 + 1.0;
130        let atom_fade = fade_in * crate::smoothstep(1.0, 0.0, fade_out);
131        let heart_glow = 3.0 * fade_out * (1.0 - heart);
132        let window_base = (1.0 - fade_out).powi(2);
133        let window_mask = match ctx.direction {
134            crate::Direction::Open => 1.0 - window_base,
135            crate::Direction::Close => window_base,
136        };
137        let tint_t = 0.2 * window_mask + 0.8;
138        let time = ctx.duration.as_secs_f64();
139        crate::render::fill(out, |x, y| {
140            let px = Vec2::<()>::new(x as f64 + 0.5, y as f64 + 0.5);
141            let uv = Vec2::<()>::new(px.x / size_px.x, px.y / size_px.y);
142            // Masks.
143            let dist = ((uv.x - 0.5).mul_add(uv.x - 0.5, (uv.y - 0.5) * (uv.y - 0.5)).sqrt()) * 4.0;
144            let mut atom_mask = crate::smoothstep(0.0, 1.0, atom_grow - dist) * atom_fade;
145            atom_mask *= crate::absolute_edge_mask(px, size_px, ctx.padding, Pixels::new(EDGE_FADE_WIDTH), 0.5, ctx.fullscreen);
146            let heart_mask = heart_glow * crate::relative_edge_mask(uv, 0.5).powi(5);
147            atom_mask = (heart_mask + atom_mask).clamp(0.0, 1.0);
148            // Window dissolves toward the tint.
149            let mut c = ctx.window.sample_bilinear(uv);
150            c = Color::<f64>::new_rgba(
151                crate::mix(self.color.r, c.r, tint_t),
152                crate::mix(self.color.g, c.g, tint_t),
153                crate::mix(self.color.b, c.b, tint_t),
154                c.a * window_mask,
155            );
156            // Atom particles.
157            let mut scaled = Vec2::<()>::new((uv.x - 0.5) * (1.0 + 0.1 * progress), (uv.y - 0.5) * (1.0 + 0.1 * progress));
158            scaled.x /= self.scale.0;
159            scaled.y /= self.scale.0;
160            let first = Vec2::<()>::new((scaled.x) * 0.010598 * 0.5 * size_px.x, (scaled.y + 0.1 * progress * time) * 0.010598 * size_px.y);
161            let mut particles = 0.2 * Simplex3D.at(Vec3::<()>::new(first.x, first.y, 0.0)).powi(3);
162            for i in 1..=3u32 {
163                let f = i as f64;
164                let auv = Vec2::<()>::new(scaled.x * 0.12154 / 1.5f64.powf(f) * size_px.x, scaled.y * 0.12154 / 1.5f64.powf(f) * size_px.y);
165                let atoms = Simplex3D.at(Vec3::<()>::new(auv.x, auv.y, 2.0 * progress * time / f));
166                particles += 0.5 * (0.2 * (1.0 / (1.0 - atoms).max(f64::EPSILON) - 1.0)).powi(2);
167            }
168            c.r += self.color.r * particles * atom_mask;
169            c.g += self.color.g * particles * atom_mask;
170            c.b += self.color.b * particles * atom_mask;
171            c.a += particles * atom_mask;
172            c
173        });
174    }
