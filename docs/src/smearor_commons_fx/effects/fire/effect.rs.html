26/// The window burns away behind a fire-colored fractal noise front.
27///
28/// Behavioral spec (source shader: `fire.frag`, re-implemented from
29/// semantics): a vertical noise field (`Simplex2DFractal`, or the 3D
30/// variant animated along the third axis) drifts upward over the run;
31/// an [`EffectMask`] splits progress into the burn-down phase (the
32/// window dissolves along the burn direction) and the after-burn phase
33/// (the remaining flame fades). The noise modulates the 5-stop
34/// [`FireColor`] ramp which is alpha-composited over the fading
35/// window.
36#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
37#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
38pub struct Fire {
39    /// Noise scale - schema `fire-scale`, default `1.0`.
40    #[serde(default = "Fire::default_scale")]
41    pub scale: ScaleFactor,
42    /// Vertical noise drift speed - schema `fire-movement-speed`, default `0.5`.
43    #[serde(default = "Fire::default_movement_speed")]
44    pub movement_speed: Factor,
45    /// Hash-derived palette instead of the gradient - schema `fire-random-color`, default `false`.
46    #[serde(default)]
47    pub random_color: bool,
48    /// Animate through 3D noise instead of 2D - schema `fire-noise-3d`, default `false`.
49    #[serde(default)]
50    pub noise_3d: bool,
51    /// Burn bottom-to-top on close - schema `fire-closing-bottom-to-top`, default `false`.
52    #[serde(default)]
53    pub closing_bottom_to_top: bool,
54    /// Burn bottom-to-top on open - schema `fire-opening-bottom-to-top`, default `false`.
55    #[serde(default)]
56    pub opening_bottom_to_top: bool,
57    /// The five-stop flame ramp - the named [`Palette`] equivalent of
58    /// BMW's `fire-color-1` .. `fire-color-5` GSettings keys. Slots are
59    /// [`FireColor`] in rising-temperature order; missing slots fall
60    /// back to the classic fire defaults.
61    #[serde(default = "Fire::default_colors")]
62    #[cfg_attr(feature = "schemars", schemars(with = "std::collections::HashMap<String, Color<f64>>"))]
63    pub colors: Palette<FireColor, f64>,
64}

66impl Fire {
67    const fn default_scale() -> ScaleFactor {
68        ScaleFactor::new(1.0)
69    }
70    const fn default_movement_speed() -> Factor {
71        Factor::new(0.5)
72    }
73
74    /// The palette form of [`DEFAULT_RAMP`].
75    fn default_colors() -> Palette<FireColor, f64> {
76        Palette::from_iter(DEFAULT_RAMP)
77    }
78
79    /// The five ramp colors in slot order, slot-missing falls back to
80    /// the classic default so an incomplete palette still renders.
81    fn ramp_colors(&self) -> [Color<f64>; 5] {
82        DEFAULT_RAMP.map(|(slot, fallback)| self.colors.get_or(&slot, fallback))
83    }
84
85    /// The `uRandomColor` palette: seed-hashed hue rotations of red at
86    /// fixed alphas - BMW's `offsetHue` chain `red -> +0.1 -> +0.1`.
87    fn random_colors(&self, ctx: &FxContext<'_>) -> [Color<f64>; 5] {
88        let hue: f64 = ctx.seed.as_f64().hash();
89        let base0 = crate::offset_hue(Color::<f64>::new(1.0, 0.0, 0.0), hue);
90        let base1 = crate::offset_hue(base0, 0.1);
91        let base2 = crate::offset_hue(base1, 0.1);
92        [
93            Color::<f64>::new_rgba(base0.r, base0.g, base0.b, 0.0),
94            Color::<f64>::new_rgba(base0.r, base0.g, base0.b, 0.5),
95            Color::<f64>::new_rgba(base1.r, base1.g, base1.b, 0.8),
96            Color::<f64>::new_rgba(base1.r, base1.g, base1.b, 0.9),
97            Color::<f64>::new_rgba(base2.r, base2.g, base2.b, 1.0),
98        ]
99    }
100
101    /// The direction-aware burn direction handed to [`EffectMask::at`].
102    fn bottom_to_top(&self, ctx: &FxContext<'_>) -> bool {
103        match ctx.direction {
104            crate::Direction::Open => self.opening_bottom_to_top,
105            crate::Direction::Close => self.closing_bottom_to_top,
106        }
107    }
108}

192impl WindowEffect for Fire {
193    fn nick(&self) -> &'static str {
194        "fire"
195    }
196
197    fn label(&self) -> &'static str {
198        "Fire"
199    }
200
201    fn params(&self) -> &'static [ParamSpec] {
202        PARAMS
203    }
204
205    fn values(&self) -> Vec<ParamValue> {
206        let mut values = vec![
207            ParamValue::Float(self.scale.0),
208            ParamValue::Float(self.movement_speed.0),
209            ParamValue::Bool(self.random_color),
210            ParamValue::Bool(self.noise_3d),
211            ParamValue::Bool(self.closing_bottom_to_top),
212            ParamValue::Bool(self.opening_bottom_to_top),
213        ];
214        values.extend(DEFAULT_RAMP.map(|(slot, fallback)| ParamValue::Color(self.colors.get_or(&slot, fallback))));
215        values
216    }
217
218    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
219        let mismatch = |index: usize| FxError::ParamMismatch(format!("fire param {index}"));
220        let float = |index: usize| values.get(index).and_then(ParamValue::as_f64).ok_or(mismatch(index));
221        let boolean = |index: usize| values.get(index).and_then(ParamValue::as_bool).ok_or(mismatch(index));
222        let color = |index: usize| values.get(index).and_then(ParamValue::as_color).ok_or(mismatch(index));
223        if values.len() != 11 {
224            return Err(FxError::ParamMismatch(format!("fire takes 11 parameters, got {}", values.len())));
225        }
226        self.scale = ScaleFactor(float(0)?);
227        self.movement_speed = Factor(float(1)?);
228        self.random_color = boolean(2)?;
229        self.noise_3d = boolean(3)?;
230        self.closing_bottom_to_top = boolean(4)?;
231        self.opening_bottom_to_top = boolean(5)?;
232        for (index, (slot, _fallback)) in DEFAULT_RAMP.iter().enumerate() {
233            self.colors.insert(*slot, color(6 + index)?);
234        }
235        Ok(())
236    }
237
238    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
239        let size = ctx.window.size();
240        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
241        let colors = if self.random_color { self.random_colors(ctx) } else { self.ramp_colors() };
242        let drift = ctx.progress.as_f64() * ctx.duration.as_secs_f64() * self.movement_speed.0;
243        let bottom_to_top = self.bottom_to_top(ctx);
244        let seed_offset = ctx.seed.offset_2d();
245        crate::render::fill(out, |x, y| {
246            let px = Vec2::<()>::new(x as f64 + 0.5, y as f64 + 0.5);
247            let uv = px / size_px;
248            // Noise domain: BMW's 400x600-normalized aspect-corrected uv.
249            let mut n_uv = px / Vec2::<()>::new(400.0, 600.0) / self.scale.0;
250            n_uv += seed_offset;
251            n_uv.y += drift;
252            let mut noise = match self.noise_3d {
253                true => Simplex3DFractal.at(Vec3::<()>::new(n_uv.x * 4.0, n_uv.y * 4.0, drift * 1.5)),
254                false => Simplex2DFractal.at(n_uv * 4.0),
255            };
256            let mask = EffectMask::at(ctx, bottom_to_top, px, size_px);
257            noise *= mask.effect;
258
259            let fire = fire_ramp(noise, &colors);
260            let mut window = ctx.window.sample_bilinear(uv);
261            window.a *= mask.window;
262            fire.alpha_over(&window)
263        });
264    }
265}

315    /// `window_mask` is monotonic along the burn direction during the
316    /// burn phase; `effect_mask` is strongest near the burn edge. The
317    /// 200x200 window keeps the sampled column inside the
318    /// `EDGE_FADE = 70` interior.
319    #[test]
320    fn fire_mask_phases() {
321        let data = vec![255u8; 200 * 200 * 4];
322        let window = BufferView::<Rgba>::from_packed(&data, Size::new(200, 200)).unwrap();
323        let size_px = Vec2::<()>::new(200.0, 200.0);
324        let ctx = ctx_for(window, Direction::Close, 0.2, Seed(1));
325        // Burn runs top -> down: the window mask is non-decreasing
326        // downward (top already burnt, bottom still intact).
327        let mut prev = 0.0;
328        let mut saw_transition = false;
329        for y in 0..200 {
330            let m = EffectMask::at(&ctx, false, Vec2::<()>::new(100.0, y as f64 + 0.5), size_px).window;
331            assert!(m >= prev - 1e-9, "window mask must not decrease downward (y={y}, {prev} -> {m})");
332            if m > 0.0 && m < 1.0 {
333                saw_transition = true;
334            }
335            prev = m;
336        }
337        assert!(saw_transition, "burn front must produce a fade zone");
338        // The effect mask peaks inside the burn region.
339        let peak = (0..200)
340            .map(|y| EffectMask::at(&ctx, false, Vec2::<()>::new(100.0, y as f64 + 0.5), size_px).effect)
341            .fold(0.0, f64::max);
342        assert!(peak > 0.5, "effect mask must peak at the burn edge, got {peak}");
343    }

345    /// Same seed + progress = identical frame; different seed differs
346    /// (via the random-color path which hashes the seed).
347    #[test]
348    fn fire_deterministic() {
349        let data = [200, 100, 50, 255].repeat(32 * 32);
350        let window = BufferView::<Rgba>::from_packed(&data, Size::new(32, 32)).unwrap();
351        let mut out_a = PixelBuffer::<Rgba>::new(Size::new(32, 32));
352        let mut out_b = PixelBuffer::<Rgba>::new(Size::new(32, 32));
353        let fire = Fire {
354            random_color: true,
355            ..Default::default()
356        };
357
358        fire.apply(&ctx_for(window, Direction::Close, 0.5, Seed(0xDEAD_BEEF_CAFE_0042)), &mut out_a.view_mut());
359        fire.apply(&ctx_for(window, Direction::Close, 0.5, Seed(0xDEAD_BEEF_CAFE_0042)), &mut out_b.view_mut());
360        assert_eq!(out_a.view().data(), out_b.view().data(), "same seed = same frame");
361
362        fire.apply(&ctx_for(window, Direction::Close, 0.5, Seed(0x0123_4567_89AB_CDEF)), &mut out_b.view_mut());
363        assert_ne!(out_a.view().data(), out_b.view().data(), "different seed differs");
364    }
