1use crate::FxContext;
2use smearor_commons_animation::EasingFunction;
3use smearor_commons_geometry::Pixels;
4use smearor_commons_geometry::Vec2;

6/// The dual-phase mask pair `effectMask` computes: `window` fades the
7/// window along the burn direction, `effect` gates the flame (strongest
8/// at the burning edge).
9#[derive(Debug, Clone, Copy)]
10pub(super) struct EffectMask {
11    /// Alpha multiplier for the window texture.
12    pub(super) window: f64,
13    /// Alpha multiplier for the fire front.
14    pub(super) effect: f64,
15}
16
17/// Edge fade width in pixels - `fire.frag`'s `EDGE_FADE`.
18const EDGE_FADE: f64 = 70.0;
19/// Relative width of the window-dissolve gradient - `FADE_WIDTH`.
20const FADE_WIDTH: f64 = 0.1;
21/// The progress fraction spent burning the window - `HIDE_TIME`.
22const HIDE_TIME: f64 = 0.4;
23
24impl EffectMask {
25    /// The paired window/effect masks at pixel `px` - the `effectMask`
26    /// helper's semantics:
27    ///
28    /// `progress` is `easeOutQuad`-eased inside. `burn` covers the
29    /// first `HIDE_TIME` of the animation (the window dissolves), the
30    /// remaining `1 - HIDE_TIME` is after-burn (the flame fades). The
31    /// burn front runs along `t` - `uv.y` shrunk by `FADE_WIDTH`, or
32    /// `1 - uv.y` when `bottom_to_top`. `window` is `1` above the
33    /// front; `effect` is densest at the front and fades by a sqrt
34    /// curve after the window is gone. The frame edges are softened by
35    /// `absolute_edge_mask`. For `Open` the window mask inverts - the
36    /// window un-burns.
37    pub(super) fn at(ctx: &FxContext<'_>, bottom_to_top: bool, px: Vec2<()>, size_px: Vec2<()>) -> Self {
38        let progress = EasingFunction::EaseOut.apply(ctx.progress).as_f64();
39        let burn = (progress / HIDE_TIME).clamp(0.0, 1.0);
40        let after_burn = ((progress - HIDE_TIME) / (1.0 - HIDE_TIME)).clamp(0.0, 1.0);
41
42        let y01 = px.y / size_px.y;
43        let t = if bottom_to_top { 1.0 - y01 } else { y01 } * (1.0 - FADE_WIDTH);
44
45        let mut window = 1.0 - ((burn - t) / FADE_WIDTH).clamp(0.0, 1.0);
46        let mut effect = (t * (1.0 - window) / burn.max(f64::EPSILON)).clamp(0.0, 1.0);
47
48        if progress > HIDE_TIME {
49            let fade = (1.0 - after_burn * after_burn).sqrt();
50            effect *= crate::mix(1.0, 1.0 - t, after_burn) * fade;
51        }
52
53        effect *= crate::absolute_edge_mask(px, size_px, ctx.padding, Pixels::new(EDGE_FADE), 0.5, ctx.fullscreen);
54
55        if ctx.direction == crate::Direction::Open {
56            window = 1.0 - window;
57        }
58
59        EffectMask { window, effect }
60    }
61}
