19/// A swirling portal opens, the window shrinks and travels through
20/// it while it wobbles elastically - five composited layers: a
21/// radial base gradient, two noise-band layers at different whirl
22/// speeds, a glowing rim and sparkles.
23///
24/// Behavioral spec (source shader: `portal.frag`, re-implemented
25/// from semantics): `getPortalScale` `easeOutBack`-grows the portal
26/// over the first `PORTAL_OPEN_TIME / duration` and shrinks over
27/// the last `PORTAL_CLOSE_TIME / duration`; `getPortalWobble`
28/// `easeInBack`-pulses around the window-travel midpoint
29/// (`WINDOW_OPEN_TIME`). Layer coords center at `(uv-0.5)*2*1.5`,
30/// divided by `scale*0.5+0.5`. Layers: radial `darken` gradient
31/// (alpha disc at `dist>1`), two whirled `simplex2D` bands at
32/// `0.6`/`1.3` detail frequencies (`alphaOver`), a rim of
33/// `mix(1,5,rand)*GLOW_EDGE_WIDTH*detailScale - 150*|dist-1|`, and
34/// additive `lighten` sparkles at `noise*rand+0.9`^50. `rand =
35/// dot(displace,displace)` from `getRandomDisplace(seed*3.7)`. The
36/// window travels on `easeInBack(progress, 1.2)`: scales to
37/// `WINDOW_SCALE`, squishes `WINDOW_SQUISH` vertically, tilts
38/// `WINDOW_TILT`, fading late (`(1-p)*3`). Output is `alphaOver
39/// (portal, window)`.
40#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
41#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
42pub struct Portal {
43    /// Portal tint - schema `portal-color`, default `rgb(121, 233, 24)`.
44    #[serde(default = "Portal::default_color")]
45    pub color: Color<f64>,
46    /// Whirl rotation speed - schema `portal-rotation_speed`, default `0.3`.
47    #[serde(default = "Portal::default_rotation_speed")]
48    pub rotation_speed: Factor,
49    /// Whirl warp amount - schema `portal-whirling`, default `1.0`.
50    #[serde(default = "Portal::default_whirling")]
51    pub whirling: Factor,
52    /// Noise detail - schema `portal-details`, default `15.0`.
53    #[serde(default = "Portal::default_details")]
54    pub details: Factor,
55}

83/// Wobble envelope - `PORTAL_WOBBLE_TIME`/`STRENGTH`.
84const PORTAL_WOBBLE_TIME: f64 = 0.8;
85const PORTAL_WOBBLE_STRENGTH: f64 = 1.2;
86/// Rim width - `GLOW_EDGE_WIDTH`.
87const GLOW_EDGE_WIDTH: f64 = 5.0;
88/// Window travel target scale - `WINDOW_SCALE`.
89const WINDOW_SCALE: f64 = 0.3;
90/// Window vertical squish - `WINDOW_SQUISH`.
91const WINDOW_SQUISH: f64 = 1.0;
92/// Window x-tilt - `WINDOW_TILT`.
93const WINDOW_TILT: f64 = -1.0;
94/// Portal open/close fractions of duration.
95const PORTAL_OPEN_TIME: f64 = 0.4;
96const PORTAL_CLOSE_TIME: f64 = 0.4;
97/// Window travel fraction of duration.
98const WINDOW_OPEN_TIME: f64 = 0.35;

140impl WindowEffect for Portal {
141    fn nick(&self) -> &'static str {
142        "portal"
143    }
144
145    fn label(&self) -> &'static str {
146        "Portal"
147    }
148
149    fn params(&self) -> &'static [ParamSpec] {
150        PARAMS
151    }
152
153    fn values(&self) -> Vec<ParamValue> {
154        vec![
155            ParamValue::Color(self.color),
156            ParamValue::Float(self.rotation_speed.0),
157            ParamValue::Float(self.whirling.0),
158            ParamValue::Float(self.details.0),
159        ]
160    }
161
162    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
163        let mismatch = |index: usize| FxError::ParamMismatch(format!("portal param {index}"));
164        if values.len() != 4 {
165            return Err(FxError::ParamMismatch(format!("portal takes 4 parameters, got {}", values.len())));
166        }
167        let float = |index: usize| values[index].as_f64().ok_or(mismatch(index));
168        self.color = values[0].as_color().ok_or(mismatch(0))?;
169        self.rotation_speed = Factor::new(float(1)?);
170        self.whirling = Factor::new(float(2)?);
171        self.details = Factor::new(float(3)?);
172        Ok(())
173    }
174
175    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
176        let size = ctx.window.size();
177        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
178        let duration = ctx.duration.as_secs_f64();
179        let u_progress = ctx.progress.as_f64();
180        let seed = ctx.seed.offset_2d();
181        // Frame constants.
182        let close_time = PORTAL_CLOSE_TIME / duration.max(f64::EPSILON);
183        let open_time = PORTAL_OPEN_TIME / duration.max(f64::EPSILON);
184        let scale = if u_progress < open_time {
185            EasingFunction::BackOut { overshoot_amount: 1.5 }
186                .apply(Fraction::new((u_progress / open_time).clamp(0.0, 1.0)))
187                .as_f64()
188        } else if u_progress > 1.0 - close_time {
189            EasingFunction::BackOut { overshoot_amount: 1.5 }
190                .apply(Fraction::new((1.0 - (u_progress - 1.0 + close_time) / close_time).clamp(0.0, 1.0)))
191                .as_f64()
192        } else {
193            1.0
194        };
195        let travel = match ctx.direction {
196            crate::Direction::Open => (1.0 - u_progress) / WINDOW_OPEN_TIME * duration,
197            crate::Direction::Close => u_progress / WINDOW_OPEN_TIME * duration,
198        };
199        let wobble_phase = (1.0 - ((travel - 1.0) / PORTAL_WOBBLE_TIME).abs()).clamp(0.0, 1.0);
200        let wobble_amount = EasingFunction::BackIn { overshoot_amount: 1.7 }.apply(Fraction::new(wobble_phase)).as_f64() * PORTAL_WOBBLE_STRENGTH;
201        let detail_scale = 10000.0 / (size_px.x + size_px.y) / self.details.0;
202        let window_prog = EasingFunction::BackIn { overshoot_amount: 1.2 }
203            .apply(Fraction::new(travel.clamp(0.0, 1.0)))
204            .as_f64();
205        let rotation_base = self.rotation_speed.0 * u_progress * duration;
206        let warp_base = self.whirling.0 * (6.0 + 1.5 * u_progress);
207        let portal_fade = scale.clamp(0.0, 1.0).powi(2);
208        let window = ctx.window;
209        let (w, h) = (size_px.x, size_px.y);
210        let whirled = |coords: Vec2<()>, speed: f64, warp: f64| crate::whirl(coords, warp_base * warp, rotation_base * speed);
211        let displace = |s: Vec2<()>, scale_f: f64| {
212            Vec2::<()>::new(
213                Simplex2D.at(Vec2::<()>::new(s.x * scale_f + seed.x, s.y * scale_f + seed.y)) - 0.5,
214                Simplex2D.at(Vec2::<()>::new(s.x * scale_f + seed.x + 7.89, s.y * scale_f + seed.y + 123.0)) - 0.5,
215            )
216        };
217        let portal_color = |uv: Vec2<()>| -> Color<f64> {
218            let mut coords = Vec2::<()>::new((uv.x - 0.5) * 3.0, (uv.y - 0.5) * 3.0);
219            let inv = 1.0 / (scale * 0.5 + 0.5).max(0.01);
220            coords.x *= inv;
221            coords.y *= inv;
222            let dist = coords.x.mul_add(coords.x, coords.y * coords.y).sqrt();
223            let wobble_scale = (1.0 - dist) * (-dist).exp() * wobble_amount;
224            let wobble = Vec2::<()>::new(coords.x * wobble_scale, coords.y * wobble_scale);
225            let alpha = if dist > 1.0 { 0.0 } else { 1.0 };
226            // Layer 1: distorted radial gradient.
227            let mut layer_coords = whirled(Vec2::<()>::new(coords.x - wobble.x, coords.y - wobble.y), 1.0, 1.0);
228            let mut disp = displace(Vec2::<()>::new(layer_coords.x * 3.7, layer_coords.y * 3.7), 3.7);
229            layer_coords = Vec2::<()>::new(layer_coords.x + disp.x * 0.1, layer_coords.y + disp.y * 0.1);
230            let grad_t = layer_coords.x.mul_add(layer_coords.x, layer_coords.y * layer_coords.y).sqrt().powi(5);
231            let rand = disp.x * disp.x + disp.y * disp.y;
232            let mut color = Color::<f64>::new_rgba(
233                crate::mix(crate::darken(self.color, 0.8).r, crate::darken(self.color, 0.2).r, grad_t),
234                crate::mix(crate::darken(self.color, 0.8).g, crate::darken(self.color, 0.2).g, grad_t),
235                crate::mix(crate::darken(self.color, 0.8).b, crate::darken(self.color, 0.2).b, grad_t),
236                alpha,
237            );
238            // Layer 2: first whirled band.
239            let noise = Simplex2D.at(Vec2::<()>::new(layer_coords.x / detail_scale + 12.3 + seed.x, layer_coords.y / detail_scale + 56.4 + seed.y));
240            let darkened = crate::darken(self.color, 0.3);
241            color = crate::alpha_over(color, Color::<f64>::new_rgba(darkened.r, darkened.g, darkened.b, if noise > 0.6 { alpha } else { 0.0 }));
242            // Layer 3: faster, brighter band.
243            let mut layer_coords = whirled(Vec2::<()>::new(coords.x - wobble.x * 1.5, coords.y - wobble.y * 1.5), 0.75, 0.5);
244            disp = displace(layer_coords, 12.2);
245            layer_coords = Vec2::<()>::new(layer_coords.x + disp.x * 0.1, layer_coords.y + disp.y * 0.1);
246            let noise = Simplex2D.at(Vec2::<()>::new(layer_coords.x / detail_scale * 1.3 + seed.x, layer_coords.y / detail_scale * 1.3 + seed.y));
247            color = crate::alpha_over(color, Color::<f64>::new_rgba(self.color.r, self.color.g, self.color.b, if noise > 0.6 { alpha } else { 0.0 }));
248            color = clamped(color);
249            // Layer 4: glowing rim.
250            let edge = crate::mix(1.0, 5.0, rand) * GLOW_EDGE_WIDTH * detail_scale - 150.0 * (dist - 1.0).abs();
251            color = crate::alpha_over(color, Color::<f64>::new_rgba(self.color.r, self.color.g, self.color.b, edge.clamp(0.0, 1.0)));
252            // Layer 5: additive sparkles.
253            let layer_coords = whirled(Vec2::<()>::new(coords.x - wobble.x * 1.8, coords.y - wobble.y * 1.8), 1.25, 0.0);
254            let noise = Simplex2D.at(Vec2::<()>::new(layer_coords.x / detail_scale * 3.0 + seed.x, layer_coords.y / detail_scale * 3.0 + seed.y));
255            let sparkle = (noise * rand + 0.9).powi(50).clamp(0.0, 1.0);
256            let light = crate::lighten(self.color, 0.8);
257            color.r += light.r * sparkle;
258            color.g += light.g * sparkle;
259            color.b += light.b * sparkle;
260            color.a *= portal_fade;
261            clamped(color)
262        };
263        crate::render::fill(out, |x, y| {
264            let uv = Vec2::<()>::new((x as f64 + 0.5) / w, (y as f64 + 0.5) / h);
265            // Window traveling through the portal.
266            let mut wc = Vec2::<()>::new(uv.x * 2.0 - 1.0, uv.y * 2.0 - 1.0);
267            let div = crate::mix(1.0, WINDOW_SCALE, window_prog).max(f64::EPSILON);
268            wc.x /= div;
269            wc.y /= div * crate::mix(1.0, 1.0 - 0.2 * WINDOW_SQUISH, window_prog).max(f64::EPSILON);
270            wc.x /= crate::mix(1.0, 1.0 - 0.1 * WINDOW_TILT * wc.y, window_prog).max(f64::EPSILON);
271            wc = Vec2::<()>::new(wc.x * 0.5 + 0.5, wc.y * 0.5 + 0.5);
272            let mut window_color = window.sample_bilinear(wc);
273            if !(0.0..=1.0).contains(&wc.x) || !(0.0..=1.0).contains(&wc.y) {
274                window_color.a = 0.0;
275            }
276            window_color.a *= ((1.0 - window_prog) * 3.0).clamp(0.0, 1.0);
277            crate::alpha_over(portal_color(uv), window_color)
278        });
279    }
280}

282/// GLSL `clamp(color, 0.0, 1.0)` per channel.
283fn clamped(c: Color<f64>) -> Color<f64> {
284    Color::<f64>::new_rgba(c.r.clamp(0.0, 1.0), c.g.clamp(0.0, 1.0), c.b.clamp(0.0, 1.0), c.a.clamp(0.0, 1.0))
