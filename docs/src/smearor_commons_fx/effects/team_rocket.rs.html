19/// The window gets sucked toward an off-center point while a rotating
20/// white sparkle flares - Team Rocket blasting off again.
21///
22/// Behavioral spec (source shader: `team-rocket.frag`, re-implemented
23/// from semantics): progress splits at `animation_split` - the first
24/// half `easeOutQuad`-warp-scales and optionally random-rotates the
25/// window uv (`uWindowRotation` -> seed-hashed xyz factors), the
26/// second half `easeInOutSine`-drives a star sparkle (`getSpark`:
27/// superellipse mask powered by `easeInExpo` brightness, faded by
28/// `fadeInOut(t, 2)`) plus edge fades `fadeInOut(uv, 8)`.
29#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
30#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
31pub struct TeamRocket {
32    /// Where warp hands off to sparkle - schema `team-rocket-animation_split`, default `0.5`.
33    #[serde(default = "TeamRocket::default_animation_split")]
34    pub animation_split: Factor,
35    /// Sparkle x offset - schema `team-rocket-horizontal_sparkle_position`, default `0.0`.
36    #[serde(default)]
37    pub horizontal_sparkle_position: Factor,
38    /// Sparkle y offset - schema `team-rocket-vertical_sparkle_position`, default `0.5`.
39    #[serde(default = "TeamRocket::default_vertical_sparkle_position")]
40    pub vertical_sparkle_position: Factor,
41    /// Sparkle rotation turns - schema `team-rocket-sparkle_rot`, default `0.33`.
42    #[serde(default = "TeamRocket::default_sparkle_rot")]
43    pub sparkle_rot: Factor,
44    /// Sparkle size - schema `team-rocket-sparkle_size`, default `0.75`.
45    #[serde(default = "TeamRocket::default_sparkle_size")]
46    pub sparkle_size: Factor,
47    /// Random xyz window warp - schema `team-rocket-window_rotation`, default `true`.
48    #[serde(default = "TeamRocket::default_window_rotation")]
49    pub window_rotation: bool,
50}

141const TAU: f64 = std::f64::consts::TAU;

143/// `fadeInOut` - `1 - ((t-0.5)/0.5)^power` clamped, a plateau easing.
144fn fade_in_out(t: f64, power: f64) -> f64 {
145    (1.0 - ((t - 0.5) / 0.5).powf(power)).clamp(0.0, 1.0)
146}
147
148/// `getSpark` - rotating superellipse star mask.
149fn spark(uv: Vec2<()>, center: Vec2<()>, brightness: f64, size: f64, rotation: f64) -> f64 {
150    let brightness = brightness.clamp(0.001, 1.0);
151    let size = size.clamp(0.001, 1.0);
152    let bn = crate::mix(0.0, 0.07, brightness);
153    let mut p = Vec2::<()>::new(uv.x + 0.5 - center.x, uv.y + 0.5 - center.y);
154    // scaleUV(uv, vec2(1.0 - size), vec2(0.0)): uv*2-1, / (1-size), *0.5+0.5.
155    // scaleUV divisor = mix(1, 0, 1-size) = size.
156    p = Vec2::<()>::new(((p.x * 2.0 - 1.0) / size) * 0.5 + 0.5, ((p.y * 2.0 - 1.0) / size) * 0.5 + 0.5);
157    // rotate around (0.5, 0.5).
158    let (sin, cos) = rotation.sin_cos();
159    p = Vec2::<()>::new(cos * (p.x - 0.5) + sin * (p.y - 0.5) + 0.5, cos * (p.y - 0.5) - sin * (p.x - 0.5) + 0.5);
160    let power = crate::mix(-1.0, 1000.0, EasingFunction::EaseInExpo.apply(Fraction::new(bn)).as_f64());
161    let m = (((p.x - 0.5).abs() * 2.0).powf(power) + ((p.y - 0.5).abs() * 2.0).powf(power)).clamp(0.0, 1.0);
162    (EasingFunction::EaseInSine.apply(Fraction::new(1.0 - (m - bn))).as_f64() - 0.004).clamp(0.0, 1.0)
163}

189    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
190        let mismatch = |index: usize| FxError::ParamMismatch(format!("team-rocket param {index}"));
191        if values.len() != 6 {
192            return Err(FxError::ParamMismatch(format!("team-rocket takes 6 parameters, got {}", values.len())));
193        }
194        self.animation_split = Factor::new(values[0].as_f64().ok_or(mismatch(0))?);
195        self.horizontal_sparkle_position = Factor::new(values[1].as_f64().ok_or(mismatch(1))?);
196        self.vertical_sparkle_position = Factor::new(values[2].as_f64().ok_or(mismatch(2))?);
197        self.sparkle_rot = Factor::new(values[3].as_f64().ok_or(mismatch(3))?);
198        self.sparkle_size = Factor::new(values[4].as_f64().ok_or(mismatch(4))?);
199        self.window_rotation = values[5].as_bool().ok_or(mismatch(5))?;
200        Ok(())
201    }

203    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
204        let size = ctx.window.size();
205        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
206        let aspect = size_px.x / size_px.y;
207        let opening = matches!(ctx.direction, crate::Direction::Open);
208        let progress = ctx.progress.as_f64();
209        let split = self.animation_split.0;
210        // Random xyz warp factors (zero when rotation is off).
211        // hash32(uSeed) scaled to [-0.1, 0.1] xyz warp factors.
212        let rot3 = Hash::<Vec3<()>>::hash(&ctx.seed.as_f64());
213        let rot = if self.window_rotation {
214            Vec3::<()>::new((rot3.x * 2.0 - 1.0) * 0.1, (rot3.y * 2.0 - 1.0) * 0.1, (rot3.z * 2.0 - 1.0) * 0.1)
215        } else {
216            Vec3::<()>::new(0.0, 0.0, 0.0)
217        };
218        let sparkle_xy = Vec2::<()>::new(self.horizontal_sparkle_position.0, self.vertical_sparkle_position.0);
219        let sparkle_rot = self.sparkle_rot.0;
220        let sparkle_size = self.sparkle_size.0;
221        let window = ctx.window;
222        crate::render::fill(out, |x, y| {
223            let uv0 = Vec2::<()>::new((x as f64 + 0.5) / size_px.x, (y as f64 + 0.5) / size_px.y);
224            let progress = if opening { 1.0 - progress } else { progress };
225            let scalep = EasingFunction::EaseOut
226                .apply(Fraction::new(crate::remap(progress, 0.0, split + 0.1, 0.0, 1.0)))
227                .as_f64();
228            let sparkp = EasingFunction::EaseInOutSine
229                .apply(Fraction::new(crate::remap(progress, split - 0.1, 1.0, 0.0, 1.0)))
230                .as_f64();
231            // Window warp: scale toward the sparkle point, rotate.
232            let scale_offset = Vec2::<()>::new(sparkle_xy.x * 0.5 * scalep, -sparkle_xy.y * 0.5 * scalep);
233            let mut uv2 = Vec2::<()>::new(uv0.x * 2.0 - 1.0 - scale_offset.x / aspect * scalep, uv0.y * 2.0 - 1.0 - scale_offset.y * scalep);
234            uv2 = Vec2::<()>::new(uv2.x / (1.0 - scalep), uv2.y / (1.0 - scalep));
235            uv2 = Vec2::<()>::new(uv2.x / crate::mix(1.0, rot.x * TAU * uv2.y, scalep), uv2.y / crate::mix(1.0, rot.y * TAU * uv2.x, scalep));
236            let (sin, cos) = (rot.z * TAU * scalep).sin_cos();
237            uv2 = Vec2::<()>::new(uv2.x * cos - uv2.y * sin, uv2.x * sin + uv2.y * cos);
238            uv2 = Vec2::<()>::new(uv2.x * 0.5 + 0.5, uv2.y * 0.5 + 0.5);
239            // The sparkle.
240            let uv = Vec2::<()>::new(uv0.x * aspect, uv0.y);
241            let offset = Vec2::<()>::new(sparkle_xy.x * 0.2, sparkle_xy.y * -0.2);
242            let fade = fade_in_out(sparkp, 2.0);
243            let mut sparkle = spark(
244                uv,
245                Vec2::<()>::new(offset.x + 0.5 * aspect, offset.y + 0.5),
246                fade * sparkle_size,
247                fade * crate::mix(25.0, 50.0, sparkle_size),
248                sparkp * TAU * sparkle_rot,
249            );
250            sparkle *= fade_in_out(uv0.y, 8.0) * fade_in_out(uv0.x, 8.0);
251            let c = window.sample_bilinear(uv2);
252            crate::alpha_over(c, Color::<f64>::new_rgba(1.0, 1.0, 1.0, sparkle))
253        });
254    }

270    fn solid(size: u32) -> PixelBuffer<Rgba> {
271        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(size, size));
272        let mut view = buffer.view_mut();
273        let c = Color::<f64>::new_rgba(0.5, 0.5, 0.5, 1.0);
274        for y in 0..size {
275            for x in 0..size {
276                view.put_color(Pixels::new(x), Pixels::new(y), c);
277            }
278        }
279        buffer
280    }

282    fn ctx_of<'a>(window: &'a PixelBuffer<Rgba>, progress: f64, direction: Direction) -> FxContext<'a> {
283        let size = window.size();
284        FxContext {
285            progress: Fraction::new(progress),
286            duration: Duration::from_secs_f64(2.0),
287            seed: Seed(0xABCD_EF01_2345_6789),
288            direction,
289            layer: Layer::Window,
290            window: window.view(),
291            surface: None,
292            assets: crate::FxAssets::default(),
293            pointer: None,
294            fullscreen: false,
295            padding: Pixels::new(0.0),
296            window_rect: Rect::from_xywh(0.0, 0.0, f64::from(size.x), f64::from(size.y)),
297        }
298    }

300    /// Open end untouched, close end warped away + sparkle faded.
301    #[test]
302    fn endpoints() {
303        let window = solid(32);
304        let mut out = PixelBuffer::<Rgba>::new(Size::new(32, 32));
305        let fx = TeamRocket::default();
306        fx.apply(&ctx_of(&window, 1.0, Direction::Open), &mut out.view_mut());
307        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 1.0);
308        fx.apply(&ctx_of(&window, 1.0, Direction::Close), &mut out.view_mut());
309        assert!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a < 0.5);
310    }

320        assert!(any, "sparkle should paint something");
