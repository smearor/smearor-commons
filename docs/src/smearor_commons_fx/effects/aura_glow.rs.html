23/// A glowing aura blooms around the window while it shrinks into a
24/// soft-edged shape - radial blur, cos-palette hue cycling and an
25/// aspect-corrected superellipse crop.
26///
27/// Behavioral spec (source shader: `aura-glow.frag`, re-implemented
28/// from semantics): for `Close` the progress is inverted (the glow
29/// grows while the window leaves). The content uv is de-padded by
30/// `uPadding` and blended toward a 1:1-aspect `oneToOneUV` as progress
31/// advances, so the final crop is circular regardless of window
32/// aspect. `gradient` is a superellipse distance (`|uv-0.5|^shape`,
33/// `shape` sweeping 2 -> 100 over `progress⁵`) plus a
34/// `Simplex2D` noise term keyed by the seed. `glowMask` rides the
35/// gradient edge; the window is radial-blurred while it zooms slightly;
36/// the glow color is a `cos(progress*speed + uv.xyx + (0,2,4))`
37/// palette hue-offset by `start_hue` (or a seed hash). The crop blends
38/// soft/hard `smoothstep` edges by `edge_hardness` and is guarded by
39/// in/out `easeInSine`/`easeOutSine` tails.
40#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
41#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
42pub struct AuraGlow {
43    /// Cos-palette cycles over progress - schema `aura-glow-speed`, default `2.5`.
44    #[serde(default = "AuraGlow::default_speed")]
45    pub speed: CycleSpeed,
46    /// Hash the start hue from the seed - schema `aura-glow-random-color`, default `false`.
47    #[serde(default)]
48    pub random_color: bool,
49    /// Start hue as a fraction of the hue circle - schema
50    /// `aura-glow-start-hue`, default `0.0`. BMW's `hueOffset` is a
51    /// `0..1` fraction, which is why this is `Fraction`, not `Hue`
52    /// (degrees).
53    #[serde(default = "AuraGlow::default_start_hue")]
54    pub start_hue: Fraction,
55    /// Glow saturation multiplier - schema `aura-glow-saturation`, default `0.75`.
56    #[serde(default = "AuraGlow::default_saturation")]
57    pub saturation: Saturation,
58    /// Glow width `0..=1` - schema `aura-glow-edge-size`, default `0.75`.
59    #[serde(default = "AuraGlow::default_edge_size")]
60    pub edge_size: Fraction,
61    /// Soft/hard crop blend `0..=1` - schema `aura-glow-edge-hardness`, default `0.25`.
62    #[serde(default = "AuraGlow::default_edge_hardness")]
63    pub edge_hardness: Fraction,
64    /// Radial blur radius in px at progress 0 - schema `aura-glow-blur`, default `15.0`.
65    #[serde(default = "AuraGlow::default_blur")]
66    pub blur: Pixels<f64>,
67}

172impl WindowEffect for AuraGlow {
173    fn nick(&self) -> &'static str {
174        "aura-glow"
175    }
176
177    fn label(&self) -> &'static str {
178        "Aura Glow"
179    }
180
181    fn params(&self) -> &'static [ParamSpec] {
182        PARAMS
183    }
184
185    fn values(&self) -> Vec<ParamValue> {
186        vec![
187            ParamValue::Float(self.speed.0),
188            ParamValue::Bool(self.random_color),
189            ParamValue::Float(self.start_hue.0),
190            ParamValue::Float(self.saturation.as_f64()),
191            ParamValue::Float(self.edge_size.0),
192            ParamValue::Float(self.edge_hardness.0),
193            ParamValue::Float(self.blur.0),
194        ]
195    }
196
197    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
198        let mismatch = |index: usize| FxError::ParamMismatch(format!("aura-glow param {index}"));
199        if values.len() != 7 {
200            return Err(FxError::ParamMismatch(format!("aura-glow takes 7 parameters, got {}", values.len())));
201        }
202        let float = |index: usize| values[index].as_f64().ok_or(mismatch(index));
203        self.speed = CycleSpeed(float(0)?);
204        self.random_color = values[1].as_bool().ok_or(mismatch(1))?;
205        self.start_hue = Fraction::new(float(2)?);
206        self.saturation = Saturation::new(float(3)?);
207        self.edge_size = Fraction::new(float(4)?);
208        self.edge_hardness = Fraction::new(float(5)?);
209        self.blur = Pixels::new(float(6)?);
210        Ok(())
211    }
212
213    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
214        let size = ctx.window.size();
215        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
216        let progress = match ctx.direction {
217            crate::Direction::Open => ctx.progress.as_f64(),
218            crate::Direction::Close => 1.0 - ctx.progress.as_f64(),
219        };
220        // Frame constants - everything below depends only on
221        // progress, size, seed or parameters, so it lifts out of the
222        // pixel loop.
223        let seed_uv = ctx.seed.offset_2d();
224        let content_fraction = Vec2::<()>::new(1.0, 1.0) - Vec2::<()>::new(2.0 * ctx.padding.0, 2.0 * ctx.padding.0) / size_px;
225        let inv_content = Vec2::<()>::new(1.0 / content_fraction.x, 1.0 / content_fraction.y);
226        let aspect = Vec2::<()>::new(
227            if size_px.x > size_px.y { 1.0 } else { size_px.x / size_px.y },
228            if size_px.x > size_px.y { size_px.y / size_px.x } else { 1.0 },
229        );
230        let shape = crate::mix(2.0, 100.0, progress.powi(5));
231        let out_fade = EasingFunction::EaseOutSine.apply(Fraction::new(((1.0 - progress) * 4.0).min(1.0))).as_f64();
232        let in_fade = EasingFunction::EaseInSine.apply(Fraction::new((progress * 2.0).min(1.0))).as_f64();
233        let zoom = crate::mix(1.1, 1.0, EasingFunction::EaseOutCubic.apply(Fraction::new(progress)).as_f64());
234        let color_offset = if self.random_color {
235            Hash::<f64>::hash(&seed_uv) + 0.1
236        } else {
237            self.start_hue.0 + 0.1
238        };
239        let phase = progress * self.speed.0;
240        let edge_width = self.edge_size.0 + 0.1;
241        let saturation = self.saturation.as_f64();
242        // The cone rays are frame-constant - precompute the offset
243        // table once instead of trig per pixel per tap. Below half a
244        // pixel of radius the blur degenerates to the plain tap.
245        let blur_px = (1.0 - progress) * self.blur.0;
246        let cone = match blur_px > 0.5 {
247            true => BufferView::<Rgba>::cone_offsets(Vec2::<()>::new(blur_px / size_px.x, blur_px / size_px.y), 15, 3),
248            false => Vec::new(),
249        };
250        crate::render::fill(out, |x, y| {
251            let i_tex = Vec2::<()>::new((x as f64 + 0.5) / size_px.x, (y as f64 + 0.5) / size_px.y);
252            // De-padded content uv: (i_tex-0.5)*size/content/size+0.5.
253            let mut uv = (i_tex - Vec2::<()>::new(0.5, 0.5)).component_mul(&inv_content) + Vec2::<()>::new(0.5, 0.5);
254
255            // 1:1 aspect blend - the crop ends circular.
256            let one_to_one = (uv - Vec2::<()>::new(0.5, 0.5)).component_mul(&aspect) + Vec2::<()>::new(0.5, 0.5);
257            uv = one_to_one * (1.0 - progress) + uv * progress;
258
259            // Superellipse gradient + noise wobble.
260            let mut gradient = ((uv.x - 0.5).abs() * 2.0).powf(shape) + ((uv.y - 0.5).abs() * 2.0).powf(shape);
261            gradient += Simplex2D.at(i_tex + seed_uv) * 0.5;
262
263            let mut glow_mask = 1.0 - ((progress - gradient) / edge_width).clamp(0.0, 1.0);
264            glow_mask *= out_fade;
265
266            // Zoom + progressive blur over the precomputed cone.
267            let window_uv = (i_tex - Vec2::<()>::new(0.5, 0.5)) * zoom + Vec2::<()>::new(0.5, 0.5);
268            let mut window_color = ctx.window.sample_cone(window_uv, &cone);
269            glow_mask *= window_color.a;
270
271            // Cos palette + hue offset, saturation-scaled.
272            let mut glow_color = Color::<f64>::new_rgba((phase + uv.x).cos(), (phase + uv.y + 2.0).cos(), (phase + uv.x + 4.0).cos(), 1.0);
273            glow_color = crate::offset_hue(glow_color, color_offset);
274            let glow = Color::<f64>::new_rgba(
275                (glow_color.r * saturation).clamp(0.0, 1.0),
276                (glow_color.g * saturation).clamp(0.0, 1.0),
277                (glow_color.b * saturation).clamp(0.0, 1.0),
278                1.0,
279            );
280
281            window_color = Color::<f64>::new_rgba(
282                window_color.r + glow.r * glow_mask,
283                window_color.g + glow.g * glow_mask,
284                window_color.b + glow.b * glow_mask,
285                window_color.a,
286            );
287            window_color = Color::<f64>::new_rgba(glow.r, glow.g, glow.b, glow_mask * 0.2).alpha_over(&window_color);
288
289            // Superellipse crop, soft/hard blended.
290            let soft_crop = 1.0 - crate::smoothstep(progress - 0.5, progress + 0.5, gradient);
291            let hard_crop = 1.0 - crate::smoothstep(progress - 0.05, progress + 0.05, gradient);
292            let mut crop_mask = crate::mix(soft_crop, hard_crop, self.edge_hardness.0);
293            crop_mask *= in_fade;
294            crop_mask = crop_mask.max(1.0 - out_fade);
295            window_color.a *= crop_mask;
296            window_color
297        });
298    }

313    fn ctx_for<'a>(window: BufferView<'a, Rgba>, progress: f64) -> FxContext<'a> {
314        FxContext {
315            progress: Fraction::new(progress),
316            duration: Duration::from_millis(1500),
317            seed: Seed(1),
318            direction: Direction::Open,
319            layer: Layer::Window,
320            window,
321            surface: None,
322            assets: crate::FxAssets::default(),
323            pointer: None,
324            fullscreen: false,
325            padding: Pixels::new(0.0),
326            window_rect: Rect::from_xywh(0.0, 0.0, 32.0, 32.0),
327        }
328    }

330    /// `edge_hardness = 1` yields a hard crop: interior alpha is either
331    /// fully kept or fully cut (no half-transparent crop zone).
332    #[test]
333    fn aura_glow_crop() {
334        let data = [200, 200, 200, 255].repeat(32 * 32);
335        let window = BufferView::<Rgba>::from_packed(&data, Size::new(32, 32)).unwrap();
336        let mut out = PixelBuffer::<Rgba>::new(Size::new(32, 32));
337        let glow = AuraGlow {
338            edge_hardness: Fraction::new(1.0),
339            ..Default::default()
340        };
341        glow.apply(&ctx_for(window, 0.6), &mut out.view_mut());
342        // At 0.6 progress the crop is hard: alpha is ~0 or ~255 modulo
343        // the glow overlay's small additive alpha.
344        let alphas: Vec<u8> = out.view().data().iter().skip(3).step_by(4).copied().collect();
345        let extreme = alphas.iter().filter(|a| **a < 40 || **a > 215).count();
346        assert!(extreme as f64 / alphas.len() as f64 > 0.6, "hard crop: most alphas near 0/255");
347    }

349    /// `blur = 0` bypasses radial blur: on a checkerboard input the
350    /// sharp sample differs from the blurred one.
351    #[test]
352    fn aura_glow_blur_radius() {
353        let mut data = Vec::with_capacity(32 * 32 * 4);
354        for y in 0..32 {
355            for x in 0..32 {
356                let v = if (x / 4 + y / 4) % 2 == 0 { 255 } else { 0 };
357                data.extend_from_slice(&[v, v, v, 255]);
358            }
359        }
360        let window = BufferView::<Rgba>::from_packed(&data, Size::new(32, 32)).unwrap();
361        let mut out_sharp = PixelBuffer::<Rgba>::new(Size::new(32, 32));
362        let mut out_blur = PixelBuffer::<Rgba>::new(Size::new(32, 32));
363        AuraGlow {
364            blur: Pixels::new(0.0),
365            ..Default::default()
366        }
367        .apply(&ctx_for(window, 0.5), &mut out_sharp.view_mut());
368        AuraGlow {
369            blur: Pixels::new(30.0),
370            ..Default::default()
371        }
372        .apply(&ctx_for(window, 0.5), &mut out_blur.view_mut());
373        assert_ne!(out_sharp.view().data(), out_blur.view().data(), "blur=0 must differ from blur=30 on sharp input");
374    }
