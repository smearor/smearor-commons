1//! Focus - a heavy radial blur that resolves into focus while the
2//! window fades in (Burn-My-Windows' `focus` effect).
3//!
4//! The blur follows BMW's `getBlurredInputColor` from `common.glsl`:
5//! `BLUR_DIRECTIONS` rays around the pixel, `blur_quality` taps along
6//! each ray at distances `{1/n, 2/n, ..., n/n} * blur_amount` -
7//! [`BufferView::cone_offsets`] + [`BufferView::sample_cone`]
8//! implement exactly this table (the GLSL `1 - s` ramp enumerates the
9//! same distances, so the mean is identical). The offsets are
10//! computed once per frame and shared by all pixels.
11//!
12//! Timing (`focus.frag`):
13//! * `progl = uProgress` when opening, `1 - uProgress` when closing
14//! * `blurAmount = mix(uBlurAmount, 0, easeInOutSine(progl))` px
15//! * `alpha = easeInOutCubic(progl)`, multiplied into the blurred
16//!   straight-alpha color afterwards
17//!
18//! At `progl = 1` the blur amount collapses to zero and the cone
19//! degenerates to a plain bilinear tap - exact rest behavior.
20//!
21//! Params (`focus.ui` ranges): `blur_amount` 0..=100 px (default 50),
22//! `blur_quality` 1..=10 taps per ray (default 3).
23

42/// Rays of the BMW blur cone (`directions = 15.0` in
43/// `getBlurredInputColor`).
44const BLUR_DIRECTIONS: u32 = 15;
45
46/// Focus - blur + fade, the simplest BMW transition.
47#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
48#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
49#[serde(default)]
50pub struct Focus {
51    /// `uBlurAmount` - the blur radius in pixels at animation start
52    /// (`focus.ui`: 0..=100, default 50). Serializes as the bare
53    /// pixel count (`"blur_amount": 50`).
54    pub blur_amount: Pixels<u32>,
55    /// `uBlurQuality` - taps per ray of the blur cone (`focus.ui`:
56    /// 1..=10, default 3). Higher values smooth the blur at linear
57    /// sampling cost.
58    pub blur_quality: BlurQuality,
59}

92impl WindowEffect for Focus {
93    fn nick(&self) -> &'static str {
94        "focus"
95    }
96
97    fn label(&self) -> &'static str {
98        "Focus"
99    }
100
101    fn layers(&self) -> Layers {
102        Layers::WindowOnly
103    }
104
105    fn params(&self) -> &'static [ParamSpec] {
106        PARAMS
107    }
108
109    fn values(&self) -> Vec<ParamValue> {
110        vec![
111            ParamValue::Int(i64::from(self.blur_amount.0)),
112            ParamValue::Int(i64::from(self.blur_quality.as_u32())),
113        ]
114    }
115
116    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), crate::FxError> {
117        let mismatch = |index: usize| crate::FxError::ParamMismatch(format!("focus param {index}"));
118        if values.len() != 2 {
119            return Err(crate::FxError::ParamMismatch(format!("focus takes 2 parameters, got {}", values.len())));
120        }
121        self.blur_amount = Pixels::new(u32::try_from(values[0].as_i64().ok_or(mismatch(0))?.clamp(0, 100)).map_err(|_| mismatch(0))?);
122        self.blur_quality = BlurQuality::new(u32::try_from(values[1].as_i64().ok_or(mismatch(1))?.clamp(1, 10)).map_err(|_| mismatch(1))?);
123        Ok(())
124    }
125
126    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
127        let size = out.size();
128        let w = f64::from(size.x).max(1.0);
129        let h = f64::from(size.y).max(1.0);
130        let opening = ctx.direction == Direction::Open;
131        let progl = if opening { ctx.progress.as_f64() } else { 1.0 - ctx.progress.as_f64() };
132
133        let blur_px = crate::mix(f64::from(self.blur_amount.0), 0.0, EasingFunction::EaseInOutSine.apply(Fraction::new(progl)).as_f64());
134        let alpha = EasingFunction::EaseInOutCubic.apply(Fraction::new(progl)).as_f64();
135
136        // The cone table once for the whole frame; `sample_cone`
137        // degenerates to a plain bilinear tap when the radius is zero.
138        let radius = Vec2::<()>::new(blur_px / w, blur_px / h);
139        let offsets = BufferView::<'_, Rgba>::cone_offsets(radius, BLUR_DIRECTIONS, self.blur_quality.as_u32().max(1));
140
141        let inv = Vec2::<()>::new(1.0 / w, 1.0 / h);
142        crate::render::fill(out, |x, y| {
143            let uv = Vec2::<()>::new((f64::from(x) + 0.5) * inv.x, (f64::from(y) + 0.5) * inv.y);
144            let mut c = ctx.window.sample_cone(uv, &offsets);
145            c.a *= alpha;
146            c
147        });
148    }
149}

162    fn solid(size: u32) -> PixelBuffer<Rgba> {
163        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(size, size));
164        let mut view = buffer.view_mut();
165        for y in 0..size {
166            for x in 0..size {
167                view.put_color(Pixels::new(x), Pixels::new(y), Color::<f64>::new_rgba(0.5, 0.5, 0.5, 1.0));
168            }
169        }
170        buffer
171    }
172
173    fn ctx_of<'a>(window: &'a PixelBuffer<Rgba>, progress: f64, direction: Direction) -> FxContext<'a> {
174        let size = window.size();
175        FxContext {
176            progress: Fraction::new(progress),
177            duration: Duration::from_secs_f64(2.0),
178            seed: Seed(0xABCD_EF01_2345_6789),
179            direction,
180            layer: Layer::Window,
181            window: window.view(),
182            surface: None,
183            assets: crate::FxAssets::default(),
184            pointer: None,
185            fullscreen: false,
186            padding: Pixels::new(0.0),
187            window_rect: smearor_commons_geometry::Rect::from_xywh(4.0, 4.0, f64::from(size.x) - 8.0, f64::from(size.y) - 8.0),
188        }
189    }
190
191    /// Open end = the window at rest, fully opaque and sharp.
192    #[test]
193    fn open_end_is_the_window() {
194        let window = solid(48);
195        let mut out = PixelBuffer::<Rgba>::new(Size::new(48, 48));
196        Focus::default().apply(&ctx_of(&window, 1.0, Direction::Open), &mut out.view_mut());
197        let c = out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5));
198        assert_eq!(c.a, 1.0);
199        // 8-bit quantization: 0.5 -> 128/255
200        assert!((c.r - 0.50196).abs() < 0.01);
201    }
202
203    /// Close end = fully transparent.
204    #[test]
205    fn close_end_is_empty() {
206        let window = solid(48);
207        let mut out = PixelBuffer::<Rgba>::new(Size::new(48, 48));
208        Focus::default().apply(&ctx_of(&window, 1.0, Direction::Close), &mut out.view_mut());
209        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 0.0);
210    }
211
212    /// Mid animation the cone actually blurs: a hard edge's straight
213    /// alpha is smeared into the transparent side.
214    #[test]
215    fn mid_animation_blurs_edges() {
216        let mut window = PixelBuffer::<Rgba>::new(Size::new(96, 96));
217        {
218            let mut view = window.view_mut();
219            for y in 0..96 {
220                for x in 0..96 {
221                    let c = if x < 48 {
222                        Color::<f64>::new_rgba(1.0, 1.0, 1.0, 1.0)
223                    } else {
224                        Color::<f64>::new_rgba(1.0, 1.0, 1.0, 0.0)
225                    };
226                    view.put_color(Pixels::new(x), Pixels::new(y), c);
227                }
228            }
229        }
230        let mut out = PixelBuffer::<Rgba>::new(Size::new(96, 96));
231        Focus::default().apply(&ctx_of(&window, 0.5, Direction::Open), &mut out.view_mut());
232        let view = out.view();
233        // the fade halves every alpha at progress 0.5
234        // (easeInOutCubic(0.5) = 0.5)
235        let deep = view.sample_bilinear(Vec2::<()>::new(0.0625, 0.5)).a;
236        assert!(deep > 0.45 && deep < 0.55, "deep alpha {deep}");
237        // blur radius 50px * (1-sine_eased(0.5)=0.5) = 25px smears
238        // alpha into the transparent half (14px in): less than `deep`
239        // but not fully dissolved
240        let smeared = view.sample_bilinear(Vec2::<()>::new(0.65, 0.5)).a;
241        assert!(smeared > 0.0 && smeared < deep, "smeared alpha {smeared} vs deep {deep}");
242    }
