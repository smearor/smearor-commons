1use smearor_commons_geometry::Pixels;
2use smearor_commons_geometry::Vec2;
3use smearor_commons_noise::Field;
4use smearor_commons_noise::Simplex2D;
5
6const FLOW_PX: Pixels<f64> = Pixels::new(30.0);
7const FLOW_SCALE: f64 = 160.0;
8const RADIAL_PX: Pixels<f64> = Pixels::new(30.0);
9const SPILL_PX: Pixels<f64> = Pixels::new(5.0);
10const CORNER_PX: Pixels<f64> = Pixels::new(24.0);
11
12/// `streamFn` - two-octave curl source.
13fn stream_fn(p: Vec2<()>, seed: Vec2<()>) -> f64 {
14    Simplex2D.at(p + seed * 100.0) + 0.5 * Simplex2D.at(p * 2.03 + Vec2::<()>::new(17.1, 5.3) + seed * 100.0)
15}
16
17pub(super) struct MorphFrame {
18    /// `uFrameRect` in layer pixels.
19    pub(super) frame: (Vec2<()>, Vec2<()>),
20    /// Texture size in pixels.
21    pub(super) size_px: Vec2<()>,
22    /// Normalized `[0,1]` seed (`uSeed`).
23    pub(super) seed: Vec2<()>,
24    /// `uStrength`.
25    pub(super) strength: f64,
26}
27
28impl MorphFrame {
29    /// `frameDist` - signed distance to the rounded frame rect.
30    fn frame_dist(&self, px: Vec2<()>) -> f64 {
31        let (o, s) = self.frame;
32        let c = o + s * 0.5;
33        let h = Vec2::<()>::new((0.5 * s.x - CORNER_PX.0).max(0.0), (0.5 * s.y - CORNER_PX.0).max(0.0));
34        let q = Vec2::<()>::new((px.x - c.x).abs() - h.x, (px.y - c.y).abs() - h.y);
35        let mx = q.x.max(0.0);
36        let my = q.y.max(0.0);
37        (mx * mx + my * my).sqrt() + q.x.max(q.y).min(0.0) - CORNER_PX.0
38    }
39
40    /// `regionMask` - 1 inside, fading to 0 `SPILL_PX.0` outside.
41    pub(super) fn region_mask(&self, px: Vec2<()>) -> f64 {
42        1.0 - crate::smoothstep(0.0, SPILL_PX.0, self.frame_dist(px))
43    }
44
45    /// `flowAt` - the full displacement field in pixels.
46    pub(super) fn flow_at(&self, px: Vec2<()>) -> Vec2<()> {
47        let m = self.region_mask(px);
48        if m <= 0.0 {
49            return Vec2::<()>::new(0.0, 0.0);
50        }
51        let p = Vec2::<()>::new(px.x / FLOW_SCALE, px.y / FLOW_SCALE);
52        const E: f64 = 0.01;
53        let dx = (stream_fn(p + Vec2::<()>::new(E, 0.0), self.seed) - stream_fn(p - Vec2::<()>::new(E, 0.0), self.seed)) / (2.0 * E);
54        let dy = (stream_fn(p + Vec2::<()>::new(0.0, E), self.seed) - stream_fn(p - Vec2::<()>::new(0.0, E), self.seed)) / (2.0 * E);
55        let mut c = Vec2::<()>::new(dy, -dx);
56        let cl = (c.x * c.x + c.y * c.y).sqrt();
57        c = c / (2.3 + 0.3 * cl);
58
59        let (o, s) = self.frame;
60        let centre = o + s * 0.5;
61        let rel = Vec2::<()>::new((px.x - centre.x) / (0.5 * s.x).max(1.0), (px.y - centre.y) / (0.5 * s.y).max(1.0));
62        let r = (rel.x * rel.x + rel.y * rel.y).sqrt();
63        let radial = if r > 1e-4 {
64            rel / r * (std::f64::consts::PI * r.min(1.0)).sin()
65        } else {
66            Vec2::<()>::new(0.0, 0.0)
67        };
68
69        Vec2::<()>::new(
70            m * self.strength * (c.x * FLOW_PX.0 + radial.x * RADIAL_PX.0),
71            m * self.strength * (c.y * FLOW_PX.0 + radial.y * RADIAL_PX.0),
72        )
73    }
74}
