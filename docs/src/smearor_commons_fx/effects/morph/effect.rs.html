25/// The desktop flows forward along a curl-noise field while the
26/// window arrives from the opposite direction and condenses out of
27/// it in noise-driven blotches - or, per `morph-style`, the layers
28/// melt along their own luminance edges / coarsen into shared pixel
29/// blocks that flip colour.
30///
31/// Behavioral spec (source shader: `morph.frag`, re-implemented from
32/// semantics): `flowAt` builds a curl-noise displacement field
33/// (`streamFn` = two simplex octaves, finite differences, soft-limited
34/// to ~unit length) plus a radial push away from the frame centre,
35/// masked to `CORNER_PX`-rounded `uFrameRect` with `SPILL_PX` feather.
36/// Timeline: the surface's displacement grows over `T_SURFACE`, the
37/// window's shrinks over `T_WINDOW`, the blotch dissolve sweeps
38/// `T_FADE` (`DISSOLVE_SCALE` noise, `DISSOLVE_EDGE` softness); both
39/// layers blur by `BLUR_PX * strength` while displaced (13-tap disc,
40/// premultiplied). Style `melt`: `meltDir` derives direction from the
41/// layer's own luminance gradient (`MELT_RADIUS` ring, `MELT_ALONG` /
42/// `MELT_ACROSS` mix, `MELT_STEPS` advection), strong window edges
43/// arriving `EDGE_FIRST` earlier. Style `pixelate`: both layers snap
44/// to blocks of the same grid anchored at the frame centre
45/// (`PIXEL_TAPS`²-tap block means); window blocks flip colour at
46/// noise+random staggered times (`PIXEL_RANDOM`, `PIXEL_SPREAD`,
47/// `PIXEL_DUR`) inside `P_FADE`, between the `P_COARSEN`/`P_SHARPEN`
48/// phases. `uSeed` is the normalized `[0,1]` seed hash.
49///
50/// Two-pass effect: `Layer::Surface` morphs the backdrop,
51/// `Layer::Window` the window - the consumer composites window over
52/// surface.
53#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
54#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
55pub struct Morph {
56    /// Morph style: `0` = liquid flow, `1` = edge-guided melt,
57    /// `2` = pixelate - schema `morph-style`, default `0`.
58    #[serde(default = "Morph::default_style")]
59    pub style: MorphStyle,
60    /// Displacement multiplier - schema `morph-strength`, default `1.0`.
61    #[serde(default = "Morph::default_strength")]
62    pub strength: Factor,
63}

83const BLUR_PX: Pixels<f64> = Pixels::new(6.0);
84const DISSOLVE_SCALE: f64 = 380.0;
85const DISSOLVE_EDGE: f64 = 0.22;
86const MELT_PX: Pixels<f64> = Pixels::new(40.0);
87const MELT_RADIUS: Pixels<f64> = Pixels::new(8.0);
88const MELT_ALONG: f64 = 0.7;
89const MELT_ACROSS: f64 = 0.5;
90const MELT_NOISE: f64 = 0.35;
91const MELT_STEPS: usize = 3;
92const EDGE_FIRST: f64 = 0.35;
93const PIXEL_PX: Pixels<f64> = Pixels::new(24.0);
94const PIXEL_TAPS: usize = 3;
95const PIXEL_RANDOM: f64 = 0.75;
96const PIXEL_SPREAD: f64 = 1.3;
97const PIXEL_DUR: f64 = 0.5;
98const PIXEL_RESIZE: f64 = 0.5;
99const P_COARSEN: Phase = Phase::new(0.0, PIXEL_RESIZE);
100const P_FADE: Phase = Phase::new(0.0, 1.0);
101const P_SHARPEN: Phase = Phase::new(1.0 - PIXEL_RESIZE, 1.0);
102const T_SURFACE: Phase = Phase::new(0.00, 0.60);
103const T_WINDOW: Phase = Phase::new(0.40, 1.00);
104const T_FADE: Phase = Phase::new(0.20, 0.85);

106/// The layer's texture read - surface clamps, the window reads
107/// transparent outside (`readLayer` in the shader).
108fn read_layer(view: BufferView<'_, Rgba>, uv: Vec2<()>, surface: bool) -> Color<f64> {
109    if !surface && (!(0.0..=1.0).contains(&uv.x) || !(0.0..=1.0).contains(&uv.y)) {
110        return Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0);
111    }
112    view.sample_bilinear(uv)
113}

115/// `blurredRead` - 13-tap disc blur, premultiplied then straightened.
116fn blurred_read(view: BufferView<'_, Rgba>, uv: Vec2<()>, radius: f64, texel: Vec2<()>, surface: bool) -> Color<f64> {
117    if radius < 0.5 {
118        return read_layer(view, uv, surface);
119    }
120    let mut acc = [0.0; 4];
121    for i in 0..13usize {
122        let off = if i == 0 {
123            Vec2::<()>::new(0.0, 0.0)
124        } else {
125            let k = (i - 1) as f64;
126            let ang = k * std::f64::consts::FRAC_PI_6 + if k < 6.0 { 0.0 } else { std::f64::consts::FRAC_PI_6 * 0.5 };
127            let rad = if k < 6.0 { 0.5 * radius } else { radius };
128            Vec2::<()>::new(ang.cos() * rad * texel.x, ang.sin() * rad * texel.y)
129        };
130        let s = read_layer(view, uv + off, surface);
131        acc[0] += s.r * s.a;
132        acc[1] += s.g * s.a;
133        acc[2] += s.b * s.a;
134        acc[3] += s.a;
135    }
136    let a = acc[3] / 13.0;
137    if a <= 0.0 {
138        return Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0);
139    }
140    Color::<f64>::new_rgba(acc[0] / 13.0 / a, acc[1] / 13.0 / a, acc[2] / 13.0 / a, a)
141}

143/// `luma` - transparent counts as mid-grey, so the window's outline
144/// is an edge too. The alpha-over-midgrey composite is a sampling
145/// policy of this edge detector, not a property of `Color`.
146fn luma(c: Color<f64>) -> f64 {
147    c.a * c.luminance_601().as_f64() + (1.0 - c.a) * 0.5
148}

150/// `meltDir` - direction from the layer's own luminance gradient.
151/// Returns (direction, edge strength).
152fn melt_dir(view: BufferView<'_, Rgba>, uv: Vec2<()>, texel: Vec2<()>, surface: bool) -> (Vec2<()>, f64) {
153    let mut g = Vec2::<()>::new(0.0, 0.0);
154    for k in 0..8 {
155        let ang = k as f64 * std::f64::consts::FRAC_PI_4;
156        let dir = Vec2::<()>::new(ang.cos(), ang.sin());
157        let l = luma(read_layer(
158            view,
159            Vec2::<()>::new(uv.x + dir.x * MELT_RADIUS.0 * texel.x, uv.y + dir.y * MELT_RADIUS.0 * texel.y),
160            surface,
161        ));
162        g += dir * l;
163    }
164    g = g * 0.25;
165    let len = (g.x * g.x + g.y * g.y).sqrt();
166    if len < 1e-4 {
167        return (Vec2::<()>::new(0.0, 0.0), 0.0);
168    }
169    let strength = crate::smoothstep(0.02, 0.25, len);
170    let n = Vec2::<()>::new(g.x / len, g.y / len);
171    (
172        Vec2::<()>::new(strength * (MELT_ALONG * -n.y + MELT_ACROSS * n.x), strength * (MELT_ALONG * n.x + MELT_ACROSS * n.y)),
173        strength,
174    )
175}

197        "morph"
198    }

200    fn label(&self) -> &'static str {
201        "Morph"
202    }

204    fn layers(&self) -> Layers {
205        Layers::SurfaceAndWindow
206    }

208    fn params(&self) -> &'static [ParamSpec] {
209        PARAMS
210    }

212    fn values(&self) -> Vec<ParamValue> {
213        vec![ParamValue::Int(i64::from(u32::from(self.style))), ParamValue::Float(self.strength.0)]
214    }
215
216    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
217        let mismatch = |index: usize| FxError::ParamMismatch(format!("morph param {index}"));
218        if values.len() != 2 {
219            return Err(FxError::ParamMismatch(format!("morph takes 2 parameters, got {}", values.len())));
220        }
221        self.style = MorphStyle::try_from(values[0].as_i64().ok_or(mismatch(0))?.clamp(0, 2) as u32).map_err(|_| mismatch(0))?;
222        self.strength = Factor::new(values[1].as_f64().ok_or(mismatch(1))?);
223        Ok(())
224    }

226    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
227        let size = out.size();
228        let size_px = Vec2::<()>::new(f64::from(size.x), f64::from(size.y));
229        let opening = ctx.direction == crate::Direction::Open;
230        let progress = if opening { ctx.progress.as_f64() } else { 1.0 - ctx.progress.as_f64() };
231        let surface_pass = ctx.layer == Layer::Surface;
232        let source = if surface_pass { ctx.surface } else { Some(ctx.window) };
233        let Some(view) = source else {
234            // No surface texture - the consumer's own backdrop shows
235            // through; the pass renders transparent.
236            crate::render::fill(out, |_, _| Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0));
237            return;
238        };
239        let frame = ctx.window_rect;
240        let mf = MorphFrame {
241            frame: (Vec2::<()>::new(frame.min.x, frame.min.y), Vec2::<()>::new(frame.size.x, frame.size.y)),
242            size_px,
243            seed: Hash::<Vec2<()>>::hash(&ctx.seed.as_f64()),
244            strength: self.strength.0,
245        };
246        let texel = Vec2::<()>::new(1.0 / size_px.x, 1.0 / size_px.y);
247
248        let blur = BLUR_PX.0 * self.strength.0.min(1.0);
249        let melt = self.style == MorphStyle::EdgeMelt;
250
251        if self.style == MorphStyle::Pixelate {
252            self.apply_pixelate(ctx, out, view, &mf, progress, surface_pass);
253            return;
254        }
255
256        crate::render::fill(out, |x, y| {
257            let px = Vec2::<()>::new(x as f64 + 0.5, y as f64 + 0.5);
258            let uv0 = Vec2::<()>::new(px.x / size_px.x, px.y / size_px.y);
259            let flow = mf.flow_at(px);
260
261            if surface_pass {
262                let s = T_SURFACE.at(progress);
263                if s <= 0.0 || (flow.x == 0.0 && flow.y == 0.0) {
264                    return view.sample_bilinear(uv0);
265                }
266                let mut tc = uv0;
267                if melt {
268                    for _ in 0..MELT_STEPS {
269                        let (d, _) = melt_dir(view, tc, texel, true);
270                        let step = mf.region_mask(px) * MELT_PX.0 * mf.strength;
271                        tc += Vec2::<()>::new(
272                            (d.x * step + flow.x * MELT_NOISE) * -s / MELT_STEPS as f64 * texel.x,
273                            (d.y * step + flow.y * MELT_NOISE) * -s / MELT_STEPS as f64 * texel.y,
274                        );
275                    }
276                } else {
277                    tc -= Vec2::<()>::new(flow.x * s * texel.x, flow.y * s * texel.y);
278                }
279                let mut c = blurred_read(view, tc, blur * s * mf.region_mask(px), texel, true);
280                c.a = 1.0;
281                c
282            } else {
283                let w = 1.0 - T_WINDOW.at(progress);
284                let f = T_FADE.at(progress);
285                if w <= 0.0 && f >= 1.0 {
286                    return read_layer(view, uv0, false);
287                }
288                let mut tc = uv0;
289                let mut edge = 0.0;
290                if melt {
291                    for _ in 0..MELT_STEPS {
292                        let (d, e) = melt_dir(view, tc, texel, false);
293                        edge = e;
294                        let step = mf.region_mask(px) * MELT_PX.0 * mf.strength;
295                        tc += Vec2::<()>::new(
296                            (d.x * step + flow.x * MELT_NOISE) * w / MELT_STEPS as f64 * texel.x,
297                            (d.y * step + flow.y * MELT_NOISE) * w / MELT_STEPS as f64 * texel.y,
298                        );
299                    }
300                } else {
301                    tc += Vec2::<()>::new(flow.x * w * texel.x, flow.y * w * texel.y);
302                }
303                let mut c = blurred_read(view, tc, blur * w * mf.region_mask(px), texel, false);
304                let mut n = Simplex2D.at(Vec2::<()>::new(
305                    px.x / DISSOLVE_SCALE + mf.seed.x * 37.0 + 3.7,
306                    px.y / DISSOLVE_SCALE + mf.seed.y * 37.0 + 3.7,
307                ));
308                n = ((n - 0.5) * 1.6 + 0.5).clamp(0.0, 1.0);
309                let early = if melt { EDGE_FIRST } else { 0.0 };
310                n -= early * edge;
311                let a = crate::smoothstep(n - DISSOLVE_EDGE, n + DISSOLVE_EDGE, crate::mix(-DISSOLVE_EDGE - early, 1.0 + DISSOLVE_EDGE, f));
312                c.a *= a;
313                c
314            }
315        });
316    }

319impl Morph {
320    /// The `pixelate` style - shared block grid, staggered colour
321    /// flips between `P_COARSEN` and `P_SHARPEN`.
322    fn apply_pixelate(
323        &self,
324        _ctx: &FxContext<'_>,
325        out: &mut BufferViewMut<'_, Rgba>,
326        view: BufferView<'_, Rgba>,
327        mf: &MorphFrame,
328        progress: f64,
329        surface_pass: bool,
330    ) {
331        let size_px = mf.size_px;
332        let texel = Vec2::<()>::new(1.0 / size_px.x, 1.0 / size_px.y);
333
334        let coarse = P_COARSEN.at(progress) * (1.0 - P_SHARPEN.at(progress));
335        let bs = crate::mix(1.0, (PIXEL_PX.0 * mf.strength).max(1.0), coarse).floor().max(1.0);
336        let (fo, fs) = mf.frame;
337        let centre = fo + fs * 0.5;
338        let peak = (PIXEL_PX.0 * mf.strength + 0.5).floor().max(1.0);
339        let f = P_FADE.at(progress);
340        let seed_px = Vec2::<()>::new((mf.seed.x * 1000.0).floor(), (mf.seed.y * 1000.0).floor());
341
342        crate::render::fill(out, |x, y| {
343            let px = Vec2::<()>::new(x as f64 + 0.5, y as f64 + 0.5);
344            let uv0 = Vec2::<()>::new(px.x / size_px.x, px.y / size_px.y);
345            let cell = Vec2::<()>::new(((px.x - centre.x) / bs).floor(), ((px.y - centre.y) / bs).floor());
346            let lo = centre + cell * bs;
347            let block_c = lo + Vec2::<()>::new(0.5 * bs, 0.5 * bs);
348            let in_region = lo.x < fo.x + fs.x && lo.y < fo.y + fs.y && lo.x + bs > fo.x && lo.y + bs > fo.y;
349
350            let mut c = if in_region && bs > 1.0 {
351                // `blockColor` - PIXEL_TAPS²-tap premultiplied mean.
352                let mut acc = [0.0; 4];
353                for j in 0..PIXEL_TAPS {
354                    for i in 0..PIXEL_TAPS {
355                        let p = Vec2::<()>::new(
356                            block_c.x + ((i as f64 + 0.5) / PIXEL_TAPS as f64 - 0.5) * bs,
357                            block_c.y + ((j as f64 + 0.5) / PIXEL_TAPS as f64 - 0.5) * bs,
358                        );
359                        let t = read_layer(view, Vec2::<()>::new(p.x * texel.x, p.y * texel.y), surface_pass);
360                        acc[0] += t.r * t.a;
361                        acc[1] += t.g * t.a;
362                        acc[2] += t.b * t.a;
363                        acc[3] += t.a;
364                    }
365                }
366                let n = (PIXEL_TAPS * PIXEL_TAPS) as f64;
367                let a = acc[3] / n;
368                if a > 0.0 {
369                    Color::<f64>::new_rgba(acc[0] / n / a, acc[1] / n / a, acc[2] / n / a, a)
370                } else {
371                    Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0)
372                }
373            } else {
374                read_layer(view, uv0, surface_pass)
375            };
376
377            if surface_pass {
378                c.a = 1.0;
379            } else if f < 1.0 {
380                let pcell = Vec2::<()>::new(((block_c.x - centre.x) / peak).floor(), ((block_c.y - centre.y) / peak).floor());
381                let pc = centre + (pcell + Vec2::<()>::new(0.5, 0.5)) * peak;
382                let mut n = Simplex2D.at(Vec2::<()>::new(
383                    pc.x / DISSOLVE_SCALE + mf.seed.x * 37.0 + 3.7,
384                    pc.y / DISSOLVE_SCALE + mf.seed.y * 37.0 + 3.7,
385                ));
386                n = ((n - 0.5) * 1.6 + 0.5).clamp(0.0, 1.0);
387                n = crate::mix(n, Hash::<f64>::hash(&(pcell + seed_px)), PIXEL_RANDOM);
388                n = ((n - 0.5) * PIXEL_SPREAD + 0.5).clamp(0.0, 1.0);
389                let t0 = n * (1.0 - PIXEL_DUR);
390                c.a *= crate::smoothstep(t0, t0 + PIXEL_DUR, f);
391            }
392            c
393        });
394    }

409    fn solid(size: u32) -> PixelBuffer<Rgba> {
410        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(size, size));
411        let mut view = buffer.view_mut();
412        let c = Color::<f64>::new_rgba(0.5, 0.5, 0.5, 1.0);
413        for y in 0..size {
414            for x in 0..size {
415                view.put_color(Pixels::new(x), Pixels::new(y), c);
416            }
417        }
418        buffer
419    }

421    fn ctx_of<'a>(window: &'a PixelBuffer<Rgba>, surface: Option<&'a PixelBuffer<Rgba>>, layer: Layer, progress: f64, direction: Direction) -> FxContext<'a> {
422        let size = window.size();
423        FxContext {
424            progress: Fraction::new(progress),
425            duration: Duration::from_secs_f64(2.0),
426            seed: Seed(0xABCD_EF01_2345_6789),
427            direction,
428            layer,
429            window: window.view(),
430            surface: surface.map(|s| s.view()),
431            assets: crate::FxAssets::default(),
432            pointer: None,
433            fullscreen: false,
434            padding: Pixels::new(0.0),
435            window_rect: Rect::from_xywh(4.0, 4.0, f64::from(size.x) - 8.0, f64::from(size.y) - 8.0),
436        }
437    }

439    /// Open end: the window dissolved fully in; close end: gone.
440    #[test]
441    fn window_pass_endpoints() {
442        let window = solid(48);
443        let surface = solid(48);
444        let mut out = PixelBuffer::<Rgba>::new(Size::new(48, 48));
445        let fx = Morph::default();
446        fx.apply(&ctx_of(&window, Some(&surface), Layer::Window, 1.0, Direction::Open), &mut out.view_mut());
447        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 1.0);
448        fx.apply(&ctx_of(&window, Some(&surface), Layer::Window, 1.0, Direction::Close), &mut out.view_mut());
449        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 0.0);

452    /// The dissolve is blotch-driven: mid-close some spots are in,
453    /// some still out.
454    #[test]
455    fn dissolve_is_uneven() {
456        let window = solid(96);
457        let surface = solid(96);
458        let mut out = PixelBuffer::<Rgba>::new(Size::new(96, 96));
459        Morph::default().apply(&ctx_of(&window, Some(&surface), Layer::Window, 0.5, Direction::Close), &mut out.view_mut());
460        let view = out.view();
461        let mut lo = false;
462        let mut hi = false;
463        for y in 0..96 {
464            for x in 0..96 {
465                let a = view.sample_bilinear(Vec2::<()>::new((x as f64 + 0.5) / 96.0, (y as f64 + 0.5) / 96.0)).a;
466                lo |= a < 0.5;
467                hi |= a > 0.5;
468            }
469        }
470        assert!(lo && hi, "dissolve should leave both states mid-run");
471    }
