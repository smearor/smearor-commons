17/// [`Pixelate`](super::Pixelate)'s cell-dropout driven by a radial
18/// wipe from the pointer position instead of uniform noise - the
19/// dissolve spreads in a circle from where the user touched.
20///
21/// Behavioral spec (source shader: `pixel-wipe.frag`, re-implemented
22/// from semantics): `circle = length(uv - start_pos)` feeds
23/// `progress = smoothstep(((1 - easeOutQuad(p)) * (1 + FADE_WIDTH) -
24/// 1 + circle) / FADE_WIDTH)`, inverted for `Close`; `start_pos` is
25/// the pointer in uv space (`uStartPos`, set from the actor-local
26/// pointer position in `PixelWipe.js` - `ctx.pointer`, window
27/// center when absent). Pixelation and noise dropout then follow
28/// `pixelate` (`simplex2DFractal(texcoord * size / 20) * 1.5 - 0.25`,
29/// `* 10` collapse).
30#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
31#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
32pub struct PixelWipe {
33    /// Maximum pixel size - schema `pixel-wipe-pixel_size`, default `10`.
34    #[serde(default = "PixelWipe::default_pixel_size")]
35    pub pixel_size: Pixels<u32>,
36}
37
38impl PixelWipe {
39    const fn default_pixel_size() -> Pixels<u32> {
40        Pixels::new(10)
41    }
42}

52/// Width of the radial transition band - `FADE_WIDTH`.
53const FADE_WIDTH: f64 = 1.0;
54
55const PARAMS: &[ParamSpec] = &[ParamSpec {
56    key: "pixel_size",
57    label: "Pixel Size",
58    kind: ParamKind::Int { min: 1, max: 50, default: 10 },
59}];
60
61impl WindowEffect for PixelWipe {
62    fn nick(&self) -> &'static str {
63        "pixel-wipe"
64    }
65
66    fn label(&self) -> &'static str {
67        "Pixel Wipe"
68    }
69
70    fn params(&self) -> &'static [ParamSpec] {
71        PARAMS
72    }
73
74    fn values(&self) -> Vec<ParamValue> {
75        vec![ParamValue::Int(self.pixel_size.0 as i64)]
76    }
77
78    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
79        match values {
80            [ParamValue::Int(size)] => {
81                self.pixel_size = Pixels::new((*size).max(1) as u32);
82                Ok(())
83            }
84            _ => Err(FxError::ParamMismatch(format!("pixel-wipe takes 1 int parameter, got {}", values.len()))),
85        }
86    }
87
88    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
89        let size = ctx.window.size();
90        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
91        // The wipe origin - pointer uv or window center.
92        let start = ctx.pointer.unwrap_or(Vec2::<()>::new(0.5, 0.5));
93        let eased = EasingFunction::EaseOut.apply(ctx.progress).as_f64();
94        let opening = matches!(ctx.direction, crate::Direction::Open);
95        // Frame constants - pixel grid uses the progress after the
96        // radial transform, so it varies per pixel.
97        crate::render::fill(out, |x, y| {
98            let uv = Vec2::<()>::new((x as f64 + 0.5) / size_px.x, (y as f64 + 0.5) / size_px.y);
99            let circle = ((uv.x - start.x).mul_add(uv.x - start.x, (uv.y - start.y) * (uv.y - start.y))).sqrt();
100            let mut progress = crate::smoothstep(0.0, 1.0, ((1.0 - eased) * (1.0 + FADE_WIDTH) - 1.0 + circle) / FADE_WIDTH);
101            if !opening {
102                progress = 1.0 - progress;
103            }
104            let pixel_size = (self.pixel_size.0 as f64 * progress + 1.0).ceil().max(1.0);
105            let grid = Vec2::<()>::new(pixel_size / size_px.x, pixel_size / size_px.y);
106            let texcoord = Vec2::<()>::new(
107                uv.x - crate::fract(uv.x / grid.x) * grid.x + grid.x * 0.5,
108                uv.y - crate::fract(uv.y / grid.y) * grid.y + grid.y * 0.5,
109            );
110            let mut c = ctx.window.sample_bilinear(texcoord);
111            let random = Simplex2DFractal.at(Vec2::<()>::new(texcoord.x * size_px.x / 20.0, texcoord.y * size_px.y / 20.0)) * 1.5 - 0.25;
112            if progress > random {
113                c.a *= (1.0 - (progress - random) * 10.0).max(0.0);
114            }
115            c
116        });
117    }
118}

133    fn solid(size: u32) -> PixelBuffer<Rgba> {
134        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(size, size));
135        let mut view = buffer.view_mut();
136        let c = Color::<f64>::new_rgba(0.5, 0.5, 0.5, 1.0);
137        for y in 0..size {
138            for x in 0..size {
139                view.put_color(Pixels::new(x), Pixels::new(y), c);
140            }
141        }
142        buffer
143    }
144
145    fn ctx_of<'a>(window: &'a PixelBuffer<Rgba>, progress: f64, direction: Direction) -> FxContext<'a> {
146        let size = window.size();
147        FxContext {
148            progress: Fraction::new(progress),
149            duration: Duration::from_secs_f64(2.0),
150            seed: Seed(0xABCD_EF01_2345_6789),
151            direction,
152            layer: Layer::Window,
153            window: window.view(),
154            surface: None,
155            assets: crate::FxAssets::default(),
156            pointer: None,
157            fullscreen: false,
158            padding: Pixels::new(0.0),
159            window_rect: Rect::from_xywh(0.0, 0.0, f64::from(size.x), f64::from(size.y)),
160        }
161    }
162
163    /// Without a pointer the wipe starts at the window center - early
164    /// close dissolves the center before the corners.
165    #[test]
166    fn wipe_starts_at_the_pointer() {
167        let window = solid(64);
168        let mut out = PixelBuffer::<Rgba>::new(Size::new(64, 64));
169        PixelWipe::default().apply(&ctx_of(&window, 0.18, Direction::Close), &mut out.view_mut());
170        let view = out.view();
171        let center = view.sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a;
172        let corner = view.sample_bilinear(Vec2::<()>::new(0.03, 0.03)).a;
173        assert!(center < corner, "center {center} dissolves before corner {corner}");
174    }
175
176    /// Full open is untouched, full close is gone.
177    #[test]
178    fn pixel_wipe_endpoints() {
179        let window = solid(32);
180        let mut out = PixelBuffer::<Rgba>::new(Size::new(32, 32));
181        let fx = PixelWipe::default();
182        fx.apply(&ctx_of(&window, 1.0, Direction::Open), &mut out.view_mut());
183        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 1.0);
184        fx.apply(&ctx_of(&window, 1.0, Direction::Close), &mut out.view_mut());
185        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 0.0);
186    }
