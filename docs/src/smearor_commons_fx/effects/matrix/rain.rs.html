1/// The rain-column domain of `matrix.frag`: per-column delay/speed
2/// randomness and the fade constants of the drop + window masks.
3///
4/// BMW's rain RNG is the *sinus* hash family (`fract(sin(x) * k)`),
5/// deliberately not the Hoskins family - kept for parity with the
6/// reference visuals.
7/// Per-column start delay - `hash11(column)` in the sinus family.
8pub(super) fn column_hash(column: f64) -> f64 {
9    crate::fract(column.sin() * 78.233)
10}
11
12/// Second column RNG from the same GLSL family (`cos` variant) -
13/// scales the drop speed.
14pub(super) fn column_speed_hash(column: f64) -> f64 {
15    crate::fract(column.cos() * 12.989)
16}
17
18/// Column-end fade width in pixels - `matrix.frag`'s `EDGE_FADE`.
19pub(super) const EDGE_FADE: f64 = 30.0;
20/// Pixel length of the window-dissolve gradient behind a drop -
21/// `FADE_WIDTH`.
22pub(super) const FADE_WIDTH: f64 = 150.0;
23/// Exponential tail length of a drop's trail - `TRAIL_LENGTH`.
24pub(super) const TRAIL_LENGTH: f64 = 0.2;
25/// Progress at which the global trail fade-out starts -
26/// `FINAL_FADE_START_TIME`.
27pub(super) const FINAL_FADE_START_TIME: f64 = 0.8;
