20/// Green rain glyphs fall down the window while it dissolves beneath
21/// them - the Matrix effect.
22///
23/// Behavioral spec (source shader: `matrix.frag`, re-implemented from
24/// semantics): the texture is vertically overscanned by `overshoot`
25/// (`uv.y` mapped to `[-overshoot/2, 1+overshoot/2]`). Per `scale`-px
26/// column a deterministic delay and speed place a "raindrop" whose
27/// exponential tail (`TRAIL_LENGTH`) fades the glyph mask; the window
28/// alpha follows the drop with a `FADE_WIDTH`-px gradient. Glyph
29/// coverage comes from a [`GlyphSource`] - the bundled
30/// [`ProceduralGlyphAtlas`] replaces BMW's `matrixFont.png` bitmap.
31/// A `FINAL_FADE_START_TIME` tail-fade extinguishes the trails at the
32/// end.
33///
34/// Note on the column RNG: BMW's rain-column hash is the *sinus* hash
35/// (`fract(sin(column) * 78.233)`), not the Hoskins family - kept as a
36/// private [`column_hash`] for deliberate parity.
37#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
38#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
39pub struct Matrix {
40    /// Glyph cell size in px - schema `matrix-scale`, default `20`.
41    #[serde(default = "Matrix::default_scale")]
42    pub scale: Pixels<u32>,
43    /// Column delay/speed jitter `0..=1` - schema `matrix-randomness`, default `1.0`.
44    #[serde(default = "Matrix::default_randomness")]
45    pub randomness: Fraction,
46    /// Head glyph color - schema `matrix-tip-color`, default `rgb(255, 255, 255)`.
47    #[serde(default = "Matrix::default_tip_color")]
48    pub tip_color: Color<f64>,
49    /// Trail color - schema `matrix-trail-color`, default `rgb(25, 255, 89)`.
50    #[serde(default = "Matrix::default_trail_color")]
51    pub trail_color: Color<f64>,
52    /// Vertical overscan `0..=1` - schema `matrix-overshoot`, default `0.25`.
53    #[serde(default = "Matrix::default_overshoot")]
54    pub overshoot: Fraction,
55}

57/// Atlas grid size - `matrix.frag`'s `LETTER_TILES`.
58const LETTER_TILES: f64 = 16.0;
59/// Tile flickers per animation second - `LETTER_FLICKER_SPEED`.
60const LETTER_FLICKER_SPEED: f64 = 2.0;
61
62impl Matrix {

64        Pixels::new(20)
65    }
66    const fn default_randomness() -> Fraction {
67        Fraction::new(1.0)
68    }
69    const fn default_tip_color() -> Color<f64> {
70        Color::<f64>::new_rgba(1.0, 1.0, 1.0, 1.0)
71    }
72    const fn default_trail_color() -> Color<f64> {
73        Color::<f64>::new_rgba(25.0 / 255.0, 1.0, 89.0 / 255.0, 1.0)
74    }
75    const fn default_overshoot() -> Fraction {
76        Fraction::new(0.25)
77    }

79    /// Flickering glyph coverage at overscanned pixel coords - the
80    /// `getText` helper: each `scale`-px block picks a `hash22`-chained
81    /// tile that flips `LETTER_FLICKER_SPEED * duration` times per run.
82    fn text_mask(&self, ctx: &FxContext<'_>, px: Vec2<()>, glyphs: &impl GlyphSource) -> f64 {
83        let letter = self.scale.0.max(1) as f64;
84        let cell_uv = Vec2::<()>::new(crate::fract(px.x / letter), crate::fract(px.y / letter));
85        let block = Vec2::<()>::new((px.x / letter).floor(), (px.y / letter).floor());
86        // Tile index: hash the block through BMW's magic constants,
87        // add the flicker-time hash, mod into the 16x16 atlas.
88        let chained: Vec2<()> = Hash::<Vec2<()>>::hash(&block);
89        let mixed = chained
90            .component_mul(&Vec2::<()>::new(12.9898, 78.233))
91            .map(|c| c.floor() + LETTER_FLICKER_SPEED * ctx.progress.as_f64() * ctx.duration.as_secs_f64() + 42.254);
92        let flicker: Vec2<()> = Hash::<Vec2<()>>::hash(&mixed);
93        let tile = flicker.map(|c| (c * LETTER_TILES).floor().min(LETTER_TILES - 1.0));
94        glyphs.glyph(tile, cell_uv)
95    }
96
97    /// The paired rain/window masks - `getRain`'s semantics: per-column
98    /// delay/speed derive a "drop position" over the *un-overscanned*
99    /// `uv.y` (BMW's `getRain` ignores `fragCoord` and uses
100    /// `iTexCoord`); the rain alpha is the drop's exponential tail, the
101    /// window alpha a `FADE_WIDTH`-px gradient trailing the drop. Both
102    /// ends of the column are shortened by a per-column `shorten`
103    /// factor so trails don't touch the edges.
104    fn rain(&self, ctx: &FxContext<'_>, uv: Vec2<()>, px: Vec2<()>, size_px: Vec2<()>) -> (f64, f64) {
105        let letter = self.scale.0.max(1) as f64;
106        let column = px.x - (px.x % letter);
107        let randomness = self.randomness.0;
108
109        let delay = rain::column_hash(column) * randomness;
110        let speed = rain::column_speed_hash(column) * randomness * 0.3 + 1.5;
111        let dist = (ctx.progress.as_f64() * 2.0 - delay) * speed - uv.y;
112
113        let mut rain_alpha = if dist >= 0.0 { (-dist / rain::TRAIL_LENGTH).exp() } else { 0.0 };
114        let mut window_alpha = 1.0 - (size_px.y * dist / rain::FADE_WIDTH).clamp(0.0, 1.0);
115
116        rain_alpha *= crate::absolute_edge_mask(px, size_px, ctx.padding, Pixels::new(rain::EDGE_FADE), 0.5, ctx.fullscreen);
117
118        let shorten = crate::fract((column + 42.0).sin() * 33.423) * self.overshoot.0 * 0.25 * randomness;
119        rain_alpha *= crate::smoothstep(0.0, 1.0, (uv.y / shorten).clamp(0.0, 1.0));
120        rain_alpha *= crate::smoothstep(0.0, 1.0, ((1.0 - uv.y) / shorten).clamp(0.0, 1.0));
121
122        if ctx.direction == crate::Direction::Open {
123            window_alpha = 1.0 - window_alpha;
124        }
125
126        (rain_alpha, window_alpha)
127    }
128}

184impl WindowEffect for Matrix {
185    fn nick(&self) -> &'static str {
186        "matrix"
187    }
188
189    fn label(&self) -> &'static str {
190        "Matrix"
191    }
192
193    fn params(&self) -> &'static [ParamSpec] {
194        PARAMS
195    }
196
197    fn values(&self) -> Vec<ParamValue> {
198        vec![
199            ParamValue::Int(self.scale.0 as i64),
200            ParamValue::Float(self.randomness.0),
201            ParamValue::Color(self.tip_color),
202            ParamValue::Color(self.trail_color),
203            ParamValue::Float(self.overshoot.0),
204        ]
205    }
206
207    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
208        let mismatch = |index: usize| FxError::ParamMismatch(format!("matrix param {index}"));
209        if values.len() != 5 {
210            return Err(FxError::ParamMismatch(format!("matrix takes 5 parameters, got {}", values.len())));
211        }
212        self.scale = Pixels::new(values[0].as_i64().ok_or(mismatch(0))?.max(0) as u32);
213        self.randomness = Fraction::new(values[1].as_f64().ok_or(mismatch(1))?);
214        self.tip_color = values[2].as_color().ok_or(mismatch(2))?;
215        self.trail_color = values[3].as_color().ok_or(mismatch(3))?;
216        self.overshoot = Fraction::new(values[4].as_f64().ok_or(mismatch(4))?);
217        Ok(())
218    }
219
220    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
221        self.apply_with_glyphs(ctx, out, &ProceduralGlyphAtlas)
222    }
223}

225impl Matrix {
226    /// `apply` against a custom [`GlyphSource`] - the point where a
227    /// consumer plugs a real bitmap font atlas. `Sync` because the
228    /// render driver shares the source across row-band workers.
229    pub fn apply_with_glyphs(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>, glyphs: &(impl GlyphSource + Sync)) {
230        let size = ctx.window.size();
231        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
232        let final_fade = 1.0 - ((ctx.progress.as_f64() - rain::FINAL_FADE_START_TIME) / (1.0 - rain::FINAL_FADE_START_TIME)).clamp(0.0, 1.0);
233        crate::render::fill(out, |x, y| {
234            let uv = Vec2::<()>::new((x as f64 + 0.5) / size_px.x, (y as f64 + 0.5) / size_px.y);
235            let mut coords = uv;
236            coords.y = coords.y * (self.overshoot.0 + 1.0) - self.overshoot.0 * 0.5;
237            let px = Vec2::<()>::new(coords.x * size_px.x, coords.y * size_px.y);
238            let px_orig = Vec2::<()>::new(uv.x * size_px.x, uv.y * size_px.y);
239
240            let (rain_alpha, window_alpha) = self.rain(ctx, uv, px_orig, size_px);
241            let text_alpha = self.text_mask(ctx, px, glyphs);
242            let rain = final_fade * rain_alpha;
243
244            let mut window = ctx.window.sample_bilinear(coords);
245            window.a *= window_alpha;
246
247            let blend = (rain + 0.1).powi(4).min(1.0);
248            let text = Color::<f64>::new_rgba(
249                crate::mix(self.trail_color.r, self.tip_color.r, blend),
250                crate::mix(self.trail_color.g, self.tip_color.g, blend),
251                crate::mix(self.trail_color.b, self.tip_color.b, blend),
252                rain * text_alpha,
253            );
254            text.alpha_over(&window)
255        });
256    }

287    /// Rain alpha is constant within a column and varies across
288    /// columns (column hash decorrelates). The window is 200px wide so
289    /// the sampled columns sit inside the `EDGE_FADE = 30` interior
290    /// where the edge mask is 1 and cannot disturb the comparison.
291    #[test]
292    fn matrix_columns() {
293        let data = [30, 30, 30, 255].repeat(200 * 64);
294        let window = BufferView::<Rgba>::from_packed(&data, Size::new(200, 64)).unwrap();
295        let m = Matrix::default();
296        let ctx = ctx_for(window, 0.5);
297        let size_px = Vec2::<()>::new(200.0, 64.0);
298        let rain_at = |x: f64| {
299            let uv = Vec2::<()>::new(x / 200.0, 0.4);
300            m.rain(&ctx, uv, Vec2::<()>::new(x, 0.4 * 64.0), size_px).0
301        };
302        // x = 50.5 and 55.5 share column block 40 (20px letters).
303        assert_eq!(rain_at(50.5), rain_at(55.5), "same column block = same rain alpha");
304        // Columns 40 and 60 decorrelate: their trails differ.
305        assert_ne!(rain_at(50.5), rain_at(70.5), "different columns differ");
306    }

308    /// The bundled atlas is deterministic per tile and covers cells.
309    #[test]
310    fn matrix_atlas_parity() {
311        let atlas = ProceduralGlyphAtlas;
312        let tile = Vec2::<()>::new(3.0, 7.0);
313        let mut filled = 0;
314        // Scan the 5x7 font grid (in its 7x7 cell frame) at centers.
315        for cy in 0..7 {
316            for cx in 0..7 {
317                let uv = Vec2::<()>::new((cx as f64 + 0.5) / 7.0, (cy as f64 + 0.5) / 7.0);
318                let a = atlas.glyph(tile, uv);
319                let b = atlas.glyph(tile, uv);
320                assert_eq!(a, b, "deterministic per tile");
321                if a > 0.0 {
322                    filled += 1;
323                }
324            }
325        }
326        assert!(filled > 3, "a glyph tile should cover several cells, got {filled}");
327    }
