1//! A tiny embedded 5x7 bitmap font for [`ProceduralGlyphAtlas`].
2//!
3//! The classic public-domain GLCD letterforms (digits `0`-`9`, then
4//! uppercase `A`-`Z`) in column-major order: each `u8` is one
5//! column, bit `row` (0 = top) marks a lit pixel. This gives the
6//! matrix rain real glyph silhouettes without shipping a font asset
7//! or depending on a font loader - the [`GlyphSource`] trait stays
8//! the plug for a consumer-supplied atlas.
9
10use crate::smoothstep;
11use smearor_commons_geometry::Vec2;
12
13/// Lit columns per glyph.
14const COLUMNS: usize = 5;
15/// Rows per glyph.
16const ROWS: usize = 7;
17/// The 5x7 grid sits inside a `7 x 7` cell grid - one empty column
18/// of padding on each side so neighboring glyphs never touch.
19const GRID: f64 = 7.0;
20/// Fraction of a cell over which the silhouette edge fades in.
21const EDGE: f64 = 0.25;
22

64/// The number of letterforms in [`GLYPHS`].
65pub(super) const GLYPH_COUNT: usize = GLYPHS.len();
66
67/// Whether the pixel at glyph grid position (`column`, `row`) is lit.
68fn lit(index: usize, column: i64, row: i64) -> bool {
69    if column < 0 || column >= COLUMNS as i64 || row < 0 || row >= ROWS as i64 {
70        return false;
71    }
72    GLYPHS[index][column as usize] & (1 << row) != 0
73}
74
75/// Coverage of glyph `index` at `uv` - the silhouette is sharp, with
76/// a short [`smoothstep`] fade only on cell sides that face an
77/// unlit neighbor, so stems stay solid while outlines stay soft.
78pub(super) fn coverage(index: usize, uv: Vec2<()>) -> f64 {
79    let cell_x = (uv.x * GRID).floor();
80    let cell_y = (uv.y * GRID).floor();
81    let (column, row) = (cell_x as i64 - 1, cell_y as i64);
82    if !lit(index, column, row) {
83        return 0.0;
84    }
85    let (cell_u, cell_v) = (uv.x * GRID - cell_x, uv.y * GRID - cell_y);
86    let mut edge = 1.0_f64;
87    if !lit(index, column - 1, row) {
88        edge = edge.min(smoothstep(0.0, EDGE, cell_u));
89    }
90    if !lit(index, column + 1, row) {
91        edge = edge.min(smoothstep(0.0, EDGE, 1.0 - cell_u));
92    }
93    if !lit(index, column, row - 1) {
94        edge = edge.min(smoothstep(0.0, EDGE, cell_v));
95    }
96    if !lit(index, column, row + 1) {
97        edge = edge.min(smoothstep(0.0, EDGE, 1.0 - cell_v));
98    }
99    edge
100}
