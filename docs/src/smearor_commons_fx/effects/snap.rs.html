18/// Thanos-snap disintegration: the window crumbles into four
19/// wind-blown dust layers.
20///
21/// Behavioral spec (source shader: `snap.frag`, re-implemented from
22/// semantics): progress is *un*inverted for `Open` (unlike most
23/// effects). A vertical+noise `progress` field (`^2`) is the per-layer
24/// dissolve threshold against the dust map's `r` channel; each of 4
25/// layers gets a seed-rotated grow/shrink direction (winding-flipped
26/// per side, flipped for half the layers) plus a small wind drift;
27/// `dustMap.g` bins particles into layers. `uDustColor` tints the
28/// window toward dust as progress grows.
29///
30/// `ctx.assets.dust` overrides the procedural Voronoi dust map
31/// (BMW `dust.png`: `r` = dissolve threshold, `g` = layer id).
32#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
33#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
34pub struct Snap {
35    /// Dust particle scale - schema `snap-scale`, default `3.0`.
36    #[serde(default = "Snap::default_scale")]
37    pub scale: Factor,
38    /// Dust tint - schema `snap-color`, default `rgb(50, 40, 35)`.
39    #[serde(default = "Snap::default_color")]
40    pub color: Color<f64>,
41}
42
43impl Snap {

45        Factor::new(3.0)
46    }
47    const fn default_color() -> Color<f64> {
48        Color::<f64>::new(50.0 / 255.0, 40.0 / 255.0, 35.0 / 255.0)

50}

81const DUST_LAYERS: f64 = 4.0;
82const GROW_INTENSITY: f64 = 0.05;
83const SHRINK_INTENSITY: f64 = 0.05;
84const WIND_INTENSITY: f64 = 0.05;
85const ACTOR_SCALE: f64 = 1.2;
86const PADDING: f64 = ACTOR_SCALE / 2.0 - 0.5;
87
88impl WindowEffect for Snap {
89    fn nick(&self) -> &'static str {
90        "snap"
91    }
92
93    fn label(&self) -> &'static str {
94        "Snap"
95    }
96
97    fn params(&self) -> &'static [ParamSpec] {
98        PARAMS
99    }
100
101    fn values(&self) -> Vec<ParamValue> {
102        vec![ParamValue::Float(self.scale.0), ParamValue::Color(self.color)]
103    }
104
105    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
106        let mismatch = |index: usize| FxError::ParamMismatch(format!("snap param {index}"));
107        if values.len() != 2 {
108            return Err(FxError::ParamMismatch(format!("snap takes 2 parameters, got {}", values.len())));
109        }
110        self.scale = Factor::new(values[0].as_f64().ok_or(mismatch(0))?.max(0.01));
111        self.color = values[1].as_color().ok_or(mismatch(1))?;
112        Ok(())
113    }
114
115    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
116        let size = ctx.window.size();
117        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
118        let opening = matches!(ctx.direction, crate::Direction::Open);
119        let seed = Hash::<Vec2<()>>::hash(&ctx.seed.as_f64());
120        let raw_progress = ctx.progress.as_f64();
121        let dust_scale = self.scale.0;
122        let dust_color = self.color;
123        let window = ctx.window;
124        let dust_texture = ctx.assets.dust;
125        crate::render::fill(out, |x, y| {
126            let uv = Vec2::<()>::new((x as f64 + 0.5) / size_px.x, (y as f64 + 0.5) / size_px.y);
127            // Un-inverted for opening (the snap runs 1 -> 0 there).
128            let progress = if opening { raw_progress } else { 1.0 - raw_progress };
129            let gradient = uv.y * ACTOR_SCALE - PADDING;
130            let mut progress = 2.0 - gradient - 2.0 * progress;
131            progress += 0.25 - 0.5 * Simplex2D.at((uv + seed) * 2.0);
132            progress = progress.max(0.0).powi(2);
133            let mut c = Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0);
134            for i in 0..DUST_LAYERS as usize {
135                let i = i as f64;
136                let factor = i / (DUST_LAYERS - 1.0);
137                let angle = 123.123 * (seed.x + factor);
138                let (sin, cos) = angle.sin_cos();
139                let mut direction = Vec2::<()>::new(cos, sin);
140                // coords relative to the actor center.
141                let mut coords = Vec2::<()>::new(uv.x * ACTOR_SCALE - PADDING - 0.5, uv.y * ACTOR_SCALE - PADDING - 0.5);
142                // Winding: flip the direction on one side of the window.
143                if direction.x * coords.y - direction.y * coords.x > 0.0 {
144                    direction = direction * -1.0;
145                }
146                if factor > 0.5 {
147                    direction = direction * -1.0;
148                }
149                // Grow along, shrink orthogonally to the direction.
150                let perp = Vec2::<()>::new(direction.y, -direction.x);
151                let plen = (perp.x * perp.x + perp.y * perp.y).sqrt();
152                let dist = ((0.0 - coords.x) * perp.x / plen + (0.0 - coords.y) * perp.y / plen).abs();
153                let grow = direction * (dist * crate::mix(0.0, GROW_INTENSITY, progress));
154                let shrink = perp * (dist * crate::mix(0.0, SHRINK_INTENSITY, progress));
155                let scale = crate::mix(1.0, 1.05, factor * progress);
156                coords = Vec2::<()>::new((coords.x + grow.x + shrink.x) / scale, (coords.y + grow.y + shrink.y) / scale);
157                // Wind.
158                let wind = WIND_INTENSITY * progress * if opening { 1.0 } else { -1.0 };
159                coords = Vec2::<()>::new(coords.x + wind, coords.y);
160                // Dust layer lookup.
161                let dust_uv = Vec2::<()>::new((coords.x + seed.x) * size_px.x / dust_scale / 100.0, (coords.y + seed.y) * size_px.y / dust_scale / 100.0);
162                let (dust_r, dust_g) = match dust_texture {
163                    Some(t) => {
164                        let s = t.sample_bilinear(dust_uv);
165                        (s.r, s.g)
166                    }
167                    None => crate::assets::procedural::dust(dust_uv),
168                };
169                let dust_group = (dust_g * DUST_LAYERS * 0.999).floor();
170                if dust_group == i {
171                    let mut window_color = window.sample_bilinear(Vec2::<()>::new(coords.x + 0.5, coords.y + 0.5));
172                    // Fade the window color to the dust color.
173                    let tint = crate::mix(window_color.r, dust_color.r, dust_color.a);
174                    let tint_g = crate::mix(window_color.g, dust_color.g, dust_color.a);
175                    let tint_b = crate::mix(window_color.b, dust_color.b, dust_color.a);
176                    window_color = Color::<f64>::new_rgba(
177                        crate::mix(window_color.r, tint, progress),
178                        crate::mix(window_color.g, tint_g, progress),
179                        crate::mix(window_color.b, tint_b, progress),
180                        window_color.a,
181                    );
182                    if dust_r - progress > 0.0 {
183                        c = window_color;
184                    }
185                }
186            }
187            c
188        });
189    }

205    fn solid(size: u32) -> PixelBuffer<Rgba> {
206        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(size, size));
207        let mut view = buffer.view_mut();
208        let c = Color::<f64>::new_rgba(0.5, 0.5, 0.5, 1.0);
209        for y in 0..size {
210            for x in 0..size {
211                view.put_color(Pixels::new(x), Pixels::new(y), c);
212            }
213        }
214        buffer
215    }
216
217    fn ctx_of<'a>(window: &'a PixelBuffer<Rgba>, progress: f64, direction: Direction) -> FxContext<'a> {
218        let size = window.size();
219        FxContext {
220            progress: Fraction::new(progress),
221            duration: Duration::from_secs_f64(2.0),
222            seed: Seed(0xABCD_EF01_2345_6789),
223            direction,
224            layer: Layer::Window,
225            window: window.view(),
226            surface: None,
227            assets: crate::FxAssets::default(),
228            pointer: None,
229            fullscreen: false,
230            padding: Pixels::new(0.0),
231            window_rect: Rect::from_xywh(0.0, 0.0, f64::from(size.x), f64::from(size.y)),
232        }
233    }
234
235    /// Full close leaves nothing.
236    #[test]
237    fn close_endpoint() {
238        let window = solid(32);
239        let mut out = PixelBuffer::<Rgba>::new(Size::new(32, 32));
240        Snap::default().apply(&ctx_of(&window, 1.0, Direction::Close), &mut out.view_mut());
241        assert_eq!(out.view().sample_bilinear(Vec2::<()>::new(0.5, 0.5)).a, 0.0);
242    }
243
244    /// Mid-snap some dust cells survive, others are gone.
245    #[test]
246    fn dust_cells_pop() {
247        let window = solid(64);
248        let mut out = PixelBuffer::<Rgba>::new(Size::new(64, 64));
249        Snap::default().apply(&ctx_of(&window, 0.4, Direction::Close), &mut out.view_mut());
250        let view = out.view();
251        let mut gone = 0;
252        let mut lit = 0;
253        for y in 0..64 {
254            for x in 0..64 {
255                if view.sample_bilinear(Vec2::<()>::new((x as f64 + 0.5) / 64.0, (y as f64 + 0.5) / 64.0)).a < 0.5 {
256                    gone += 1
257                } else {
258                    lit += 1
259                }
260            }
261        }
262        assert!(gone > 0 && lit > 0, "dust should dissolve cell-wise: {gone} gone, {lit} lit");
263    }
