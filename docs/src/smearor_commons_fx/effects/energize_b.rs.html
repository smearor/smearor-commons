22/// A shower of particles sweeps top-to-bottom, trailing streak
23/// lines, while glimmering atoms condense where the window
24/// dissolved - Energize A's sibling with directional sweep instead
25/// of a radial burst.
26///
27/// Behavioral spec (source shader: `energize-b.frag`, re-implemented
28/// from semantics): `progress = easeOutQuad(uProgress)`. `getMasks`
29/// returns four components - `shower` (a `SHOWER_WIDTH` band
30/// sweeping down over `SHOWER_TIME`), `streak` (everything the
31/// shower passed, over `STREAK_TIME`), `atom` (the relative-edge
32/// mask minus the shower, streak-gated, fading as
33/// `sqrt(1-fade²)`), `window` (`(1-fade)²`, inverted for `Open`).
34/// Masks clip at a 50-px absolute edge fade and a top-down
35/// `streakProgress` gradient. Particles: shower =
36/// `simplex2D(uv·size/scale)^10` drifting up at `0.7p/SHOWER_TIME`,
37/// streaks = `simplex2DFractal(uv·(0.05,0.001)·size/scale)*0.5`
38/// drifting at `p/SHOWER_TIME`, atoms = `simplex3D(uv·0.2·size/
39/// scale, p·duration)^5` drifting at `0.025p/SHOWER_TIME`.
40#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
41#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
42pub struct EnergizeB {
43    /// Particle/tint color - schema `energize-b-color`, default `rgb(160, 180, 255)`.
44    #[serde(default = "EnergizeB::default_color")]
45    pub color: Color<f64>,
46    /// Noise scale - schema `energize-b-scale`, default `1.0`.
47    #[serde(default = "EnergizeB::default_scale")]
48    pub scale: ScaleFactor,
49}

69/// Shower sweep duration - `SHOWER_TIME`.
70const SHOWER_TIME: f64 = 0.3;
71/// Shower band width - `SHOWER_WIDTH`.
72const SHOWER_WIDTH: f64 = 0.3;
73/// Streak persistence - `STREAK_TIME`.
74const STREAK_TIME: f64 = 0.6;
75/// Edge fade in px - `EDGE_FADE`.
76const EDGE_FADE: f64 = 50.0;
77
78/// The four `getMasks` components for one pixel.
79struct Masks {
80    /// Leading shower band.
81    shower: f64,
82    /// Trailing streak region.
83    streak: f64,
84    /// Glimmering atom field.
85    atom: f64,
86    /// Remaining window opacity.
87    window: f64,
88}
89

110impl WindowEffect for EnergizeB {
111    fn nick(&self) -> &'static str {
112        "energize-b"
113    }
114
115    fn label(&self) -> &'static str {
116        "Energize B"
117    }
118
119    fn params(&self) -> &'static [ParamSpec] {
120        PARAMS
121    }
122
123    fn values(&self) -> Vec<ParamValue> {
124        vec![ParamValue::Color(self.color), ParamValue::Float(self.scale.0)]
125    }
126
127    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError> {
128        match values {
129            [ParamValue::Color(color), ParamValue::Float(scale)] => {
130                self.color = *color;
131                self.scale = ScaleFactor::new(*scale);
132                Ok(())
133            }
134            _ => Err(FxError::ParamMismatch(format!("energize-b takes color + float, got {values:?}"))),
135        }
136    }
137
138    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>) {
139        let size = ctx.window.size();
140        let size_px = Vec2::<()>::new(size.x as f64, size.y as f64);
141        let progress = EasingFunction::EaseOut.apply(ctx.progress).as_f64();
142        // Frame constants (getMasks' time schedule).
143        let shower_progress = progress / SHOWER_TIME;
144        let streak_progress = ((progress - SHOWER_TIME) / STREAK_TIME).clamp(0.0, 1.0);
145        let fade_progress = ((progress - SHOWER_TIME) / (1.0 - SHOWER_TIME)).clamp(0.0, 1.0);
146        let atom_fade = (1.0 - fade_progress * fade_progress).max(0.0).sqrt();
147        let window_base = (1.0 - fade_progress).powi(2);
148        let window_mask = match ctx.direction {
149            crate::Direction::Open => 1.0 - window_base,
150            crate::Direction::Close => window_base,
151        };
152        let tint_t = 0.5 * window_mask + 0.5;
153        let time = ctx.duration.as_secs_f64();
154        let masks = |uv: Vec2<()>, px: Vec2<()>| -> Masks {
155            let edge = crate::absolute_edge_mask(px, size_px, ctx.padding, Pixels::new(EDGE_FADE), 0.5, ctx.fullscreen);
156            let mut shower = crate::smoothstep(1.0, 0.0, ((shower_progress - uv.y - SHOWER_WIDTH) / SHOWER_WIDTH).abs());
157            let mut streak = if shower_progress - uv.y - SHOWER_WIDTH > 0.0 { 1.0 } else { 0.0 };
158            let atom = (crate::relative_edge_mask(uv, 0.2) - shower).max(0.0) * streak * atom_fade;
159            shower += 0.05 * streak;
160            streak = streak.max(shower);
161            streak *= edge;
162            shower *= edge;
163            let fade = crate::smoothstep(0.0, 1.0, 1.0 + uv.y - 2.0 * streak_progress);
164            streak *= fade;
165            shower *= fade;
166            // The atom mask is already relative-edge bounded.
167            Masks {
168                shower,
169                streak,
170                atom,
171                window: window_mask,
172            }
173        };
174        crate::render::fill(out, |x, y| {
175            let px = Vec2::<()>::new(x as f64 + 0.5, y as f64 + 0.5);
176            let uv = Vec2::<()>::new(px.x / size_px.x, px.y / size_px.y);
177            let m = masks(uv, px);
178            let mut c = ctx.window.sample_bilinear(uv);
179            c = Color::<f64>::new_rgba(
180                crate::mix(self.color.r, c.r, tint_t),
181                crate::mix(self.color.g, c.g, tint_t),
182                crate::mix(self.color.b, c.b, tint_t),
183                c.a * m.window,
184            );
185            // Shower band particles.
186            let shower_uv = Vec2::<()>::new(
187                uv.x * 0.02 * size_px.x / self.scale.0,
188                (uv.y - 0.7 * progress / SHOWER_TIME) * 0.02 * size_px.y / self.scale.0,
189            );
190            let shower = Simplex2D.at(shower_uv).max(0.0).powi(10);
191            // Trailing streak lines.
192            let streak_uv = Vec2::<()>::new(uv.x * 0.05 * size_px.x / self.scale.0, (uv.y - progress / SHOWER_TIME) * 0.001 * size_px.y / self.scale.0);
193            let streaks = Simplex2DFractal.at(streak_uv) * 0.5;
194            // Glimmering atoms.
195            let atom_uv = Vec2::<()>::new(
196                uv.x * 0.2 * size_px.x / self.scale.0,
197                (uv.y - 0.025 * progress / SHOWER_TIME) * 0.2 * size_px.y / self.scale.0,
198            );
199            let atoms = Simplex3D.at(Vec3::<()>::new(atom_uv.x, atom_uv.y, progress * time)).powi(5);
200            let (pr, pg, pb) = (self.color.r, self.color.g, self.color.b);
201            c.r += pr * shower * m.shower + pr * streaks * m.streak + pr * atoms * m.atom;
202            c.g += pg * shower * m.shower + pg * streaks * m.streak + pg * atoms * m.atom;
203            c.b += pb * shower * m.shower + pb * streaks * m.streak + pb * atoms * m.atom;
204            c.a += shower * m.shower + streaks * m.streak + atoms * m.atom;
205            c
206        });
207    }
