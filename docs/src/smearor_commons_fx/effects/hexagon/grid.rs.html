3/// BMW `getHexagons` - the procedural hexagonal grid: `p` in grid
4/// space, returns cell-relative `coords` ([0,0] = cell center),
5/// `edge` = distance to the closest cell edge (`1` at the center,
6/// `0` at the edge), and `glow` = squared distance to the closest
7/// cell center (`/1.5`). Two overlapping cell lattices offset by
8/// half a period tile the plane.
9pub(super) struct HexCell {
10    /// Cell-relative coordinates - `result.xy`.
11    pub coords: Vec2<()>,
12    /// Distance to the closest edge - `result.z`.
13    pub edge: f64,
14    /// Squared distance to the closest center - `result.w`.
15    pub glow: f64,
16}
17
18/// Edge length of one hexagonal cell - `sqrt(4/3)`.
19const EDGE_LENGTH: f64 = 1.154_700_538_379_251_5; // sqrt(4.0/3.0)
20
21/// See [`HexCell`].
22pub(super) fn hex_cell(p: Vec2<()>) -> HexCell {
23    let scale = Vec2::<()>::new(3.0 * EDGE_LENGTH, 2.0);
24    let wrap = |q: Vec2<()>, offset: f64| {
25        Vec2::<()>::new(
26            crate::fract((q.x + offset * scale.x) / scale.x) * scale.x - scale.x * 0.5,
27            crate::fract((q.y + offset * scale.y) / scale.y) * scale.y - scale.y * 0.5,
28        )
29    };
30    let a = wrap(p, 0.0);
31    let b = wrap(p, 0.5);
32    let dist_a = (a.x.abs() / EDGE_LENGTH + a.y.abs() * 0.5).max(a.y.abs());
33    let dist_b = (b.x.abs() / EDGE_LENGTH + b.y.abs() * 0.5).max(b.y.abs());
34    HexCell {
35        coords: if dist_a < dist_b { a } else { b },
36        edge: 1.0 - dist_a.min(dist_b),
37        glow: (a.x * a.x + a.y * a.y).min(b.x * b.x + b.y * b.y) / 1.5,
38    }
39}

41#[cfg(test)]
42mod tests {
43    use super::*;
44
45    /// Cell centers report `coords ≈ 0`, `edge = 1`, `glow = 0`.
46    #[test]
47    fn cell_center_is_the_origin() {
48        let cell = hex_cell(Vec2::<()>::new(0.0, 0.0));
49        assert!(cell.coords.x.abs() < 1e-9 && cell.coords.y.abs() < 1e-9);
50        assert!((cell.edge - 1.0).abs() < 1e-9, "edge {} at the center", cell.edge);
51        assert!(cell.glow.abs() < 1e-9);
52    }
53
54    /// The grid tiles: `p` and `p + scale` produce identical cells.
55    #[test]
56    fn grid_repeats_at_scale() {
57        let scale = Vec2::<()>::new(3.0 * 1.154_700_538_379_251_5, 2.0);
58        let p = Vec2::<()>::new(0.37, 0.71);
59        let a = hex_cell(p);
60        let b = hex_cell(p + scale);
61        assert!((a.edge - b.edge).abs() < 1e-9);
62        assert!((a.coords.x - b.coords.x).abs() < 1e-9 && (a.coords.y - b.coords.y).abs() < 1e-9);
63    }
64
65    /// `edge` falls to 0 at cell borders, `glow` peaks there too.
66    #[test]
67    fn edge_drops_at_borders() {
68        // Scan a horizontal line - edge must hit ~0 somewhere.
69        let min_edge = (0..200).map(|i| hex_cell(Vec2::<()>::new(i as f64 / 40.0, 0.0)).edge).fold(1.0_f64, f64::min);
70        assert!(min_edge < 0.05, "border never reached: {min_edge}");
71    }
