1use thiserror::Error;

3/// Errors of the fx spec/dispatch layer - unknown effect tags,
4/// missing layered context, parameter mismatches.
5#[derive(Debug, Error)]
6pub enum FxError {
7    /// The `"effect"` tag names no shipped variant.
8    #[error("unknown effect `{0}`")]
9    UnknownEffect(String),
10
11    /// A [`crate::Layers::SurfaceAndWindow`] effect ran a pass without
12    /// `ctx.surface`.
13    #[error("layered effect requires ctx.surface, got none")]
14    MissingSurface,
15
16    /// Live [`crate::ParamValue`]s don't line up with the effect's
17    /// `params()` (wrong length or wrong kind).
18    #[error("parameter mismatch: {0}")]
19    ParamMismatch(String),
20
21    /// [`crate::gtk::snapshot`] got a zero-size or node-less widget.
22    #[cfg(feature = "gtk4")]
23    #[error("widget snapshot produced no pixels")]
24    SnapshotEmpty,
25
26    /// A cairo status error surfaced by the GTK adapter - carried as
27    /// text so the error type stays GTK-free.
28    #[cfg(feature = "cairo")]
29    #[error("cairo error: {0}")]
30    Cairo(String),
31}
