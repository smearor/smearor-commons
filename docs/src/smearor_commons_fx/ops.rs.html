1//! `ops` - the `common.glsl` helper family effects actually use.
2//!
3//! Easing is deliberately absent: `animation::EasingFunction` covers
4//! `easeOutQuad`/`easeInSine`/etc., and `fract`'s noise-specific form
5//! lives in `smearor-commons-noise` - these are the remaining
6//! mask/blend primitives.

15/// GLSL `fract(x) = x - x.floor()` - exact for negatives.
16pub fn fract(x: f64) -> f64 {
17    x - x.floor()
18}
19
20/// GLSL `mix(a, b, t)` - linear blend (`t` unclamped).
21pub fn mix(a: f64, b: f64, t: f64) -> f64 {
22    a + (b - a) * t
23}
24
25/// GLSL `smoothstep(edge0, edge1, x)` - hermite ramp `t²(3-2t)`
26/// clamped to `[0, 1]`.
27pub fn smoothstep(edge0: f64, edge1: f64, x: f64) -> f64 {
28    let t = ((x - edge0) / (edge1 - edge0)).clamp(0.0, 1.0);
29    t * t * (3.0 - 2.0 * t)
30}
31
32/// BMW `remap` - `x` from `[old_min, old_max]` into `[new_min,
33/// new_max]`, result clamped to the new range.
34pub fn remap(x: f64, old_min: f64, old_max: f64, new_min: f64, new_max: f64) -> f64 {
35    let t = ((x - old_min) / (old_max - old_min)).clamp(0.0, 1.0);
36    new_min + (new_max - new_min) * t
37}

39/// BMW `getAbsoluteEdgeMask` - the context-aware edge fade: fades
40/// `uv` (pixel space) inside the layer rect shrunk by the effective
41/// padding `max(0, padding - fade * offset)`; fullscreen content is
42/// never faded.
43///
44/// `offset` positions the fade zone relative to the window border:
45/// `0` fully inside, `0.5` centered on the border (the value every
46/// shipped effect uses), `1` fully outside.
47///
48/// Delegates to [`Rect::edge_fade`] for the pure rect math - this
49/// wrapper only knows where the window sits inside the padded layer.
50pub fn absolute_edge_mask(uv: Vec2<()>, size: Vec2<()>, padding: Pixels<f64>, fade: Pixels<f64>, offset: f64, fullscreen: bool) -> f64 {
51    if fullscreen {
52        return 1.0;
53    }
54    let p = (padding.0 - fade.0 * offset).max(0.0);
55    Rect::new(Vec2::<()>::new(p, p), Vec2::<()>::new(size.x - 2.0 * p, size.y - 2.0 * p)).edge_fade(uv, fade.0)
56}

58/// BMW `offsetHue` - rotates the color's hue by `offset` circle
59/// fractions (`0.5` = complementary), saturation and alpha preserved.
60/// `offset` is a `[0, 1)` fraction, not degrees - matching BMW's
61/// `hueOffset` uniform.
62pub fn offset_hue(color: Color<f64>, offset: f64) -> Color<f64> {
63    let mut hsv = Hsv::from(color);
64    hsv.h = Hue::new(hsv.h.as_f64() + offset * 360.0);
65    Color::<f64>::from(hsv)
66}

68/// BMW `whirl` - swirl distortion: rotates `coords` around the
69/// origin by an angle which falls off quadratically with distance
70/// (`warping` scales the falloff, `rotation` adds a constant spin).
71/// The geeks3d swirl-post-processing filter.
72pub fn whirl(coords: Vec2<()>, warping: f64, rotation: f64) -> Vec2<()> {
73    // (1-len)^2 stays positive outside the radius - powi(2), not max(0).
74    let angle = (1.0 - coords.x.mul_add(coords.x, coords.y * coords.y).sqrt()).powi(2) * warping + rotation;
75    let (sin, cos) = angle.sin_cos();
76    Vec2::<()>::new(coords.x * cos - coords.y * sin, coords.x * sin + coords.y * cos)
77}

79/// BMW `getRelativeEdgeMask` - fades `uv` (unit space) to zero at the
80/// actor boundaries over `fade` relative units; pure uv math, no
81/// pixel context.
82pub fn relative_edge_mask(uv: Vec2<()>, fade: f64) -> f64 {
83    Rect::new(Vec2::<()>::new(0.0, 0.0), Vec2::<()>::new(1.0, 1.0)).edge_fade(uv, fade)
84}

86/// BMW `darken` - scales color toward black by `fac`.
87pub fn darken(color: Color<f64>, fac: f64) -> Color<f64> {
88    let t = 1.0 - fac;
89    Color::<f64>::new_rgba(color.r * t, color.g * t, color.b * t, color.a)
90}
91
92/// BMW `lighten` - lerps color toward white by `fac`.
93pub fn lighten(color: Color<f64>, fac: f64) -> Color<f64> {
94    Color::<f64>::new_rgba(color.r + (1.0 - color.r) * fac, color.g + (1.0 - color.g) * fac, color.b + (1.0 - color.b) * fac, color.a)
95}

97/// BMW `tritone` - maps `val` through a three-stop gradient:
98/// shadows → midtones over the lower half, midtones → highlights
99/// over the upper half, each leg smoothed. Used by `incinerate`
100/// and `wisps`.
101pub fn tritone(val: f64, shadows: Color<f64>, midtones: Color<f64>, highlights: Color<f64>) -> Color<f64> {
102    let mix_rgb = |a: Color<f64>, b: Color<f64>, t: f64| Color::<f64>::new(mix(a.r, b.r, t), mix(a.g, b.g, t), mix(a.b, b.b, t));
103    if val < 0.5 {
104        mix_rgb(shadows, midtones, smoothstep(0.0, 1.0, val * 2.0))
105    } else {
106        mix_rgb(midtones, highlights, smoothstep(0.0, 1.0, val * 2.0 - 1.0))
107    }
108}

110/// BMW `alphaOver` - source-over compositing: `src` over `dst`,
111/// straight-alpha semantics.
112pub fn alpha_over(dst: Color<f64>, src: Color<f64>) -> Color<f64> {
113    let a = src.a + dst.a * (1.0 - src.a);
114    if a <= 0.0 {
115        return Color::<f64>::new_rgba(0.0, 0.0, 0.0, 0.0);
116    }
117    let over = |s: f64, d: f64| (s * src.a + d * dst.a * (1.0 - src.a)) / a;
118    Color::<f64>::new_rgba(over(src.r, dst.r), over(src.g, dst.g), over(src.b, dst.b), a)
119}

121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    #[test]
126    fn remap_matches_the_glsl_formula() {
127        assert_eq!(remap(0.5, 0.0, 1.0, 0.0, 10.0), 5.0);
128        assert_eq!(remap(2.0, 0.0, 1.0, 0.0, 10.0), 10.0, "clamps above");
129        assert_eq!(remap(-1.0, 0.0, 1.0, 0.0, 10.0), 0.0, "clamps below");
130        assert_eq!(remap(0.5, 1.0, 0.0, 0.0, 10.0), 5.0, "inverted range");
131    }
132
133    #[test]
134    fn smoothstep_matches_glsl_on_a_grid() {
135        for i in -10..=20 {
136            let x = i as f64 / 10.0;
137            let t = x.clamp(0.0, 1.0);
138            let expected = t * t * (3.0 - 2.0 * t);
139            assert!((smoothstep(0.0, 1.0, x) - expected).abs() < 1e-12);
140        }
141    }
142
143    #[test]
144    fn absolute_edge_mask_respects_padding_and_fullscreen() {
145        let size = Vec2::<()>::new(100.0, 100.0);
146        // Inside the padded rect -> 1.0; at the outer edge -> faded.
147        assert_eq!(absolute_edge_mask(Vec2::<()>::new(50.0, 50.0), size, Pixels::new(10.0), Pixels::new(5.0), 0.5, false), 1.0);
148        assert!(absolute_edge_mask(Vec2::<()>::new(0.0, 50.0), size, Pixels::new(10.0), Pixels::new(5.0), 0.5, false) < 0.5);
149        assert_eq!(absolute_edge_mask(Vec2::<()>::new(0.0, 0.0), size, Pixels::new(10.0), Pixels::new(5.0), 0.5, true), 1.0);
150        // A bigger offset pushes the fade zone outward: at uv 8 the
151        // offset-1.0 mask (fade starts at padding 5) is well inside
152        // while offset-0.0 (fade starts at padding 10) is still 0.
153        let outward = absolute_edge_mask(Vec2::<()>::new(8.0, 50.0), size, Pixels::new(10.0), Pixels::new(5.0), 1.0, false);
154        let inward = absolute_edge_mask(Vec2::<()>::new(8.0, 50.0), size, Pixels::new(10.0), Pixels::new(5.0), 0.0, false);
155        assert!(outward > inward, "offset 1.0 keeps more mask inside the window");
156    }
