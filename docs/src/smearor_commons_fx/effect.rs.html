9/// A per-pixel effect over frozen textures - the analogue of one
10/// Burn-My-Windows shader.
11///
12/// Implementors are plain serde structs (the `EffectSpec` variant
13/// payloads); the trait adds identity, parameter metadata and the
14/// frame application.
15pub trait WindowEffect {
16    /// Stable serialized identifier, e.g. `"fire"` - the spec's
17    /// `"effect"` tag.
18    fn nick(&self) -> &'static str;
19
20    /// Human label, e.g. `"Fire"`.
21    fn label(&self) -> &'static str;
22
23    /// The layers this effect renders - [`Layers::SurfaceAndWindow`]
24    /// makes the consumer run two passes and supply `ctx.surface`.
25    fn layers(&self) -> Layers {
26        Layers::WindowOnly
27    }
28
29    /// Parameter metadata - drives UI rows, serde defaults, codegen.
30    fn params(&self) -> &'static [ParamSpec];
31
32    /// The live parameter values in `params()` order - the
33    /// serialization and UI currency.
34    fn values(&self) -> Vec<ParamValue>;
35
36    /// Applies `values` (in `params()` order) onto the typed fields;
37    /// length/kind mismatches yield [`FxError::ParamMismatch`].
38    fn set_values(&mut self, values: &[ParamValue]) -> Result<(), FxError>;
39
40    /// Pre-flight check for a pass: `Err(MissingSurface)` when this
41    /// is a two-pass effect and `ctx.surface` is `None`.
42    fn check(&self, ctx: &FxContext<'_>) -> Result<(), FxError> {
43        match (self.layers(), ctx.surface.is_some()) {
44            (Layers::SurfaceAndWindow, false) => Err(FxError::MissingSurface),
45            _ => Ok(()),
46        }
47    }
48
49    /// Render one frame: read `ctx`, write `out` fully (every pixel).
50    fn apply(&self, ctx: &FxContext<'_>, out: &mut BufferViewMut<'_, Rgba>);
51}

87            Vec::new()

91            Ok(())
92        }

95    }

97    /// A `SurfaceAndWindow` effect surfaces `MissingSurface` when
98    /// `ctx.surface` is `None`, and passes `check` when it is `Some`.
99    #[test]
100    fn layered_context_requires_surface() {
101        let data = [0u8; 4];
102        let window = BufferView::<Rgba>::from_packed(&data, Size::new(1, 1)).unwrap();
103        let ctx = FxContext {
104            progress: Fraction::ZERO,
105            duration: Duration::ZERO,
106            seed: Seed(0),
107            direction: Direction::Open,
108            layer: Layer::Window,
109            window,
110            surface: None,
111            assets: crate::FxAssets::default(),
112            pointer: None,
113            fullscreen: false,
114            padding: Pixels::new(0.0),
115            window_rect: Rect::from_xywh(0.0, 0.0, 1.0, 1.0),
116        };
117        assert!(matches!(DummyLayered.check(&ctx), Err(FxError::MissingSurface)));
118
119        let ctx = FxContext { surface: Some(window), ..ctx };
120        assert!(DummyLayered.check(&ctx).is_ok());
121    }
