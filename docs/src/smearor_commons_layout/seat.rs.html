12/// A seat on a tabletop surface: which [`TableSide`] it faces plus
13/// where along that edge it sits.
14///
15/// A seat is more than a side - users can sit anywhere along an edge
16/// (two on one long side, off-center), so the offset carries the
17/// position along the edge axis. `offset` `None` means "assigned to
18/// the side, a strategy may distribute the item along it";
19/// `Some(fraction)` pins the item to that edge position
20/// (`HALF` = centered). Which *user* sits there stays consumer domain.
21///
22/// No `zvariant::Type` impl: `Option<Fraction>` has no D-Bus signature
23/// (the format knows no nullable) - wire consumers transport a seat as
24/// a `(TableSide, f64)` pair with `NaN` for an unpinned offset.
25#[derive(Debug, Default, Clone, Copy, PartialEq, Serialize, Deserialize)]
26#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject, async_graphql::InputObject))]
27#[cfg_attr(feature = "async-graphql", graphql(input_name = "SeatInput"))]
28#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
29pub struct Seat {
30    /// The table edge the seat faces.
31    pub side: TableSide,
32    /// Position along the edge axis (`0.0` = edge start, `1.0` = edge
33    /// end, `0.5` = centered). `None` leaves distribution along the
34    /// side to the strategy.
35    pub offset: Option<Fraction>,
36}
37
38impl Seat {
39    /// A seat assigned to `side` without a pinned position - a
40    /// strategy distributes the item along the edge.
41    pub const fn new(side: TableSide) -> Self {
42        Self { side, offset: None }
43    }
44
45    /// A seat pinned to `offset` along the edge of `side`.
46    pub const fn at(side: TableSide, offset: Fraction) -> Self {
47        Self { side, offset: Some(offset) }
48    }
49
50    /// A seat centered on `side`'s edge.
51    pub const fn center(side: TableSide) -> Self {
52        Self::at(side, Fraction::HALF)
53    }
54
55    /// The seat one step clockwise (`side` steps, `offset` keeps).
56    pub fn clockwise(&self) -> Self {
57        Self {
58            side: self.side.clockwise(),
59            ..*self
60        }
61    }
62
63    /// The seat one step counter-clockwise.
64    pub fn counter_clockwise(&self) -> Self {
65        Self {
66            side: self.side.counter_clockwise(),
67            ..*self
68        }
69    }
70
71    /// The centered seat nearest a (free) rotation.
72    pub fn nearest(rotation: RotationRad) -> Self {
73        Self::center(TableSide::nearest(rotation))
74    }

76    /// The seat a position inside `bounds` marks: [`TableSide::nearest_edge`]
77    /// picks the edge, `offset` is the position's projection onto that
78    /// edge's axis, clamped to `0.0..=1.0` - the inverse of the anchor
79    /// math in [`Self::placement`].
80    ///
81    /// Used to derive a seat from a physical marker position (see
82    /// [`crate::SeatPose`]). Degenerate (zero-extent) bounds pin the
83    /// seat to the edge's center.
84    pub fn from_position<U>(position: Vec2<U>, bounds: Rect<U>) -> Self {
85        let side = TableSide::nearest_edge(position, bounds);
86        let (coordinate, start, extent) = match side {
87            TableSide::Bottom | TableSide::Top => (position.x, bounds.min.x, bounds.size.x),
88            TableSide::Left | TableSide::Right => (position.y, bounds.min.y, bounds.size.y),
89        };
90        let offset = if extent > 0.0 {
91            Fraction::new((coordinate - start) / extent).clamped()
92        } else {
93            Fraction::HALF
94        };
95        Self::at(side, offset)
96    }

98    /// The seat's position along the table perimeter, clockwise in
99    /// edge-length units `0.0..4.0`: the bottom edge spans `0..1`
100    /// (right-to-left, `offset` decreases), the left edge `1..2`
101    /// (bottom-to-top), the top edge `2..3` (left-to-right), the right
102    /// edge `3..4` (top-to-bottom) - continuous at the corners, so
103    /// seats on neighboring sides compare directly. An unpinned
104    /// `offset` resolves to `HALF`, like [`Self::placement`].
105    ///
106    /// The coordinate orders seats for "pass to the next occupied
107    /// seat" and lets consumers position seat indicators around the
108    /// table.
109    pub fn perimeter_position(&self) -> f64 {
110        let offset = self.offset.unwrap_or(Fraction::HALF).as_f64();
111        match self.side {
112            TableSide::Bottom => 1.0 - offset,
113            TableSide::Left => 2.0 - offset,
114            TableSide::Top => 2.0 + offset,
115            TableSide::Right => 3.0 + offset,
116        }
117    }

119    /// The occupied seat one step clockwise along the table perimeter:
120    /// the roster seat with the smallest positive distance in
121    /// [`Self::perimeter_position`]. Empty sides are skipped, several
122    /// seats on one side are visited in order, and a seat at this
123    /// seat's own position does not count.
124    ///
125    /// Returns `None` when `roster` is empty or holds no position
126    /// other than this seat's.
127    pub fn next_in(&self, roster: &[Seat]) -> Option<Seat> {
128        self.step_in(roster, true)
129    }
130
131    /// The occupied seat one step counter-clockwise - the mirror of
132    /// [`Self::next_in`].
133    pub fn previous_in(&self, roster: &[Seat]) -> Option<Seat> {
134        self.step_in(roster, false)
135    }

137    /// Shared walk of `next_in`/`previous_in`: picks the roster seat
138    /// with the smallest strictly-positive perimeter distance in the
139    /// chosen direction.
140    fn step_in(&self, roster: &[Seat], clockwise: bool) -> Option<Seat> {
141        let here = self.perimeter_position();
142        let mut best: Option<(Seat, f64)> = None;
143        for seat in roster {
144            let position = seat.perimeter_position();
145            let distance = if clockwise { position - here } else { here - position }.rem_euclid(4.0);
146            if distance <= f64::EPSILON {
147                continue;
148            }
149            if best.is_none_or(|(_, shortest)| distance < shortest) {
150                best = Some((*seat, distance));
151            }
152        }
153        best.map(|(seat, _)| seat)
154    }

156    /// Places `footprint` in front of this seat: `margin` (scalar in
157    /// the space of `bounds`) off the edge, positioned along the
158    /// edge at `offset` (unpinned offsets resolve to `HALF`),
159    /// rotated so a user sitting there reads it correctly.
160    ///
161    /// `footprint` is the item's extent *as displayed at the seat's
162    /// rotation* - compute it with [`crate::footprint_for`] when the
163    /// placement changes orientation.
164    pub fn placement<U>(&self, bounds: Rect<U>, footprint: Vec2<U>, margin: impl Into<Distance<U>>) -> Placement<U> {
165        let margin = margin.into().as_f64();
166        let offset = self.offset.unwrap_or(Fraction::HALF).as_f64();
167        let center = match self.side {
168            TableSide::Bottom => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + bounds.size.y - margin - footprint.y / 2.0),
169            TableSide::Left => Vec2::new(bounds.min.x + margin + footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
170            TableSide::Top => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + margin + footprint.y / 2.0),
171            TableSide::Right => Vec2::new(bounds.min.x + bounds.size.x - margin - footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
172        };
173        Placement::new(center, Rotation::QuarterTurn(self.side.to_quarter_turn()), footprint)
174    }

176    /// Places the item in the best free spot nearest this seat: like
177    /// [`Self::placement`], but `taken` items are treated as physical
178    /// obstacles and the item slips into the closest free lateral gap
179    /// instead of stacking on them.
180    ///
181    /// `taken` holds each occupying item as `(center, footprint)` -
182    /// its live center and displayed extent; *every* item near the
183    /// seat counts, regardless of any seat assignment. Spots are
184    /// evaluated as lanes perpendicular to the edge: lane 0 is the
185    /// edge lane at `margin` inset (the preferred position), deeper
186    /// lanes stack behind it. Within a lane the free center with the
187    /// smallest lateral distance wins, so slots freed by removed items
188    /// are reused and a bigger item only takes a middle slot when it
189    /// actually fits between its neighbors with `gap` clearance.
190    /// Candidates are scored by seat-space distance
191    /// `hypot(offset, ROW_JUMP_WEIGHT * depth)`, so a card may leave
192    /// the edge lane when the queue there reaches notably farther
193    /// from the seat than a spot behind the first row.
194    pub fn fanned_placement<U>(
195        &self,
196        bounds: Rect<U>,
197        footprint: Vec2<U>,
198        margin: impl Into<Distance<U>>,
199        taken: &[(Vec2<U>, Vec2<U>)],
200        gap: impl Into<Distance<U>>,
201    ) -> Placement<U> {
202        let margin = margin.into().as_f64();
203        let gap = gap.into().as_f64();
204        /// Perpendicular depth counts this much when scoring lanes -
205        /// staying on the edge is preferred unless the queue there is
206        /// notably farther from the seat.
207        const ROW_JUMP_WEIGHT: f64 = 2.0;
208
209        let mut placement = self.placement(bounds, footprint, margin);
210        let right: Vec2<U> = self.side.right_direction();
211        let inward = Vec2::new(right.y, -right.x);
212        let anchor = placement.center;
213        let own_t = self.tangential_extent(footprint) / 2.0;
214        let own_p = self.perpendicular_extent(footprint) / 2.0;
215        let anchor_inset = margin + own_p;
216
217        let (mut best_offset, mut best_depth, mut best_score) = (0.0, 0.0, f64::MAX);
218        let mut lane_inset = anchor_inset;
219        loop {
220            let mut blocked = Vec::new();
221            let mut inner_edge = lane_inset;
222            for &(center, item_footprint) in taken {
223                let item_inset = self.inset_of(center, bounds);
224                let item_p = self.perpendicular_extent(item_footprint) / 2.0;
225                if (item_inset - lane_inset).abs() >= item_p + own_p + gap {
226                    continue;
227                }
228                inner_edge = inner_edge.max(item_inset + item_p);
229                let delta = center - anchor;
230                let offset = delta.x * right.x + delta.y * right.y;
231                let item_t = self.tangential_extent(item_footprint) / 2.0;
232                blocked.push((offset - item_t - gap - own_t, offset + item_t + gap + own_t));
233            }
234            let depth = lane_inset - anchor_inset;
235            let offset = nearest_free_slot(&mut blocked);
236            let weighted = ROW_JUMP_WEIGHT * depth;
237            let score = (offset * offset + weighted * weighted).sqrt();
238            if score < best_score {
239                (best_offset, best_depth, best_score) = (offset, depth, score);
240            }
241            // An empty lane has no inner edge to stack the next lane
242            // behind; deeper lanes only score worse once their depth
243            // alone already beats the best candidate.
244            if blocked.is_empty() || weighted >= best_score {
245                break;
246            }
247            lane_inset = inner_edge + gap + own_p;
248        }
249        placement.center += right * best_offset + inward * best_depth;
250        placement
251    }

253    /// The item's extent along this seat's edge axis - x for the
254    /// bottom/top edges, y for the left/right edges.
255    fn tangential_extent<U>(&self, footprint: Vec2<U>) -> f64 {
256        match self.side {
257            TableSide::Bottom | TableSide::Top => footprint.x,
258            TableSide::Left | TableSide::Right => footprint.y,
259        }
260    }
261
262    /// The item's extent perpendicular to this seat's edge - y for the
263    /// bottom/top edges, x for the left/right edges.
264    fn perpendicular_extent<U>(&self, footprint: Vec2<U>) -> f64 {
265        match self.side {
266            TableSide::Bottom | TableSide::Top => footprint.y,
267            TableSide::Left | TableSide::Right => footprint.x,
268        }
269    }
270
271    /// The distance of a center point inward from this seat's edge.
272    fn inset_of<U>(&self, center: Vec2<U>, bounds: Rect<U>) -> f64 {
273        match self.side {
274            TableSide::Bottom => bounds.min.y + bounds.size.y - center.y,
275            TableSide::Top => center.y - bounds.min.y,
276            TableSide::Left => center.x - bounds.min.x,
277            TableSide::Right => bounds.min.x + bounds.size.x - center.x,
278        }
279    }
280}

282/// The center coordinate on a 1D lane nearest `0` where `blocked`
283/// intervals (in newcomer-center space, already expanded by gap and
284/// own half-extent) leave room. Interval boundaries count as free -
285/// landing exactly on a boundary means exactly `gap` clearance.
286/// Prefers the positive direction on ties.
287fn nearest_free_slot(blocked: &mut [(f64, f64)]) -> f64 {
288    blocked.sort_by(|left, right| left.0.partial_cmp(&right.0).unwrap_or(std::cmp::Ordering::Equal));
289    let mut merged: Vec<(f64, f64)> = Vec::with_capacity(blocked.len());
290    for &(left, right) in blocked.iter() {
291        match merged.last_mut() {
292            Some((_, merged_right)) if left <= *merged_right => *merged_right = merged_right.max(right),
293            _ => merged.push((left, right)),
294        }
295    }
296    let is_free = |candidate: f64| merged.iter().all(|&(left, right)| candidate <= left || candidate >= right);
297    std::iter::once(0.0)
298        .chain(merged.iter().flat_map(|&(left, right)| [left, right]))
299        .filter(|&candidate| is_free(candidate))
300        .min_by(|left, right| {
301            left.abs()
302                .partial_cmp(&right.abs())
303                .unwrap_or(std::cmp::Ordering::Equal)
304                .then(right.partial_cmp(left).unwrap_or(std::cmp::Ordering::Equal))
305        })
306        .unwrap_or(0.0)
307}

531    /// `from_position` derives side and offset from a marker position -
532    /// the inverse of `placement`'s edge anchoring.
533    #[test]
534    fn from_position_derives_seat() {
535        let cases = [
536            // Quarter along the bottom edge, near it -> Bottom at 0.25.
537            (Vec2::new(100.0, 290.0), Seat::at(TableSide::Bottom, Fraction::new(0.25))),
538            // Three-quarters down the left edge -> Left at 0.75.
539            (Vec2::new(10.0, 225.0), Seat::at(TableSide::Left, Fraction::new(0.75))),
540            // Edge centers.
541            (Vec2::new(200.0, 10.0), Seat::at(TableSide::Top, Fraction::new(0.5))),
542            (Vec2::new(395.0, 150.0), Seat::at(TableSide::Right, Fraction::new(0.5))),
543        ];
544        for (position, expected) in cases {
545            assert_eq!(Seat::from_position(position, bounds()), expected, "{position:?}");
546        }
547    }

549    /// Positions projecting past an edge's end clamp the offset into
550    /// `0.0..=1.0` instead of extrapolating. `nearest_edge` picks a
551    /// side whose span contains the projection, so clamping only
552    /// bites on outside-corner ties and barely-outside positions.
553    #[test]
554    fn from_position_clamps_offset() {
555        // Left of and above the corner: Left wins the Left/Top
556        // distance tie (ALL order), the y-projection past the edge
557        // start clamps to 0.
558        assert_eq!(Seat::from_position(Vec2::new(-1.0, -1.0), bounds()), Seat::at(TableSide::Left, Fraction::ZERO));
559        // Outside-corner tie: Bottom and Right equally near - Bottom
560        // wins ALL order, the x-projection past the edge end clamps
561        // to 1.
562        assert_eq!(Seat::from_position(Vec2::new(450.0, 350.0), bounds()), Seat::at(TableSide::Bottom, Fraction::ONE));
563    }
