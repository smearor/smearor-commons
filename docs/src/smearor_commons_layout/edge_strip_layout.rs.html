16/// Edge-strip arrangement: items line up as a strip *outside* the
17/// `bounds` rectangle, along each edge named by `sides`.
18///
19/// Meant for host-attached chrome (docked tool strips, satellites):
20/// the strip hovers `gap` off the host outline, so the host's own
21/// extent is untouched and the consumer can clip-union both
22/// footprints if needed. The strip works in the host's local frame -
23/// rotating the host is the consumer's transform to apply afterwards.
24///
25/// Each side's group is centered on the edge and ordered along the
26/// side's reading direction ([`TableSide::right_direction`]), so item
27/// order reads naturally for a user seated there. Placements carry
28/// the side's quarter-turn orientation; consumers may override
29/// `rotation` afterwards. Overlapping strips (long strips on
30/// adjacent edges colliding at a corner) are reported
31/// [`LayoutError::DoesNotFit`].
32#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
33pub struct EdgeStripLayout<U = ()> {
34    /// The edges to dock to - one to four, non-empty by `SideSet`'s
35    /// invariant.
36    pub sides: SideSet,
37    /// How items spread over `sides`.
38    pub distribution: EdgeDistribution,
39    /// Distance between the host outline and the strip, and between
40    /// adjacent items, in the `U` coordinate space of `bounds`.
41    pub gap: Distance<U>,
42}
43
44impl<U> Default for EdgeStripLayout<U> {
45    fn default() -> Self {
46        Self {
47            sides: SideSet::default(),
48            distribution: EdgeDistribution::default(),
49            gap: Distance::new(12.0),
50        }
51    }
52}
53
54impl<U> EdgeStripLayout<U> {
55    /// Whether `side`'s edge runs horizontally (`Bottom`/`Top`).
56    fn horizontal(side: TableSide) -> bool {
57        matches!(side, TableSide::Bottom | TableSide::Top)
58    }
59
60    /// The midpoint of `side`'s edge of `bounds`.
61    fn edge_midpoint(bounds: Rect<U>, side: TableSide) -> Vec2<U> {
62        let center = bounds.center();
63        match side {
64            TableSide::Bottom => Vec2::new(center.x, bounds.min.y + bounds.size.y),
65            TableSide::Left => Vec2::new(bounds.min.x, center.y),
66            TableSide::Top => Vec2::new(center.x, bounds.min.y),
67            TableSide::Right => Vec2::new(bounds.min.x + bounds.size.x, center.y),
68        }
69    }
70
71    /// Groups items onto the sides per `distribution`: each group's
72    /// item indices in input order.
73    fn groups(&self, bounds: Rect<U>, items: &[LayoutItem<U>]) -> Result<Vec<(TableSide, Vec<usize>)>, LayoutError> {
74        match self.distribution {
75            EdgeDistribution::Sequential => self.sequential_groups(bounds, items),
76            EdgeDistribution::Balanced => Ok(self.balanced_groups(items.len())),
77        }
78    }
79
80    /// Greedy side-by-side fill: each side takes as many leading
81    /// items as fit its edge, the rest spills to the next side.
82    fn sequential_groups(&self, bounds: Rect<U>, items: &[LayoutItem<U>]) -> Result<Vec<(TableSide, Vec<usize>)>, LayoutError> {
83        let mut pending: Vec<usize> = (0..items.len()).collect();
84        let mut groups = Vec::new();
85        for side in self.sides.iter() {
86            if pending.is_empty() {
87                break;
88            }
89            let edge_length = if Self::horizontal(side) { bounds.size.x } else { bounds.size.y };
90            let gap = self.gap.as_f64();
91            let mut used = 0.0_f64;
92            let mut taken = 0_usize;
93            for &index in &pending {
94                let footprint = footprint_for(items[index].size, items[index].rotation, side.to_quarter_turn());
95                let tangential = if Self::horizontal(side) { footprint.x } else { footprint.y };
96                let needed = if taken == 0 { tangential } else { gap + tangential };
97                if used + needed > edge_length {
98                    break;
99                }
100                used += needed;
101                taken += 1;
102            }
103            groups.push((side, pending.drain(..taken).collect()));
104        }
105        if let Some(&index) = pending.first() {
106            return Err(LayoutError::DoesNotFit { index, items: items.len() });
107        }
108        Ok(groups)
109    }
110
111    /// Even split over all sides, largest remainder to earlier sides.
112    fn balanced_groups(&self, count: usize) -> Vec<(TableSide, Vec<usize>)> {
113        let sides: Vec<TableSide> = self.sides.iter().collect();
114        let mut groups = Vec::with_capacity(sides.len());
115        let mut cursor = 0_usize;
116        for (position, side) in sides.iter().enumerate() {
117            let group_count = count / sides.len() + usize::from(position < count % sides.len());
118            groups.push((*side, (cursor..cursor + group_count).collect()));
119            cursor += group_count;
120        }
121        groups
122    }
123
124    /// Places one side's group centered on its edge and returns the
125    /// group's error when it does not fit.
126    fn place_group(&self, bounds: Rect<U>, items: &[LayoutItem<U>], side: TableSide, group: &[usize]) -> Result<Vec<(usize, Placement<U>)>, LayoutError> {
127        let gap = self.gap.as_f64();
128        let turn = side.to_quarter_turn();
129        let footprints: Vec<Vec2<U>> = group
130            .iter()
131            .map(|&index| footprint_for(items[index].size, items[index].rotation, turn))
132            .collect();
133        let edge_length = if Self::horizontal(side) { bounds.size.x } else { bounds.size.y };
134        let span = footprints
135            .iter()
136            .map(|footprint| if Self::horizontal(side) { footprint.x } else { footprint.y })
137            .sum::<f64>()
138            + gap * group.len().saturating_sub(1) as f64;
139        if span > edge_length {
140            return Err(LayoutError::DoesNotFit {
141                index: group.first().copied().unwrap_or(0),
142                items: items.len(),
143            });
144        }
145
146        let edge_midpoint = Self::edge_midpoint(bounds, side);
147        let right: Vec2<U> = side.right_direction();
148        let outward = Vec2::<U>::new(-right.y, right.x);
149        let rotation = Rotation::QuarterTurn(turn);
150        let mut cursor = -span / 2.0;
151        let mut placed = Vec::with_capacity(group.len());
152        for (&index, footprint) in group.iter().zip(&footprints) {
153            let (tangential, perpendicular) = if Self::horizontal(side) {
154                (footprint.x, footprint.y)
155            } else {
156                (footprint.y, footprint.x)
157            };
158            let center = edge_midpoint + right * (cursor + tangential / 2.0) + outward * (gap + perpendicular / 2.0);
159            placed.push((index, Placement::new(center, rotation, *footprint)));
160            cursor += tangential + gap;
161        }
162        Ok(placed)
163    }
164}
165
166/// Whether two placement footprints overlap (strictly - touching
167/// edges do not collide).
168fn overlaps<U>(a: &Placement<U>, b: &Placement<U>) -> bool {
169    (a.center.x - b.center.x).abs() * 2.0 < a.size.x + b.size.x && (a.center.y - b.center.y).abs() * 2.0 < a.size.y + b.size.y
170}

172impl<U> LayoutStrategy<U> for EdgeStripLayout<U> {
173    /// Per [`EdgeDistribution`]: `Sequential` fills side by side with
174    /// overflow spilling clockwise, `Balanced` centers an even share
175    /// on every selected edge. Placement rotation is the side's
176    /// quarter turn; `seat` is ignored. Strips overlapping at a
177    /// corner - or a group exceeding its edge - yield
178    /// [`LayoutError::DoesNotFit`].
179    fn arrange(&self, bounds: Rect<U>, items: &[LayoutItem<U>]) -> Result<Vec<Placement<U>>, LayoutError> {
180        let mut placements: Vec<Option<Placement<U>>> = vec![None; items.len()];
181        for (side, group) in self.groups(bounds, items)? {
182            for (index, placement) in self.place_group(bounds, items, side, &group)? {
183                placements[index] = Some(placement);
184            }
185        }
186        let placements: Vec<Placement<U>> = placements.into_iter().collect::<Option<Vec<_>>>().ok_or(LayoutError::DoesNotFit {
187            index: items.len(),
188            items: items.len(),
189        })?;
190        for (index, placement) in placements.iter().enumerate() {
191            if placements[..index].iter().any(|earlier| overlaps(earlier, placement)) {
192                return Err(LayoutError::DoesNotFit { index, items: items.len() });
193            }
194        }
195        Ok(placements)
196    }
197}

226    /// A single side docks one centered strip below the host, items
227    /// in input order along the side's reading direction.
228    #[test]
229    fn strip_docks_centered_below() {
230        let layout = EdgeStripLayout {
231            sides: SideSet::single(TableSide::Bottom),
232            distribution: EdgeDistribution::Sequential,
233            gap: Distance::new(10.0),
234        };
235        let placements = layout.arrange(bounds(), &[item(60.0, 30.0), item(40.0, 30.0)]).expect("fits");
236        // span = 60 + 10 + 40 = 110, centered on x=300: first item at 245+30=...
237        assert!((placements[0].center.x - (300.0 - 55.0 + 30.0)).abs() < 1e-9);
238        assert!((placements[1].center.x - (300.0 - 55.0 + 60.0 + 10.0 + 20.0)).abs() < 1e-9);
239        // Outside the bottom edge: 320 + gap 10 + half height 15.
240        assert!((placements[0].center.y - 345.0).abs() < 1e-9);
241        assert_eq!(placements[0].rotation, Rotation::QuarterTurn(QuarterTurn::Deg0));
242    }

244    /// Left and right strips order along the edge axis in each
245    /// side's own reading direction.
246    #[test]
247    fn vertical_strips_follow_reading_direction() {
248        let layout = EdgeStripLayout {
249            sides: SideSet::try_from_sides([TableSide::Left, TableSide::Right]).expect("non-empty"),
250            distribution: EdgeDistribution::Balanced,
251            gap: Distance::new(10.0),
252        };
253        let placements = layout.arrange(bounds(), &[item(30.0, 60.0), item(30.0, 60.0)]).expect("fits");
254        // Deg90/Deg270 swap the footprint axes: the 30x60 items display
255        // 60x30, so the strips sit 30 (perpendicular half) + gap off
256        // the edges - Left at x = 100 - 10 - 30 = 60, Right at 540.
257        assert!((placements[0].center.x - 60.0).abs() < 1e-9);
258        assert!((placements[1].center.x - 540.0).abs() < 1e-9);
259        // Left's right_direction is +y (down), Right's is -y (up).
260        assert!((placements[0].center.y - 200.0).abs() < 1e-9);
261        assert!((placements[1].center.y - 200.0).abs() < 1e-9);
262        assert_eq!(placements[0].rotation, Rotation::QuarterTurn(QuarterTurn::Deg90));
263        assert_eq!(placements[1].rotation, Rotation::QuarterTurn(QuarterTurn::Deg270));
264    }
