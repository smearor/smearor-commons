14/// Four-seat arrangement: every item lands in front of a table side,
15/// rotated for the user sitting there.
16///
17/// Items sharing a side spread along its edge with `gap` spacing
18/// (centered as a group); an explicit [`LayoutItem::seat`] wins over
19/// the `assignment` fallback.
20#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
21pub struct SeatLayout {
22    /// Assignment for items without an explicit `seat`.
23    pub assignment: SeatAssignment,
24    /// Footprint distance to the table edge - scalar in the
25    /// coordinate space of `bounds` (see [`crate::FlowLayout::gap`]).
26    pub edge_margin: f64,
27    /// Spacing between items sharing the same side - same space
28    /// as `bounds`.
29    pub gap: f64,
30}
31
32impl Default for SeatLayout {
33    fn default() -> Self {
34        Self {
35            assignment: SeatAssignment::RoundRobin,
36            edge_margin: 24.0,
37            gap: 24.0,
38        }
39    }
40}
41
42impl SeatLayout {
43    /// Whether the side's edge runs horizontally (`Bottom`/`Top`) or
44    /// vertically (`Left`/`Right`).
45    fn horizontal_edge(side: TableSide) -> bool {
46        matches!(side, TableSide::Bottom | TableSide::Top)
47    }
48}
49
50impl<U> LayoutStrategy<U> for SeatLayout {
51    /// Per side: pinned seats take their [`Seat::offset`], unpinned
52    /// items spread along the edge centered as a group, each
53    /// `edge_margin` off the edge, rotated to the side's orientation.
54    /// Extents along the edge use the footprint at the target
55    /// rotation ([`footprint_for`]). A spread group wider than its
56    /// edge reports [`LayoutError::DoesNotFit`].
57    fn arrange(&self, bounds: Rect<U>, items: &[LayoutItem<U>]) -> Result<Vec<Placement<U>>, LayoutError> {
58        let count = items.len();
59        let seats: Vec<Seat> = items
60            .iter()
61            .enumerate()
62            .map(|(index, item)| item.seat.unwrap_or_else(|| Seat::new(TableSide::ALL[index % TableSide::ALL.len()])))
63            .collect();
64
65        let mut placements: Vec<Option<Placement<U>>> = vec![None; count];
66        for side in TableSide::ALL {
67            // Pinned seats land verbatim on their offsets.
68            for (index, seat_ref) in seats.iter().enumerate() {
69                if seat_ref.side == side && seat_ref.offset.is_some() {
70                    let footprint = footprint_for(items[index].size, items[index].rotation, side.to_quarter_turn());
71                    placements[index] = Some(seat_ref.placement(bounds, footprint, self.edge_margin));
72                }
73            }
74
75            // The unpinned pool spreads along the edge as a group.
76            let group: Vec<usize> = (0..count).filter(|&index| seats[index].side == side && seats[index].offset.is_none()).collect();
77            if group.is_empty() {
78                continue;
79            }
80            // Footprints at the target rotation drive every extent.
81            let target: Vec<Vec2<U>> = group
82                .iter()
83                .map(|&index| footprint_for(items[index].size, items[index].rotation, side.to_quarter_turn()))
84                .collect();
85            let edge_extents: Vec<f64> = target
86                .iter()
87                .map(|footprint| if Self::horizontal_edge(side) { footprint.x } else { footprint.y })
88                .collect();
89            let span = edge_extents.iter().sum::<f64>() + self.gap * group.len().saturating_sub(1) as f64;
90            let edge_length = if Self::horizontal_edge(side) { bounds.size.x } else { bounds.size.y };
91            if span > edge_length {
92                return Err(LayoutError::DoesNotFit { index: group[0], items: count });
93            }
94
95            // Walk the edge from the group's centered start position.
96            let mut cursor = {
97                let center = if Self::horizontal_edge(side) {
98                    bounds.min.x + bounds.size.x / 2.0
99                } else {
100                    bounds.min.y + bounds.size.y / 2.0
101                };
102                center - span / 2.0
103            };
104            for (&index, footprint) in group.iter().zip(&target) {
105                let edge_extent = if Self::horizontal_edge(side) { footprint.x } else { footprint.y };
106                let mut placement = Seat::new(side).placement(bounds, *footprint, self.edge_margin);
107                placement.center = if Self::horizontal_edge(side) {
108                    Vec2::new(cursor + edge_extent / 2.0, placement.center.y)
109                } else {
110                    Vec2::new(placement.center.x, cursor + edge_extent / 2.0)
111                };
112                placements[index] = Some(placement);
113                cursor += edge_extent + self.gap;
114            }
115        }
116        placements
117            .into_iter()
118            .collect::<Option<Vec<_>>>()
119            .ok_or(LayoutError::DoesNotFit { index: count, items: count })
120    }
121}

136    fn bounds() -> Rect<Pixels> {
137        Rect {
138            min: Vec2::new(0.0, 0.0),
139            size: Vec2::new(400.0, 300.0),
140        }
141    }
142
143    fn item(width: f64, height: f64) -> LayoutItem<Pixels> {
144        LayoutItem::new(Vec2::new(width, height), Rotation::ZERO)
145    }
146
147    /// Round-robin deals items over `TableSide::ALL` in input order.
148    #[test]
149    fn round_robin_deals_to_all_sides() {
150        let items = [item(50.0, 50.0); 5];
151        let placements = SeatLayout::default().arrange(bounds(), &items).expect("fits");
152        for (index, placement) in placements.iter().enumerate() {
153            let expected = Rotation::QuarterTurn(TableSide::ALL[index % 4].to_quarter_turn());
154            assert_eq!(placement.rotation, expected, "item {index}");
155        }
156    }
157
158    /// An explicit seat wins over the round-robin order.
159    #[test]
160    fn explicit_seat_wins() {
161        let items = [item(100.0, 50.0).with_seat(Seat::new(TableSide::Right))];
162        let placements = SeatLayout::default().arrange(bounds(), &items).expect("fits");
163        assert_eq!(placements[0].rotation, Rotation::QuarterTurn(QuarterTurn::Deg270));
164        // Target footprint is 50x100 after the Deg270 axis swap.
165        assert!((placements[0].center.x - (400.0 - 24.0 - 25.0)).abs() < 1e-9);
166    }
167
168    /// Items sharing a side spread along its edge without overlap.
169    #[test]
170    fn same_side_items_spread() {
171        let items = [
172            item(100.0, 50.0).with_seat(Seat::new(TableSide::Bottom)),
173            item(100.0, 50.0).with_seat(Seat::new(TableSide::Bottom)),
174        ];
175        let placements = SeatLayout::default().arrange(bounds(), &items).expect("fits");
176        // Two 100-wide items + 24 gap centered on the 400-wide edge.
177        assert!((placements[0].center.x - (200.0 - 12.0 - 50.0)).abs() < 1e-9);
178        assert!((placements[1].center.x - (200.0 + 12.0 + 50.0)).abs() < 1e-9);
179    }
180
181    /// A group wider than its edge reports `DoesNotFit`.
182    #[test]
183    fn overflowing_side_errors() {
184        let items = [
185            item(300.0, 50.0).with_seat(Seat::new(TableSide::Top)),
186            item(300.0, 50.0).with_seat(Seat::new(TableSide::Top)),

188        assert!(SeatLayout::default().arrange(bounds(), &items).is_err());
189    }
190
191    /// Rotated items occupy their *target* footprint on the edge:
192    /// a 200-wide item at Left (Deg90) needs 200 of vertical edge.
193    #[test]
194    fn target_footprint_drives_edge_extents() {
195        let items = [
196            item(200.0, 50.0).with_seat(Seat::new(TableSide::Left)),
197            item(200.0, 50.0).with_seat(Seat::new(TableSide::Left)),
198        ];
199        // Two 50x200 footprints + 24 gap = 424 > 300 edge length.
200        assert!(SeatLayout::default().arrange(bounds(), &items).is_err());
201        let ok = [
202            item(100.0, 50.0).with_seat(Seat::new(TableSide::Left)),
203            item(100.0, 50.0).with_seat(Seat::new(TableSide::Left)),
204        ];
205        assert!(SeatLayout::default().arrange(bounds(), &ok).is_ok());
206    }
