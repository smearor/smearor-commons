6/// An arrangement strategy: maps items onto target placements inside
7/// `bounds`.
8///
9/// Strategies are pure functions over technical types - they see
10/// footprints, not widgets or domain objects, and stay consumer-
11/// extensible: any crate can implement this trait and hand the
12/// strategy to a shared caller (e.g. `FloatingBoard::apply_layout`).
13///
14/// Input and output are index-stable: `placements[i]` answers
15/// `items[i]`, so the caller maps back onto its own item list by
16/// position. Strategies decide per item which input fields they
17/// honor (`rotation`, `seat`) and which they override.
18pub trait LayoutStrategy<U> {
19    /// Computes the target [`Placement`] for each item.
20    ///
21    /// Returns [`LayoutError::DoesNotFit`] when the requested options
22    /// cannot fit every item inside `bounds`; the consumer decides
23    /// how to react.
24    fn arrange(&self, bounds: Rect<U>, items: &[LayoutItem<U>]) -> Result<Vec<Placement<U>>, LayoutError>;
25}
