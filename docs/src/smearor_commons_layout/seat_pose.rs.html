11/// A [`Seat`] plus the free radial orientation of the user sitting
12/// there.
13///
14/// `Seat` alone carries the *organizational* alignment - which edge,
15/// where along it. `orientation` carries the direction the seated
16/// user faces as a free angle, not snapped to quarter turns, so a
17/// seat in a table corner can face diagonally into the surface.
18/// Placements made through this type set `Rotation::Free(orientation)`
19/// instead of the edge's `QuarterTurn` - that is what makes corner
20/// seats usable.
21///
22/// Technical like `Seat`: which *user* sits there stays consumer
23/// domain. No `zvariant::Type` impl - a `SeatPose` travels D-Bus as a
24/// `(TableSide, f64, f64)` triple with `NaN` for an unpinned offset
25/// (see [`Seat`]).
26#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
27#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject, async_graphql::InputObject))]
28#[cfg_attr(feature = "async-graphql", graphql(input_name = "SeatPoseInput"))]
29#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
30pub struct SeatPose {
31    /// Organizational alignment - the table edge plus the position
32    /// along it.
33    pub seat: Seat,
34    /// Radial orientation the seated user faces, in radians - free
35    /// angle, not snapped to [`crate::TableSide`]'s quarter turns.
36    pub orientation: RotationRad,
37}
38
39impl Default for SeatPose {
40    /// The bottom edge, unpinned, facing `RotationRad::ZERO`.
41    fn default() -> Self {
42        Self {
43            seat: Seat::default(),
44            orientation: RotationRad::ZERO,
45        }
46    }
47}
48
49impl SeatPose {
50    /// A pose for `seat` facing `orientation`.
51    pub const fn new(seat: Seat, orientation: RotationRad) -> Self {
52        Self { seat, orientation }
53    }
54
55    /// The whole pose derived from a physical marker pose: the edge
56    /// alignment from [`Seat::from_position`], `orientation` passed
57    /// through - the one-step derivation for tangible seat markers.
58    pub fn from_position<U>(position: Vec2<U>, bounds: Rect<U>, orientation: RotationRad) -> Self {
59        Self {
60            seat: Seat::from_position(position, bounds),
61            orientation,
62        }
63    }
64
65    /// The pose one step clockwise (`side` steps, `offset` and
66    /// `orientation` keep).
67    pub fn clockwise(&self) -> Self {
68        Self {
69            seat: self.seat.clockwise(),
70            ..*self
71        }
72    }
73
74    /// The pose one step counter-clockwise.
75    pub fn counter_clockwise(&self) -> Self {
76        Self {
77            seat: self.seat.counter_clockwise(),
78            ..*self
79        }
80    }
81
82    /// The seat's position along the table perimeter - delegates to
83    /// [`Seat::perimeter_position`].
84    pub fn perimeter_position(&self) -> f64 {
85        self.seat.perimeter_position()
86    }
87
88    /// The occupied seat one step clockwise along the table perimeter -
89    /// delegates to [`Seat::next_in`]. Roster walking works on
90    /// `&[Seat]`; consumers holding `&[SeatPose]` map `pose.seat`.
91    pub fn next_in(&self, roster: &[Seat]) -> Option<Seat> {
92        self.seat.next_in(roster)
93    }
94
95    /// The occupied seat one step counter-clockwise - delegates to
96    /// [`Seat::previous_in`].
97    pub fn previous_in(&self, roster: &[Seat]) -> Option<Seat> {
98        self.seat.previous_in(roster)
99    }
100
101    /// Places `footprint` in front of the seat like [`Seat::placement`],
102    /// but rotated to `Rotation::Free(self.orientation)` instead of
103    /// the edge's quarter turn - a corner seat's content reads
104    /// diagonally to its user.
105    pub fn placement<U>(&self, bounds: Rect<U>, footprint: Vec2<U>, margin: impl Into<Distance<U>>) -> Placement<U> {
106        let mut placement = self.seat.placement(bounds, footprint, margin);
107        placement.rotation = Rotation::Free(self.orientation);
108        placement
109    }
110
111    /// Places `footprint` in the best free spot nearest the seat like
112    /// [`Seat::fanned_placement`], with the same `Rotation::Free`
113    /// override as [`Self::placement`].
114    pub fn fanned_placement<U>(
115        &self,
116        bounds: Rect<U>,
117        footprint: Vec2<U>,
118        margin: impl Into<Distance<U>>,
119        taken: &[(Vec2<U>, Vec2<U>)],
120        gap: impl Into<Distance<U>>,
121    ) -> Placement<U> {
122        let mut placement = self.seat.fanned_placement(bounds, footprint, margin, taken, gap);
123        placement.rotation = Rotation::Free(self.orientation);
124        placement
125    }
126}

141    fn bounds() -> Rect<Pixels> {
142        Rect {
143            min: Vec2::new(0.0, 0.0),
144            size: Vec2::new(400.0, 300.0),
145        }
146    }
147
148    /// `from_position` derives the seat from the position and carries
149    /// the free orientation through.
150    #[test]
151    fn from_position_derives_pose() {
152        let pose = SeatPose::from_position(Vec2::new(100.0, 290.0), bounds(), RotationRad::from_degrees(30.0));
153        assert_eq!(pose.seat, Seat::at(TableSide::Bottom, Fraction::new(0.25)));
154        assert_eq!(pose.orientation, RotationRad::from_degrees(30.0));
155    }
156
157    /// A corner marker resolves to the nearer edge while the
158    /// orientation keeps the diagonal angle.
159    #[test]
160    fn from_position_corner_keeps_diagonal() {
161        let pose = SeatPose::from_position(Vec2::new(395.0, 290.0), bounds(), RotationRad::from_degrees(135.0));
162        assert_eq!(pose.seat.side, TableSide::Right);
163        assert_eq!(pose.orientation, RotationRad::from_degrees(135.0));
164    }
165
166    /// `placement` matches the seat's anchor but rotates freely.
167    #[test]
168    fn placement_rotates_freely() {
169        let pose = SeatPose::new(Seat::center(TableSide::Bottom), RotationRad::from_degrees(45.0));
170        let placement = pose.placement(bounds(), Vec2::new(100.0, 50.0), 24.0);
171        assert_eq!(placement.rotation, Rotation::Free(RotationRad::from_degrees(45.0)));
172        assert_eq!(placement.center, Seat::center(TableSide::Bottom).placement(bounds(), Vec2::new(100.0, 50.0), 24.0).center);
173    }
174
175    /// `fanned_placement` applies the same free-rotation override.
176    #[test]
177    fn fanned_placement_rotates_freely() {
178        let pose = SeatPose::new(Seat::center(TableSide::Top), RotationRad::from_degrees(10.0));
179        let placement = pose.fanned_placement(bounds(), Vec2::new(50.0, 50.0), 10.0, &[], 10.0);
180        assert_eq!(placement.rotation, Rotation::Free(RotationRad::from_degrees(10.0)));
181    }
182
183    /// Stepping keeps offset and orientation while the side moves.
184    #[test]
185    fn clockwise_keeps_offset_and_orientation() {
186        let pose = SeatPose::new(Seat::at(TableSide::Bottom, Fraction::new(0.25)), RotationRad::from_degrees(70.0));
187        assert_eq!(pose.clockwise().seat, Seat::at(TableSide::Left, Fraction::new(0.25)));
188        assert_eq!(pose.clockwise().orientation, RotationRad::from_degrees(70.0));
189        assert_eq!(pose.counter_clockwise().seat, Seat::at(TableSide::Right, Fraction::new(0.25)));
190    }
191
192    /// Roster walking delegates to the inner seat.
193    #[test]
194    fn roster_walk_delegates() {
195        let pose = SeatPose::new(Seat::center(TableSide::Bottom), RotationRad::ZERO);
196        let roster = [Seat::center(TableSide::Top)];
197        assert_eq!(pose.next_in(&roster), Some(Seat::center(TableSide::Top)));
198        assert_eq!(pose.previous_in(&roster), Some(Seat::center(TableSide::Top)));
199    }
200
201    /// The default pose sits unpinned at the bottom edge facing zero.
202    #[test]
203    fn default_is_bottom_zero() {
204        let pose = SeatPose::default();
205        assert_eq!(pose.seat, Seat::default());
206        assert_eq!(pose.orientation, RotationRad::ZERO);
207        assert_eq!(pose.seat.side.to_quarter_turn(), QuarterTurn::Deg0);
208    }
