1//! `Simplex3DFractal` - the fixed four-octave variant of
2//! [`Simplex3D`].
3//!
4//! Based on the MIT-licensed variant at
5//! <https://www.shadertoy.com/view/XsX3zB> (Copyright (c) 2013 Nikita
6//! Miropolskiy): the domain is rotated by three fixed `mat3` matrices
7//! between octaves to break up directional artifacts, weights
8//! `0.5333/0.2667/0.1333/0.0667`.
9
10use super::Simplex3D;
11use crate::Field;
12use smearor_commons_geometry::Vec3;
13
14/// The fixed four-octave fractal variant of [`Simplex3D`] - rotates the
15/// domain by three fixed `mat3` matrices between octaves to break up
16/// directional artifacts, weights `0.5333/0.2667/0.1333/0.0667`.
17#[derive(Debug, Clone, Copy, Default)]
18pub struct Simplex3DFractal;
19
20/// mat3-by-column rotation: `cols[0]*v.x + cols[1]*v.y + cols[2]*v.z`.
21fn rotate(cols: [Vec3<()>; 3], v: Vec3<()>) -> Vec3<()> {
22    cols[0] * v.x + cols[1] * v.y + cols[2] * v.z
23}
24
25/// The three octave rotations (GLSL mat3 constructors are
26/// column-major).
27const ROT1: [Vec3<()>; 3] = [
28    Vec3::<()>::new(-0.37, 0.36, 0.85),
29    Vec3::<()>::new(-0.14, -0.93, 0.34),
30    Vec3::<()>::new(0.92, 0.01, 0.4),
31];
32const ROT2: [Vec3<()>; 3] = [
33    Vec3::<()>::new(-0.55, -0.39, 0.74),
34    Vec3::<()>::new(0.33, -0.91, -0.24),
35    Vec3::<()>::new(0.77, 0.12, 0.63),
36];
37const ROT3: [Vec3<()>; 3] = [
38    Vec3::<()>::new(-0.71, 0.52, -0.47),
39    Vec3::<()>::new(-0.08, -0.72, -0.68),
40    Vec3::<()>::new(-0.7, -0.45, 0.56),
41];
42
43impl Field<Vec3<()>> for Simplex3DFractal {
44    fn at(&self, m: Vec3<()>) -> f64 {
45        0.5333333 * Simplex3D.at(rotate(ROT1, m))
46            + 0.2666667 * Simplex3D.at(rotate(ROT2, m * 2.0))
47            + 0.1333333 * Simplex3D.at(rotate(ROT3, m * 4.0))
48            + 0.0666667 * Simplex3D.at(m * 8.0)
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55
56    #[test]
57    fn fractal_rotates_octaves() {
58        let a = Simplex3DFractal.at(Vec3::<()>::new(0.31, -0.7, 0.2));
59        assert_ne!(a, Simplex3D.at(Vec3::<()>::new(0.31, -0.7, 0.2)));
60        assert!(a.is_finite());
61    }
