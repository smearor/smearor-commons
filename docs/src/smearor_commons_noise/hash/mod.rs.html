1//! `Hash` - the Hoskins "hash without sine" family as a generic trait.
2//!
3//! MIT License, <https://www.shadertoy.com/view/4djSRW> - Copyright (c)
4//! 2014 David Hoskins. The recipe is `fract`/multiply/add only: no trig,
5//! no tables - bitwise identical output on every platform.
6
7/// The Hoskins fract-hash family.
8///
9/// `p.hash()` returns the `Out`-shaped hash of `p` with every component
10/// in `[0, 1)`. The 13 impls mirror the GLSL `hashNN` naming (input
11/// dimension + output dimension); each impl's rustdoc names its
12/// `hashNN` designation. This is *not* [`std::hash::Hash`] - the
13/// no-arg `Out`-returning signature never collides at call sites.
14pub trait Hash<Out> {
15    /// The deterministic hash value in `[0, 1)` per component.
16    fn hash(&self) -> Out;
17}
18
19/// GLSL `fract(x) = x - x.floor()` - exact for negatives (unlike
20/// `rem_euclid` variants that behave differently around integers).
21pub(crate) fn fract(x: f64) -> f64 {
22    x - x.floor()
23}

30#[cfg(test)]
31mod tests {
32
33    use super::*;
34    use smearor_commons_geometry::Vec2;
35    use smearor_commons_geometry::Vec3;
36    use smearor_commons_geometry::Vec4;
37
38    /// Regression pins against the ported Hoskins formula - if any of
39    /// these drift, the recipe constants changed.
40    #[test]
41    fn known_values() {
42        let h11: f64 = 1.0f64.hash();
43        assert!((h11 - 0.7629645911716167).abs() < 1e-12);
44        let h12: f64 = Vec2::<()>::new(0.5, -1.25).hash();
45        assert!((h12 - 0.1989645971389109).abs() < 1e-12);
46        let h13: f64 = Vec3::<()>::new(1.0, 2.0, 3.0).hash();
47        assert!((h13 - 0.17883083253718723).abs() < 1e-12);
48        let h22: Vec2<()> = Vec2::<()>::new(0.5, -1.25).hash();
49        assert!((h22.x - 0.538394836124553).abs() < 1e-12 && (h22.y - 0.28917051798407556).abs() < 1e-12);
50        let h44: Vec4<()> = Vec4::<()>::new(1.0, 2.0, 3.0, 4.0).hash();
51        assert!((h44.x - 0.5798467050076397).abs() < 1e-12 && (h44.w - 0.12897491332751088).abs() < 1e-12);
52    }
53
54    /// Rough uniformity: mean ~ 0.5, variance ~ 1/12 over a 10k grid.
55    #[test]
56    fn hash_is_roughly_uniform() {
57        let mut sum = 0.0;
58        let mut sq = 0.0;
59        let n = 10000.0;
60        for i in 0..10000 {
61            let v: f64 = (i as f64 * 0.173).hash();
62            sum += v;
63            sq += v * v;
64        }
65        let mean = sum / n;
66        let variance = sq / n - mean * mean;
67        assert!((mean - 0.5).abs() < 0.02, "mean {mean}");
68        assert!((variance - 1.0 / 12.0).abs() < 0.01, "variance {variance}");
69    }
