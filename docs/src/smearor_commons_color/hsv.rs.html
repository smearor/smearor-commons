10/// An HSV color: hue in degrees (`0-360`), saturation and value in
11/// `0.0-1.0`.
12///
13/// `Hsv` is the technical representation used by color-wheel
14/// interfaces: hue walks the wheel, saturation runs from gray to
15/// full chroma, value from black to full brightness. The `From`
16/// conversions perform the standard sRGB-space HSV round trip; a
17/// saturation of `0` has no hue and round-trips as `h = 0`.
18#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
19#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
20pub struct Hsv {
21    /// Hue in degrees (`0-360`, wraps around).
22    pub h: Hue,
23    /// Saturation (`0.0` gray to `1.0` full chroma).
24    pub s: Saturation,
25    /// Value (`0.0` black to `1.0` full brightness).
26    pub v: Value,
27}
28
29impl Hsv {
30    /// Creates an HSV color from its channels.
31    pub fn new(h: impl Into<Hue>, s: impl Into<Saturation>, v: impl Into<Value>) -> Self {
32        Self {
33            h: h.into(),
34            s: s.into(),
35            v: v.into(),
36        }
37    }
38}

40impl Display for Hsv {
41    /// `hsv(h, s%, v%)` - saturation and value in percent.
42    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
43        write!(f, "hsv({:.1}, {:.1}%, {:.1}%)", self.h.as_f64(), self.s.as_f64() * 100.0, self.v.as_f64() * 100.0)
44    }
45}

47/// The sRGB value of an HSV color (sector decomposition, alpha
48/// opaque).
49impl From<Hsv> for Color<f64> {
50    fn from(hsv: Hsv) -> Self {
51        let (h, s, v) = (hsv.h.as_f64() / 60.0, hsv.s.as_f64(), hsv.v.as_f64());
52        let c = v * s;
53        let x = c * (1.0 - (h.rem_euclid(2.0) - 1.0).abs());
54        let m = v - c;
55        let (r, g, b) = match h as u32 {
56            0 => (c, x, 0.0),
57            1 => (x, c, 0.0),
58            2 => (0.0, c, x),
59            3 => (0.0, x, c),
60            4 => (x, 0.0, c),
61            _ => (c, 0.0, x),
62        };
63        Color::<f64>::new(r + m, g + m, b + m)
64    }
65}

81/// The HSV decomposition of an sRGB color: `h` from max-channel
82/// sector, `s = delta / max`, `v = max`. Achromatic colors yield
83/// `h = 0, s = 0`.
84impl From<Color<f64>> for Hsv {
85    fn from(color: Color<f64>) -> Self {
86        let max = color.r.max(color.g).max(color.b);
87        let min = color.r.min(color.g).min(color.b);
88        let delta = max - min;
89        let h = if delta <= 0.0 {
90            0.0
91        } else if max == color.r {
92            60.0 * ((color.g - color.b) / delta).rem_euclid(6.0)
93        } else if max == color.g {
94            60.0 * ((color.b - color.r) / delta + 2.0)
95        } else {
96            60.0 * ((color.r - color.g) / delta + 4.0)
97        };
98        Self::new(h, if max > 0.0 { delta / max } else { 0.0 }, max)
99    }
100}

116#[cfg(test)]
117mod tests {
118    use super::*;
119
120    #[test]
121    fn primaries_map_to_their_sectors() {
122        assert_eq!(Hsv::from(Color::<u8>::new(255, 0, 0)).h.as_f64(), 0.0);
123        assert_eq!(Hsv::from(Color::<u8>::new(0, 255, 0)).h.as_f64(), 120.0);
124        assert_eq!(Hsv::from(Color::<u8>::new(0, 0, 255)).h.as_f64(), 240.0);
125        assert_eq!(Hsv::from(Color::<u8>::new(255, 255, 0)).h.as_f64(), 60.0);
126    }
127
128    #[test]
129    fn achromatic_has_no_hue() {
130        let hsv = Hsv::from(Color::<u8>::new(128, 128, 128));
131        assert_eq!(hsv.h, Hue::ZERO);
132        assert_eq!(hsv.s, Saturation::MIN);
133        assert!((hsv.v.as_f64() - 128.0 / 255.0).abs() < 1e-9);
134    }
135
136    #[test]
137    fn hue_wraps_and_channels_clamp() {
138        assert_eq!(Hsv::new(370.0, 0.5, 0.5).h.as_f64(), 10.0);
139        assert_eq!(Hsv::new(-10.0, 0.5, 0.5).h.as_f64(), 350.0);
140        assert_eq!(
141            Hsv::new(0.0, 2.0, -1.0),
142            Hsv {
143                h: Hue::ZERO,
144                s: Saturation::MAX,
145                v: Value::MIN,
146            }
147        );
148    }
149
150    #[test]
151    fn round_trip_keeps_the_color() {
152        for color in [Color::<u8>::new(255, 0, 0), Color::<u8>::new(18, 130, 200), Color::<u8>::new(200, 40, 180)] {
153            let back = Color::<u8>::from(Hsv::from(color));
154            assert!((i32::from(back.r) - i32::from(color.r)).abs() <= 1);
155            assert!((i32::from(back.g) - i32::from(color.g)).abs() <= 1);
156            assert!((i32::from(back.b) - i32::from(color.b)).abs() <= 1);
157        }
158    }
