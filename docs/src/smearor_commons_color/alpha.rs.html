1/// A normalized alpha value (0.0-1.0), clamped on construction.
2///
3/// `Alpha` is a unit: the canonical representation is always `f64`;
4/// `u8`/`f32` channel values are encodings of it and convert via
5/// `From`/`as_u8`/`as_f32`. The `a` channel of [`crate::Color<T>`]
6/// stays `T` — `Alpha` is the standalone scalar for APIs that take
7/// or return an opacity.
8use std::ops::Div;
9use std::ops::Mul;
10use std::ops::Sub;
11
12#[derive(Debug, Clone, Copy, PartialEq, PartialOrd)]
13pub struct Alpha(f64);
14
15impl Alpha {
16    /// Fully transparent.
17    pub const TRANSPARENT: Self = Self(0.0);
18
19    /// Fully opaque.
20    pub const OPAQUE: Self = Self(1.0);
21
22    /// Creates an alpha value, clamped to `0.0-1.0`.
23    pub const fn new(value: f64) -> Self {
24        Self(if value < 0.0 {
25            0.0
26        } else if value > 1.0 {
27            1.0
28        } else {
29            value
30        })
31    }
32
33    /// The numeric value.
34    pub const fn as_f64(self) -> f64 {
35        self.0
36    }
37
38    /// The value as an `f32` channel.
39    pub fn as_f32(self) -> f32 {
40        self.0 as f32
41    }
42
43    /// The value as a `u8` channel (0-255, rounded).
44    pub fn as_u8(self) -> u8 {
45        (self.0 * 255.0).round() as u8
46    }
47
48    /// Creates an alpha from a percent value (`0-100`, clamped).
49    pub const fn from_percent(percent: u8) -> Self {
50        Self::new(if percent > 100 { 1.0 } else { percent as f64 / 100.0 })
51    }
52
53    /// The value in percent (`0-100`, rounded).
54    pub fn percent(self) -> u8 {
55        (self.0 * 100.0).round() as u8
56    }
57
58    /// Whether the alpha is fully transparent - the rasterizer can
59    /// skip the pixel entirely.
60    pub fn is_transparent(self) -> bool {
61        self.0 <= 0.0
62    }
63
64    /// Whether the alpha is fully opaque - blending can write the
65    /// source color directly.
66    pub fn is_opaque(self) -> bool {
67        self.0 >= 1.0
68    }
69}
70
71/// Saturating difference - a negative result clamps to [`Alpha::TRANSPARENT`].
72impl Sub for Alpha {
73    type Output = Self;
74
75    fn sub(self, rhs: Self) -> Self {
76        Self::new(self.0 - rhs.0)
77    }
78}
79
80/// Scales the alpha by a plain factor (e.g. per-pixel coverage),
81/// clamped back to the unit range.
82impl Mul<f64> for Alpha {
83    type Output = Self;
84
85    fn mul(self, rhs: f64) -> Self {
86        Self::new(self.0 * rhs)
87    }
88}
89
90/// Ratio of two alpha values as a plain factor (e.g. a blend amount).
91impl Div for Alpha {
92    type Output = f64;
93
94    fn div(self, rhs: Self) -> Self::Output {
95        self.0 / rhs.0
96    }
97}
98
99impl Default for Alpha {
100    fn default() -> Self {
101        Self::OPAQUE
102    }
103}
104
105impl From<f64> for Alpha {
106    fn from(value: f64) -> Self {
107        Self::new(value)
108    }
109}
110
111impl From<f32> for Alpha {
112    fn from(value: f32) -> Self {
113        Self::new(f64::from(value))
114    }
115}
116
117impl From<u8> for Alpha {
118    fn from(value: u8) -> Self {
119        Self::new(f64::from(value) / 255.0)
120    }
121}
122
123impl From<Alpha> for f64 {
124    fn from(alpha: Alpha) -> Self {
125        alpha.0
126    }
127}
128
129impl From<Alpha> for f32 {
130    fn from(alpha: Alpha) -> Self {
131        alpha.0 as f32
132    }
133}
134
135impl From<Alpha> for u8 {
136    fn from(alpha: Alpha) -> Self {
137        alpha.as_u8()
138    }
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144
145    #[test]
146    fn new_clamps_to_unit_range() {
147        assert_eq!(Alpha::new(1.5), Alpha::OPAQUE);
148        assert_eq!(Alpha::new(-0.5), Alpha::TRANSPARENT);
149    }
150
151    #[test]
152    fn u8_round_trip() {
153        assert_eq!(Alpha::from(128u8).as_u8(), 128);
154        assert_eq!(u8::from(Alpha::OPAQUE), 255);
155    }
156
157    #[test]
158    fn percent_round_trip() {
159        assert_eq!(Alpha::from_percent(75).percent(), 75);
160        assert!((Alpha::from_percent(75).as_f64() - 0.75).abs() < 1e-9);
161        assert_eq!(Alpha::from_percent(255), Alpha::OPAQUE);
162        assert_eq!(Alpha::OPAQUE.percent(), 100);
163    }
164
165    #[test]
166    fn default_is_opaque() {
167        assert_eq!(Alpha::default(), Alpha::OPAQUE);
168    }
169
170    #[test]
171    fn sub_saturates_at_transparent() {
172        let delta = Alpha::OPAQUE - Alpha::new(0.25);
173        assert!((delta.as_f64() - 0.75).abs() < 1e-9);
174        assert_eq!(Alpha::new(0.25) - Alpha::OPAQUE, Alpha::TRANSPARENT);
175    }
176
177    #[test]
178    fn predicates() {
179        assert!(Alpha::TRANSPARENT.is_transparent());
180        assert!(Alpha::OPAQUE.is_opaque());
181        assert!(!Alpha::OPAQUE.is_transparent());
182        assert!(!Alpha::TRANSPARENT.is_opaque());
183    }
184
185    #[test]
186    fn mul_scales_and_clamps() {
187        assert_eq!(Alpha::OPAQUE * 0.5, Alpha::new(0.5));
188        assert_eq!(Alpha::OPAQUE * 2.0, Alpha::OPAQUE);
189    }
190
191    #[test]
192    fn div_yields_ratio() {
193        assert!((Alpha::new(0.5) / Alpha::new(0.25) - 2.0).abs() < 1e-9);
194        assert_eq!(Alpha::new(0.5) / Alpha::OPAQUE, 0.5);
195    }
196}
