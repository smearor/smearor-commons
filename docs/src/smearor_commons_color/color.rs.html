15/// Generic RGBA color with four components of type `T`.
16///
17/// Use `Color<f64>` or `Color<f32>` for normalized channel values in
18/// `[0.0, 1.0]` (CSS/GTK/cairo operations) and `Color<u8>` for
19/// pixel-native channel values in `[0, 255]` (software rendering).
20///
21/// Conversions between the instantiations are provided via `From`/`Into`.
22/// Parsing and formatting use `FromStr`/`Display`. Parsing accepts all
23/// common hex forms (`#RGB`, `#RGBA`, `#RRGGBB` and `#RRGGBBAA` - the
24/// leading `#` is mandatory, see [`HexColor`]) and the CSS Color Level 4
25/// named colors (`"red"`, `"rebeccapurple"`, `"transparent"`, ASCII
26/// case-insensitive, see [`CssNamedColor`]).
27///
28/// Semantic color types (`FigureColor`, `LineColor`, `PawnColor`, ...)
29/// stay in consumer crates as thin newtype wrappers around `Color<T>`.
30#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
31pub struct Color<T> {
32    /// Red component.
33    pub r: T,
34    /// Green component.
35    pub g: T,
36    /// Blue component.
37    pub b: T,
38    /// Alpha component.
39    pub a: T,
40}

46impl Color<f64> {
47    /// Creates a new opaque color from RGB components (alpha = 1.0).
48    pub const fn new(r: f64, g: f64, b: f64) -> Self {
49        Self { r, g, b, a: 1.0 }
50    }
51
52    /// Creates a new color from RGBA components.
53    pub const fn new_rgba(r: f64, g: f64, b: f64, a: f64) -> Self {
54        Self { r, g, b, a }
55    }
56
57    /// Black.
58    pub const BLACK: Self = Self::new(0.0, 0.0, 0.0);
59
60    /// White.
61    pub const WHITE: Self = Self::new(1.0, 1.0, 1.0);
62
63    /// The perceived luminance with custom `weights` (0.0-1.0).
64    pub fn luminance(&self, weights: LuminanceWeights) -> Luminance {
65        Luminance::new(self.r * weights.r + self.g * weights.g + self.b * weights.b)
66    }
67
68    /// The perceived luminance with ITU-R BT.709 weights (0.0-1.0).
69    pub fn luminance_709(&self) -> Luminance {
70        self.luminance(LuminanceWeights::REC_709)
71    }
72
73    /// The perceived luminance with ITU-R BT.601 weights (0.0-1.0).
74    pub fn luminance_601(&self) -> Luminance {
75        self.luminance(LuminanceWeights::REC_601)
76    }
77
78    /// Linear blend toward `other` by `t` (0 = self, 1 = other).
79    /// All four channels are interpolated; `t` is clamped to `[0, 1]`.
80    pub fn lerp(&self, other: &Self, t: f64) -> Self {
81        let t = t.clamp(0.0, 1.0);
82        Self {
83            r: self.r + (other.r - self.r) * t,
84            g: self.g + (other.g - self.g) * t,
85            b: self.b + (other.b - self.b) * t,
86            a: self.a + (other.a - self.a) * t,
87        }
88    }
89
90    /// Porter-Duff `over` with the alpha carried by both colors:
91    /// `self` composited over `under`. The sibling of `Color<u8>::alpha_over`,
92    /// which instead takes an external coverage alpha - here the source's
93    /// own alpha channel decides how much it covers (GLSL `alphaOver`
94    /// semantics for straight-alpha shader compositing).
95    ///
96    /// `out.a = src.a + dst.a * (1 - src.a)`; channels are weighted by
97    /// their own alpha, unpremultiplied by `out.a`. A fully transparent
98    /// pair yields transparent black.
99    pub fn alpha_over(&self, under: &Self) -> Self {
100        let out_a = self.a + under.a * (1.0 - self.a);
101        if out_a < 1e-6 {
102            return Self::new_rgba(0.0, 0.0, 0.0, 0.0);
103        }
104        let blend = |s: f64, d: f64| (s * self.a + d * under.a * (1.0 - self.a)) / out_a;
105        Self::new_rgba(blend(self.r, under.r), blend(self.g, under.g), blend(self.b, under.b), out_a)
106    }
107
108    /// The color scaled so its largest channel is `1.0` - the
109    /// brightest displayable color at this hue ratio (the gamut
110    /// edge). Black stays black.
111    pub fn normalized(&self) -> Self {
112        let max = self.r.max(self.g).max(self.b);
113        if max <= 0.0 {
114            *self
115        } else {
116            Self::new(self.r / max, self.g / max, self.b / max)
117        }
118    }
119
120    /// Lightens the color by blending each channel toward white by
121    /// `factor` (0 = unchanged, 1 = white). Alpha is preserved.
122    pub fn lighten(&self, factor: f64) -> Self {
123        let factor = factor.clamp(0.0, 1.0);
124        Self {
125            r: self.r + (1.0 - self.r) * factor,
126            g: self.g + (1.0 - self.g) * factor,
127            b: self.b + (1.0 - self.b) * factor,
128            a: self.a,
129        }
130    }
131
132    /// Adds `amount` to every color channel - an additive lift,
133    /// unlike [`Self::lighten`]'s blend toward white. Channels are
134    /// not clamped; use [`Self::clamp`] when the result must stay in
135    /// `[0, 1]`.
136    pub fn brighten(&self, amount: f64) -> Self {
137        Self {
138            r: self.r + amount,
139            g: self.g + amount,
140            b: self.b + amount,
141            a: self.a,
142        }
143    }
144
145    /// The color lifted to at least `min` luminance (BT.709) by
146    /// blending toward white - dark colors stay readable on dark
147    /// backgrounds. Alpha is preserved.
148    pub fn with_min_luminance(&self, min: impl Into<Luminance>) -> Self {
149        let min = min.into();
150        let luminance = self.luminance_709();
151        if luminance >= min {
152            return *self;
153        }
154        self.lighten((min - luminance) / (Luminance::MAX - luminance))
155    }
156
157    /// Clamps all four channels to `[0.0, 1.0]`.
158    pub fn clamp(&self) -> Self {
159        Self {
160            r: self.r.clamp(0.0, 1.0),
161            g: self.g.clamp(0.0, 1.0),
162            b: self.b.clamp(0.0, 1.0),
163            a: self.a.clamp(0.0, 1.0),
164        }
165    }
166
167    /// The color with its alpha channel replaced by `alpha` - for
168    /// fades driven by an external opacity (disabled state, hover
169    /// highlight) rather than the color's own alpha.
170    pub fn with_alpha(&self, alpha: impl Into<Alpha>) -> Self {
171        Self::new_rgba(self.r, self.g, self.b, alpha.into().as_f64())
172    }
173
174    /// The color with its alpha channel scaled by `factor` (0.0-1.0) -
175    /// relative dimming that keeps the color's own alpha. Unlike
176    /// [`Self::with_alpha`], a faded color stays faded when the slot
177    /// already carries transparency.
178    pub fn fade(&self, factor: impl Into<Alpha>) -> Self {
179        Self::new_rgba(self.r, self.g, self.b, self.a * factor.into().as_f64())
180    }
181}

310impl Color<u8> {
311    /// Creates a new opaque color from RGB components (alpha = 255).
312    pub const fn new(r: u8, g: u8, b: u8) -> Self {
313        Self { r, g, b, a: 255 }
314    }
315
316    /// Creates a new color from RGBA components.
317    pub const fn new_rgba(r: u8, g: u8, b: u8, a: u8) -> Self {
318        Self { r, g, b, a }
319    }
320
321    /// Black.
322    pub const BLACK: Self = Self::new(0, 0, 0);
323
324    /// White.
325    pub const WHITE: Self = Self::new(255, 255, 255);
326
327    /// Creates an opaque color from a packed `0xRRGGBB` value -
328    /// `const`-usable, e.g. in `static_palette!` definitions.
329    pub const fn from_rgb_u32(value: u32) -> Self {
330        Self {
331            r: ((value >> 16) & 0xFF) as u8,
332            g: ((value >> 8) & 0xFF) as u8,
333            b: (value & 0xFF) as u8,
334            a: 255,
335        }
336    }
337
338    /// Creates a color from a packed `0xRRGGBBAA` value -
339    /// `const`-usable, e.g. in `static_palette!` definitions.
340    pub const fn from_rgba_u32(value: u32) -> Self {
341        Self {
342            r: ((value >> 24) & 0xFF) as u8,
343            g: ((value >> 16) & 0xFF) as u8,
344            b: ((value >> 8) & 0xFF) as u8,
345            a: (value & 0xFF) as u8,
346        }
347    }
348
349    /// Porter-Duff "over" compositing: `self` modulated to the effective
350    /// alpha `alpha` blended on top of `dst`. Rasterizers use `alpha` to
351    /// fold per-pixel coverage into the source color's own alpha.
352    /// Returns fully transparent when the combined alpha is negligible.
353    pub fn alpha_over(&self, dst: &Self, alpha: impl Into<Alpha>) -> Self {
354        let src_alpha = alpha.into().as_f32();
355        let dst_alpha = Alpha::from(dst.a).as_f32();
356        let out_alpha = src_alpha + dst_alpha * (1.0 - src_alpha);
357        if out_alpha < 0.001 {
358            return Self::default();
359        }
360        let blend = |s: u8, d: u8| ((s as f32 * src_alpha + d as f32 * dst_alpha * (1.0 - src_alpha)) / out_alpha).round() as u8;
361        Self::new_rgba(blend(self.r, dst.r), blend(self.g, dst.g), blend(self.b, dst.b), (out_alpha * 255.0).round() as u8)
362    }
363
364    /// The color with its alpha channel replaced by `alpha` - for
365    /// fades driven by an external opacity (disabled state, hover
366    /// highlight) rather than the color's own alpha.
367    pub fn with_alpha(&self, alpha: impl Into<Alpha>) -> Self {
368        Self::new_rgba(self.r, self.g, self.b, alpha.into().as_u8())
369    }
370
371    /// The color with its alpha channel scaled by `factor` (0.0-1.0) -
372    /// relative dimming that keeps the color's own alpha. Unlike
373    /// [`Self::with_alpha`], a faded color stays faded when the slot
374    /// already carries transparency.
375    pub fn fade(&self, factor: impl Into<Alpha>) -> Self {
376        Self::new_rgba(self.r, self.g, self.b, (Alpha::from(self.a) * factor.into().as_f64()).as_u8())
377    }
378
379    /// The perceived luminance with custom `weights` (0.0-1.0).
380    ///
381    /// Computed via the normalized `f64` channels.
382    pub fn luminance(&self, weights: LuminanceWeights) -> Luminance {
383        Color::<f64>::from(*self).luminance(weights)
384    }
385
386    /// The perceived luminance with ITU-R BT.709 weights (0.0-1.0).
387    pub fn luminance_709(&self) -> Luminance {
388        Color::<f64>::from(*self).luminance_709()
389    }
390
391    /// The perceived luminance with ITU-R BT.601 weights (0.0-1.0).
392    pub fn luminance_601(&self) -> Luminance {
393        Color::<f64>::from(*self).luminance_601()
394    }
395
396    /// Linear blend toward `other` by `t` (0 = self, 1 = other), via
397    /// the normalized `f64` channels. All four channels are
398    /// interpolated; `t` is clamped to `[0, 1]`.
399    pub fn lerp(&self, other: &Self, t: f64) -> Self {
400        Self::from(Color::<f64>::from(*self).lerp(&Color::<f64>::from(*other), t))
401    }
402
403    /// Lightens the color by blending each channel toward white by
404    /// `factor` (0 = unchanged, 1 = white). Alpha is preserved.
405    pub fn lighten(&self, factor: f64) -> Self {
406        Self::from(Color::<f64>::from(*self).lighten(factor))
407    }
408
409    /// Adds `amount` (in normalized channel units, `1.0` = full
410    /// range) to every color channel - an additive lift, unlike
411    /// [`Self::lighten`]'s blend toward white. Out-of-range results
412    /// saturate at the channel bounds.
413    pub fn brighten(&self, amount: f64) -> Self {
414        Self::from(Color::<f64>::from(*self).brighten(amount).clamp())
415    }
416
417    /// The color lifted to at least `min` luminance (BT.709) by
418    /// blending toward white - dark colors stay readable on dark
419    /// backgrounds. Alpha is preserved.
420    pub fn with_min_luminance(&self, min: impl Into<Luminance>) -> Self {
421        Self::from(Color::<f64>::from(*self).with_min_luminance(min))
422    }
423
424    /// The squared Euclidean distance to `other` over the RGB
425    /// channels (alpha ignored).
426    ///
427    /// Squared distances are monotone with true distances - suitable
428    /// for nearest-color comparisons without a square root.
429    pub const fn squared_rgb_distance(&self, other: &Self) -> u32 {
430        let r = self.r as i32 - other.r as i32;
431        let g = self.g as i32 - other.g as i32;
432        let b = self.b as i32 - other.b as i32;
433        (r * r + g * g + b * b) as u32
434    }
435}

637/// The schema mirrors the serde wire format: the hex string
638/// `"#rgb"`, `"#rgba"`, `"#rrggbb"` or `"#rrggbbaa"`.
639#[cfg(feature = "schemars")]
640impl<T> schemars::JsonSchema for Color<T> {
641    fn schema_name() -> std::borrow::Cow<'static, str> {
642        "Color".into()
643    }
644
645    fn json_schema(_generator: &mut schemars::SchemaGenerator) -> schemars::Schema {
646        schemars::json_schema!({
647            "type": "string",
648            "pattern": "^#([0-9a-fA-F]{3,4}|[0-9a-fA-F]{6}([0-9a-fA-F]{2})?)$",
649        })
650    }
651
652    fn inline_schema() -> bool {
653        true
654    }
655}

681    #[test]
682    fn alpha_over_full_source_wins() {
683        let red = Color::<u8>::new(255, 0, 0);
684        let dst = Color::<u8>::new_rgba(0, 0, 255, 128);
685        // Full effective alpha replaces dst entirely.
686        assert_eq!(red.alpha_over(&dst, 255u8), Color::<u8>::new_rgba(255, 0, 0, 255));
687        // Zero effective alpha keeps dst (channels unmixed, dst alpha kept).
688        assert_eq!(red.alpha_over(&dst, 0.0), dst);
