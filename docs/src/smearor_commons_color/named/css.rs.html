12/// A CSS `<named-color>` keyword from the W3C
13/// [CSS Color Module Level 4](https://www.w3.org/TR/css-color-4/#named-colors)
14/// spec (the former "CSS3 color names").
15///
16/// The enum covers all 148 keywords - including the Level 4 addition
17/// `rebeccapurple` and the `gray`/`grey` spelling variants - plus the
18/// `transparent` keyword. `currentcolor` is context-dependent and
19/// intentionally not represented.
20///

22/// [`Display`] and the wire format use the canonical lowercase keyword.

635    /// The color as byte components - [`Self::Transparent`] maps to
636    /// `rgba(0, 0, 0, 0)`.
637    pub const fn to_color(&self) -> Color<u8> {
638        match self {
639            Self::Transparent => Color::<u8>::new_rgba(0, 0, 0, 0),
640            _ => Color::from_rgb_u32(self.packed_rgb()),
641        }
642    }

644    /// The packed `0xRRGGBB` value of the keyword.
645    const fn packed_rgb(&self) -> u32 {
646        match self {
647            Self::AliceBlue => 0xF0F8FF,
648            Self::AntiqueWhite => 0xFAEBD7,
649            Self::Aqua => 0x00FFFF,

800impl FromStr for CssNamedColor {
801    type Err = ParseColorError;
802
803    fn from_str(s: &str) -> Result<Self, Self::Err> {
804        Self::ALL
805            .iter()
806            .find(|color| s.eq_ignore_ascii_case(color.as_str()))
807            .copied()
808            .ok_or_else(|| ParseColorError::UnknownNamedColor { input: s.into() })
809    }
810}

813    /// Formats the canonical lowercase CSS keyword.
814    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
815        f.write_str(self.as_str())
816    }
817}
