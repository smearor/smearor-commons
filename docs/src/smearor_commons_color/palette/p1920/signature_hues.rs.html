7palette_definition! {
8    /// The signature hues of the 1920s - the lead swatch of each
9    /// chapter palette of "Pantone: The Twentieth Century in Color"
10    /// (Eiseman & Recker, 2011), in chapter order.
11    pub enum SignatureHues1920Color {
12        /// The lead swatch of the "Art Deco" palette - PANTONE 14-5002 TCX `Silver`.
13        ArtDeco => "art-deco",
14        /// The lead swatch of the "Tutmania" palette - PANTONE 16-1836 TCX `Rich Gold`. The caption reads 16-1836; the official code of `Rich Gold` is 16-0836.
15        Tutmania => "tutmania",
16        /// The lead swatch of the "Cocktails and Laughter" palette - PANTONE 19-2118 TCX `Winetasting`.
17        CocktailsAndLaughter => "cocktails-and-laughter",
18        /// The lead swatch of the "Destinations" palette - Pantone 7410 C.
19        Destinations => "destinations",
20        /// The lead swatch of the "The Leyendecker Man" palette - PANTONE 16-1331 TCX `Toast`.
21        LeyendeckerMan => "leyendecker-man",
22        /// The lead swatch of the "Bauhaus" palette - PANTONE 19-4203 TCX `Moonless Night`.
23        Bauhaus => "bauhaus",
24        /// The lead swatch of the "Modern Pleasures" palette - PANTONE 14-0244 TCX `Bright Lime Green`.
25        ModernPleasures => "modern-pleasures",
26        /// The lead swatch of the "A Rose Is a Rose" palette - PANTONE 18-1629 TCX `Faded Rose`.
27        RoseIsARose => "rose-is-a-rose"
28    }
29}
