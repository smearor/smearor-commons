7palette_definition! {
8    /// The signature hues of the 1980s - the lead swatch of each
9    /// chapter palette of "Pantone: The Twentieth Century in Color"
10    /// (Eiseman & Recker, 2011), in chapter order.
11    pub enum SignatureHues1980Color {
12        /// The lead swatch of the "Memphis, Michael, and Philippe" palette - PANTONE 19-3847 TCX `Deep Blue`.
13        MemphisMichaelAndPhilippe => "memphis-michael-and-philippe",
14        /// The lead swatch of the "To the Manor Born" palette - PANTONE 19-6110 TCX `Deep Forest`. Caption spells `Deep Forrest`.
15        ToTheManorBorn => "to-the-manor-born",
16        /// The lead swatch of the "Urban Cowboys" palette - PANTONE 19-4118 TCX `Dark Denim`.
17        UrbanCowboys => "urban-cowboys",
18        /// The lead swatch of the "Signs and Symbols" palette - Pantone 583 C.
19        SignsAndSymbols => "signs-and-symbols",
20        /// The lead swatch of the "Miami Vice" palette - PANTONE 13-2805 TCX `Pink Mist`.
21        MiamiVice => "miami-vice",
22        /// The lead swatch of the "Majorelle and Morocco" palette - PANTONE 18-4043 TCX `Palace Blue`.
23        MajorelleAndMorocco => "majorelle-and-morocco",
24        /// The lead swatch of the "Santa Fe" palette - Pantone 4735 C.
25        SantaFe => "santa-fe",
26        /// The lead swatch of the "Personal Colors" palette - PANTONE 19-1220 TCX `Cappuccino`.
27        PersonalColors => "personal-colors",
28        /// The lead swatch of the "Japonais" palette - PANTONE 19-4005 TCX `Stretch Limo`.
29        Japonais => "japonais"
30    }
31}
