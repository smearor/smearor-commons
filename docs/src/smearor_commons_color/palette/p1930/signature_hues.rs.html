8palette_definition! {
9    /// The signature hues of the 1930s - the lead swatch of each
10    /// chapter palette of "Pantone: The Twentieth Century in Color"
11    /// (Eiseman & Recker, 2011), in chapter order.
12    pub enum SignatureHues1930Color {
13        /// The lead swatch of the "Deco Architecture" palette - Pantone `Silver` metallic chip (closest catalog color is Pantone 877 C); not in the PANTONE solid-coated catalog - the sRGB value is sampled from the book page.
14        DecoArchitecture => "deco-architecture",
15        /// The lead swatch of the "Illusions" palette - PANTONE 18-0000 TCX `Smoked Pearl`.
16        Illusions => "illusions",
17        /// The lead swatch of the "Fantastic Plastic" palette - PANTONE 19-4305 TCX `Pirate Black`.
18        FantasticPlastic => "fantastic-plastic",
19        /// The lead swatch of the "Diversions" palette - Pantone 5473 C.
20        Diversions => "diversions",
21        /// The lead swatch of the "Parks and Recreation" palette - Pantone Black 6 C.
22        ParksAndRecreation => "parks-and-recreation",
23        /// The lead swatch of the "Roseville" palette - PANTONE 17-0215 TCX `Aspen Green`.
24        Roseville => "roseville",
25        /// The lead swatch of the "The Wizard of Oz" palette - PANTONE 14-4121 TCX `Blue Bell`.
26        WizardOfOz => "wizard-of-oz",
27        /// The lead swatch of the "The World of Tomorrow" palette - PANTONE 19-3847 TCX `Deep Blue`.
28        WorldOfTomorrow => "world-of-tomorrow"
29    }
30}

32/// The sRGB values of the [`SignatureHues1930Color`] slots.

34    SignatureHues1930Color::DecoArchitecture => Color::from_rgb_u32(0xABAFB0),

37    SignatureHues1930Color::Diversions => PantoneNamedColor::Pms5473C.to_color(),
38    SignatureHues1930Color::ParksAndRecreation => PantoneNamedColor::Black6C.to_color(),
39    SignatureHues1930Color::Roseville => PantoneFhiNamedColor::AspenGreen.to_color(),
40    SignatureHues1930Color::WizardOfOz => PantoneFhiNamedColor::BlueBell.to_color(),
41    SignatureHues1930Color::WorldOfTomorrow => PantoneFhiNamedColor::DeepBlue.to_color(),
