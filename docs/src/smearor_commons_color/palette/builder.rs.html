1//! A fluent builder for [`Palette`]s - expressive assembly plus a
2//! completeness gate for [`PaletteDefinition`] keys.
3
4use crate::Color;
5use crate::Palette;
6use crate::PaletteDefinition;
7use std::collections::HashMap;
8use std::hash::Hash;
9
10/// A fluent builder assembling a [`Palette`] name by name.
11///
12/// ```
13/// use smearor_commons_color::Color;
14/// use smearor_commons_color::Palette;
15/// use smearor_commons_color::palette_definition;
16///
17/// palette_definition! {
18///     /// Square colors of a board theme.
19///     pub enum BoardSlot {
20///         /// Light squares.
21///         Light => "light",
22///         /// Dark squares.
23///         Dark => "dark"
24///     }
25/// }
26///
27/// let palette = Palette::<BoardSlot, u8>::builder()
28///     .color(BoardSlot::Light, Color::<u8>::new(0xED, 0xD4, 0x9E))
29///     .color(BoardSlot::Dark, Color::<u8>::new(0x73, 0x4D, 0x2E))
30///     .build();
31/// assert!(palette.is_complete());
32/// ```
33#[derive(Debug, Default)]
34pub struct PaletteBuilder<N: Eq + Hash + 'static, T: 'static> {
35    /// Name/color pairs collected so far.
36    colors: HashMap<N, Color<T>>,
37}
38
39impl<N: Eq + Hash + 'static, T: 'static> PaletteBuilder<N, T> {
40    /// Starts an empty builder.
41    pub fn new() -> Self {
42        Self { colors: HashMap::new() }
43    }
44
45    /// Registers `color` under `name` - later duplicates win.
46    ///
47    /// Names go through `Into`, so enum variants can be used
48    /// directly and [`crate::NamedColor`] keys can be written as
49    /// plain `&str`.
50    pub fn color(mut self, name: impl Into<N>, color: Color<T>) -> Self {
51        self.colors.insert(name.into(), color);
52        self
53    }
54
55    /// Registers all `(name, color)` pairs - later duplicates win.
56    pub fn colors(mut self, pairs: impl IntoIterator<Item = (N, Color<T>)>) -> Self {
57        self.colors.extend(pairs);
58        self
59    }
60
61    /// Finishes the palette, complete or not.
62    pub fn build(self) -> Palette<N, T> {
63        Palette::new(self.colors)
64    }
65}
66
67impl<N: PaletteDefinition, T: 'static> PaletteBuilder<N, T> {
68    /// Finishes the palette only when every declared slot carries a
69    /// color - the completeness gate [`Palette::is_complete`]
70    /// enforces after the fact.
71    pub fn build_complete(self) -> Result<Palette<N, T>, PaletteBuildError<N>> {
72        let palette = self.build();
73        let missing = palette.missing();
74        if missing.is_empty() {
75            Ok(palette)
76        } else {
77            Err(PaletteBuildError::Missing(missing))
78        }
79    }
80}
81
82/// The palette could not be built because declared slots have no
83/// color - the failing slots are part of the error.
84#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
85pub enum PaletteBuildError<N> {
86    /// These [`PaletteDefinition`] slots were left without a color.
87    #[error("missing colors for palette slots: {0:?}")]
88    Missing(Vec<N>),
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94    use crate::NamedColor;
95    use crate::palette_definition;
96
97    palette_definition! {
98        /// Colors of a chess board theme.
99        pub enum BoardSlot {
100            /// Light squares.
101            Light => "light",
102            /// Dark squares.
103            Dark => "dark",
104            /// Frame around the board.
105            Frame => "frame"
106        }
107    }
108
109    #[test]
110    fn color_registers_by_name() {
111        let palette = Palette::<BoardSlot, u8>::builder()
112            .color(BoardSlot::Light, Color::<u8>::new(0xED, 0xD4, 0x9E))
113            .color(BoardSlot::Dark, Color::<u8>::new(0x73, 0x4D, 0x2E))
114            .build();
115        assert_eq!(palette.get(&BoardSlot::Light), Some(&Color::<u8>::new(0xED, 0xD4, 0x9E)));
116        assert_eq!(palette.len(), 2);
117    }
118
119    #[test]
120    fn names_go_through_into() {
121        let palette = Palette::<NamedColor, u8>::builder().color("light", Color::<u8>::new(0xED, 0xD4, 0x9E)).build();
122        assert_eq!(palette.get("light"), Some(&Color::<u8>::new(0xED, 0xD4, 0x9E)));
123    }
124
125    #[test]
126    fn later_duplicates_win() {
127        let palette = Palette::<BoardSlot, u8>::builder()
128            .color(BoardSlot::Light, Color::<u8>::WHITE)
129            .color(BoardSlot::Light, Color::<u8>::BLACK)
130            .build();
131        assert_eq!(palette.get(&BoardSlot::Light), Some(&Color::<u8>::BLACK));
132    }
133
134    #[test]
135    fn colors_registers_pairs() {
136        let palette = Palette::<BoardSlot, u8>::builder()
137            .colors([(BoardSlot::Light, Color::<u8>::WHITE), (BoardSlot::Dark, Color::<u8>::BLACK)])
138            .build();
139        assert_eq!(palette.len(), 2);
140    }
141
142    #[test]
143    fn build_complete_succeeds_when_every_slot_is_set() {
144        let palette = Palette::<BoardSlot, u8>::builder()
145            .color(BoardSlot::Light, Color::<u8>::WHITE)
146            .color(BoardSlot::Dark, Color::<u8>::BLACK)
147            .color(BoardSlot::Frame, Color::<u8>::new(0x80, 0x60, 0x40))
148            .build_complete()
149            .unwrap_or_else(|error| panic!("unexpected build error: {error}"));
150        assert!(palette.is_complete());
151    }
152
153    #[test]
154    fn build_complete_reports_the_missing_slots() {
155        let error = Palette::<BoardSlot, u8>::builder().color(BoardSlot::Light, Color::<u8>::WHITE).build_complete();
156        assert_eq!(error, Err(PaletteBuildError::Missing(vec![BoardSlot::Dark, BoardSlot::Frame])));
157        assert_eq!(error.err().map(|e| e.to_string()).as_deref(), Some("missing colors for palette slots: [Dark, Frame]"));
158    }
159}
