1//! The 1970s chapter palettes of "Pantone: The Twentieth
2//! Century in Color" (Eiseman & Recker, 2011).
3//!
4//! The slot names and order match the book's swatch rows. TCX
5//! swatches resolve to the official PANTONE Fashion, Home +
6//! Interiors (FHI) color via [`crate::PantoneFhiNamedColor`];
7//! swatches numbered in the PANTONE solid-coated system resolve to
8//! [`crate::PantoneNamedColor`]. Premium-metallic and pastel chips
9//! outside those catalogs carry the sRGB value sampled from the
10//! book page and say so in their doc comments.

47    #[test]
48    fn palettes_are_complete() {

51        assert!(FEATHERS_AND_LEATHERS.is_complete());

54        assert!(DAY_THE_WORLD_TURNED_DAY_GLO.is_complete());

57    }

59    #[test]
60    fn palettes_hold_the_book_chip_counts() {
61        assert_eq!(COLORS_AND_COORDINATES.len(), 8);
62        assert_eq!(AVOCADO_AND_HARVEST_GOLD.len(), 6);
63        assert_eq!(FEATHERS_AND_LEATHERS.len(), 6);
64        assert_eq!(PROVENCE.len(), 7);
65        assert_eq!(LAND_ART.len(), 7);
66        assert_eq!(DAY_THE_WORLD_TURNED_DAY_GLO.len(), 8);
67        assert_eq!(NIGHT_LIFE.len(), 7);
68        assert_eq!(HOTEL_CALIFORNIA.len(), 7);
69    }
70
71    #[test]
72    fn signature_hues_are_complete() {
73        assert!(SIGNATURE_HUES_1970.is_complete());
74    }
75
76    #[test]
77    fn signature_hues_hold_chapter_leads_in_order() {
78        assert_eq!(SIGNATURE_HUES_1970.len(), 8);
79        assert_eq!(SignatureHues1970Color::slots()[0], SignatureHues1970Color::ColorsAndCoordinates);
80    }
81
82    #[test]
