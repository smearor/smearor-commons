8    /// The signature hues of the 1970s - the lead swatch of each
9    /// chapter palette of "Pantone: The Twentieth Century in Color"
10    /// (Eiseman & Recker, 2011), in chapter order.
11    pub enum SignatureHues1970Color {
12        /// The lead swatch of the "Colors and Coordinates" palette - Pantone 7752 C.
13        ColorsAndCoordinates => "colors-and-coordinates",
14        /// The lead swatch of the "Avocado and Harvest Gold" palette - PANTONE 18-0430 TCX `Avocado`.
15        AvocadoAndHarvestGold => "avocado-and-harvest-gold",
16        /// The lead swatch of the "Feathers and Leathers" palette - PANTONE 19-4329 TCX `Corsair`.
17        FeathersAndLeathers => "feathers-and-leathers",
18        /// The lead swatch of the "Provence" palette - PANTONE 16-4032 TCX `Provence`.
19        Provence => "provence",
20        /// The lead swatch of the "Land Art" palette - Pantone 284 C.
21        LandArt => "land-art",
22        /// The lead swatch of the "The Day the World Turned Day-Glo" palette - Pantone Process Blue C.
23        DayTheWorldTurnedDayGlo => "day-the-world-turned-day-glo",
24        /// The lead swatch of the "Night Life" palette - Pantone White C.
25        NightLife => "night-life",
26        /// The lead swatch of the "Hotel California" palette - Pantone 3005 C.
27        HotelCalifornia => "hotel-california"
28    }
29}
