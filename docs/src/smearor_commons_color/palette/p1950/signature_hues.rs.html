8palette_definition! {
9    /// The signature hues of the 1950s - the lead swatch of each
10    /// chapter palette of "Pantone: The Twentieth Century in Color"
11    /// (Eiseman & Recker, 2011), in chapter order.
12    pub enum SignatureHues1950Color {
13        /// The lead swatch of the "Happy at Home" palette - PANTONE 12-0736 TCX `Lemon Drop`.
14        HappyAtHome => "happy-at-home",
15        /// The lead swatch of the "Teen Angels" palette - Pantone 7471 C.
16        TeenAngels => "teen-angels",
17        /// The lead swatch of the "Mid-Century Modernists" palette - PANTONE 14-0848 TCX `Mimosa`.
18        MidCenturyModernists => "mid-century-modernists",
19        /// The lead swatch of the "Movie Goddesses" palette - Pantone 9286 C; not in the PANTONE solid-coated catalog - the sRGB value is sampled from the book page.
20        MovieGoddesses => "movie-goddesses",
21        /// The lead swatch of the "Cosmetic Superstars" palette - PANTONE 11-1404 TCX `Powder Puff`.
22        CosmeticSuperstars => "cosmetic-superstars",
23        /// The lead swatch of the "Coast-to-Coast Woolens" palette - PANTONE 19-1759 TCX `American Beauty`.
24        CoastToCoastWoolens => "coast-to-coast-woolens",
25        /// The lead swatch of the "Fantasyland" palette - Pantone 211 C.
26        Fantasyland => "fantasyland",
27        /// The lead swatch of the "Abstract Expressionists" palette - Pantone 7579 C.
28        AbstractExpressionists => "abstract-expressionists"
29    }
30}
31
32/// The sRGB values of the [`SignatureHues1950Color`] slots.
33pub static SIGNATURE_HUES_1950: StaticPalette<SignatureHues1950Color, u8> = static_palette! {
34    SignatureHues1950Color::HappyAtHome => PantoneFhiNamedColor::LemonDrop.to_color(),
35    SignatureHues1950Color::TeenAngels => PantoneNamedColor::Pms7471C.to_color(),
36    SignatureHues1950Color::MidCenturyModernists => PantoneFhiNamedColor::Mimosa.to_color(),
37    SignatureHues1950Color::MovieGoddesses => Color::from_rgb_u32(0xFCDCDD),
38    SignatureHues1950Color::CosmeticSuperstars => PantoneFhiNamedColor::PowderPuff.to_color(),
39    SignatureHues1950Color::CoastToCoastWoolens => PantoneFhiNamedColor::AmericanBeauty.to_color(),
40    SignatureHues1950Color::Fantasyland => PantoneNamedColor::Pms211C.to_color(),
41    SignatureHues1950Color::AbstractExpressionists => PantoneNamedColor::Pms7579C.to_color(),
42};
