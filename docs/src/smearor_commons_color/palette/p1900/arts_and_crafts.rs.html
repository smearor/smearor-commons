6palette_definition! {
7    /// The "Arts and Crafts" palette - 1900s chapter of "Pantone: The
8    /// Twentieth Century in Color" (Eiseman & Recker, 2011).
9    pub enum ArtsAndCraftsColor {
10        /// PANTONE 17-1410 TCX `Pine Bark`.
11        PineBark => "pine-bark",
12        /// PANTONE 19-1618 TCX `Beech`. The caption reads 19-1618; the official code of `Beech` is 19-0618.
13        Beech => "beech",
14        /// PANTONE 11-0105 TCX `Antique White`.
15        AntiqueWhite => "antique-white",
16        /// PANTONE 13-1108 TCX `Cream Tan`.
17        CreamTan => "cream-tan",
18        /// PANTONE 18-1142 TCX `Leather Brown`.
19        LeatherBrown => "leather-brown",
20        /// PANTONE 17-1347 TCX `Autumn Leaf`.
21        AutumnLeaf => "autumn-leaf",
22        /// PANTONE 18-5610 TCX `Brittany Blue`.
23        BrittanyBlue => "brittany-blue",
24        /// PANTONE 18-0422 TCX `Loden Green`.
25        LodenGreen => "loden-green"
26    }
27}
28
29/// The sRGB values of the [`ArtsAndCraftsColor`] slots.
30pub static ARTS_AND_CRAFTS: StaticPalette<ArtsAndCraftsColor, u8> = static_palette! {
31    ArtsAndCraftsColor::PineBark => PantoneFhiNamedColor::PineBark.to_color(),
32    ArtsAndCraftsColor::Beech => PantoneFhiNamedColor::Beech.to_color(),
33    ArtsAndCraftsColor::AntiqueWhite => PantoneFhiNamedColor::AntiqueWhite.to_color(),
34    ArtsAndCraftsColor::CreamTan => PantoneFhiNamedColor::CreamTan.to_color(),
35    ArtsAndCraftsColor::LeatherBrown => PantoneFhiNamedColor::LeatherBrown.to_color(),
36    ArtsAndCraftsColor::AutumnLeaf => PantoneFhiNamedColor::AutumnLeaf.to_color(),
37    ArtsAndCraftsColor::BrittanyBlue => PantoneFhiNamedColor::BrittanyBlue.to_color(),
38    ArtsAndCraftsColor::LodenGreen => PantoneFhiNamedColor::LodenGreen.to_color(),
39};
