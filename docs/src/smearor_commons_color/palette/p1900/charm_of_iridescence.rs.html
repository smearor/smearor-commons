6palette_definition! {
7    /// The "The Charm of Iridescence" palette - 1900s chapter of "Pantone: The
8    /// Twentieth Century in Color" (Eiseman & Recker, 2011).
9    pub enum CharmOfIridescenceColor {
10        /// PANTONE 15-1927 TCX `Pale Gold`. The caption reads 15-1927; the official code of `Pale Gold` is 15-0927.
11        PaleGold => "pale-gold",
12        /// PANTONE 16-0730 TCX `Antique Gold`.
13        AntiqueGold => "antique-gold",
14        /// PANTONE 18-6330 TCX `Juniper`.
15        Juniper => "juniper",
16        /// PANTONE 18-1928 TCX `Sepia`. The caption reads 18-1928; the official code of `Sepia` is 18-0928.
17        Sepia => "sepia",
18        /// PANTONE 19-4914 TCX `Deep Teal`.
19        DeepTeal => "deep-teal",
20        /// PANTONE 15-3817 TCX `Lavender`.
21        Lavender => "lavender",
22        /// PANTONE 14-1311 TCX `Evening Sand`.
23        EveningSand => "evening-sand",
24        /// PANTONE 12-2103 TCX `Almost Mauve`.
25        AlmostMauve => "almost-mauve"
26    }
27}
28
29/// The sRGB values of the [`CharmOfIridescenceColor`] slots.
30pub static CHARM_OF_IRIDESCENCE: StaticPalette<CharmOfIridescenceColor, u8> = static_palette! {
31    CharmOfIridescenceColor::PaleGold => PantoneFhiNamedColor::PaleGold.to_color(),
32    CharmOfIridescenceColor::AntiqueGold => PantoneFhiNamedColor::AntiqueGold.to_color(),
33    CharmOfIridescenceColor::Juniper => PantoneFhiNamedColor::Juniper.to_color(),
34    CharmOfIridescenceColor::Sepia => PantoneFhiNamedColor::Sepia.to_color(),
35    CharmOfIridescenceColor::DeepTeal => PantoneFhiNamedColor::DeepTeal.to_color(),
36    CharmOfIridescenceColor::Lavender => PantoneFhiNamedColor::Lavender.to_color(),
37    CharmOfIridescenceColor::EveningSand => PantoneFhiNamedColor::EveningSand.to_color(),
38    CharmOfIridescenceColor::AlmostMauve => PantoneFhiNamedColor::AlmostMauve.to_color(),
39};
