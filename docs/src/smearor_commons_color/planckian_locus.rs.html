2//! radiator from 1000 K to 20000 K in 100 K steps.
3//!
4//! Computed from Planck's radiation law weighted with the CIE 1931
5//! 2-degree color matching functions (same source as
6//! [`crate::SPECTRAL_LOCUS`]): `xy` of `sum M(lambda,T) * (xbar,ybar,zbar)`.
7

208#[cfg(test)]
209mod tests {
210    use super::*;
211
212    #[test]
213    fn t6500_sits_near_d65() {
214        let Some((_, x, y)) = PLANCKIAN_LOCUS.iter().find(|(k, _, _)| *k == 6500) else {
215            panic!("6500K missing");
216        };
217        assert!((*x - 0.3135).abs() < 0.005 && (*y - 0.3237).abs() < 0.005);
218    }
219
220    #[test]
221    fn the_locus_moves_from_red_to_blue() {
222        let (first, last) = (PLANCKIAN_LOCUS.first(), PLANCKIAN_LOCUS.last());
223        assert!(matches!((first, last), (Some(f), Some(l)) if f.1 > l.1 && f.2 > l.2));
224    }
