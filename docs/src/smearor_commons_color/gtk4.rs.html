1//! `gdk::RGBA` conversions, gated behind the `gtk4` feature.
2//!
3//! `gdk::RGBA` stores `f32` channels, so `Color<f32>` converts
4//! losslessly; `Color<f64>`/`Color<u8>` convert via `Color<f32>`.
5
6use crate::Color;
7use gtk4::gdk;
8
9impl From<Color<f32>> for gdk::RGBA {
10    fn from(color: Color<f32>) -> Self {
11        Self::new(color.r, color.g, color.b, color.a)
12    }
13}
14
15impl From<&Color<f32>> for gdk::RGBA {
16    fn from(color: &Color<f32>) -> Self {
17        Self::new(color.r, color.g, color.b, color.a)
18    }
19}
20
21impl From<gdk::RGBA> for Color<f32> {
22    fn from(color: gdk::RGBA) -> Self {
23        Self::new_rgba(color.red(), color.green(), color.blue(), color.alpha())
24    }
25}
26
27impl From<Color<f64>> for gdk::RGBA {
28    fn from(color: Color<f64>) -> Self {
29        Self::from(Color::<f32>::from(color))
30    }
31}
32
33impl From<gdk::RGBA> for Color<f64> {
34    fn from(color: gdk::RGBA) -> Self {
35        Self::from(Color::<f32>::from(color))
36    }
37}
38
39impl From<Color<u8>> for gdk::RGBA {
40    fn from(color: Color<u8>) -> Self {
41        Self::from(Color::<f32>::from(color))
42    }
43}

45#[cfg(test)]
46mod tests {
47    use super::*;
48
49    #[test]
50    fn f32_round_trip() {
51        let color = Color::<f32>::new_rgba(0.25, 0.5, 0.75, 0.8);
52        let rgba = gdk::RGBA::from(color);
53        assert_eq!(rgba.red(), 0.25);
54        assert_eq!(rgba.alpha(), 0.8);
55        assert_eq!(Color::<f32>::from(rgba), color);
56    }
57
58    #[test]
59    fn f64_and_u8_convert() {
60        let rgba = gdk::RGBA::from(Color::<u8>::new_rgba(255, 0, 0, 255));
61        assert_eq!(rgba.red(), 1.0);
62        let color = Color::<f64>::from(gdk::RGBA::new(0.0, 1.0, 0.0, 1.0));
63        assert_eq!(color.g, 1.0);
64    }
