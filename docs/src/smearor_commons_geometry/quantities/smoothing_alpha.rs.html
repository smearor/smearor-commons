8/// Exponential-moving-average weight of the newest sample
9/// (`pos = alpha*raw + (1-alpha)*prev`).
10///
11/// Averages out the differential-latency distortion sensors report during
12/// motion. `0.0` disables smoothing, `1.0` passes raw values through.
13#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
14#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
15#[serde(transparent)]
16pub struct SmoothingAlpha(f64);
17
18impl SmoothingAlpha {
19    /// Creates a smoothing factor.
20    pub const fn new(alpha: f64) -> Self {
21        Self(alpha)
22    }
23
24    /// The EMA weight of the newest sample.
25    pub fn as_f64(&self) -> f64 {
26        self.0
27    }
28
29    /// Whether running the smoothing algorithm changes anything -
30    /// `0.0` freezes the value (disabled), `1.0` passes samples
31    /// through (identity); only `0 < alpha < 1` blends.
32    pub fn is_effective(&self) -> bool {
33        self.0 > 0.0 && self.0 < 1.0
34    }
35
36    /// The complementary weight `1 − alpha` - the previous value's
37    /// share of an EMA blend (`pos * alpha + prev * alpha.complement()`).
38    pub fn complement(self) -> Self {
39        Self(1.0 - self.0)
40    }
41
42    /// Frame-rate independent smoothing factor for one rendered frame.
43    ///
44    /// Computes `alpha = 1 - exp(-frame_time / tau)` so the same `tau`
45    /// time constant closes the same fraction of the remaining gap per
46    /// second at any frame rate. A zero `tau` yields `alpha = 1.0`
47    /// (instant tracking).
48    pub fn from_frame_time(frame_time: Duration, tau: Duration) -> Self {
49        if tau.is_zero() {
50            return Self(1.0);
51        }
52        Self(1.0 - (-frame_time.as_secs_f64() / tau.as_secs_f64()).exp())
53    }
54}
55
56impl<U> Mul<SmoothingAlpha> for Vec2<U> {
57    type Output = Vec2<U>;
58
59    /// Scales the vector by the weight (`v * alpha`).
60    fn mul(self, alpha: SmoothingAlpha) -> Vec2<U> {
61        self * alpha.as_f64()
62    }
63}

65impl Default for SmoothingAlpha {

69}

91#[cfg(test)]
92mod tests {
93    use super::SmoothingAlpha;
94    use std::time::Duration;
95
96    #[test]
97    fn frame_time_scales_with_tau() {
98        // One 16 ms frame at tau = 80 ms closes ~18% of the gap.
99        let alpha = SmoothingAlpha::from_frame_time(Duration::from_millis(16), Duration::from_millis(80));
100        assert!((alpha.as_f64() - 0.181).abs() < 0.001);
101        // Zero frame time blends nothing.
102        assert_eq!(SmoothingAlpha::from_frame_time(Duration::ZERO, Duration::from_millis(80)).as_f64(), 0.0);
103        // Zero tau tracks instantly.
104        assert_eq!(SmoothingAlpha::from_frame_time(Duration::from_millis(16), Duration::ZERO).as_f64(), 1.0);
105    }
