5/// The four corners of a [`crate::Rect`], as named fields instead of an
6/// index-addressed array.
7///
8/// Field names follow the y-down screen convention: `top_left` is the
9/// smallest corner, `bottom_right` the largest.
10#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
11pub struct Corners<U> {
12    /// Top-left corner (the rect's `min`).
13    pub top_left: Vec2<U>,
14    /// Top-right corner (`right`, `min.y`).
15    pub top_right: Vec2<U>,
16    /// Bottom-right corner (the rect's `max`).
17    pub bottom_right: Vec2<U>,
18    /// Bottom-left corner (`min.x`, `bottom`).
19    pub bottom_left: Vec2<U>,
20}
21
22impl<U> Corners<U> {
23    /// The corners as an array in declaration order
24    /// (`top_left`, `top_right`, `bottom_right`, `bottom_left`).
25    pub fn into_array(self) -> [Vec2<U>; 4] {
26        [self.top_left, self.top_right, self.bottom_right, self.bottom_left]
27    }
28}
29
30impl<U> From<Corners<U>> for [Vec2<U>; 4] {
31    /// Same order as [`Corners::into_array`].
32    fn from(corners: Corners<U>) -> Self {
33        corners.into_array()
34    }
35}
36
37impl<U> IntoIterator for Corners<U> {
38    type Item = Vec2<U>;
39    type IntoIter = std::array::IntoIter<Vec2<U>, 4>;
40
41    /// Iterates in declaration order (starting at `top_left`).
42    fn into_iter(self) -> Self::IntoIter {
43        self.into_array().into_iter()
44    }
45}
