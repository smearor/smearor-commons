13/// `sin(60°) = sqrt(3) / 2` - the inradius (apothem) of a regular
14/// hexagon with circumradius `1`.
15const SIN_60: f64 = 0.866_025_403_784_438_6;
16
17/// A regular hexagon tagged with its coordinate unit `U` and
18/// numeric component type `T`.
19///
20/// A regular hexagon is fully determined by a single extent: the
21/// **circumradius** (center to vertex) equals its edge length, and
22/// the **inradius** (center to edge, also called *apothem*) is
23/// `circumradius * sin(60°)`. Use [`Self::with_inradius`] or
24/// [`Self::with_side_length`] when the hexagon is specified by one
25/// of the other extents.
26///
27/// `T` defaults to `f64` for domain math. The type is a technical
28/// primitive without validation: `circumradius` should be
29/// non-negative, stricter bounds belong to consumer wrappers.
30#[derive(Debug, Serialize, Deserialize)]
31pub struct Hexagon<U = (), T = f64> {
32    /// The center point.
33    pub center: Vec2<U, T>,
34    /// The circumradius - distance from `center` to each vertex.
35    /// For a regular hexagon this equals the edge length.
36    pub circumradius: T,
37    /// Whether the hexagon has a horizontal top edge
38    /// ([`PolygonOrientation::FlatTop`]) or a vertex on top
39    /// ([`PolygonOrientation::PointyTop`]).
40    pub orientation: PolygonOrientation,
41}
42
43impl<U, T: Copy> Copy for Hexagon<U, T> {}

45impl<U, T: Clone> Clone for Hexagon<U, T> {
46    fn clone(&self) -> Self {
47        Self::new(self.center.clone(), self.circumradius.clone(), self.orientation)
48    }
49}

51impl<U, T: PartialEq> PartialEq for Hexagon<U, T> {
52    fn eq(&self, other: &Self) -> bool {
53        self.center == other.center && self.circumradius == other.circumradius && self.orientation == other.orientation
54    }
55}

57impl<U, T: Eq> Eq for Hexagon<U, T> {}

59impl<U, T: Hash> Hash for Hexagon<U, T> {

62        self.circumradius.hash(state);

64    }

67impl<U, T> Hexagon<U, T> {
68    /// Creates a hexagon from its center, circumradius and
69    /// orientation.
70    pub const fn new(center: Vec2<U, T>, circumradius: T, orientation: PolygonOrientation) -> Self {
71        Self {
72            center,
73            circumradius,
74            orientation,
75        }
76    }
77
78    /// Creates a [`PolygonOrientation::FlatTop`] hexagon.
79    pub const fn flat_top(center: Vec2<U, T>, circumradius: T) -> Self {
80        Self::new(center, circumradius, PolygonOrientation::FlatTop)
81    }
82
83    /// Creates a [`PolygonOrientation::PointyTop`] hexagon.
84    pub const fn pointy_top(center: Vec2<U, T>, circumradius: T) -> Self {
85        Self::new(center, circumradius, PolygonOrientation::PointyTop)
86    }
87}

89impl<U> Hexagon<U, f64> {
90    /// Creates a hexagon from its center, edge length and
91    /// orientation. The edge length of a regular hexagon equals
92    /// its circumradius.
93    pub const fn with_side_length(center: Vec2<U, f64>, side_length: f64, orientation: PolygonOrientation) -> Self {
94        Self::new(center, side_length, orientation)
95    }
96
97    /// Creates a hexagon from its center, inradius and
98    /// orientation. The inradius (apothem, center to edge) is
99    /// `circumradius * sin(60°)`.
100    pub fn with_inradius(center: Vec2<U, f64>, inradius: f64, orientation: PolygonOrientation) -> Self {
101        Self::new(center, inradius / SIN_60, orientation)
102    }
103
104    /// The edge length - for a regular hexagon equal to the
105    /// circumradius.
106    pub fn side_length(&self) -> f64 {
107        self.circumradius
108    }
109
110    /// The inradius (apothem) - distance from `center` to each
111    /// edge, `circumradius * sin(60°)`.
112    pub fn inradius(&self) -> f64 {
113        self.circumradius * SIN_60
114    }
115
116    /// The six vertices in counter-clockwise order starting from
117    /// angle `0°` (`FlatTop`) resp. `30°` (`PointyTop`) in steps of
118    /// `60°`.
119    pub fn vertices(&self) -> [Vec2<U, f64>; 6] {
120        let offset = match self.orientation {
121            PolygonOrientation::FlatTop => 0.0,
122            PolygonOrientation::PointyTop => FRAC_PI_6,
123        };
124        [0, 1, 2, 3, 4, 5].map(|i| self.center + Vec2::from_polar(self.circumradius, RotationRad::new(offset + f64::from(i) * FRAC_PI_3)))
125    }
126
127    /// The hexagon as a generic closed [`Polygon`] - the bridge to
128    /// ring algorithms like `segments` and `offset`.
129    pub fn to_polygon(&self) -> Polygon<U> {
130        Polygon::new(self.vertices().to_vec())
131    }
132
133    /// Whether `point` lies inside the hexagon - the boundary
134    /// counts as inside.
135    ///
136    /// A regular hexagon is the intersection of three strips
137    /// perpendicular to its three edge normals: `y` (`FlatTop`)
138    /// resp. `x` (`PointyTop`) and the two `±60°` axes.
139    pub fn contains(&self, point: &Vec2<U, f64>) -> bool {
140        let (dx, dy) = (point.x - self.center.x, point.y - self.center.y);
141        let inradius = self.inradius();
142        match self.orientation {
143            PolygonOrientation::FlatTop => dy.abs() <= inradius && (SIN_60 * dx + 0.5 * dy).abs() <= inradius && (SIN_60 * dx - 0.5 * dy).abs() <= inradius,
144            PolygonOrientation::PointyTop => dx.abs() <= inradius && (0.5 * dx + SIN_60 * dy).abs() <= inradius && (0.5 * dx - SIN_60 * dy).abs() <= inradius,
145        }
146    }
147
148    /// The axis-aligned bounding box: `2R x sqrt(3)*R` for
149    /// `FlatTop`, `sqrt(3)*R x 2R` for `PointyTop`.
150    pub fn bounding_box(&self) -> Rect<U, f64> {
151        let (half_width, half_height) = match self.orientation {
152            PolygonOrientation::FlatTop => (self.circumradius, self.inradius()),
153            PolygonOrientation::PointyTop => (self.inradius(), self.circumradius),
154        };
155        Rect::new(self.center - Vec2::new(half_width, half_height), Vec2::new(half_width * 2.0, half_height * 2.0))
156    }
157
158    /// How far the outline extends along `direction` measured from
159    /// `center` - the exact support function of the convex hexagon:
160    /// the largest projection of any vertex.
161    pub fn support(&self, direction: RotationRad) -> f64 {
162        let (sin, cos) = direction.sin_cos();
163        let ray = Vec2::<U>::new(cos, sin);
164        self.vertices().iter().map(|vertex| (*vertex - self.center).dot(&ray)).fold(0.0, f64::max)
165    }
166}

175    /// The three extent specifications produce the same hexagon.
176    #[test]
177    fn circumradius_inradius_and_side_length_agree() {
178        let center = Vec2::<()>::new(10.0, 20.0);
179        let by_radius = Hexagon::flat_top(center, 100.0);
180        let by_side = Hexagon::with_side_length(center, 100.0, PolygonOrientation::FlatTop);
181        let by_inradius = Hexagon::with_inradius(center, 100.0 * super::SIN_60, PolygonOrientation::FlatTop);
182
183        assert_eq!(by_radius, by_side);
184        assert!((by_inradius.circumradius - 100.0).abs() < 1e-9);
185        assert!((by_radius.inradius() - 100.0 * super::SIN_60).abs() < 1e-9);
186        assert!((by_radius.side_length() - 100.0).abs() < 1e-9);
187    }

189    /// `FlatTop` vertices sit on the circumcircle at `0°`, `60°`, ...
190    /// and `PointyTop` vertices sit on it offset by `30°`.
191    #[test]
192    fn vertices_follow_the_orientation() {
193        let flat = Hexagon::<()>::flat_top(Vec2::new(0.0, 0.0), 10.0).vertices();
194        assert!((flat[0].x - 10.0).abs() < 1e-9 && flat[0].y.abs() < 1e-9, "flat-top vertex at 0 degrees");
195        assert!(flat.iter().all(|v| v.magnitude_f64() - 10.0 < 1e-9), "all on the circumcircle");
196
197        let pointy = Hexagon::<()>::pointy_top(Vec2::new(0.0, 0.0), 10.0).vertices();
198        assert!((pointy[1].x.abs() < 1e-9) && (pointy[1].y - 10.0).abs() < 1e-9, "pointy-top vertex at 90 degrees");
199        assert!(pointy.iter().all(|v| v.magnitude_f64() - 10.0 < 1e-9), "all on the circumcircle");
200    }

202    /// `contains` accepts center, edge midpoints and vertices and
203    /// rejects points outside the strips both orientations span.
204    #[test]
205    fn contains_the_strips() {
206        let hex = Hexagon::<()>::flat_top(Vec2::new(0.0, 0.0), 10.0);
207        assert!(hex.contains(&Vec2::new(0.0, 0.0)));
208        assert!(hex.contains(&Vec2::new(10.0, 0.0)), "vertex counts as inside");
209        assert!(!hex.contains(&Vec2::new(10.1, 0.0)));
210        let inradius = hex.inradius();
211        assert!(!hex.contains(&Vec2::new(0.0, inradius + 0.1)), "above the top edge");
212        assert!(hex.contains(&Vec2::new(0.0, inradius)), "top edge midpoint counts as inside");
213        // The upper-right edge at x = 7 sits at y = 5.196:
214        // (7, 4) inside, (7, 5.5) outside.
215        assert!(hex.contains(&Vec2::new(7.0, 4.0)));
216        assert!(!hex.contains(&Vec2::new(7.0, 5.5)));
217        assert!(!hex.contains(&Vec2::new(9.0, inradius - 1.0)));
218
219        let pointy = Hexagon::<()>::pointy_top(Vec2::new(0.0, 0.0), 10.0);
220        assert!(pointy.contains(&Vec2::new(0.0, 10.0)), "top vertex inside");
221        assert!(!pointy.contains(&Vec2::new(0.0, 10.1)));
222        assert!(pointy.contains(&Vec2::new(pointy.inradius(), 0.0)), "right edge midpoint inside");
223    }

225    /// `bounding_box` covers all vertices and matches the flat/pointy
226    /// extents.
227    #[test]
228    fn bounding_box_covers_the_vertices() {
229        let flat = Hexagon::<()>::flat_top(Vec2::new(0.0, 0.0), 10.0);
230        let bbox = flat.bounding_box();
231        assert!((bbox.size.x - 20.0).abs() < 1e-9);
232        assert!((bbox.size.y - 20.0 * super::SIN_60).abs() < 1e-9);
233        assert!(
234            flat.vertices()
235                .iter()
236                .all(|v| v.x >= bbox.min.x - 1e-9 && v.x <= bbox.right() + 1e-9 && v.y >= bbox.min.y - 1e-9 && v.y <= bbox.bottom() + 1e-9),
237            "vertices inside the bbox (float boundary tolerance)"
238        );
239
240        let pointy = Hexagon::<()>::pointy_top(Vec2::new(0.0, 0.0), 10.0).bounding_box();
241        assert!((pointy.size.x - 20.0 * super::SIN_60).abs() < 1e-9);
242        assert!((pointy.size.y - 20.0).abs() < 1e-9);
243    }

245    /// `support` equals the circumradius along vertex directions and
