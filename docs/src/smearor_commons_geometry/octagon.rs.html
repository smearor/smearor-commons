15/// `cos(22.5°) ≈ 0.9239` - the inradius (apothem) of a regular
16/// octagon with circumradius `1`.
17const COS_22_5: f64 = 0.923_879_532_511_286_7;
18
19/// `sin(22.5°) ≈ 0.3827` - half the edge length of a regular
20/// octagon with circumradius `1`.
21const SIN_22_5: f64 = 0.382_683_432_365_089_8;
22
23/// `sqrt(2)` - the diagonal of the unit square.
24const SQRT_2: f64 = std::f64::consts::SQRT_2;
25
26/// A regular octagon tagged with its coordinate unit `U` and
27/// numeric component type `T`.
28///
29/// A regular octagon is fully determined by a single extent: the
30/// **circumradius** (center to vertex), the **inradius** (center
31/// to edge, also called *apothem*, `circumradius * cos(22.5°)`)
32/// and the **edge length** (`2 * circumradius * sin(22.5°)`). Use
33/// [`Self::with_inradius`] or [`Self::with_side_length`] when the
34/// octagon is specified by one of the other extents.
35///
36/// `T` defaults to `f64` for domain math. The type is a technical
37/// primitive without validation: `circumradius` should be
38/// non-negative, stricter bounds belong to consumer wrappers.
39#[derive(Debug, Serialize, Deserialize)]
40pub struct Octagon<U = (), T = f64> {
41    /// The center point.
42    pub center: Vec2<U, T>,
43    /// The circumradius - distance from `center` to each vertex.
44    pub circumradius: T,
45    /// Whether the octagon has a horizontal top edge
46    /// ([`PolygonOrientation::FlatTop`]) or a vertex on top
47    /// ([`PolygonOrientation::PointyTop`]).
48    pub orientation: PolygonOrientation,
49}
50

55        Self::new(self.center.clone(), self.circumradius.clone(), self.orientation)
56    }
57}

61        self.center == other.center && self.circumradius == other.circumradius && self.orientation == other.orientation
62    }

79        Self {

97impl<U> Octagon<U, f64> {
98    /// Creates an octagon from its center, edge length and
99    /// orientation. The edge length of a regular octagon is
100    /// `2 * circumradius * sin(22.5°)`.
101    pub fn with_side_length(center: Vec2<U, f64>, side_length: f64, orientation: PolygonOrientation) -> Self {
102        Self::new(center, side_length / (2.0 * SIN_22_5), orientation)
103    }
104
105    /// Creates an octagon from its center, inradius and
106    /// orientation. The inradius (apothem, center to edge) is
107    /// `circumradius * cos(22.5°)`.
108    pub fn with_inradius(center: Vec2<U, f64>, inradius: f64, orientation: PolygonOrientation) -> Self {
109        Self::new(center, inradius / COS_22_5, orientation)
110    }
111
112    /// The edge length - `2 * circumradius * sin(22.5°)`.
113    pub fn side_length(&self) -> f64 {
114        self.circumradius * 2.0 * SIN_22_5
115    }
116
117    /// The inradius (apothem) - distance from `center` to each
118    /// edge, `circumradius * cos(22.5°)`.
119    pub fn inradius(&self) -> f64 {
120        self.circumradius * COS_22_5
121    }
122
123    /// The eight vertices in counter-clockwise order starting from
124    /// angle `22.5°` (`FlatTop`) resp. `0°` (`PointyTop`) in steps
125    /// of `45°`.
126    pub fn vertices(&self) -> [Vec2<U, f64>; 8] {
127        let offset = match self.orientation {
128            PolygonOrientation::FlatTop => FRAC_PI_8,
129            PolygonOrientation::PointyTop => 0.0,
130        };
131        [0, 1, 2, 3, 4, 5, 6, 7].map(|i| self.center + Vec2::from_polar(self.circumradius, RotationRad::new(offset + f64::from(i) * FRAC_PI_4)))
132    }
133
134    /// The octagon as a generic closed [`Polygon`] - the bridge to
135    /// ring algorithms like `segments` and `offset`.
136    pub fn to_polygon(&self) -> Polygon<U> {
137        Polygon::new(self.vertices().to_vec())
138    }
139
140    /// Whether `point` lies inside the octagon - the boundary
141    /// counts as inside.
142    ///
143    /// A regular octagon is the intersection of four strips
144    /// perpendicular to its edge normals: `x` and `y` plus the two
145    /// `±45°` diagonals for `FlatTop`, the `22.5°`-rotated
146    /// counterparts for `PointyTop`.
147    pub fn contains(&self, point: &Vec2<U, f64>) -> bool {
148        let (dx, dy) = (point.x - self.center.x, point.y - self.center.y);
149        let inradius = self.inradius();
150        match self.orientation {
151            PolygonOrientation::FlatTop => {
152                dx.abs() <= inradius && dy.abs() <= inradius && (dx + dy).abs() <= inradius * SQRT_2 && (dx - dy).abs() <= inradius * SQRT_2
153            }
154            PolygonOrientation::PointyTop => {
155                (COS_22_5 * dx + SIN_22_5 * dy).abs() <= inradius
156                    && (SIN_22_5 * dx + COS_22_5 * dy).abs() <= inradius
157                    && (SIN_22_5 * dx - COS_22_5 * dy).abs() <= inradius
158                    && (COS_22_5 * dx - SIN_22_5 * dy).abs() <= inradius
159            }
160        }
161    }
162
163    /// The axis-aligned bounding box: a square `2*inradius x
164    /// 2*inradius` for `FlatTop`, `2R x 2R` for `PointyTop`.
165    pub fn bounding_box(&self) -> Rect<U, f64> {
166        let half = Self::bounding_side_for_inradius(self.inradius(), self.orientation) / 2.0;
167        Rect::new(self.center - Vec2::new(half, half), Vec2::new(half * 2.0, half * 2.0))
168    }
169
170    /// Side of the largest centered axis-aligned square inscribed
171    /// in the octagon, expressed in the inradius.
172    ///
173    /// `FlatTop` keeps a square's corners on the four diagonal
174    /// edges (`side = inradius * sqrt(2)`); `PointyTop` puts the
175    /// corners against the 22.5-degree-slanted edges and fits the
176    /// larger `2 * inradius / (cos(22.5deg) + sin(22.5deg))`.
177    pub fn inscribed_square_side(&self) -> f64 {
178        Self::inscribed_square_side_for_inradius(self.inradius(), self.orientation)
179    }
180
181    /// Side of the largest centered axis-aligned square inscribed
182    /// in an octagon of `inradius` -
183    /// [`Octagon::inscribed_square_side`] without constructing the
184    /// octagon.
185    pub fn inscribed_square_side_for_inradius(inradius: f64, orientation: PolygonOrientation) -> f64 {
186        match orientation {
187            PolygonOrientation::FlatTop => inradius * SQRT_2,
188            PolygonOrientation::PointyTop => 2.0 * inradius / (COS_22_5 + SIN_22_5),
189        }
190    }
191
192    /// The octagon whose inscribed square has `side` - the exact
193    /// inverse of [`Octagon::inscribed_square_side`].
194    pub fn with_inscribed_square(center: Vec2<U>, side: f64, orientation: PolygonOrientation) -> Self {
195        Self::with_inradius(center, Self::inradius_for_inscribed_square(side, orientation), orientation)
196    }
197
198    /// The inradius of the octagon whose inscribed square has
199    /// `side` - [`Octagon::with_inscribed_square`] without
200    /// constructing the octagon.
201    pub fn inradius_for_inscribed_square(side: f64, orientation: PolygonOrientation) -> f64 {
202        match orientation {
203            PolygonOrientation::FlatTop => side / SQRT_2,
204            PolygonOrientation::PointyTop => side * (COS_22_5 + SIN_22_5) / 2.0,
205        }
206    }
207
208    /// Side of the bounding square of the octagon of `inradius`:
209    /// `2 * inradius` for `FlatTop`, `2 * inradius / cos(22.5deg)`
210    /// for `PointyTop`.
211    pub fn bounding_side_for_inradius(inradius: f64, orientation: PolygonOrientation) -> f64 {
212        match orientation {
213            PolygonOrientation::FlatTop => 2.0 * inradius,
214            PolygonOrientation::PointyTop => 2.0 * inradius / COS_22_5,
215        }
216    }
217
218    /// The edge length of a regular octagon of `inradius`:
219    /// `2 * inradius * tan(22.5deg)` - orientation-independent,
220    /// since a regular octagon's apothem is `a = R * cos(22.5deg)`.
221    /// The exact inverse of [`Octagon::inradius_for_side_length`].
222    pub fn side_length_for_inradius(inradius: f64) -> f64 {
223        inradius * 2.0 * SIN_22_5 / COS_22_5
224    }
225
226    /// The inradius of a regular octagon of edge `side` -
227    /// [`Octagon::with_side_length`] takes the same measure via the
228    /// circumradius. The exact inverse of
229    /// [`Octagon::side_length_for_inradius`].
230    pub fn inradius_for_side_length(side: f64) -> f64 {
231        side * COS_22_5 / (2.0 * SIN_22_5)
232    }
233
234    /// The trapezoidal wedge on the `edge`-th side: the outer side is
235    /// the `outer`-scaled edge `squeeze`-narrowed around its midpoint,
236    /// the inner side the parallel edge of the `inner`-scaled octagon.
237    /// `outer`/`inner` are fractions of the inradius: `Fraction::ONE`
238    /// on `outer` puts the outer side on the rim edge, `Fraction::ZERO`
239    /// on `inner` reaches the center, `squeeze = Fraction::ONE` keeps
240    /// the outer side's full edge length.
241    pub fn wedge(&self, edge: usize, outer: Fraction, inner: Fraction, squeeze: Fraction) -> Polygon<U> {
242        let vertices = self.vertices();
243        let scaled = |index: usize, fraction: Fraction| self.center + (vertices[index % 8] - self.center) * fraction.as_f64();
244        let (outer_a, outer_b) = (scaled(edge, outer), scaled(edge + 1, outer));
245        let mid = (outer_a + outer_b) * 0.5;
246        let squeezed = |point: Vec2<U>| mid + (point - mid) * squeeze.as_f64();
247        Polygon::new(vec![squeezed(outer_a), scaled(edge, inner), scaled(edge + 1, inner), squeezed(outer_b)])
248    }
249
250    /// The octagon's eight edge wedges, one trapezoid per side - see
251    /// [`Octagon::wedge`].
252    ///
253    /// ```
254    /// use smearor_commons_geometry::{Fraction, Octagon, PolygonOrientation, Vec2};
255    /// let octagon = Octagon::<()>::flat_top(Vec2::new(0.0, 0.0), 100.0);
256    /// let wedges: Vec<_> = octagon.wedges(Fraction::new(0.97), Fraction::new(0.66), Fraction::new(0.7)).collect();
257    /// assert_eq!(wedges.len(), 8);
258    /// ```
259    pub fn wedges(&self, outer: Fraction, inner: Fraction, squeeze: Fraction) -> OctagonWedges<U> {
260        OctagonWedges::new(*self, outer, inner, squeeze)
261    }
262
263    /// The inradius of the octagon whose bounding square has
264    /// `side` - the exact inverse of
265    /// [`Octagon::bounding_side_for_inradius`].
266    pub fn inradius_for_bounding_side(side: f64, orientation: PolygonOrientation) -> f64 {
267        match orientation {
268            PolygonOrientation::FlatTop => side / 2.0,
269            PolygonOrientation::PointyTop => side * COS_22_5 / 2.0,
270        }
271    }
272
273    /// The largest octagon centered in `bounds` - its square
274    /// bounding box touches the tighter side.
275    pub fn inscribed_in(bounds: &Rect<U, f64>, orientation: PolygonOrientation) -> Self {
276        Self::with_inradius(
277            bounds.center(),
278            Self::inradius_for_bounding_side(bounds.width().min(bounds.height()), orientation),
279            orientation,
280        )
281    }
282
283    /// How far the outline extends along `direction` measured from
284    /// `center` - the exact support function of the convex octagon:
285    /// the largest projection of any vertex.
286    pub fn support(&self, direction: RotationRad) -> f64 {
287        let (sin, cos) = direction.sin_cos();
288        let ray = Vec2::<U>::new(cos, sin);
289        self.vertices().iter().map(|vertex| (*vertex - self.center).dot(&ray)).fold(0.0, f64::max)
290    }
291}

303    /// The three extent specifications produce the same octagon.
304    #[test]
305    fn circumradius_inradius_and_side_length_agree() {
306        let center = Vec2::<()>::new(10.0, 20.0);
307        let by_radius = Octagon::flat_top(center, 100.0);
308        let by_side = Octagon::with_side_length(center, 200.0 * SIN_22_5, PolygonOrientation::FlatTop);
309        let by_inradius = Octagon::with_inradius(center, 100.0 * COS_22_5, PolygonOrientation::FlatTop);
310
311        assert!((by_side.circumradius - 100.0).abs() < 1e-9);
312        assert!((by_inradius.circumradius - 100.0).abs() < 1e-9);
313        assert!((by_radius.inradius() - 100.0 * COS_22_5).abs() < 1e-9);
314        assert!((by_radius.side_length() - 200.0 * SIN_22_5).abs() < 1e-9);
315    }

317    /// `FlatTop` vertices sit on the circumcircle at `22.5°`,
318    /// `67.5°`, ... and `PointyTop` vertices on `0°`, `45°`, ...
319    #[test]
320    fn vertices_follow_the_orientation() {
321        let flat = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0).vertices();
322        assert!((flat[0].x - 100.0 * COS_22_5).abs() < 1e-9 && (flat[0].y - 100.0 * SIN_22_5).abs() < 1e-9);
323        let pointy = Octagon::pointy_top(Vec2::<()>::new(0.0, 0.0), 100.0).vertices();
324        assert!((pointy[0].x - 100.0).abs() < 1e-9 && pointy[0].y.abs() < 1e-9);
325        assert_eq!(flat.len(), 8);
326    }

328    /// The flat-top bounding box is the square `2*inradius` wide.

337    /// The four strips cut the flat-top octagon's corners.
338    #[test]
339    fn contains_the_strips() {
340        let octagon = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0);
341        let apothem = octagon.inradius();
342        // The bbox corner (apothem, apothem) lies outside - the
343        // diagonal edge cuts it.
344        assert!(!octagon.contains(&Vec2::new(apothem, apothem)));
345        // The top edge's midpoint and the right edge's midpoint are
346        // on the boundary - inside.
347        assert!(octagon.contains(&Vec2::new(0.0, apothem)));
348        assert!(octagon.contains(&Vec2::new(apothem, 0.0)));
349        // A diagonal-edge midpoint: the slant between the right and
350        // the top-right vertex centers at 45°.
351        assert!(octagon.contains(&Vec2::new(apothem / SQRT_2, apothem / SQRT_2)));
352        // The center is inside, a point past the vertex is outside.
353        assert!(octagon.contains(&Vec2::new(0.0, 0.0)));
354        assert!(!octagon.contains(&Vec2::new(0.0, 101.0)));
355    }

357    /// `support` matches the largest vertex projection.
358    #[test]
359    fn support_reaches_the_outline() {
360        let octagon = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0);
361        // Along 0° and 45° an edge is perpendicular to the ray - the
362        // support is the apothem; along 22.5° a vertex sits on the
363        // ray - the support is the circumradius.
364        assert!((octagon.support(RotationRad::new(0.0)) - 100.0 * COS_22_5).abs() < 1e-9);
365        assert!((octagon.support(RotationRad::from_degrees(45.0)) - 100.0 * COS_22_5).abs() < 1e-9);
366        assert!((octagon.support(RotationRad::from_degrees(22.5)) - 100.0).abs() < 1e-9);
367    }
