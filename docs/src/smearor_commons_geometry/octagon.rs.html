13/// `cos(22.5°) ≈ 0.9239` - the inradius (apothem) of a regular
14/// octagon with circumradius `1`.
15const COS_22_5: f64 = 0.923_879_532_511_286_7;
16
17/// `sin(22.5°) ≈ 0.3827` - half the edge length of a regular
18/// octagon with circumradius `1`.
19const SIN_22_5: f64 = 0.382_683_432_365_089_8;
20
21/// `sqrt(2)` - the diagonal of the unit square.
22const SQRT_2: f64 = std::f64::consts::SQRT_2;
23
24/// A regular octagon tagged with its coordinate unit `U` and
25/// numeric component type `T`.
26///
27/// A regular octagon is fully determined by a single extent: the
28/// **circumradius** (center to vertex), the **inradius** (center
29/// to edge, also called *apothem*, `circumradius * cos(22.5°)`)
30/// and the **edge length** (`2 * circumradius * sin(22.5°)`). Use
31/// [`Self::with_inradius`] or [`Self::with_side_length`] when the
32/// octagon is specified by one of the other extents.
33///
34/// `T` defaults to `f64` for domain math. The type is a technical
35/// primitive without validation: `circumradius` should be
36/// non-negative, stricter bounds belong to consumer wrappers.
37#[derive(Debug, Serialize, Deserialize)]
38pub struct Octagon<U = (), T = f64> {
39    /// The center point.
40    pub center: Vec2<U, T>,
41    /// The circumradius - distance from `center` to each vertex.
42    pub circumradius: T,
43    /// Whether the octagon has a horizontal top edge
44    /// ([`PolygonOrientation::FlatTop`]) or a vertex on top
45    /// ([`PolygonOrientation::PointyTop`]).
46    pub orientation: PolygonOrientation,
47}
48

53        Self::new(self.center.clone(), self.circumradius.clone(), self.orientation)
54    }

60    }

73impl<U, T> Octagon<U, T> {
74    /// Creates an octagon from its center, circumradius and
75    /// orientation.
76    pub const fn new(center: Vec2<U, T>, circumradius: T, orientation: PolygonOrientation) -> Self {
77        Self {
78            center,
79            circumradius,
80            orientation,
81        }
82    }
83
84    /// Creates a [`PolygonOrientation::FlatTop`] octagon.
85    pub const fn flat_top(center: Vec2<U, T>, circumradius: T) -> Self {
86        Self::new(center, circumradius, PolygonOrientation::FlatTop)
87    }
88
89    /// Creates a [`PolygonOrientation::PointyTop`] octagon.
90    pub const fn pointy_top(center: Vec2<U, T>, circumradius: T) -> Self {
91        Self::new(center, circumradius, PolygonOrientation::PointyTop)
92    }
93}

95impl<U> Octagon<U, f64> {
96    /// Creates an octagon from its center, edge length and
97    /// orientation. The edge length of a regular octagon is
98    /// `2 * circumradius * sin(22.5°)`.
99    pub fn with_side_length(center: Vec2<U, f64>, side_length: f64, orientation: PolygonOrientation) -> Self {
100        Self::new(center, side_length / (2.0 * SIN_22_5), orientation)
101    }
102
103    /// Creates an octagon from its center, inradius and
104    /// orientation. The inradius (apothem, center to edge) is
105    /// `circumradius * cos(22.5°)`.
106    pub fn with_inradius(center: Vec2<U, f64>, inradius: f64, orientation: PolygonOrientation) -> Self {
107        Self::new(center, inradius / COS_22_5, orientation)
108    }
109
110    /// The edge length - `2 * circumradius * sin(22.5°)`.
111    pub fn side_length(&self) -> f64 {
112        self.circumradius * 2.0 * SIN_22_5
113    }
114
115    /// The inradius (apothem) - distance from `center` to each
116    /// edge, `circumradius * cos(22.5°)`.
117    pub fn inradius(&self) -> f64 {
118        self.circumradius * COS_22_5
119    }
120
121    /// The eight vertices in counter-clockwise order starting from
122    /// angle `22.5°` (`FlatTop`) resp. `0°` (`PointyTop`) in steps
123    /// of `45°`.
124    pub fn vertices(&self) -> [Vec2<U, f64>; 8] {
125        let offset = match self.orientation {
126            PolygonOrientation::FlatTop => FRAC_PI_8,
127            PolygonOrientation::PointyTop => 0.0,
128        };
129        [0, 1, 2, 3, 4, 5, 6, 7].map(|i| self.center + Vec2::from_polar(self.circumradius, RotationRad::new(offset + f64::from(i) * FRAC_PI_4)))
130    }
131
132    /// The octagon as a generic closed [`Polygon`] - the bridge to
133    /// ring algorithms like `segments` and `offset`.
134    pub fn to_polygon(&self) -> Polygon<U> {
135        Polygon::new(self.vertices().to_vec())
136    }
137
138    /// Whether `point` lies inside the octagon - the boundary
139    /// counts as inside.
140    ///
141    /// A regular octagon is the intersection of four strips
142    /// perpendicular to its edge normals: `x` and `y` plus the two
143    /// `±45°` diagonals for `FlatTop`, the `22.5°`-rotated
144    /// counterparts for `PointyTop`.
145    pub fn contains(&self, point: &Vec2<U, f64>) -> bool {
146        let (dx, dy) = (point.x - self.center.x, point.y - self.center.y);
147        let inradius = self.inradius();
148        match self.orientation {
149            PolygonOrientation::FlatTop => {
150                dx.abs() <= inradius && dy.abs() <= inradius && (dx + dy).abs() <= inradius * SQRT_2 && (dx - dy).abs() <= inradius * SQRT_2
151            }
152            PolygonOrientation::PointyTop => {
153                (COS_22_5 * dx + SIN_22_5 * dy).abs() <= inradius
154                    && (SIN_22_5 * dx + COS_22_5 * dy).abs() <= inradius
155                    && (SIN_22_5 * dx - COS_22_5 * dy).abs() <= inradius
156                    && (COS_22_5 * dx - SIN_22_5 * dy).abs() <= inradius
157            }
158        }
159    }
160
161    /// The axis-aligned bounding box: a square `2*inradius x
162    /// 2*inradius` for `FlatTop`, `2R x 2R` for `PointyTop`.
163    pub fn bounding_box(&self) -> Rect<U, f64> {
164        let half = Self::bounding_side_for_inradius(self.inradius(), self.orientation) / 2.0;
165        Rect::new(self.center - Vec2::new(half, half), Vec2::new(half * 2.0, half * 2.0))
166    }
167
168    /// Side of the largest centered axis-aligned square inscribed
169    /// in the octagon, expressed in the inradius.
170    ///
171    /// `FlatTop` keeps a square's corners on the four diagonal
172    /// edges (`side = inradius * sqrt(2)`); `PointyTop` puts the
173    /// corners against the 22.5-degree-slanted edges and fits the
174    /// larger `2 * inradius / (cos(22.5deg) + sin(22.5deg))`.
175    pub fn inscribed_square_side(&self) -> f64 {
176        Self::inscribed_square_side_for_inradius(self.inradius(), self.orientation)
177    }
178
179    /// Side of the largest centered axis-aligned square inscribed
180    /// in an octagon of `inradius` -
181    /// [`Octagon::inscribed_square_side`] without constructing the
182    /// octagon.
183    pub fn inscribed_square_side_for_inradius(inradius: f64, orientation: PolygonOrientation) -> f64 {
184        match orientation {
185            PolygonOrientation::FlatTop => inradius * SQRT_2,
186            PolygonOrientation::PointyTop => 2.0 * inradius / (COS_22_5 + SIN_22_5),
187        }
188    }
189
190    /// The octagon whose inscribed square has `side` - the exact
191    /// inverse of [`Octagon::inscribed_square_side`].
192    pub fn with_inscribed_square(center: Vec2<U>, side: f64, orientation: PolygonOrientation) -> Self {
193        Self::with_inradius(center, Self::inradius_for_inscribed_square(side, orientation), orientation)
194    }
195
196    /// The inradius of the octagon whose inscribed square has
197    /// `side` - [`Octagon::with_inscribed_square`] without
198    /// constructing the octagon.
199    pub fn inradius_for_inscribed_square(side: f64, orientation: PolygonOrientation) -> f64 {
200        match orientation {
201            PolygonOrientation::FlatTop => side / SQRT_2,
202            PolygonOrientation::PointyTop => side * (COS_22_5 + SIN_22_5) / 2.0,
203        }
204    }
205
206    /// Side of the bounding square of the octagon of `inradius`:
207    /// `2 * inradius` for `FlatTop`, `2 * inradius / cos(22.5deg)`
208    /// for `PointyTop`.
209    pub fn bounding_side_for_inradius(inradius: f64, orientation: PolygonOrientation) -> f64 {
210        match orientation {
211            PolygonOrientation::FlatTop => 2.0 * inradius,
212            PolygonOrientation::PointyTop => 2.0 * inradius / COS_22_5,
213        }
214    }
215
216    /// The inradius of the octagon whose bounding square has
217    /// `side` - the exact inverse of
218    /// [`Octagon::bounding_side_for_inradius`].
219    pub fn inradius_for_bounding_side(side: f64, orientation: PolygonOrientation) -> f64 {
220        match orientation {
221            PolygonOrientation::FlatTop => side / 2.0,
222            PolygonOrientation::PointyTop => side * COS_22_5 / 2.0,
223        }
224    }
225
226    /// How far the outline extends along `direction` measured from
227    /// `center` - the exact support function of the convex octagon:
228    /// the largest projection of any vertex.
229    pub fn support(&self, direction: RotationRad) -> f64 {
230        let (sin, cos) = direction.sin_cos();
231        let ray = Vec2::<U>::new(cos, sin);
232        self.vertices().iter().map(|vertex| (*vertex - self.center).dot(&ray)).fold(0.0, f64::max)
233    }
234}

246    /// The three extent specifications produce the same octagon.
247    #[test]
248    fn circumradius_inradius_and_side_length_agree() {
249        let center = Vec2::<()>::new(10.0, 20.0);
250        let by_radius = Octagon::flat_top(center, 100.0);
251        let by_side = Octagon::with_side_length(center, 200.0 * SIN_22_5, PolygonOrientation::FlatTop);
252        let by_inradius = Octagon::with_inradius(center, 100.0 * COS_22_5, PolygonOrientation::FlatTop);
253
254        assert!((by_side.circumradius - 100.0).abs() < 1e-9);
255        assert!((by_inradius.circumradius - 100.0).abs() < 1e-9);
256        assert!((by_radius.inradius() - 100.0 * COS_22_5).abs() < 1e-9);
257        assert!((by_radius.side_length() - 200.0 * SIN_22_5).abs() < 1e-9);
258    }

260    /// `FlatTop` vertices sit on the circumcircle at `22.5°`,
261    /// `67.5°`, ... and `PointyTop` vertices on `0°`, `45°`, ...
262    #[test]
263    fn vertices_follow_the_orientation() {
264        let flat = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0).vertices();
265        assert!((flat[0].x - 100.0 * COS_22_5).abs() < 1e-9 && (flat[0].y - 100.0 * SIN_22_5).abs() < 1e-9);
266        let pointy = Octagon::pointy_top(Vec2::<()>::new(0.0, 0.0), 100.0).vertices();
267        assert!((pointy[0].x - 100.0).abs() < 1e-9 && pointy[0].y.abs() < 1e-9);
268        assert_eq!(flat.len(), 8);
269    }

271    /// The flat-top bounding box is the square `2*inradius` wide.
272    #[test]
273    fn flat_top_bounding_box_is_a_square() {

275        let bbox = octagon.bounding_box();
276        assert!((bbox.size.x - bbox.size.y).abs() < 1e-9);

280    /// The four strips cut the flat-top octagon's corners.
281    #[test]
282    fn contains_the_strips() {
283        let octagon = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0);
284        let apothem = octagon.inradius();
285        // The bbox corner (apothem, apothem) lies outside - the
286        // diagonal edge cuts it.
287        assert!(!octagon.contains(&Vec2::new(apothem, apothem)));
288        // The top edge's midpoint and the right edge's midpoint are
289        // on the boundary - inside.
290        assert!(octagon.contains(&Vec2::new(0.0, apothem)));
291        assert!(octagon.contains(&Vec2::new(apothem, 0.0)));
292        // A diagonal-edge midpoint: the slant between the right and
293        // the top-right vertex centers at 45°.
294        assert!(octagon.contains(&Vec2::new(apothem / SQRT_2, apothem / SQRT_2)));
295        // The center is inside, a point past the vertex is outside.
296        assert!(octagon.contains(&Vec2::new(0.0, 0.0)));
297        assert!(!octagon.contains(&Vec2::new(0.0, 101.0)));
298    }

300    /// `support` matches the largest vertex projection.
301    #[test]
302    fn support_reaches_the_outline() {
303        let octagon = Octagon::flat_top(Vec2::<()>::new(0.0, 0.0), 100.0);
304        // Along 0° and 45° an edge is perpendicular to the ray - the
305        // support is the apothem; along 22.5° a vertex sits on the
306        // ray - the support is the circumradius.
307        assert!((octagon.support(RotationRad::new(0.0)) - 100.0 * COS_22_5).abs() < 1e-9);
308        assert!((octagon.support(RotationRad::from_degrees(45.0)) - 100.0 * COS_22_5).abs() < 1e-9);
309        assert!((octagon.support(RotationRad::from_degrees(22.5)) - 100.0).abs() < 1e-9);
310    }

325        for orientation in [PolygonOrientation::FlatTop, PolygonOrientation::PointyTop] {
326            let octagon = Octagon::with_inradius(Vec2::<()>::new(0.0, 0.0), 50.0, orientation);
