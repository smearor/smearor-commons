10/// A closed disk / circle tagged with its coordinate unit `U` and
11/// numeric component type `T`.
12///
13/// `T` defaults to `f64` for domain math. The type is a technical
14/// primitive without validation: `radius` should be non-negative,
15/// stricter bounds belong to consumer wrappers.
16#[derive(Debug, Serialize, Deserialize)]
17pub struct Circle<U = (), T = f64> {
18    /// The center point.
19    pub center: Vec2<U, T>,
20    /// The radius.
21    pub radius: T,
22}
23
24impl<U, T: Copy> Copy for Circle<U, T> {}
25
26impl<U, T: Clone> Clone for Circle<U, T> {
27    fn clone(&self) -> Self {
28        Self::new(self.center.clone(), self.radius.clone())
29    }
30}
31
32impl<U, T: PartialEq> PartialEq for Circle<U, T> {
33    fn eq(&self, other: &Self) -> bool {
34        self.center == other.center && self.radius == other.radius
35    }
36}
37
38impl<U, T: Eq> Eq for Circle<U, T> {}
39
40impl<U, T: Hash> Hash for Circle<U, T> {
41    fn hash<H: Hasher>(&self, state: &mut H) {
42        self.center.hash(state);
43        self.radius.hash(state);
44    }
45}

47impl<U, T> Circle<U, T> {
48    /// Creates a circle from its center and radius.
49    pub const fn new(center: Vec2<U, T>, radius: T) -> Self {
50        Self { center, radius }
51    }
52}
53
54impl<U> Circle<U, f64> {
55    /// Whether `point` lies inside the disk - the boundary counts
56    /// as inside (`distance <= radius`).
57    pub fn contains(&self, point: &Vec2<U, f64>) -> bool {
58        self.center.distance_f64(point) <= self.radius
59    }
60
61    /// Whether the two disks share interior points - boundary touch
62    /// alone does not count (`distance < r1 + r2`).
63    pub fn overlaps(&self, other: &Self) -> bool {
64        self.center.distance_f64(&other.center) < self.radius + other.radius
65    }
66
67    /// The axis-aligned square fully containing the disk
68    /// (`center ± radius` on both axes).
69    pub fn bounding_box(&self) -> Rect<U, f64> {
70        let diameter = self.diameter_f64();
71        Rect::centered(self.center, Vec2::new(diameter, diameter))
72    }
73
74    /// The radius in the unit `U` - the unit-carrying counterpart of
75    /// the raw `radius` field, like [`Vec2::x_unit`].
76    pub fn radius_unit(&self) -> U
77    where
78        U: From<f64>,
79    {
80        U::from(self.radius)
81    }
82
83    /// `2 * radius`, in the unit `U`.
84    pub fn diameter(&self) -> U
85    where
86        U: From<f64>,
87    {
88        U::from(self.diameter_f64())
89    }
90
91    /// `2 * radius`, as a plain `f64` - the unit-erased counterpart
92    /// for `Circle<()>` and unit-free math.
93    pub fn diameter_f64(&self) -> f64 {
94        self.radius * 2.0
95    }
96
97    /// `PI * radius^2` - the caller picks the area type `P` (like
98    /// [`Vec2::dot`]'s product type): [`crate::SquareMillimeters`] for
99    /// `Circle<Millimeters>`, `f64` for `Circle<()>` and unit-free math.
100    pub fn area<P: From<f64>>(&self) -> P {
101        P::from(PI * self.radius * self.radius)
102    }
103
104    /// `2 * PI * radius`, in the unit `U`.
105    pub fn circumference(&self) -> U
106    where
107        U: From<f64>,
108    {
109        U::from(self.circumference_f64())
110    }
111
112    /// `2 * PI * radius`, as a plain `f64` - see
113    /// [`Circle::diameter_f64`].
114    pub fn circumference_f64(&self) -> f64 {
115        2.0 * PI * self.radius
116    }
117
118    /// Side of the largest axis-aligned square inscribed in the
119    /// circle: its four corners rest on the outline, so
120    /// `side = radius * sqrt(2)`.
121    pub fn inscribed_square_side(&self) -> f64 {
122        self.radius * SQRT_2
123    }
124
125    /// The circle circumscribing a centered axis-aligned square of
126    /// `side` - the exact inverse of
127    /// [`Circle::inscribed_square_side`].
128    pub fn with_inscribed_square(center: Vec2<U>, side: f64) -> Self {
129        Self::new(center, side / SQRT_2)
130    }
131
132    /// Rasterizes the disk into 1-px-high integer rectangles - one
133    /// [`Rect<U, i32>`] per covered scanline, in the circle's own
134    /// coordinate space.
135    ///
136    /// Each row uses the chord at its center-nearest edge (the widest
137    /// the circle reaches inside that row), so the union covers every
138    /// point [`contains`](Self::contains) would hit - an outer
139    /// rasterization suitable for rectangle-union regions like
140    /// `cairo::Region`.
141    pub fn region_rects(&self) -> Vec<Rect<U, i32>> {
142        if self.radius <= 0.0 {
143            return Vec::new();
144        }
145        let bounds = self.bounding_box();
146        let top = bounds.min.y.floor() as i32;
147        // `floor + 1` keeps the row containing the bottom edge inside
148        // the range even when it lands exactly on a scanline boundary
149        // (`ceil` would exclude it there).
150        let bottom = bounds.bottom().floor() as i32 + 1;
151        (top..bottom)
152            .filter_map(|y| {
153                // Distance from the center to the row's nearest edge -
154                // zero when the row straddles the center line.
155                let nearest = (f64::from(y) - self.center.y).max(self.center.y - f64::from(y + 1)).max(0.0);
156                if nearest > self.radius {
157                    return None;
158                }
159                let half = (self.radius * self.radius - nearest * nearest).sqrt().max(0.5);
160                let left = (self.center.x - half).floor() as i32;
161                // Same edge rule as `bottom`: include the column the
162                // right edge lands on when it is exactly integer.
163                let right = (self.center.x + half).floor() as i32 + 1;
164                Some(Rect::from_xywh(left, y, right - left, 1))
165            })
166            .collect()
167    }
168
169    /// Rasterizes the disk into a cairo [`Region`](cairo::Region) in
170    /// the circle's own coordinate space - one
171    /// [`RectangleInt`](cairo::RectangleInt) per
172    /// [`region_rects`](Self::region_rects) strip. Used e.g. for
173    /// circular dead zones inside a wayland surface's input region.
174    ///
175    /// # Errors
176    ///
177    /// Propagates [`cairo::Error`]s from `Region` construction.
178    #[cfg(feature = "cairo")]
179    pub fn region(&self) -> Result<cairo::Region, cairo::Error> {
180        let region = cairo::Region::create();
181        for rect in self.region_rects() {
182            region.union_rectangle(&rect.into())?;
183        }
184        Ok(region)
185    }
186}

197    /// Points inside, on and outside the disk.
198    #[test]
199    fn contains_includes_boundary() {
200        let circle = Circle::<Pixels>::new(Vec2::new(10.0, 20.0), 5.0);
201        assert!(circle.contains(&Vec2::new(10.0, 20.0)));
202        assert!(circle.contains(&Vec2::new(15.0, 20.0)));
203        assert!(!circle.contains(&Vec2::new(16.0, 20.0)));
204    }

206    /// Strict interior overlap: touching boundaries do not count.
207    #[test]
208    fn overlaps_is_strict() {
209        let a = Circle::<()>::new(Vec2::new(0.0, 0.0), 1.0);
210        let touching = Circle::<()>::new(Vec2::new(2.0, 0.0), 1.0);
211        let overlapping = Circle::<()>::new(Vec2::new(1.9, 0.0), 1.0);
212        let separate = Circle::<()>::new(Vec2::new(3.0, 0.0), 1.0);
213        assert!(!a.overlaps(&touching));
214        assert!(a.overlaps(&overlapping));
215        assert!(!a.overlaps(&separate));
216        assert!(a.overlaps(&a));
217    }

219    /// The bounding box is the square `center ± radius`.
220    #[test]
221    fn bounding_box_is_circumscribed_square() {
222        let circle = Circle::<Pixels>::new(Vec2::new(10.0, 20.0), 5.0);
223        let bounds = circle.bounding_box();
224        assert_eq!(bounds.min, Vec2::new(5.0, 15.0));
225        assert_eq!(bounds.max(), Vec2::new(15.0, 25.0));
226        assert_eq!(bounds.center(), Vec2::new(10.0, 20.0));
227    }

229    /// The classic measures follow the radius - plain `f64` on the
230    /// unit-erased `Circle<()>`.
231    #[test]
232    fn circle_measures() {
233        let circle = Circle::<()>::new(Vec2::new(0.0, 0.0), 2.0);
234        assert_eq!(circle.diameter_f64(), 4.0);
235        let area: f64 = circle.area();
236        assert!((area - 4.0 * PI).abs() < 1e-9);
237        assert!((circle.circumference_f64() - 4.0 * PI).abs() < 1e-9);
238        let unit = Circle::<()>::new(Vec2::new(0.0, 0.0), FRAC_1_SQRT_2);
239        let unit_area: f64 = unit.area();
240        assert!((unit_area - PI / 2.0).abs() < 1e-9);
241    }

243    /// The scanline rasterization covers every point `contains`
244    /// reports inside - an outer rasterization, so every integer point
245    /// of a contained point's pixel is unioned.
246    #[test]
247    fn region_rects_cover_contained_points() {
248        let circle = Circle::<Pixels>::new(Vec2::new(10.0, 10.0), 5.0);
249        let rects = circle.region_rects();
250        assert!(!rects.is_empty());
251        for dy in -6..=6 {
252            for dx in -6..=6 {
253                let point = Vec2::new(10.0 + f64::from(dx), 10.0 + f64::from(dy));
254                if circle.contains(&point) {
255                    let covered = rects.iter().any(|rect| {
256                        point.x >= f64::from(rect.min.x)
257                            && point.x < f64::from(rect.min.x + rect.size.x)
258                            && point.y >= f64::from(rect.min.y)
259                            && point.y < f64::from(rect.min.y + rect.size.y)
260                    });
261                    assert!(covered, "contained point {point:?} not covered");
262                }
263            }
264        }
265    }

267    /// Measures carry the coordinate unit when `U` is a unit newtype.
268    #[test]
269    fn measures_carry_the_unit() {
270        let circle = Circle::<Pixels>::new(Vec2::new(0.0, 0.0), 2.0);
271        assert_eq!(circle.radius_unit(), Pixels::new(2.0));
272        assert_eq!(circle.diameter(), Pixels::new(4.0));
273        assert_eq!(circle.circumference(), Pixels::new(4.0 * PI));

278    /// The inscribed square's corners rest on the circle.
279    #[test]
280    fn inscribed_square_corners_touch_the_circle() {
281        let circle = Circle::<()>::new(Vec2::new(0.0, 0.0), 10.0);
282        let side = circle.inscribed_square_side();
283        // Corner distance: sqrt(2) * side/2 = radius.
284        assert!((side * std::f64::consts::FRAC_1_SQRT_2 - 10.0).abs() < 1e-9);
285        let back = Circle::<()>::with_inscribed_square(Vec2::new(0.0, 0.0), side);
286        assert!((back.radius - 10.0).abs() < 1e-9);
287    }
