9/// A half-disk ("dome") - the upper half of a circle, bounded below
10/// by the diameter edge through [`Semicircle::center`].
11///
12/// `center` is the midpoint of the flat (diameter) edge - the center
13/// of the virtual full circle - and the dome extends to smaller `y`
14/// in the usual screen convention (`point.y <= center.y`). Rotate or
15/// mirror the coordinate frame for domes opening in other directions.
16///
17/// `T` defaults to `f64` for domain math. The type is a technical
18/// primitive without validation: `radius >= 0` for meaningful
19/// geometry, stricter bounds belong to consumer wrappers.
20#[derive(Debug, Serialize, Deserialize)]
21pub struct Semicircle<U = (), T = f64> {
22    /// The center of the virtual full circle - the midpoint of the
23    /// flat diameter edge the dome sits on.
24    pub center: Vec2<U, T>,
25    /// The radius of the virtual full circle.
26    pub radius: T,
27}
28
29impl<U, T: Copy> Copy for Semicircle<U, T> {}
30
31impl<U, T: Clone> Clone for Semicircle<U, T> {
32    fn clone(&self) -> Self {
33        Self::new(self.center.clone(), self.radius.clone())
34    }
35}
36
37impl<U, T: PartialEq> PartialEq for Semicircle<U, T> {
38    fn eq(&self, other: &Self) -> bool {
39        self.center == other.center && self.radius == other.radius
40    }
41}
42
43impl<U, T: Eq> Eq for Semicircle<U, T> {}
44
45impl<U, T: Hash> Hash for Semicircle<U, T> {
46    fn hash<H: Hasher>(&self, state: &mut H) {
47        self.center.hash(state);
48        self.radius.hash(state);
49    }
50}

52impl<U, T> Semicircle<U, T> {
53    /// Creates a dome sitting on the diameter edge through `center`.
54    pub const fn new(center: Vec2<U, T>, radius: T) -> Self {
55        Self { center, radius }
56    }
57}
58
59impl<U> Semicircle<U, f64> {
60    /// Whether `point` lies in the dome - inside the virtual circle
61    /// and on its flat edge's dome side (`point.y <= center.y`,
62    /// boundary inclusive).
63    pub fn contains(&self, point: &Vec2<U, f64>) -> bool {
64        self.center.distance_f64(point) <= self.radius && point.y <= self.center.y
65    }
66
67    /// The axis-aligned bounding box: `2r` wide and `r` tall, the
68    /// diameter edge as its bottom side.
69    pub fn bounding_box(&self) -> Rect<U, f64> {
70        Rect::new(self.center - Vec2::new(self.radius, self.radius), Vec2::new(self.radius * 2.0, self.radius))
71    }
72
73    /// How far the outline extends along `direction` measured from
74    /// `center` - the support function of the half-disk: `r` for any
75    /// direction into the dome half (the arc is part of the full
76    /// circle), `r * |cos|` along the flat edge and below it (only
77    /// the diameter endpoints bound those directions).
78    pub fn support(&self, direction: RotationRad) -> f64 {
79        let (sin, cos) = direction.sin_cos();
80        if sin < 0.0 { self.radius } else { self.radius * cos.abs() }
81    }
82
83    /// Side of the largest axis-aligned square standing on the
84    /// diameter edge with both top corners on the arc: `2r / sqrt(5)`.
85    pub fn inscribed_square_side(&self) -> f64 {
86        self.radius * 2.0 / 5.0_f64.sqrt()
87    }
88
89    /// The dome whose inscribed square stands on the diameter edge
90    /// through `center` with `side` - the exact inverse of
91    /// [`Semicircle::inscribed_square_side`].
92    pub fn with_inscribed_square(center: Vec2<U>, side: f64) -> Self {
93        Self::new(center, side * 5.0_f64.sqrt() / 2.0)
94    }
95}

97#[cfg(test)]
98mod tests {
99    use super::Semicircle;
100    use crate::Pixels;
101    use crate::RotationRad;
102    use crate::Vec2;
103
104    /// Points inside the dome, past the flat edge and off the sides.
105    #[test]
106    fn contains_uses_the_diameter_edge() {
107        let dome = Semicircle::<Pixels>::new(Vec2::new(0.0, 0.0), 10.0);
108        assert!(dome.contains(&Vec2::new(0.0, 0.0)));
109        assert!(dome.contains(&Vec2::new(0.0, -10.0)));
110        assert!(dome.contains(&Vec2::new(9.9, -0.5)));
111        assert!(!dome.contains(&Vec2::new(0.0, 1.0)));
112        assert!(!dome.contains(&Vec2::new(11.0, -0.5)));
113    }
114
115    /// The bounding box is `2r x r` with the flat edge at the bottom.
116    #[test]
117    fn bounding_box_is_wide_and_half_tall() {
118        let dome = Semicircle::<()>::new(Vec2::new(50.0, 60.0), 20.0);
119        let bounds = dome.bounding_box();
120        assert_eq!(bounds.width(), 40.0);
121        assert_eq!(bounds.height(), 20.0);
122        assert_eq!(bounds.bottom(), 60.0);
123    }
124
125    /// The inscribed square and its inverse agree.
126    #[test]
127    fn inscribed_square_roundtrips() {
128        let dome = Semicircle::<()>::new(Vec2::new(0.0, 0.0), 10.0);
129        let side = dome.inscribed_square_side();
130        // Top corners rest on the arc: (s/2)^2 + s^2 = r^2.
131        assert!((side.powi(2) / 4.0 + side.powi(2) - 100.0).abs() < 1e-9);
132        let back = Semicircle::<()>::with_inscribed_square(Vec2::new(0.0, 0.0), side);
133        assert!((back.radius - 10.0).abs() < 1e-9);
134    }
135
136    /// Support reaches `r` upward and `r|cos|` along the flat edge.
137    #[test]
138    fn support_follows_the_dome_side() {
139        let dome = Semicircle::<()>::new(Vec2::new(0.0, 0.0), 10.0);
140        assert!((dome.support(RotationRad::new(-std::f64::consts::FRAC_PI_2)) - 10.0).abs() < 1e-9);
141        assert!((dome.support(RotationRad::ZERO) - 10.0).abs() < 1e-9);
142        assert!(dome.support(RotationRad::QUARTER_TURN).abs() < 1e-9);
143    }
