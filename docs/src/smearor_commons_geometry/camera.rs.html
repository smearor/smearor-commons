9/// An orthographic orbit camera: a `yaw`/`pitch` view rotation plus
10/// a world-to-screen `scale`, projecting [`Vec3`] world points to
11/// screen pixels together with a painter-sort depth.
12///
13/// The camera carries no viewport - pass the viewport extent per
14/// call to [`Self::project`].
15#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
16#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
17pub struct Camera {
18    /// Azimuth around the `y` axis, applied first.
19    pub yaw: RotationRad,
20    /// Elevation around the `x` axis, applied after `yaw`.
21    pub pitch: RotationRad,
22    /// Screen pixels per world unit.
23    pub scale: ScaleFactor,
24}
25
26impl Camera {
27    /// The unrotated camera at unit scale.
28    pub const IDENTITY: Self = Self {
29        yaw: RotationRad::ZERO,
30        pitch: RotationRad::ZERO,
31        scale: ScaleFactor::ONE,
32    };
33
34    /// The point in camera space: rotated by `yaw`, then `pitch`.
35    /// `+z` points toward the viewer.
36    pub fn view<U>(&self, point: Vec3<U>) -> Vec3<U> {
37        point.rotated_y(self.yaw).rotated_x(self.pitch)
38    }
39
40    /// The inverse of [`Self::view`]: a camera-space point back to
41    /// world space - undoes the pitch, then the yaw.
42    pub fn unview<U>(&self, point: Vec3<U>) -> Vec3<U> {
43        point.rotated_x(-self.pitch).rotated_y(-self.yaw)
44    }
45
46    /// The screen position (in `Pixels`, `x` rightward and `y`
47    /// downward from the viewport center) and the painter depth -
48    /// larger means further away - of `point` inside a `viewport`.
49    pub fn project<U>(&self, point: Vec3<U>, viewport: Vec2<Pixels>) -> (Vec2<Pixels>, f64) {
50        let view = self.view(point);
51        let screen = Vec2::<Pixels>::new(viewport.x / 2.0 + view.x * self.scale.as_f64(), viewport.y / 2.0 - view.y * self.scale.as_f64());
52        (screen, -view.z)
53    }
54}

56#[cfg(test)]
57mod tests {
58    use super::*;
59    use std::f64::consts::FRAC_PI_2;
60
61    #[test]
62    fn identity_maps_center_to_viewport_center() {
63        let (screen, depth) = Camera::IDENTITY.project(Vec3::<()>::new(0.0, 0.0, 0.0), Vec2::<Pixels>::new(100.0, 80.0));
64        assert_eq!(screen, Vec2::new(50.0, 40.0));
65        assert_eq!(depth, 0.0);
66    }
67
68    #[test]
69    fn project_scales_and_flips_y() {
70        let camera = Camera {
71            scale: ScaleFactor::new(2.0),
72            ..Camera::IDENTITY
73        };
74        let (screen, _) = camera.project(Vec3::<()>::new(10.0, 5.0, 0.0), Vec2::<Pixels>::new(100.0, 100.0));
75        assert_eq!(screen, Vec2::new(70.0, 40.0), "x+20, y-10 from center");
76    }
77
78    #[test]
79    fn yaw_90_turns_z_into_view_x() {
80        let camera = Camera {
81            yaw: RotationRad::new(FRAC_PI_2),
82            ..Camera::IDENTITY
83        };
84        let view = camera.view(Vec3::<()>::new(0.0, 0.0, 1.0));
85        assert!((view.x - 1.0).abs() < 1e-12 && view.z.abs() < 1e-12);
86    }
87
88    #[test]
89    fn unview_inverts_view() {
90        let cam = Camera {
91            yaw: RotationRad::new(0.7),
92            pitch: RotationRad::new(-0.4),
93            scale: ScaleFactor::new(2.0),
94        };
95        let point = Vec3::<()>::new(1.0, 2.0, 3.0);
96        let back = cam.unview(cam.view(point));
97        for (a, b) in [(back.x, point.x), (back.y, point.y), (back.z, point.z)] {
98            assert!((a - b).abs() < 1e-12);
99        }
100    }
101
102    #[test]
103    fn depth_grows_away_from_the_viewer() {
104        let (_, near) = Camera::IDENTITY.project(Vec3::<()>::new(0.0, 0.0, 1.0), Vec2::<Pixels>::new(10.0, 10.0));
105        let (_, far) = Camera::IDENTITY.project(Vec3::<()>::new(0.0, 0.0, -1.0), Vec2::<Pixels>::new(10.0, 10.0));
106        assert!(far > near);
107    }
