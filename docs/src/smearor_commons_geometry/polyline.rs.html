13/// An open line of points in unit `U`: river courses, streets,
14/// decorative chords, edge routes - the last point does not connect
15/// back to the first. Derefs to the `[Vec2]` slice so every
16/// `&[Vec2]` geometry helper keeps working.
17///
18/// Serializes flat: `[x0, y0, x1, y1, ...]`.
19#[derive(Debug, Clone, Default, PartialEq)]
20pub struct Polyline<U>(pub Vec<Vec2<U>>);
21
22impl<U> Polyline<U> {
23    /// A polyline from its points.
24    pub fn new(points: Vec<Vec2<U>>) -> Self {
25        Self(points)
26    }
27
28    /// The polyline's points.
29    pub fn points(&self) -> &[Vec2<U>] {
30        &self.0
31    }
32}
33
34impl<U: Copy> Polyline<U> {
35    /// The polyline's point-to-point segments.
36    pub fn segments(&self) -> impl Iterator<Item = Segment<U>> + '_ {
37        self.0.windows(2).map(|pair| Segment::new(pair[0], pair[1]))
38    }
39
40    /// Whether `other` strictly crosses the polyline.
41    pub fn crosses(&self, other: &Polyline<U>) -> bool {
42        self.segments().any(|a| other.segments().any(|b| a.crosses(&b)))
43    }
44
45    /// Corner-cutting subdivision (Chaikin): each round replaces
46    /// every segment by points at 1/4 and 3/4, rounding corners
47    /// and roughly doubling the point count - denser spline
48    /// checkpoints and no miter spikes at sharp bends.
49    /// `pinned` lists point indices (of the original polyline)
50    /// that stay on the course exactly.
51    pub fn smoothed(&self, rounds: usize, pinned: &[usize]) -> Self {
52        let mut flags: Vec<bool> = self.0.iter().enumerate().map(|(index, _)| pinned.contains(&index)).collect();
53        let mut current = self.0.clone();
54        for _ in 0..rounds {
55            if current.len() < 3 {
56                break;
57            }
58            let mut next = Vec::with_capacity(current.len() * 2);
59            let mut next_flags = Vec::with_capacity(flags.len() * 2);
60            next.push(current[0]);
61            next_flags.push(flags[0]);
62            for (index, window) in current.windows(2).enumerate() {
63                let (from, to) = (window[0], window[1]);
64                next.push(Vec2::new(from.x * 0.75 + to.x * 0.25, from.y * 0.75 + to.y * 0.25));
65                next.push(Vec2::new(from.x * 0.25 + to.x * 0.75, from.y * 0.25 + to.y * 0.75));
66                next_flags.extend_from_slice(&[false, false]);
67                if flags.get(index + 1).copied().unwrap_or(false) {
68                    next.push(to);
69                    next_flags.push(true);
70                }
71            }
72            if let (Some(last), Some(flag)) = (current.last(), flags.last()) {
73                next.push(*last);
74                next_flags.push(*flag);
75            }
76            (current, flags) = (next, next_flags);
77        }
78        Self::new(current)
79    }
80
81    /// The polyline resampled to `count` evenly spaced points along
82    /// its arc length - the $-family recognizers' canonical
83    /// normalization step: kills sampling-rate noise and makes
84    /// pointwise comparisons (e.g. discrete Fréchet) well-defined.
85    /// Fewer than two source points or a degenerate (zero-length)
86    /// course return the polyline unchanged; `count == 0` yields an
87    /// empty polyline.
88    pub fn resampled(&self, count: usize) -> Self {
89        if count == 0 {
90            return Self::new(Vec::new());
91        }
92        if self.0.len() < 2 {
93            return self.clone();
94        }
95        let total: f64 = self.0.windows(2).map(|pair| pair[0].distance_f64(&pair[1])).sum();
96        if total < 1e-6 {
97            return self.clone();
98        }
99        let step = total / (count - 1).max(1) as f64;
100        let mut out = Vec::with_capacity(count);
101        let mut previous = self.0[0];
102        out.push(previous);
103        let mut walked = 0.0;
104        for point in &self.0[1..] {
105            let mut seg = previous.distance_f64(point);
106            while walked + seg >= step && out.len() < count {
107                let t = (step - walked) / seg;
108                let interpolated = Vec2::new(previous.x + (point.x - previous.x) * t, previous.y + (point.y - previous.y) * t);
109                out.push(interpolated);
110                previous = interpolated;
111                walked = 0.0;
112                seg = previous.distance_f64(point);
113            }
114            walked += seg;
115            previous = *point;
116        }
117        while out.len() < count {
118            out.push(self.0.last().copied().unwrap_or(previous));
119        }
120        Self::new(out)
121    }
122
123    /// The polyline translated so its first point sits at the
124    /// origin - position-free shape comparison (e.g. stroke
125    /// matching, where a gesture is drawable anywhere).
126    pub fn translated_to_origin(&self) -> Self {
127        let Some(first) = self.0.first() else {
128            return self.clone();
129        };
130        Self::new(self.0.iter().map(|point| *point - *first).collect())
131    }
132}

134impl<U> Polyline<U>
135where
136    U: Copy + From<f64> + PartialOrd + Add<Output = U> + Mul<f64, Output = U>,
137    f64: From<U>,
138{
139    /// The axis parameter `t` where the line `center + axis * t`
140    /// meets this polyline - `None` when they do not intersect
141    /// within `reach`.
142    pub fn axis_param(&self, center: Vec2<U>, axis: Vec2<U>, reach: U) -> Option<U> {
143        let half = f64::from(reach);
144        let line = Segment::new(center - axis * half, center + axis * half);
145        self.segments()
146            .find_map(|segment| line.intersection(&segment).map(|point| U::from((point - center).dot::<f64>(&axis))))
147    }
148
149    /// The polyline smoothed `rounds` times except the points
150    /// within `within` of `anchors`, which stay pinned.
151    pub fn smoothed_anchored(&self, anchors: &[Vec2<U>], within: U, rounds: usize) -> Self {
152        self.smoothed(rounds, &self.indices_within(anchors, within))
153    }
154
155    /// The indices of points lying within `within` of any
156    /// `positions` - e.g. the waypoints pinned through a
157    /// `smoothed` call.
158    pub fn indices_within(&self, positions: &[Vec2<U>], within: U) -> Vec<usize> {
159        self.0
160            .iter()
161            .enumerate()
162            .filter(|(_, point)| positions.iter().any(|pos| pos.distance(point) < within))
163            .map(|(index, _)| index)
164            .collect()
165    }
166
167    /// The arc length along the polyline up to the line parameter
168    /// `at`: `at`'s integer part is a segment index, its fraction
169    /// the position along that segment. Zero on fewer than two
170    /// points.
171    pub fn arc_length(&self, at: f64) -> U {
172        if self.0.len() < 2 {
173            return U::from(0.0);
174        }
175        let index = (at.max(0.0).floor() as usize).min(self.0.len() - 1);
176        let full = self.0.windows(2).take(index).fold(U::from(0.0), |arc, pair| arc + pair[0].distance(&pair[1]));
177        let (a, b) = (self.0[index], self.0[(index + 1).min(self.0.len() - 1)]);
178        full + a.distance(&b) * at.max(0.0).fract()
179    }
180
181    /// The polyline shifted `distance` to the left of its
182    /// direction: bend vertices average the adjacent segment
183    /// normals, so the offset stays near-constant without miter
184    /// spikes.
185    pub fn offset(&self, distance: U) -> Self {
186        let normal = |from: Vec2<U>, to: Vec2<U>| {
187            let perpendicular = (to - from).normalized().perpendicular();
188            (perpendicular.x, perpendicular.y)
189        };
190        let distance = f64::from(distance);
191        let points: Vec<Vec2<U>> = self
192            .0
193            .iter()
194            .enumerate()
195            .map(|(index, point)| {
196                let (mut normal_x, mut normal_y) = (0.0, 0.0);
197                if index > 0 {
198                    let (x, y) = normal(self.0[index - 1], *point);
199                    normal_x += x;
200                    normal_y += y;
201                }
202                if index + 1 < self.0.len() {
203                    let (x, y) = normal(*point, self.0[index + 1]);
204                    normal_x += x;
205                    normal_y += y;
206                }
207                let norm = normal_x.hypot(normal_y).max(1e-6);
208                Vec2::new(point.x + normal_x / norm * distance, point.y + normal_y / norm * distance)
209            })
210            .collect();
211        Self::new(points)
212    }
213
214    /// The segment index and projection nearest to `point` -
215    /// `None` for an empty polyline.
216    pub fn nearest_segment(&self, point: &Vec2<U>) -> Option<(usize, Projection<U>)> {
217        self.segments()
218            .enumerate()
219            .filter_map(|(index, segment)| segment.nearest(point).map(|projection| (index, projection)))
220            .min_by(|(_, a), (_, b)| f64::from(point.distance(&a.point)).total_cmp(&f64::from(point.distance(&b.point))))
221    }
222}

259impl<U> Serialize for Polyline<U> {
260    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
261        let mut flat = Vec::with_capacity(self.0.len() * 2);
262        for p in &self.0 {
263            flat.push(p.x);
264            flat.push(p.y);
265        }
266        flat.serialize(serializer)
267    }
268}
269
270impl<'de, U> Deserialize<'de> for Polyline<U> {
271    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
272        let flat = <Vec<f64>>::deserialize(deserializer)?;
273        if flat.len() % 2 != 0 {
274            return Err(D::Error::custom("polyline must contain an even number of coordinates"));
275        }
276        let (pairs, _rest) = flat.as_chunks::<2>();
277        let points = pairs.iter().map(|xy| Vec2::new(xy[0], xy[1])).collect();
278        Ok(Self(points))
279    }
