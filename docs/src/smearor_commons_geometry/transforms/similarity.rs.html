7/// Least-squares similarity transform - rotation, uniform scale and
8/// translation via centroids - between two corresponding point sets.
9///
10/// Multi-touch gestures and marker tracking produce pairs of point
11/// sets in known point-wise correspondence: [`Self::fit`] computes
12/// the optimal angle and spread ratio in closed form, and
13/// [`Self::apply`] then maps any baseline-space point - e.g. the
14/// center of a dragged object - into the current space.
15#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
16pub struct SimilarityTransform<U> {
17    /// Rotation mapping the baseline points onto the current points,
18    /// in the coordinate handedness of the input.
19    pub rotation: RotationRad,
20    /// Uniform scale ratio (current spread / baseline spread); `1.0`
21    /// when the baseline has no spread to compare against.
22    pub scale: Fraction,
23    /// Centroid of the baseline point set.
24    pub baseline_centroid: Vec2<U>,
25    /// Centroid of the current point set.
26    pub current_centroid: Vec2<U>,
27}
28
29impl<U> SimilarityTransform<U> {
30    /// Fits the least-squares rotation and uniform scale between two
31    /// corresponding point sets in closed form.
32    ///
33    /// Each corresponding pair votes for the angle with its squared
34    /// distance from the centroid as weight (`atan2` of the summed
35    /// cross- and dot-products, in [`Vec2::rotated`] sign convention);
36    /// the scale is the ratio of the spreads' root-mean-square radii.
37    ///
38    /// `baseline` and `current` must have the same length in
39    /// correspondence order (e.g. same touch index); `None` is
40    /// returned for length mismatches and empty inputs - the cases
41    /// where no correspondence exists at all. A baseline without
42    /// spread (all points coincident) underdetermines rotation and
43    /// scale; [`RotationRad::ZERO`] and [`Fraction::ONE`] are
44    /// substituted so [`Self::apply`] still reproduces the centroid
45    /// translation.
46    pub fn fit(baseline: &[Vec2<U>], current: &[Vec2<U>]) -> Option<Self> {
47        if baseline.len() != current.len() || baseline.is_empty() {
48            return None;
49        }
50        let baseline_centroid = Vec2::centroid(baseline);
51        let current_centroid = Vec2::centroid(current);
52        let (mut numerator, mut denominator, mut radius_current, mut radius_baseline) = (0.0, 0.0, 0.0, 0.0);
53        for (current_point, baseline_point) in current.iter().zip(baseline) {
54            let current_offset = *current_point - current_centroid;
55            let baseline_offset = *baseline_point - baseline_centroid;
56            numerator += baseline_offset.cross::<f64>(&current_offset);
57            denominator += current_offset.dot::<f64>(&baseline_offset);
58            radius_current += current_offset.dot::<f64>(&current_offset);
59            radius_baseline += baseline_offset.dot::<f64>(&baseline_offset);
60        }
61        let scale = if radius_baseline > f64::EPSILON {
62            (radius_current / radius_baseline).sqrt()
63        } else {
64            1.0
65        };
66        Some(Self {
67            rotation: RotationRad::new(numerator.atan2(denominator)),
68            scale: Fraction::new(scale),
69            baseline_centroid,
70            current_centroid,
71        })
72    }
73
74    /// A declared similarity around a fixed anchor - rotation and
75    /// scale about `center`. The [`Self::fit`] counterpart for
76    /// author-specified transforms where no point correspondence
77    /// exists: both centroids coincide at the anchor, so
78    /// [`Self::apply`] maps `center` onto itself.
79    pub fn about(center: Vec2<U>, rotation: RotationRad, scale: Fraction) -> Self {
80        Self {
81            rotation,
82            scale,
83            baseline_centroid: center,
84            current_centroid: center,
85        }
86    }
87
88    /// Maps a baseline-space point into the current space: shifted to
89    /// the baseline centroid, rotated and scaled, then re-anchored at
90    /// the current centroid.
91    pub fn apply(&self, point: Vec2<U>) -> Vec2<U> {
92        self.current_centroid + (point - self.baseline_centroid).rotated(self.rotation) * self.scale.as_f64()
93    }
94}

99    use crate::Pixels;

101    fn baseline_square() -> Vec<Vec2<Pixels>> {
102        vec![Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0), Vec2::new(10.0, 10.0), Vec2::new(0.0, 10.0)]
103    }
104
105    fn assert_vec2_eq(actual: Vec2<Pixels>, expected: Vec2<Pixels>) {
106        assert!((actual.x - expected.x).abs() < 1e-9 && (actual.y - expected.y).abs() < 1e-9, "{actual:?} != {expected:?}");
107    }
108
109    /// `about` anchors both centroids: the center stays fixed and
110    /// other points rotate and scale around it.
111    #[test]
112    fn about_rotates_and_scales_around_center() {
113        let center = Vec2::<Pixels>::new(5.0, 5.0);
114        let transform = SimilarityTransform::about(center, RotationRad::QUARTER_TURN, Fraction::new(2.0));
115        assert_vec2_eq(transform.apply(center), center);
116        let point = Vec2::<Pixels>::new(10.0, 5.0);
117        let mapped = transform.apply(point);
118        assert_vec2_eq(mapped, Vec2::new(5.0, 15.0));
119    }
120
121    #[test]
122    fn fit_same_points_is_identity() {
123        let points = baseline_square();
124        let transform = SimilarityTransform::fit(&points, &points).unwrap();
125        assert_eq!(transform.rotation, RotationRad::ZERO);
126        assert_eq!(transform.scale, Fraction::ONE);
127        for point in &points {
128            assert_vec2_eq(transform.apply(*point), *point);
129        }
130    }
131
132    #[test]
133    fn fit_recovers_rotation() {
134        let baseline = baseline_square();
135        let rotation = RotationRad::from_degrees(90.0);
136        let centroid = Vec2::centroid(&baseline);
137        let current: Vec<_> = baseline.iter().map(|point| centroid + (*point - centroid).rotated(rotation)).collect();
138        let transform = SimilarityTransform::fit(&baseline, &current).unwrap();
139        assert!((transform.rotation.to_degrees() - 90.0).abs() < 1e-9);
140        assert!((transform.scale.as_f64() - 1.0).abs() < 1e-9);
141        for (point, expected) in baseline.iter().zip(&current) {
142            assert_vec2_eq(transform.apply(*point), *expected);
143        }
144    }
145
146    #[test]
147    fn fit_recovers_uniform_scale() {
148        let baseline = baseline_square();
149        let centroid = Vec2::centroid(&baseline);
150        let current: Vec<_> = baseline.iter().map(|point| centroid + (*point - centroid) * 2.0).collect();
151        let transform = SimilarityTransform::fit(&baseline, &current).unwrap();
152        assert_eq!(transform.rotation, RotationRad::ZERO);
153        assert!((transform.scale.as_f64() - 2.0).abs() < 1e-9);
154        for (point, expected) in baseline.iter().zip(&current) {
155            assert_vec2_eq(transform.apply(*point), *expected);
156        }
157    }
158
159    #[test]
160    fn fit_translation_moves_centroid() {
161        let baseline = baseline_square();
162        let offset = Vec2::<Pixels>::new(30.0, -20.0);
163        let current: Vec<_> = baseline.iter().map(|point| *point + offset).collect();
164        let transform = SimilarityTransform::fit(&baseline, &current).unwrap();
165        assert_eq!(transform.rotation, RotationRad::ZERO);
166        assert_eq!(transform.scale, Fraction::ONE);
167        assert_vec2_eq(transform.current_centroid - transform.baseline_centroid, offset);
168    }
169
170    #[test]
171    fn fit_rejects_mismatched_lengths() {
172        assert_eq!(SimilarityTransform::<Pixels>::fit(&baseline_square(), &baseline_square()[..2]), None);
173    }

175    #[test]
176    fn fit_rejects_empty_inputs() {
177        let empty: Vec<Vec2<Pixels>> = vec![];
178        assert_eq!(SimilarityTransform::fit(&empty, &empty), None);
179    }

182    fn fit_degenerate_baseline_yields_translation() {
183        let baseline = vec![Vec2::<Pixels>::new(5.0, 5.0); 4];
184        let offset = Vec2::<Pixels>::new(10.0, 0.0);
185        let current: Vec<_> = baseline.iter().map(|point| *point + offset).collect();
