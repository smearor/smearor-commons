7/// Least-squares rigid 2D transform - rotation and translation via
8/// centroids - between two corresponding point sets; the
9/// scale-forced-to-`1` counterpart of [`SimilarityTransform`].
10///
11/// Use [`Self::fit`] where both point sets share a physical unit
12/// (e.g. a declared pin layout against measured touches, both in
13/// millimeters): a fitted scale would absorb measurement noise as a
14/// bogus size change, while rotation and translation are the actual
15/// degrees of freedom of the placement.
16#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
17pub struct RigidTransform<U> {
18    /// Rotation mapping the baseline points onto the current points,
19    /// in the coordinate handedness of the input.
20    pub rotation: RotationRad,
21    /// Centroid of the baseline point set.
22    pub baseline_centroid: Vec2<U>,
23    /// Centroid of the current point set.
24    pub current_centroid: Vec2<U>,
25}
26
27impl<U> RigidTransform<U> {
28    /// Fits the least-squares rotation and translation - scale fixed
29    /// at `1` - between two corresponding point sets in closed form,
30    /// via [`SimilarityTransform::fit`] with the spread ratio dropped.
31    ///
32    /// `baseline` and `current` must have the same length in
33    /// correspondence order; `None` is returned for length mismatches
34    /// and empty inputs. A baseline without spread (all points
35    /// coincident) underdetermines rotation; [`RotationRad::ZERO`] is
36    /// substituted so [`Self::apply`] still reproduces the centroid
37    /// translation.
38    pub fn fit(baseline: &[Vec2<U>], current: &[Vec2<U>]) -> Option<Self> {
39        let fit = SimilarityTransform::fit(baseline, current)?;
40        Some(Self {
41            rotation: fit.rotation,
42            baseline_centroid: fit.baseline_centroid,
43            current_centroid: fit.current_centroid,
44        })
45    }
46
47    /// Maps a baseline-space point into the current space:
48    /// `current_centroid + R(rotation) · (point - baseline_centroid)`.
49    pub fn apply(&self, point: Vec2<U>) -> Vec2<U> {
50        self.current_centroid + (point - self.baseline_centroid).rotated(self.rotation)
51    }
52}

54#[cfg(test)]
55mod tests {
56    use super::*;
57    use crate::Pixels;
58
59    /// A quarter-turn baseline maps each point around the centroid -
60    /// `apply` reproduces the rotated set exactly.
61    #[test]
62    fn fit_recovers_rotation() {
63        let baseline = [Vec2::<Pixels>::new(0.0, 0.0), Vec2::new(10.0, 0.0), Vec2::new(10.0, 10.0), Vec2::new(0.0, 10.0)];
64        let centroid = Vec2::centroid(&baseline);
65        let current: Vec<_> = baseline
66            .iter()
67            .map(|point| Vec2::new(100.0, 50.0) + (*point - centroid).rotated(RotationRad::QUARTER_TURN))
68            .collect();
69        let transform = RigidTransform::fit(&baseline, &current).expect("fit");
70        assert!((transform.rotation - RotationRad::QUARTER_TURN).abs().as_f64() < 1e-9);
71        for (point, expected) in baseline.iter().zip(&current) {
72            let mapped = transform.apply(*point);
73            assert!((mapped.x - expected.x).abs() < 1e-9 && (mapped.y - expected.y).abs() < 1e-9);
74        }
75    }
76
77    /// Mismatched and empty inputs admit no correspondence.
78    #[test]
79    fn fit_rejects_mismatched_inputs() {
80        let baseline = [Vec2::<()>::new(0.0, 0.0), Vec2::new(1.0, 0.0)];
81        assert!(RigidTransform::fit(&baseline, &[]).is_none());
82        assert!(RigidTransform::fit(&baseline, &baseline[..1]).is_none());
83    }
