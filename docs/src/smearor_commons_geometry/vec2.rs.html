21/// A 2D point or vector tagged with its coordinate unit `U` and
22/// numeric component type `T`.
23///
24/// The phantom unit distinguishes positions in millimeters
25/// (`Vec2<Millimeters>`), raw device units and screen pixels
26/// (`Vec2<Pixels>`) so they cannot be mixed up; `T` defaults to `f64`
27/// for domain math but can be any component type the boundaries need
28/// (`Vec2<(), i32>` for smithay `Point<i32, K>` and textures,
29/// `Vec2<(), f32>` for graphene). The *unit-erased* form `Vec2<(), T>`
30/// is the boundary representation - it deliberately carries no
31/// coordinate-space claim and converts into any framework space;
32/// [`crate::Position`] and [`crate::Size`] are readable aliases for it.
33/// Serializes as a flat `{"x": .., "y": ..}` map; the unit marker is
34/// skipped. With the `async-graphql` feature, `Vec2<Millimeters>` is
35/// exposed as the GraphQL object type `Point`.
36#[derive(Debug, Serialize, Deserialize)]
37#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
38pub struct Vec2<U = (), T = f64> {
39    /// X coordinate in the unit `U`.
40    pub x: T,
41    /// Y coordinate in the unit `U`.
42    pub y: T,
43    /// Phantom marker tagging the coordinate unit; not serialized.
44    #[serde(skip)]
45    unit: PhantomData<U>,
46}
47
48/// `Vec2<Millimeters>` is exposed as the GraphQL object type `Point`.
49/// A `SimpleObject` derive cannot express the doubly-generic struct
50/// (it would require `T: OutputType` on every instantiation), so the
51/// concrete object is declared as resolvers on the `f64` millimeter
52/// instantiation only.
53#[cfg(feature = "async-graphql")]
54#[async_graphql::Object(name = "Point")]
55impl Vec2<Millimeters> {
56    /// X coordinate in millimeters.
57    async fn x(&self) -> f64 {
58        self.x
59    }
60
61    /// Y coordinate in millimeters.
62    async fn y(&self) -> f64 {
63        self.y
64    }
65}

118impl<U, T> Vec2<U, T> {
119    /// Creates a new 2D point in the unit `U`.
120    pub const fn new(x: T, y: T) -> Self {
121        Self { x, y, unit: PhantomData }
122    }
123
124    /// `y,x` swizzle - swaps the components.
125    pub fn yx(&self) -> Self
126    where
127        T: Clone,
128    {
129        Self::new(self.y.clone(), self.x.clone())
130    }
131
132    /// `x,y,x` swizzle - lifts to three components (GLSL `p.xyx`).
133    pub fn xyx(&self) -> crate::Vec3<U, T>
134    where
135        T: Clone,
136    {
137        crate::Vec3::new(self.x.clone(), self.y.clone(), self.x.clone())
138    }
139
140    /// Reinterprets this vector in the unit `V` - the inverse of the
141    /// unit-erased boundary forms ([`crate::Size`], [`crate::Position`]).
142    /// The components are unchanged; use this when a framework boundary
143    /// knows the coordinate space that the erased `()` tag does not
144    /// express.
145    pub fn with_unit<V>(&self) -> Vec2<V, T>
146    where
147        T: Copy,
148    {
149        Vec2::new(self.x, self.y)
150    }
151
152    /// Applies `f` to each component - component-wise transforms like
153    /// clamping, rounding or numeric conversion. The component type may
154    /// change (`v.map(|c| c.round() as i32)` yields `Vec2<U, i32>`); the
155    /// unit `U` is always preserved.
156    pub fn map<V>(&self, f: impl Fn(T) -> V) -> Vec2<U, V>
157    where
158        T: Copy,
159    {
160        Vec2::new(f(self.x), f(self.y))
161    }
162
163    /// Component-wise product (Hadamard product): each axis scaled by the
164    /// other vector's matching component - per-axis weights, not a dot
165    /// product.
166    pub fn component_mul(&self, other: &Vec2<U, T>) -> Self
167    where
168        T: Copy + Mul<Output = T>,
169    {
170        Self::new(self.x * other.x, self.y * other.y)
171    }
172
173    /// Component-wise maximum of two points in the same unit.
174    pub fn max(&self, other: &Self) -> Self
175    where
176        T: Copy + Ord,
177    {
178        Self::new(self.x.max(other.x), self.y.max(other.y))
179    }
180
181    /// Component-wise minimum of two points in the same unit.
182    pub fn min(&self, other: &Self) -> Self
183    where
184        T: Copy + Ord,
185    {
186        Self::new(self.x.min(other.x), self.y.min(other.y))
187    }
188
189    /// The vector rotated 90° counterclockwise: `(-y, x)` - the
190    /// perpendicular used for offset and normal computations.
191    pub fn perpendicular(&self) -> Self
192    where
193        T: Copy + Neg<Output = T>,
194    {
195        Self::new(-self.y, self.x)
196    }
197
198    /// The axis-aligned rectangle anchored at this point with the given
199    /// `size` extent.
200    pub fn rect(&self, size: Vec2<U, T>) -> crate::Rect<U, T>
201    where
202        T: Copy,
203    {
204        crate::Rect::new(*self, size)
205    }
206}

209    /// The origin `(0, 0)` - both components `T::default()`.
210    fn default() -> Self {
211        Self::new(T::default(), T::default())
212    }
213}

303// ---------------------------------------------------------------------------
304
305impl<U> Vec2<U, f64> {
306    /// The origin `(0, 0)` in the unit `U`.
307    pub const ZERO: Self = Self {
308        x: 0.0,
309        y: 0.0,
310        unit: PhantomData,
311    };
312
313    /// The x-component as a scalar in the unit `U` - the typed
314    /// counterpart of the raw `x` field for boundaries that need the
315    /// unit attached (`v.x` is `f64`, `v.x_unit()` is `U`).
316    pub fn x_unit(&self) -> U
317    where
318        U: From<f64>,
319    {
320        U::from(self.x)
321    }
322
323    /// The y-component as a scalar in the unit `U` - see
324    /// [`Self::x_unit`].
325    pub fn y_unit(&self) -> U
326    where
327        U: From<f64>,
328    {
329        U::from(self.y)
330    }
331
332    /// Both components set to the same value - `Vec2::splat(margin)` is
333    /// the vector `(margin, margin)` (e.g. a uniform margin around a
334    /// rectangle).
335    pub fn splat(value: U) -> Self
336    where
337        U: Into<f64>,
338    {
339        let v = value.into();
340        Self::new(v, v)
341    }
342
343    /// Creates a vector from polar coordinates: `radius` in the unit `U`
344    /// at `angle` (counter-clockwise from the positive x-axis).
345    pub fn from_polar(radius: f64, angle: RotationRad) -> Self {
346        let (sin, cos) = angle.sin_cos();
347        Vec2::new(radius * cos, radius * sin)
348    }
349
350    /// Euclidean distance to another point in the same unit.
351    pub fn distance(&self, other: &Vec2<U>) -> U
352    where
353        U: From<f64>,
354    {
355        U::from(self.distance_f64(other))
356    }
357
358    /// Euclidean distance to another point, as a plain `f64`.
359    ///
360    /// Unlike [`Vec2::distance`] this needs no unit-carrying result
361    /// type, so it also works on the unit-erased `Vec2<()>` where the
362    /// distance itself has no meaningful unit.
363    pub fn distance_f64(&self, other: &Vec2<U>) -> f64 {
364        ((self.x - other.x).powi(2) + (self.y - other.y).powi(2)).sqrt()
365    }
366
367    /// Dot product `x·x + y·y` of two vectors in the same unit.
368    ///
369    /// The caller picks the product type `T` (like [`Vec2::distance`]'s
370    /// unit return): `f64` for unit-free math or a unit type such as
371    /// [`crate::SquareMillimeters`] for `Vec2<Millimeters>` products.
372    pub fn dot<T: From<f64>>(&self, other: &Vec2<U>) -> T {
373        T::from(self.x * other.x + self.y * other.y)
374    }
375
376    /// 2D determinant `x·y' − y·x'` (z component of the 3D cross
377    /// product): positive when `other` turns left from `self`, negative
378    /// right; zero for (anti-)parallel vectors.
379    ///
380    /// Same caller-chosen product type as [`Vec2::dot`].
381    pub fn cross<T: From<f64>>(&self, other: &Vec2<U>) -> T {
382        T::from(self.x * other.y - self.y * other.x)
383    }
384
385    /// Euclidean magnitude of the vector, in the unit `U`.
386    pub fn magnitude(&self) -> U
387    where
388        U: From<f64>,
389    {
390        self.distance(&Vec2::ZERO)
391    }
392
393    /// Euclidean magnitude of the vector, as a plain `f64` - the
394    /// unit-erased counterpart of [`Vec2::magnitude`] for `Vec2<()>`.
395    pub fn magnitude_f64(&self) -> f64 {
396        self.distance_f64(&Vec2::ZERO)
397    }
398
399    /// The unit vector pointing the same way; a vector shorter
400    /// than `1e-9` normalizes to itself.
401    pub fn normalized(&self) -> Self {
402        let magnitude = self.x.hypot(self.y);
403        if magnitude < 1e-9 {
404            return Vec2::new(self.x, self.y);
405        }
406        Vec2::new(self.x / magnitude, self.y / magnitude)
407    }
408
409    /// The direction the vector points, as a `RotationRad` angle -
410    /// the inverse of [`Vec2::from_polar`].
411    pub fn direction(&self) -> RotationRad {
412        RotationRad::new(self.y.atan2(self.x))
413    }
414
415    /// The vector rotated by `rotation`.
416    pub fn rotated(&self, rotation: RotationRad) -> Self {
417        let (sin, cos) = rotation.as_f64().sin_cos();
418        Vec2::new(self.x * cos - self.y * sin, self.x * sin + self.y * cos)
419    }
420
421    /// The bounding-box size of an axis-aligned rectangle of this size
422    /// rotated by `rotation` around its center - the forward map
423    /// `bb(v) = A·v` with `A = [[cos, sin], [sin, cos]]` over the
424    /// absolute sine/cosine.
425    ///
426    /// [`Rect::rotated_bounds`](crate::Rect::rotated_bounds) delegates
427    /// to this method on its `size`.
428    pub fn rotated_bounds(&self, rotation: RotationRad) -> Self {
429        let (sin, cos) = rotation.sin_cos();
430        let (abs_sin, abs_cos) = (sin.abs(), cos.abs());
431        Vec2::new(self.x * abs_cos + self.y * abs_sin, self.x * abs_sin + self.y * abs_cos)
432    }
433
434    /// Inverse of [`Self::rotated_bounds`]: the axis-aligned size that,
435    /// rotated by `rotation`, yields this bounding box - the map
436    /// `A⁻¹·v` with `det(A) = cos(2θ)`.
437    ///
438    /// The determinant vanishes at `45° + k·90°` where no finite
439    /// solution exists for non-square boxes; components then hold
440    /// `±inf`/`NaN` and callers must apply their own fallback.
441    pub fn unrotated_bounds(&self, rotation: RotationRad) -> Self {
442        let (sin, cos) = rotation.sin_cos();
443        let (abs_sin, abs_cos) = (sin.abs(), cos.abs());
444        let determinant = abs_cos * abs_cos - abs_sin * abs_sin;
445        Vec2::new((self.x * abs_cos - self.y * abs_sin) / determinant, (self.y * abs_cos - self.x * abs_sin) / determinant)
446    }
447
448    /// The largest rectangle of aspect ratio `aspect` (width/height)
449    /// whose `rotation`-rotated bounding box fits inside a rectangle of
450    /// size `self` - the inscribed, aspect-preserving counterpart of
451    /// [`Vec2::unrotated_bounds`].
452    ///
453    /// Solves `A·(aspect·h, h) <= self` for the largest `h` over the
454    /// absolute sin/cos matrix `A`: the tighter of the two axis
455    /// constraints wins. Unlike the exact inverse solve this never
456    /// diverges and never goes negative - rotating a rectangle by any
457    /// angle keeps such an inscribed solution bounded. At `aspect == 1`
458    /// it degenerates to the square fit `min(w, h) / (|cos| + |sin|)`.
459    ///
460    /// `aspect` is strictly positive by construction
461    /// ([`AspectRatio`]); negative components of `self` are clamped to
462    /// [`Vec2::ZERO`].
463    pub fn unrotated_fit(&self, aspect: AspectRatio, rotation: RotationRad) -> Self {
464        let aspect = aspect.as_f64();
465        let (sin, cos) = rotation.sin_cos();
466        let (abs_sin, abs_cos) = (sin.abs(), cos.abs());
467        let height = (self.x / (aspect * abs_cos + abs_sin)).min(self.y / (aspect * abs_sin + abs_cos)).max(0.0);
468        Vec2::new(aspect * height, height)
469    }
470
471    /// The smallest rectangle of aspect ratio `aspect` (width/height)
472    /// whose `rotation`-rotated bounding box contains a rectangle of
473    /// size `self` - the covering, aspect-preserving counterpart of
474    /// [`Vec2::unrotated_bounds`], dual to [`Vec2::unrotated_fit`].
475    ///
476    /// Solves `A·(aspect·h, h) >= self` for the smallest `h` over the
477    /// absolute sin/cos matrix `A`: the looser of the two axis
478    /// constraints wins, so the result covers `self` on both axes and
479    /// is tight on at least one of them. Like `unrotated_fit` this
480    /// never diverges and never goes negative. At `aspect == 1` it
481    /// degenerates to the square cover `max(w, h) / (|cos| + |sin|)`.
482    ///
483    /// Covering with the aspect of a feasible
484    /// [`Vec2::unrotated_bounds`] solve reproduces that solve
485    /// exactly - both axes then constrain the same height. That
486    /// makes the aspect-clamped cover a continuous fallback for the
487    /// exact solve: the cover meets it at the clamp edge and relaxes
488    /// into overhang beyond it.
489    ///
490    /// `aspect` is strictly positive by construction
491    /// ([`AspectRatio`]); negative components of `self` are clamped to
492    /// [`Vec2::ZERO`].
493    pub fn unrotated_cover(&self, aspect: AspectRatio, rotation: RotationRad) -> Self {
494        let aspect = aspect.as_f64();
495        let (sin, cos) = rotation.sin_cos();
496        let (abs_sin, abs_cos) = (sin.abs(), cos.abs());
497        let height = (self.x / (aspect * abs_cos + abs_sin)).max(self.y / (aspect * abs_sin + abs_cos)).max(0.0);
498        Vec2::new(aspect * height, height)
499    }
500
501    /// Half of this vector component-wise - the midpoint between the
502    /// origin and `self`. For a [`crate::Size`] extent this is its
503    /// center offset.
504    pub fn half(&self) -> Self {
505        self.map(|v| v / 2.0)
506    }
507
508    /// The mean of the two components - a scalar average, not a
509    /// vector. For per-axis factors (e.g. mm-per-pixel on x and y)
510    /// this is the single factor absorbing axis jitter.
511    pub fn component_mean(&self) -> f64 {
512        (self.x + self.y) / 2.0
513    }
514
515    /// The aspect ratio `x / y` - `width / height` for a
516    /// [`crate::Size`] extent. Returns `None` when either component is
517    /// non-positive or the ratio degenerates (`0` or non-finite): a
518    /// degenerate extent has no meaningful aspect.
519    pub fn aspect(&self) -> Option<AspectRatio> {
520        (self.x > 0.0 && self.y > 0.0).then_some(self.x / self.y).and_then(AspectRatio::new)
521    }
522
523    /// The raw aspect ratio `x / y` without validation - preserves the
524    /// sign, so degenerate extents report negative, infinite or `NaN`
525    /// values that [`Vec2::aspect`] would reject. Useful when the sign
526    /// itself carries meaning, e.g. distinguishing the infeasible
527    /// negative branches of [`Vec2::unrotated_bounds`].
528    pub fn signed_aspect(&self) -> f64 {
529        self.x / self.y
530    }
531
532    /// Geometric centroid of a non-empty point set.
533    ///
534    /// Returns [`Vec2::new(0.0, 0.0)`] for an empty slice instead of panicking.
535    pub fn centroid(points: &[Vec2<U>]) -> Self {
536        let n = points.len() as f64;
537        if n == 0.0 {
538            return Vec2::new(0.0, 0.0);
539        }
540        Vec2::new(points.iter().map(|p| p.x).sum::<f64>() / n, points.iter().map(|p| p.y).sum::<f64>() / n)
541    }
542}

592impl<U, T: Add<Output = T>> Add for Vec2<U, T> {
593    type Output = Self;
594
595    /// Component-wise sum of two points in the same unit.
596    fn add(self, other: Self) -> Self {
597        Self::new(self.x + other.x, self.y + other.y)
598    }
599}
600
601impl<U, T: AddAssign> AddAssign for Vec2<U, T> {
602    fn add_assign(&mut self, other: Self) {
603        self.x += other.x;
604        self.y += other.y;
605    }
606}
607
608impl<U, T: Sub<Output = T>> Sub for Vec2<U, T> {
609    type Output = Self;
610
611    /// Component-wise difference of two points in the same unit.
612    fn sub(self, other: Self) -> Self {
613        Self::new(self.x - other.x, self.y - other.y)
614    }
615}
616
617impl<U, T: SubAssign> SubAssign for Vec2<U, T> {
618    fn sub_assign(&mut self, other: Self) {
619        self.x -= other.x;
620        self.y -= other.y;
621    }
622}

624impl<U, T: Neg<Output = T>> Neg for Vec2<U, T> {
625    type Output = Self;
626
627    /// The point mirrored at the origin.
628    fn neg(self) -> Self {
629        Self::new(-self.x, -self.y)
630    }
631}

633impl<U, T: Copy + Mul<Output = T>> Mul<T> for Vec2<U, T> {
634    type Output = Self;
635
636    /// The vector scaled by a dimensionless factor.
637    fn mul(self, factor: T) -> Self {
638        Self::new(self.x * factor, self.y * factor)
639    }
640}

642impl<U> Mul<Fraction> for Vec2<U> {
643    type Output = Self;
644
645    /// The vector scaled by a progress fraction - the natural
646    /// interpolation operator `point + direction * t`.
647    fn mul(self, fraction: Fraction) -> Self {
648        self * fraction.as_f64()
649    }
650}

655    /// The vector scaled by a dimensionless factor.

658    }

662    type Output = Self;
663
664    /// The vector divided by a dimensionless factor.
665    fn div(self, divisor: T) -> Self {
666        Self::new(self.x / divisor, self.y / divisor)
667    }
668}

670/// Component-wise quotient `v / w`: each axis divided by the other
671/// vector's matching component - per-axis scale factors, not a
672/// projection. Unlike `*` (where `v * w` could read as a dot product)
673/// vector-by-vector division has no competing meaning, so the operator
674/// is elementwise.
675impl<U, T: Copy + Div<Output = T>> Div for Vec2<U, T> {
676    type Output = Self;
677
678    fn div(self, rhs: Self) -> Self {
679        Self::new(self.x / rhs.x, self.y / rhs.y)
680    }
681}

811        [v.x, v.y].serialize(serializer)
812    }

814    /// Deserializes `[x, y]` into a `Vec2` in the unit `U`.

820/// Serde adapter for `#[serde(with)]`: `Option<Vec2<U>>` as a
821/// compact `[x, y]` pair - the absent-tolerant form of
822/// [`serde_pair`]. Serializes `Some` as `[x, y]` and `None` as
823/// `null`; combine with `#[serde(default)]` on the field so a
824/// missing key also yields `None`.
825pub mod serde_pair_option {
826    use super::Vec2;
827    use serde::Deserialize;
828    use serde::Deserializer;
829    use serde::Serialize;
830    use serde::Serializer;

832    /// Serializes `Some(p)` as `[p.x, p.y]`, `None` as `null`.
833    pub fn serialize<U, S: Serializer>(pos: &Option<Vec2<U>>, serializer: S) -> Result<S::Ok, S::Error> {
834        match pos {
835            Some(point) => [point.x, point.y].serialize(serializer),
836            None => serializer.serialize_none(),
837        }
838    }
839
840    /// Deserializes `[x, y]` into `Some(Vec2)`, `null` into `None`.
841    pub fn deserialize<'de, U, D: Deserializer<'de>>(deserializer: D) -> Result<Option<Vec2<U>>, D::Error> {
842        <Option<[f64; 2]>>::deserialize(deserializer).map(|pair| pair.map(Vec2::from))
843    }
