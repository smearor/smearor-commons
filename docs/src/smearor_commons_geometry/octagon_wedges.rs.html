1//! Iterator over the eight edge wedges of an [`Octagon`].
2
3use crate::Fraction;
4use crate::Octagon;
5use crate::Polygon;
6
7/// The eight edge wedges of an [`Octagon`], created by
8/// [`Octagon::wedges`].
9///
10/// Each item is the trapezoidal insert on one edge: its outer side is
11/// the `outer`-scaled edge `squeeze`-narrowed around its midpoint,
12/// its inner side the parallel edge of the `inner`-scaled octagon -
13/// the shape chip inserts, rim marks and border studs take.
14#[derive(Debug, Clone)]
15pub struct OctagonWedges<U> {
16    /// The octagon whose edges carry the wedges.
17    octagon: Octagon<U>,
18    /// Radial fraction of the outer side, relative to the inradius.
19    outer: Fraction,
20    /// Radial fraction of the inner side, relative to the inradius.
21    inner: Fraction,
22    /// The outer side's shrink relative to the edge's half-length.
23    squeeze: Fraction,
24    /// The edge index the iterator returns next.
25    next: usize,
26}
27
28impl<U> OctagonWedges<U> {
29    /// An `OctagonWedges` over `octagon`'s edges - see
30    /// [`Octagon::wedges`].
31    pub(crate) fn new(octagon: Octagon<U>, outer: Fraction, inner: Fraction, squeeze: Fraction) -> Self {
32        Self {
33            octagon,
34            outer,
35            inner,
36            squeeze,
37            next: 0,
38        }
39    }
40}
41
42impl<U: Copy> Iterator for OctagonWedges<U> {
43    type Item = Polygon<U>;
44
45    fn next(&mut self) -> Option<Self::Item> {
46        if self.next >= 8 {
47            return None;
48        }
49        let wedge = self.octagon.wedge(self.next, self.outer, self.inner, self.squeeze);
50        self.next += 1;
51        Some(wedge)
52    }
53
54    fn size_hint(&self) -> (usize, Option<usize>) {
55        let remaining = 8 - self.next;
56        (remaining, Some(remaining))
57    }
58}
59
60impl<U: Copy> ExactSizeIterator for OctagonWedges<U> {}
