1//! A regular grid partitioning of a rectangular extent.
2//!
3//! ```
4//! use smearor_commons_geometry::{Cell, Col, Grid, Rect, Row, Vec2};
5//!
6//! let grid = Grid::for_items(9, 4); // 4 cols, 3 rows
7//! let cell = grid.cell(5).expect("in range");
8//! assert_eq!(cell, Cell::new(Col(1), Row(1)));
9//!
10//! let extent = Vec2::<(), u32>::new(800, 600);
11//! assert_eq!(grid.rect(cell, extent), Rect::new(Vec2::new(200, 200), Vec2::new(200, 200)));
12//! ```
13

25/// A regular grid of `cols` × `rows` equal-sized cells.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
27pub struct Grid {
28    cols: u32,
29    rows: u32,
30}
31
32impl Grid {
33    /// A grid of `cols` columns and `rows` rows.
34    ///
35    /// `0` clamps to `1`: an empty grid partitions nothing and would only
36    /// surface as division by zero downstream.
37    pub const fn new(cols: u32, rows: u32) -> Self {
38        Self {
39            cols: if cols == 0 { 1 } else { cols },
40            rows: if rows == 0 { 1 } else { rows },
41        }
42    }
43
44    /// A grid of `cols` columns with just enough rows to hold
45    /// `item_count` items in row-major order.
46    ///
47    /// `0` items still yields one row: the grid describes a partition,
48    /// not the items.
49    pub fn for_items(item_count: u32, cols: u32) -> Self {
50        let cols = cols.max(1);
51        Self::new(cols, item_count.div_ceil(cols).max(1))
52    }
53
54    /// Number of columns.
55    pub const fn cols(&self) -> u32 {
56        self.cols
57    }
58
59    /// Number of rows.
60    pub const fn rows(&self) -> u32 {
61        self.rows
62    }
63
64    /// Total number of cells.
65    pub const fn capacity(&self) -> u32 {
66        self.cols * self.rows
67    }
68
69    /// The [`Cell`] at linear `index` in row-major order, or `None`
70    /// when `index` exceeds [`capacity`](Self::capacity).
71    pub fn cell(&self, index: u32) -> Option<Cell> {
72        (index < self.capacity()).then(|| Cell::new(Col(index % self.cols), Row(index / self.cols)))
73    }
74
75    /// The linear row-major index of `cell`, or `None` when `cell`
76    /// addresses outside this grid.
77    pub fn index(&self, cell: Cell) -> Option<u32> {
78        (cell.col.0 < self.cols && cell.row.0 < self.rows).then(|| cell.row.0 * self.cols + cell.col.0)
79    }
80
81    /// The grid rotated by `turn`. Quarter turns transpose the
82    /// partition: `Grid::new(4, 3).rotated(Deg90)` is 3 × 4.
83    pub fn rotated(&self, turn: QuarterTurn) -> Self {
84        match turn {
85            QuarterTurn::Deg0 | QuarterTurn::Deg180 => *self,
86            QuarterTurn::Deg90 | QuarterTurn::Deg270 => Self::new(self.rows, self.cols),
87        }
88    }
89
90    /// The cell rotated by `turn` — shorthand for
91    /// `cell.rotated(turn, self)`.
92    pub fn rotate_cell(&self, turn: QuarterTurn, cell: Cell) -> Cell {
93        cell.rotated(turn, self)
94    }
95
96    /// The extent of a single cell when `extent` is partitioned by this
97    /// grid. Integer extents keep the remainder unassigned at the
98    /// right/bottom edge.
99    pub fn cell_size<U, T>(&self, extent: Vec2<U, T>) -> Vec2<U, T>
100    where
101        T: Copy + Div<Output = T> + From<u32>,
102    {
103        Vec2::new(extent.x / T::from(self.cols), extent.y / T::from(self.rows))
104    }
105
106    /// The [`Rect`] of `cell` inside `extent`, with the partition
107    /// anchored at the origin — shorthand for
108    /// `rect_within(cell, Rect::new(origin, extent))`.
109    pub fn rect<U, T>(&self, cell: Cell, extent: Vec2<U, T>) -> Rect<U, T>
110    where
111        T: Copy + Default + Add<Output = T> + Sub<Output = T> + Div<Output = T> + Mul<Output = T> + From<u32>,
112    {
113        self.rect_within(cell, Rect::new(Vec2::default(), extent))
114    }
115
116    /// The [`Rect`] of `cell` inside `frame` — like [`rect`](Self::rect),
117    /// but the partition is anchored at `frame.min` (e.g. a board
118    /// rectangle offset inside a surface).
119    ///
120    /// Both cell edges are anchored independently at `frame.min`
121    /// (`min + k*cell_size`), so horizontally and vertically adjacent
122    /// cells share bit-identical boundaries — relevant for consumers
123    /// that assert disjointness of a produced field set.
124    pub fn rect_within<U, T>(&self, cell: Cell, frame: Rect<U, T>) -> Rect<U, T>
125    where
126        T: Copy + Add<Output = T> + Sub<Output = T> + Div<Output = T> + Mul<Output = T> + From<u32>,
127    {
128        let size = self.cell_size(frame.size);
129        let min = frame.min + Vec2::new(T::from(cell.col.0) * size.x, T::from(cell.row.0) * size.y);
130        let max = frame.min + Vec2::new(T::from(cell.col.0 + 1) * size.x, T::from(cell.row.0 + 1) * size.y);
131        Rect::new(min, max - min)
132    }
133
134    /// The cell containing `point` inside `frame` - the inverse of
135    /// [`rect_within`](Self::rect_within). `None` when `point` lies
136    /// outside the frame; a point exactly on the right or bottom edge
137    /// maps to a column/row beyond the grid and yields `None` too, so
138    /// the far edge is half-open and every point belongs to at most
139    /// one cell.
140    pub fn cell_at<U, T>(&self, point: Vec2<U, T>, frame: Rect<U, T>) -> Option<Cell>
141    where
142        T: Copy + PartialOrd + Add<Output = T> + Sub<Output = T> + Div<Output = T> + From<u32> + Into<f64>,
143    {
144        if !frame.contains(&point) {
145            return None;
146        }
147        let size = self.cell_size(frame.size);
148        let offset = point - frame.min;
149        let cell = Cell::new(Col((offset.x / size.x).into() as u32), Row((offset.y / size.y).into() as u32));
150        self.index(cell).map(|_| cell)
151    }
152
153    /// The linear row-major index of the cell containing `point` -
154    /// [`cell_at`](Self::cell_at) resolved through [`index`](Self::index).
155    pub fn index_at<U, T>(&self, point: Vec2<U, T>, frame: Rect<U, T>) -> Option<u32>
156    where
157        T: Copy + PartialOrd + Add<Output = T> + Sub<Output = T> + Div<Output = T> + From<u32> + Into<f64>,
158    {
159        self.cell_at(point, frame).and_then(|cell| self.index(cell))
160    }
161}

173    #[test]
174    fn for_items_rounds_rows_up() {
175        let grid = Grid::for_items(9, 4);
176        assert_eq!(grid.cols(), 4);
177        assert_eq!(grid.rows(), 3);
178        assert_eq!(grid.capacity(), 12);
179    }

181    #[test]
182    fn for_items_zero_items_keeps_one_row() {
183        assert_eq!(Grid::for_items(0, 4).rows(), 1);
184    }

186    #[test]
187    fn new_clamps_zero_dimensions() {
188        let grid = Grid::new(0, 0);
189        assert_eq!(grid.cols(), 1);
190        assert_eq!(grid.rows(), 1);
191    }

193    #[test]
194    fn cell_maps_row_major() {
195        let grid = Grid::new(4, 3);
196        assert_eq!(grid.cell(0), Some(Cell::new(Col(0), Row(0))));
197        assert_eq!(grid.cell(5), Some(Cell::new(Col(1), Row(1))));
198        assert_eq!(grid.cell(11), Some(Cell::new(Col(3), Row(2))));
199        assert_eq!(grid.cell(12), None);
200    }
201
202    #[test]
203    fn index_is_the_inverse_of_cell() {
204        let grid = Grid::new(4, 3);
205        for index in 0..grid.capacity() {
206            let cell = grid.cell(index).expect("in range");
207            assert_eq!(grid.index(cell), Some(index));
208        }
209        assert_eq!(grid.index(Cell::new(Col(4), Row(0))), None);
210    }
211
212    #[test]
213    fn cell_size_divides_extent() {
214        let grid = Grid::new(4, 3);
215        assert_eq!(grid.cell_size(Vec2::<(), u32>::new(800, 600)), Vec2::new(200, 200));
216        assert_eq!(Grid::new(8, 8).cell_size(Vec2::<(), f64>::new(80.0, 80.0)), Vec2::new(10.0, 10.0));
217    }

219    #[test]
220    fn rect_places_cell_in_extent() {
221        let grid = Grid::new(4, 3);
222        let rect = grid.rect(Cell::new(Col(1), Row(1)), Vec2::<(), u32>::new(800, 600));
223        assert_eq!(rect, Rect::new(Vec2::new(200, 200), Vec2::new(200, 200)));
224    }

226    #[test]
227    fn rect_within_offsets_by_frame_origin() {
228        let grid = Grid::new(8, 8);
229        let frame = Rect::new(Vec2::<(), f64>::new(10.0, 20.0), Vec2::new(80.0, 80.0));
230        let rect = grid.rect_within(Cell::new(Col(2), Row(1)), frame);
231        assert_eq!(rect, Rect::new(Vec2::new(30.0, 30.0), Vec2::new(10.0, 10.0)));
232    }
233
234    #[test]
235    fn adjacent_cells_share_bit_identical_boundaries() {
236        // `frame.min + k*size` on both sides: neighbours must meet
237        // exactly, not merely within an ulp.
238        let grid = Grid::new(8, 8);
239        let frame = Rect::new(Vec2::<(), f64>::new(12.5, 7.25), Vec2::new(83.0, 83.0));
240        for index in 0..grid.capacity() {
241            let cell = grid.cell(index).expect("in range");
242            let rect = grid.rect_within(cell, frame);
243            if cell.col.0 + 1 < grid.cols() {
244                let east = grid.rect_within(Cell::new(Col(cell.col.0 + 1), cell.row), frame);
245                assert_eq!(rect.right(), east.min.x);
246            }
247            if cell.row.0 + 1 < grid.rows() {
248                let south = grid.rect_within(Cell::new(cell.col, Row(cell.row.0 + 1)), frame);
249                assert_eq!(rect.bottom(), south.min.y);
250            }
251        }
252    }

254    #[test]
255    fn cell_at_is_the_inverse_of_rect_within() {
256        let grid = Grid::new(8, 8);
257        let frame = Rect::new(Vec2::<(), f64>::new(10.0, 20.0), Vec2::new(80.0, 80.0));
258        assert_eq!(grid.cell_at(Vec2::new(15.0, 25.0), frame), Some(Cell::new(Col(0), Row(0))));
259        assert_eq!(grid.cell_at(Vec2::new(35.0, 55.0), frame), Some(Cell::new(Col(2), Row(3))));
260        // Cell-internal point maps to the same cell whose rect covers it.
261        let cell = grid.cell_at(Vec2::new(49.9, 79.9), frame).expect("inside");
262        assert!(grid.rect_within(cell, frame).contains(&Vec2::new(49.9, 79.9)));
263    }

265    #[test]
266    fn cell_at_rejects_outside_and_far_edge() {
267        let grid = Grid::new(8, 8);
268        let frame = Rect::new(Vec2::<(), f64>::new(0.0, 0.0), Vec2::new(80.0, 80.0));
269        assert_eq!(grid.cell_at(Vec2::new(-0.5, 40.0), frame), None);
270        // Half-open far edge: 80.0 is on the frame's right/bottom edge.
271        assert_eq!(grid.cell_at(Vec2::new(80.0, 40.0), frame), None);
272        assert_eq!(grid.cell_at(Vec2::new(40.0, 80.0), frame), None);
273        assert_eq!(grid.index_at(Vec2::new(40.0, 40.0), frame), Some(4 * 8 + 4));
274        assert_eq!(grid.index_at(Vec2::new(90.0, 40.0), frame), None);
275    }

278    fn rotated_transposes_on_quarter_turns() {
279        let grid = Grid::new(4, 3);
280        assert_eq!(grid.rotated(QuarterTurn::Deg90), Grid::new(3, 4));
281        assert_eq!(grid.rotated(QuarterTurn::Deg270), Grid::new(3, 4));
