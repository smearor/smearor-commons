4/// One horizontal strip of a rasterized rotated rectangle.
5///
6/// Produced by [`crate::Rect::rotated_region_scanlines`]. Coordinates are
7/// relative to the rect's center and one pixel high; callers add the
8/// actual center to map strips onto integer regions such as
9/// `cairo::RectangleInt`.
10#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
11pub struct Scanline {
12    /// Scanline y-coordinate relative to the rect's center.
13    pub y: i32,
14    /// Left x edge relative to the center (inclusive).
15    pub x_start: i32,
16    /// Right x edge relative to the center (inclusive).
17    pub x_end: i32,
18}
19
20impl Scanline {
21    /// Width of the strip in pixels (`x_end - x_start`).
22    pub fn width(&self) -> i32 {
23        self.x_end - self.x_start
24    }
25}
