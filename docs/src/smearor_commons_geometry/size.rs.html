1//! `Size<T>` - the unit-erased extent form for framework boundaries.
2//!
3//! `Size<T>` is an alias of [`Vec2`]`<(), T>`: a width/height pair
4//! (`size.x` = width, `size.y` = height, same convention as
5//! [`crate::Rect`]'s `size` field) that deliberately carries no
6//! coordinate-space claim. It converts into smithay `Size<i32, K>` of
7//! any space kind or describes texture extents at the GTK boundary.
8
9use crate::Vec2;
10
11/// A 2D extent on an integer or float coordinate grid, generic over the
12/// numeric type `T`. Alias of [`Vec2`]`<(), T>` - the unit-erased
13/// boundary form; `x` is the width, `y` the height.
14pub type Size<T> = Vec2<(), T>;
15

20/// The pixel extent of a `gdk::Texture`.
21#[cfg(feature = "gtk4")]
22impl From<&gtk4::gdk::Texture> for Size<i32> {
23    fn from(texture: &gtk4::gdk::Texture) -> Self {
24        use gtk4::prelude::TextureExt;
25        Self::new(texture.width(), texture.height())
26    }
27}

29/// Convenience geometry accessors for `gtk4::Widget`s.
30///
31/// Import this trait to call [`WidgetSizeExt::allocated_size`] on any
32/// widget. Named after `gtk_widget_get_allocated_*` rather than `size`,
33/// which collides with `gtk4::prelude::WidgetExt::size`.
34#[cfg(feature = "gtk4")]
35pub trait WidgetSizeExt {
36    /// The widget's allocated pixel extent.
37    fn allocated_size(&self) -> Size<i32>;
38
39    /// The widget's measured minimum and natural pixel extents as
40    /// `(minimum, natural)`, measured unconstrained (`for_size = -1`)
41    /// in both orientations.
42    fn measure_size(&self) -> (Size<i32>, Size<i32>);
43
44    /// The widget's measured minimum and natural extent on one axis
45    /// (`x` = minimum, `y` = natural) for a concrete `for_size`
46    /// constraint - e.g. the height a widget reports given a fixed
47    /// width, which `measure_size` cannot express.
48    fn measure_axis(&self, orientation: gtk4::Orientation, for_size: i32) -> Vec2<(), i32>;
49
50    /// The center of the widget's allocated area in pixel
51    /// coordinates - the natural origin for pointer hit tests and
52    /// radial layouts.
53    fn center(&self) -> Vec2<crate::Pixels, f64> {
54        self.allocated_size().with_unit::<crate::Pixels>().to_f64().half()
55    }
56}
57
58#[cfg(feature = "gtk4")]
59impl<W: gtk4::prelude::IsA<gtk4::Widget>> WidgetSizeExt for W {
60    fn allocated_size(&self) -> Size<i32> {
61        use gtk4::prelude::Cast;
62        use gtk4::prelude::WidgetExt;
63        let widget = self.upcast_ref::<gtk4::Widget>();
64        Size::new(widget.width(), widget.height())
65    }
66
67    fn measure_size(&self) -> (Size<i32>, Size<i32>) {
68        use gtk4::prelude::Cast;
69        use gtk4::prelude::WidgetExt;
70        let widget = self.upcast_ref::<gtk4::Widget>();
71        let (min_w, nat_w, _, _) = widget.measure(gtk4::Orientation::Horizontal, -1);
72        let (min_h, nat_h, _, _) = widget.measure(gtk4::Orientation::Vertical, -1);
73        (Size::new(min_w, min_h), Size::new(nat_w, nat_h))
74    }
75
76    fn measure_axis(&self, orientation: gtk4::Orientation, for_size: i32) -> Vec2<(), i32> {
77        use gtk4::prelude::Cast;
78        use gtk4::prelude::WidgetExt;
79        let widget = self.upcast_ref::<gtk4::Widget>();
80        let (minimum, natural, _, _) = widget.measure(orientation, for_size);
81        Vec2::new(minimum, natural)
82    }
83}
