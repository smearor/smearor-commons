1//! Image decode through gdk — the generalized form of the
2//! chess `decode_texture` helper: gdk handles PNG/JPEG/etc. decoding,
3//! the texture pixels land in a `PixelBuffer`.
4
5use gtk4::gdk::Texture;
6use gtk4::glib;
7
8use crate::PixelBuffer;
9use crate::Rgba;
10use crate::gtk::TextureError;
11
12impl PixelBuffer<Rgba> {
13    /// Decodes `bytes` (any image format gdk understands: PNG, JPEG, …)
14    /// into a straight-alpha [`PixelBuffer<Rgba>`].
15    ///
16    /// Fails when gdk cannot decode the data ([`TextureError::Decode`]).
17    /// Consumers needing a cairo surface can follow with their own
18    /// buffer→surface bridge downstream.
19    pub fn decode_image(bytes: &[u8]) -> Result<Self, TextureError> {
20        let texture = Texture::from_bytes(&glib::Bytes::from(bytes))?;
21        Ok(Self::from_texture(&texture))
22    }
23}
