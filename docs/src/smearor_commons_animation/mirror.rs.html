1//! A yo-yo trajectory - plays the inner trajectory forward, then
2//! backward, over the same progress range.
3
4use crate::Trajectory;
5use serde::Deserialize;
6use serde::Serialize;
7use smearor_commons_geometry::Fraction;
8
9/// Mirrors the `inner` trajectory: `0 -> 0.5` plays `inner`
10/// forward, `0.5 -> 1` plays it backward - `A -> B -> A` on one
11/// clock, the yo-yo shape.
12///
13/// `Mirror<Span<ScaleFactor>>` is a breathing animation;
14/// `Mirror<Span<RotationRad>>` swings out and back. Chaining
15/// `Mirror` in a [`crate::Repeat`] gives a continuous back-and-forth
16/// without a seam.
17#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
18#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
19pub struct Mirror<S> {
20    /// The trajectory played forward, then backward.
21    pub inner: S,
22}
23
24impl<S> Mirror<S> {
25    /// Creates a mirrored trajectory.
26    pub const fn new(inner: S) -> Self {
27        Self { inner }
28    }
29}
30
31impl<S: Trajectory> Trajectory for Mirror<S> {
32    type Value = S::Value;
33
34    fn value_at(&self, t: Fraction) -> S::Value {
35        let t = t.as_f64();
36        self.inner.value_at(Fraction::new(if t <= 0.5 { 2.0 * t } else { 2.0 - 2.0 * t }))
37    }
38}
39
40#[cfg(test)]
41mod tests {
42    use super::Mirror;
43    use crate::Span;
44    use crate::Trajectory;
45    use smearor_commons_geometry::Fraction;
46
47    /// The mirror reaches the inner end at t = 0.5 and returns.
48    #[test]
49    fn the_value_turns_around_at_the_midpoint() {
50        let mirror = Mirror::new(Span::new(0.0_f64, 10.0));
51        assert_eq!(mirror.value_at(Fraction::new(0.5)), 10.0);
52        assert_eq!(mirror.value_at(Fraction::new(0.25)), 5.0);
53        assert_eq!(mirror.value_at(Fraction::new(0.75)), 5.0);
54        assert_eq!(mirror.value_at(Fraction::ONE), 0.0);
55    }
56}
