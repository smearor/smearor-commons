1//! A piecewise-linear trajectory through named progress points -
2//! the general case that [`crate::Span`] (two keys) and
3//! [`crate::ThreePhaseSpan`] (four keys) specialize.
4
5use crate::Interpolate;
6use crate::Trajectory;
7use serde::Deserialize;
8use serde::Serialize;
9use smearor_commons_geometry::Fraction;
10
11/// A trajectory through `keys` - `(progress, value)` pairs sorted
12/// ascending by progress. `value_at` lerps between the surrounding
13/// keys; before the first and after the last key the boundary
14/// value holds.
15///
16/// The first key is a separate constructor argument, so a
17/// `Keyframes` can never be empty. `Keyframes::new((0.0, a), [(0.5,
18/// b), (1.0, c)])` is the generalization every other shape is a
19/// special case of: a `Span` is two keys, a `ThreePhaseSpan` is
20/// four `(0, lead, trail, 1)`.
21#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
22#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
23pub struct Keyframes<T> {
24    /// `(progress, value)` pairs - sorted ascending on
25    /// construction, never empty.
26    pub keys: Vec<(Fraction, T)>,
27}
28
29impl<T> Keyframes<T> {
30    /// Creates a keyframe trajectory from its `first` key and the
31    /// `rest`. Keys are sorted ascending by progress.
32    pub fn new(first: (Fraction, T), rest: impl IntoIterator<Item = (Fraction, T)>) -> Self {
33        let mut keys: Vec<(Fraction, T)> = rest.into_iter().collect();
34        keys.push(first);
35        keys.sort_by(|(a, _), (b, _)| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
36        Self { keys }
37    }
38}
39
40impl<T: Interpolate + Copy> Trajectory for Keyframes<T> {
41    type Value = T;
42
43    fn value_at(&self, t: Fraction) -> T {
44        // The constructor guarantees a first key - index 0 is the
45        // boundary fallback before and after the key range.
46        let first = self.keys[0].1;
47        for window in self.keys.windows(2) {
48            let [(t0, v0), (t1, v1)] = window else { continue };
49            if t >= *t0 && t <= *t1 {
50                let span = t1.as_f64() - t0.as_f64();
51                if span <= 0.0 {
52                    return *v1;
53                }
54                return v0.interpolate(*v1, Fraction::new((t.as_f64() - t0.as_f64()) / span));
55            }
56        }
57        let last = self.keys[self.keys.len() - 1].1;
58        if t <= self.keys[0].0 { first } else { last }
59    }
60}

62#[cfg(test)]
63mod tests {
64    use super::Keyframes;
65    use crate::Trajectory;
66    use smearor_commons_geometry::Fraction;
67
68    /// Outside the key range the boundary values hold.
69    #[test]
70    fn boundary_values_hold_outside_the_keys() {
71        let keys = Keyframes::new((Fraction::new(0.25), 10.0_f64), [(Fraction::new(0.75), 20.0)]);
72        assert_eq!(keys.value_at(Fraction::ZERO), 10.0);
73        assert_eq!(keys.value_at(Fraction::ONE), 20.0);
74    }
75
76    /// Between two keys the value lerps linearly.
77    #[test]
78    fn values_lerp_between_neighbouring_keys() {
79        let keys = Keyframes::new((Fraction::ZERO, 0.0_f64), [(Fraction::HALF, 10.0), (Fraction::ONE, 20.0)]);
80        assert_eq!(keys.value_at(Fraction::new(0.25)), 5.0);
81        assert_eq!(keys.value_at(Fraction::new(0.75)), 15.0);
82    }
83
84    /// Unsorted keys land in progress order.
85    #[test]
86    fn keys_are_sorted_on_construction() {
87        let keys = Keyframes::new((Fraction::ONE, 20.0_f64), [(Fraction::ZERO, 0.0), (Fraction::HALF, 10.0)]);
88        assert_eq!(keys.keys[0].0, Fraction::ZERO);
89        assert_eq!(keys.keys[2].0, Fraction::ONE);
90    }
