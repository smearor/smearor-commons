8/// Progress curve of an animation: maps linear time progress
9/// (`0.0..=1.0`) to eased progress. Eased values may exceed the
10/// range (e.g. [`Overshoot`](Self::Overshoot)) - trajectories decide
11/// whether extrapolation is meaningful for their value type.
12#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
13#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
14pub enum EasingFunction {
15    /// Constant speed - eased progress equals time progress.
16    Linear,
17    /// Slow start, fast finish (quadratic ease-in).
18    EaseIn,
19    /// Fast start, slow finish (quadratic ease-out).
20    EaseOut,
21    /// Slow start and end, fast middle (quadratic ease-in-out).
22    EaseInOut,
23    /// Slow start, fast finish (cubic ease-in).
24    EaseInCubic,
25    /// Fast start, slow finish (cubic ease-out) - the classic
26    /// "arrive" curve for camera and pose transitions.
27    EaseOutCubic,
28    /// Slow start and end, fast middle (cubic ease-in-out).
29    EaseInOutCubic,
30    /// Goes below 0, then beyond the target and settles back
31    /// (Penner's back-in-out).
32    Overshoot {
33        /// Overshoot strength - higher values overshoot further.
34        overshoot_amount: f64,
35    },
36    /// Slow start, fast finish (quartic ease-in, `t^4`).
37    EaseInQuart,
38    /// Fast start, slow finish (quartic ease-out).
39    EaseOutQuart,
40    /// Slow start and end, fast middle (quartic ease-in-out).
41    EaseInOutQuart,
42    /// Slow start, fast finish (quintic ease-in, `t^5`).
43    EaseInQuint,
44    /// Fast start, slow finish (quintic ease-out).
45    EaseOutQuint,
46    /// Slow start and end, fast middle (quintic ease-in-out).
47    EaseInOutQuint,
48    /// Slow start, fast finish (sinusoidal ease-in).
49    EaseInSine,
50    /// Fast start, slow finish (sinusoidal ease-out) - the gentlest
51    /// curve of the family.
52    EaseOutSine,
53    /// Slow start and end, fast middle (sinusoidal ease-in-out).
54    EaseInOutSine,
55    /// Very slow start, abrupt finish (exponential ease-in).
56    EaseInExpo,
57    /// Abrupt start, very slow finish (exponential ease-out).
58    EaseOutExpo,
59    /// Abrupt middle between very slow ends (exponential
60    /// ease-in-out).
61    EaseInOutExpo,
62    /// Slow start, fast finish (circular ease-in).
63    EaseInCirc,
64    /// Fast start, slow finish (circular ease-out).
65    EaseOutCirc,
66    /// Slow start and end, fast middle (circular ease-in-out).
67    EaseInOutCirc,
68    /// Dips below the start value, then accelerates (Penner's
69    /// back-in).
70    BackIn {
71        /// Overshoot strength - higher values dip further.
72        overshoot_amount: f64,
73    },
74    /// Overshoots the target, then settles back (Penner's back-out).
75    /// [`Self::Overshoot`] is the in-out combination of this and
76    /// [`Self::BackIn`].
77    BackOut {
78        /// Overshoot strength - higher values overshoot further.
79        overshoot_amount: f64,
80    },
81    /// Oscillates around the target and settles - the springy
82    /// ease-out of the easing literature.
83    Elastic {
84        /// Oscillation amplitude - `1.0` is the classic curve.
85        amplitude: f64,
86        /// Oscillation period in progress units - the classic
87        /// curve uses `0.3`.
88        period: f64,
89    },
90    /// Bounces off the target with decreasing heights - the
91    /// physical drop-in of the easing literature.
92    Bounce,
93    /// A custom cubic bezier curve - CSS `cubic-bezier()`
94    /// compatible: `x1`, `y1` are the outgoing control point, `x2`,
95    /// `y2` the incoming one. `y` may exceed `0..1`, `x` is clamped
96    /// to it.
97    CubicBezier {
98        /// X of the outgoing control point.
99        x1: f64,
100        /// Y of the outgoing control point.
101        y1: f64,
102        /// X of the incoming control point.
103        x2: f64,
104        /// Y of the incoming control point.
105        y2: f64,
106    },
107    /// Discrete progress: `steps` equally sized jumps, CSS
108    /// `steps()` compatible.
109    Steps {
110        /// The number of discrete steps.
111        steps: u32,
112        /// When the jump happens within each step interval.
113        jump: StepsJump,
114    },
115}

117impl Default for EasingFunction {
118    /// The identity curve - `Default` makes a plain
119    /// `#[serde(default)]` mean "no easing".
120    fn default() -> Self {
121        Self::Linear
122    }
123}

125/// When a [`EasingFunction::Steps`] interval jumps to the next
126/// value - the CSS `steps()` jump terms.
127#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
128#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
129pub enum StepsJump {
130    /// The step value holds at the interval start - the first jump
131    /// happens at `t = 0`.
132    Start,
133    /// The step value holds at the interval end - the last jump
134    /// happens at `t = 1`.
135    End,
136}

138impl EasingFunction {
139    /// Penner's canonical back-easing strength (`s = 1.70158`) -
140    /// the standard "visible but tasteful" overshoot from the
141    /// easing literature.
142    pub const DEFAULT_OVERSHOOT: f64 = 1.70158;
143
144    /// The canonical overshoot easing - [`Self::Overshoot`] with
145    /// [`Self::DEFAULT_OVERSHOOT`].
146    pub const fn overshoot() -> Self {
147        Self::Overshoot {
148            overshoot_amount: Self::DEFAULT_OVERSHOOT,
149        }
150    }
151
152    /// The eased progress for linear time progress `t`.
153    pub fn apply(&self, t: Fraction) -> Fraction {
154        let t = t.as_f64();
155        // The endpoints are contractually fixed - trigonometric
156        // curves would otherwise return rounding residue.
157        if t <= 0.0 {
158            return Fraction::ZERO;
159        }
160        if t >= 1.0 {
161            return Fraction::ONE;
162        }
163        Fraction::new(match self {
164            Self::Linear => t,
165            Self::EaseIn => t * t,
166            Self::EaseOut => 1.0 - (1.0 - t) * (1.0 - t),
167            Self::EaseInOut => {
168                if t < 0.5 {
169                    2.0 * t * t
170                } else {
171                    -1.0 + (4.0 - 2.0 * t) * t
172                }
173            }
174            Self::EaseInCubic => t * t * t,
175            Self::EaseOutCubic => 1.0 - (1.0 - t).powi(3),
176            Self::EaseInOutCubic => {
177                if t < 0.5 {
178                    4.0 * t * t * t
179                } else {
180                    1.0 - (-2.0 * t + 2.0).powi(3) / 2.0
181                }
182            }
183            Self::Overshoot { overshoot_amount } => {
184                let s = overshoot_amount * 1.525;
185                if t < 0.5 {
186                    let p = t * 2.0;
187                    0.5 * (p * p * ((s + 1.0) * p - s))
188                } else {
189                    let p = t * 2.0 - 2.0;
190                    0.5 * (p * p * ((s + 1.0) * p + s) + 2.0)
191                }
192            }
193            Self::EaseInQuart => t.powi(4),
194            Self::EaseOutQuart => 1.0 - (1.0 - t).powi(4),
195            Self::EaseInOutQuart => {
196                if t < 0.5 {
197                    8.0 * t.powi(4)
198                } else {
199                    1.0 - (-2.0 * t + 2.0).powi(4) / 2.0
200                }
201            }
202            Self::EaseInQuint => t.powi(5),
203            Self::EaseOutQuint => 1.0 - (1.0 - t).powi(5),
204            Self::EaseInOutQuint => {
205                if t < 0.5 {
206                    16.0 * t.powi(5)
207                } else {
208                    1.0 - (-2.0 * t + 2.0).powi(5) / 2.0
209                }
210            }
211            Self::EaseInSine => 1.0 - (t * FRAC_PI_2).cos(),
212            Self::EaseOutSine => (t * FRAC_PI_2).sin(),
213            Self::EaseInOutSine => -((t * PI).cos() - 1.0) / 2.0,
214            Self::EaseInExpo => {
215                if t <= 0.0 {
216                    0.0
217                } else {
218                    (10.0 * t - 10.0).exp2()
219                }
220            }
221            Self::EaseOutExpo => {
222                if t >= 1.0 {
223                    1.0
224                } else {
225                    1.0 - (-10.0 * t).exp2()
226                }
227            }
228            Self::EaseInOutExpo => {
229                if t <= 0.0 {
230                    0.0
231                } else if t >= 1.0 {
232                    1.0
233                } else if t < 0.5 {
234                    (20.0 * t - 10.0).exp2() / 2.0
235                } else {
236                    (2.0 - (-20.0 * t + 10.0).exp2()) / 2.0
237                }
238            }
239            Self::EaseInCirc => 1.0 - (1.0 - t * t).sqrt(),
240            Self::EaseOutCirc => (1.0 - (t - 1.0).powi(2)).sqrt(),
241            Self::EaseInOutCirc => {
242                if t < 0.5 {
243                    (1.0 - (1.0 - 4.0 * t * t).sqrt()) / 2.0
244                } else {
245                    ((1.0 - (-2.0 * t + 2.0).powi(2)).sqrt() + 1.0) / 2.0
246                }
247            }
248            Self::BackIn { overshoot_amount } => {
249                let s = overshoot_amount + 1.0;
250                s * t * t * t - overshoot_amount * t * t
251            }
252            Self::BackOut { overshoot_amount } => {
253                let s = overshoot_amount + 1.0;
254                let p = t - 1.0;
255                1.0 + s * p * p * p + overshoot_amount * p * p
256            }
257            Self::Elastic { amplitude, period } => {
258                if t <= 0.0 {
259                    0.0
260                } else if t >= 1.0 {
261                    1.0
262                } else {
263                    // The phase shift starts the oscillation on the
264                    // target side so the curve passes through 0.
265                    let phase = period / TAU * (amplitude.max(1.0).asin());
266                    amplitude * (-10.0 * t).exp2() * (TAU / period * (t - phase)).sin() + 1.0
267                }
268            }
269            Self::Bounce => {
270                // The classic ease-out bounce: parabolic arcs of
271                // decreasing height, mirrored into `0..1`.
272                let t = 1.0 - t;
273                1.0 - if t < 1.0 / 2.75 {
274                    7.5625 * t * t
275                } else if t < 2.0 / 2.75 {
276                    let t = t - 1.5 / 2.75;
277                    7.5625 * t * t + 0.75
278                } else if t < 2.5 / 2.75 {
279                    let t = t - 2.25 / 2.75;
280                    7.5625 * t * t + 0.9375
281                } else {
282                    let t = t - 2.625 / 2.75;
283                    7.5625 * t * t + 0.984375
284                }
285            }
286            Self::CubicBezier { x1, y1, x2, y2 } => {
287                let bezier = |s: f64, a: f64, b: f64| (1.0 - s).powi(3) * 0.0 + 3.0 * (1.0 - s).powi(2) * s * a + 3.0 * (1.0 - s) * s * s * b + s.powi(3);
288                // Invert x(s) = t via Newton iterations, bisection
289                // fallback - the CSS `cubic-bezier()` semantics.
290                let mut param = t;
291                for _ in 0..8 {
292                    let x = bezier(param, *x1, *x2) - t;
293                    if x.abs() < 1e-6 {
294                        break;
295                    }
296                    let dx = 3.0 * (1.0 - param).powi(2) * x1 + 6.0 * (1.0 - param) * param * (x2 - x1) + 3.0 * param * param * (1.0 - x2);
297                    if dx.abs() < 1e-6 {
298                        break;
299                    }
300                    param = (param - x / dx).clamp(0.0, 1.0);
301                }
302                bezier(param, *y1, *y2)
303            }
304            Self::Steps { steps, jump } => {
305                let n = (*steps).max(1) as f64;
306                let step = (t * n).floor() / n;
307                match jump {
308                    StepsJump::End => step,
309                    StepsJump::Start => {
310                        if t <= 0.0 {
311                            1.0 / n
312                        } else {
313                            (step + 1.0 / n).min(1.0)
314                        }
315                    }
316                }
317            }
318        })
319    }
320}

379    /// Overshoot exceeds 1.0 somewhere in the second half - the
380    /// "settle back" is the point of the curve.
381    #[test]
382    fn overshoot_exceeds_the_target() {
383        let easing = EasingFunction::Overshoot { overshoot_amount: 1.7 };
384        let exceeded = (1..100).any(|i| easing.apply(Fraction::new(i as f64 / 100.0)) > Fraction::ONE);
385        assert!(exceeded);
386    }

388    /// EaseIn is slow at the start, EaseOut is fast - mirror curves.
389    #[test]
390    fn ease_in_and_out_mirror_each_other() {
391        for i in 0..=10 {
392            let t = i as f64 / 10.0;
393            let sum = EasingFunction::EaseIn.apply(Fraction::new(t)).as_f64() + EasingFunction::EaseOut.apply(Fraction::new(1.0 - t)).as_f64();
394            assert!((sum - 1.0).abs() < 1e-9, "t={t}");
395        }
396    }

399/// BackIn dips below the start, BackOut overshoots the target -
400/// the two halves of Overshoot.
401#[test]
402fn back_curves_exceed_their_ends() {
403    let back_in = EasingFunction::BackIn { overshoot_amount: 1.7 };
404    let back_out = EasingFunction::BackOut { overshoot_amount: 1.7 };
405    assert!((1..50).any(|i| back_in.apply(Fraction::new(i as f64 / 100.0)) < Fraction::ZERO));
406    assert!((50..100).any(|i| back_out.apply(Fraction::new(i as f64 / 100.0)) > Fraction::ONE));
407}

409/// The CSS `ease` curve `cubic-bezier(0.25, 0.1, 0.25, 1.0)` is
410/// a gentle ease-out - its midpoint sits above 0.5.
411#[test]
412fn cubic_bezier_matches_the_css_ease_curve() {
413    let bezier = EasingFunction::CubicBezier {
414        x1: 0.25,
415        y1: 0.1,
416        x2: 0.25,
417        y2: 1.0,
418    };
419    let mid = bezier.apply(Fraction::HALF).as_f64();

423/// Steps only ever produce the discrete jump values.
424#[test]
425fn steps_quantizes_the_progress() {
426    let steps = EasingFunction::Steps {
427        steps: 4,
428        jump: crate::easing::StepsJump::End,
429    };
430    for i in 0..100 {
431        let eased = steps.apply(Fraction::new(i as f64 / 100.0)).as_f64();
432        assert!((eased * 4.0).fract().abs() < 1e-9, "t={i}: {eased}");
433    }
434    assert_eq!(steps.apply(Fraction::ZERO), Fraction::ZERO);
435    assert_eq!(steps.apply(Fraction::new(0.99)), Fraction::new(0.75));
