1use crate::Interpolate;

3use serde::Deserialize;

7/// A three-phase trajectory: interpolates `from -> mid` over the
8/// leading `lead` fraction of progress, holds `mid` until `trail`,
9/// then interpolates `mid -> to` over the remainder.
10///
11/// The zoom-during-rotation pattern (zoom out, hold, zoom back in)
12/// is `ThreePhaseSpan<ScaleFactor>` with `from = to = 1.0`.
13#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
14#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
15pub struct ThreePhaseSpan<T> {
16    /// The value at progress `0`.
17    pub from: T,
18    /// The value held between `lead` and `trail`.
19    pub mid: T,
20    /// The value at progress `1`.
21    pub to: T,
22    /// End of the leading phase (progress fraction `0.0..trail`).
23    pub lead: Fraction,
24    /// End of the hold phase (progress fraction `lead..1.0`).
25    pub trail: Fraction,
26}
27
28impl<T> ThreePhaseSpan<T> {
29    /// Creates a three-phase trajectory. `lead` ends the `from ->
30    /// mid` phase; `trail` ends the `mid` hold and starts `mid ->
31    /// to`. Both fractions clamp into `0.0..=1.0` with
32    /// `lead <= trail`.
33    pub fn new(from: T, mid: T, to: T, lead: Fraction, trail: Fraction) -> Self {
34        let lead = lead.clamped();
35        let trail = trail.clamped();
36        let trail = if trail < lead { lead } else { trail };
37        Self { from, mid, to, lead, trail }
38    }
39}
40
41impl<T: Interpolate + Copy> Trajectory for ThreePhaseSpan<T> {
42    type Value = T;
43
44    fn value_at(&self, t: Fraction) -> T {
45        if t < self.lead {
46            // `t < lead` implies `lead > 0` - the division is safe.
47            self.from.interpolate(self.mid, Fraction::new(t / self.lead))
48        } else if t <= self.trail {
49            self.mid
50        } else {
51            let tail = Fraction::new((t - self.trail) / (Fraction::ONE - self.trail));
52            self.mid.interpolate(self.to, tail)
53        }
54    }
55}

57#[cfg(test)]
58mod tests {
59    use super::ThreePhaseSpan;
60    use crate::Trajectory;
61    use smearor_commons_geometry::Fraction;
62
63    /// `from -> mid` over the lead, `mid` hold, `mid -> to` over
64    /// the tail - each phase interpolates linearly.
65    #[test]
66    fn phases_interpolate_and_hold() {
67        let span = ThreePhaseSpan::new(1.0_f64, 0.9, 1.0, Fraction::new(1.0 / 3.0), Fraction::new(2.0 / 3.0));
68        assert_eq!(span.value_at(Fraction::new(0.0)), 1.0);
69        assert!((span.value_at(Fraction::new(1.0 / 6.0)) - 0.95).abs() < 1e-9);
70        assert!((span.value_at(Fraction::new(1.0 / 3.0)) - 0.9).abs() < 1e-9);
71        assert!((span.value_at(Fraction::new(0.5)) - 0.9).abs() < 1e-9);
72        assert!((span.value_at(Fraction::new(2.0 / 3.0)) - 0.9).abs() < 1e-9);
73        assert!((span.value_at(Fraction::new(5.0 / 6.0)) - 0.95).abs() < 1e-9);
74        assert!((span.value_at(Fraction::new(1.0)) - 1.0).abs() < 1e-9);
75    }
76
77    /// Out-of-range fractions clamp so `lead <= trail` always.
78    #[test]
79    fn out_of_range_phase_fractions_clamp() {
80        let span = ThreePhaseSpan::new(0.0_f64, 1.0, 2.0, Fraction::new(0.8), Fraction::new(0.4));
81        assert_eq!(span.lead, Fraction::new(0.8));
82        assert_eq!(span.trail, Fraction::new(0.8));
83    }
