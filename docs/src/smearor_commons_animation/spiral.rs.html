1//! A spiral trajectory - `Orbit` with a radius growing over time.
2
3use crate::Trajectory;
4use serde::Deserialize;
5use serde::Serialize;
6use smearor_commons_geometry::Fraction;
7use smearor_commons_geometry::Vec2;
8use std::f64::consts::TAU;
9
10/// An Archimedean spiral: the value circles `center` `turns` times
11/// while its distance grows linearly from `0` to `radius` -
12/// winding in or out depending on the direction of travel.
13#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
14#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
15pub struct Spiral<U = ()> {
16    /// The point the spiral winds out of.
17    pub center: Vec2<U>,
18    /// The final radius at progress `1`, in the unit of `center`.
19    pub radius: f64,
20    /// Revolutions over `0..=1` - `1.0` is a single winding.
21    pub turns: f64,
22}
23
24impl<U> Spiral<U> {
25    /// Creates a single-winding spiral from `center` to `radius`.
26    pub fn new(center: Vec2<U>, radius: f64) -> Self {
27        Self { center, radius, turns: 1.0 }
28    }
29
30    /// Returns the spiral with `turns` windings per run.
31    pub fn with_turns(mut self, turns: f64) -> Self {
32        self.turns = turns;
33        self
34    }
35}
36
37impl<U> Trajectory for Spiral<U> {
38    type Value = Vec2<U>;
39
40    fn value_at(&self, t: Fraction) -> Vec2<U> {
41        let t = t.as_f64();
42        let radius = self.radius * t;
43        let angle = t * self.turns * TAU;
44        Vec2::new(self.center.x + radius * angle.cos(), self.center.y + radius * angle.sin())
45    }
46}

48#[cfg(test)]
49mod tests {
50    use super::Spiral;
51    use crate::Trajectory;
52    use smearor_commons_geometry::Fraction;
53    use smearor_commons_geometry::Vec2;
54
55    /// The spiral starts at the center and ends a full radius out.
56    #[test]
57    fn the_spiral_grows_from_the_center() {
58        let spiral = Spiral::new(Vec2::<()>::new(0.5, 0.5), 0.4);
59        assert_eq!(spiral.value_at(Fraction::ZERO), Vec2::<()>::new(0.5, 0.5));
60        let end = spiral.value_at(Fraction::ONE);
61        assert!((end.x - 0.9).abs() < 1e-9);
62        assert!((end.y - 0.5).abs() < 1e-9);
63    }
