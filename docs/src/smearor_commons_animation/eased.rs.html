1//! Easing as a trajectory decorator - shapes progress before the
2//! inner trajectory sees it.
3
4use crate::EasingFunction;
5use crate::Trajectory;
6use serde::Deserialize;
7use serde::Serialize;
8use smearor_commons_geometry::Fraction;
9
10/// Applies `easing` to the progress before delegating to `inner`:
11/// `value_at(t) = inner.value_at(easing.apply(t))`.
12///
13/// Lets each part of a composite trajectory keep its own curve -
14/// `(Eased<Span>, Eased<Span>)` eases a composite's members
15/// differently while `TimedAnimation`'s outer easing stays shared.
16/// Endpoints are unaffected - easing snaps `0`/`1` through.
17#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
18#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
19pub struct Eased<S> {
20    /// The easing applied before the inner trajectory.
21    pub easing: EasingFunction,
22    /// The wrapped trajectory.
23    pub inner: S,
24}
25
26impl<S> Eased<S> {
27    /// Wraps `inner` so progress passes `easing` first.
28    pub fn new(inner: S, easing: EasingFunction) -> Self {
29        Self { easing, inner }
30    }
31}
32
33impl<S: Trajectory> Trajectory for Eased<S> {
34    type Value = S::Value;
35
36    fn value_at(&self, t: Fraction) -> S::Value {
37        self.inner.value_at(self.easing.apply(t))
38    }
39}
40
41#[cfg(test)]
42mod tests {
43    use super::Eased;
44    use crate::EasingFunction;
45    use crate::Span;
46    use crate::Trajectory;
47    use smearor_commons_geometry::Fraction;
48
49    /// The easing shapes the inner progress, endpoints pass through.
50    #[test]
51    fn the_easing_shapes_the_inner_progress() {
52        let eased = Eased::new(Span::new(0.0_f64, 1.0), EasingFunction::EaseIn);
53        assert_eq!(eased.value_at(Fraction::ZERO), 0.0);
54        assert_eq!(eased.value_at(Fraction::HALF), 0.25); // t^2 at .5
55        assert_eq!(eased.value_at(Fraction::ONE), 1.0);
56    }
57}
