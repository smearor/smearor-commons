1//! A closed path through vertices - the value travels each edge
2//! in equal time.
3
4use crate::Interpolate;
5use crate::Trajectory;
6use serde::Deserialize;
7use serde::Serialize;
8use smearor_commons_geometry::Fraction;
9use smearor_commons_geometry::Vec2;
10
11/// A polygon path: the value travels the edges `points[0] →
12/// points[1] → … → points[n-1] → points[0]` at equal time per
13/// edge - patrol routes, marquee loops, shape tracing.
14///
15/// Generic over the point type - `PolygonPath<Vec2<Pixels>>` walks
16/// screen vertices, `PolygonPath<Vec3>` a spatial polygon. Sharper
17/// than [`crate::CatmullRom`]: corners stay corners.
18#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
19#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
20pub struct PolygonPath<P = Vec2<()>> {
21    /// The vertices to travel - two or more.
22    pub points: Vec<P>,
23}
24
25impl<P> PolygonPath<P> {
26    /// Creates a closed path over `points`.
27    pub fn new(points: impl IntoIterator<Item = P>) -> Self {
28        let points: Vec<P> = points.into_iter().collect();
29        Self { points }
30    }
31}
32
33impl<P: Interpolate + Copy> Trajectory for PolygonPath<P> {
34    type Value = P;
35
36    fn value_at(&self, t: Fraction) -> P {
37        let edges = self.points.len().max(1);
38        let position = t.as_f64().clamp(0.0, 1.0) * edges as f64;
39        let edge = (position as usize).min(edges - 1);
40        let from = self.points[edge];
41        let to = self.points[(edge + 1) % edges];
42        from.interpolate(to, Fraction::new(position - edge as f64))
43    }
44}

46#[cfg(test)]
47mod tests {
48    use super::PolygonPath;
49    use crate::Trajectory;
50    use smearor_commons_geometry::Fraction;
51    use smearor_commons_geometry::Vec2;
52
53    /// A triangle closes: start and end share the first vertex.
54    #[test]
55    fn the_path_closes_onto_itself() {
56        let triangle = PolygonPath::new([Vec2::<()>::new(0.0, 0.0), Vec2::<()>::new(1.0, 0.0), Vec2::<()>::new(0.5, 1.0)]);
57        assert_eq!(triangle.value_at(Fraction::ZERO), Vec2::<()>::new(0.0, 0.0));
58        assert_eq!(triangle.value_at(Fraction::ONE), Vec2::<()>::new(0.0, 0.0));
59    }
60
61    /// Equal time per edge: the first third reaches vertex two.
62    #[test]
63    fn edges_get_equal_time() {
64        let triangle = PolygonPath::new([Vec2::<()>::new(0.0, 0.0), Vec2::<()>::new(1.0, 0.0), Vec2::<()>::new(0.5, 1.0)]);
65        assert_eq!(triangle.value_at(Fraction::new(1.0 / 3.0)), Vec2::<()>::new(1.0, 0.0));
66    }
