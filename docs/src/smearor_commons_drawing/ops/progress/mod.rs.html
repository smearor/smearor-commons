12/// Fraction-based indicator drawing operations: progress bars and
13/// circular gauges. Values are clamped to `[0.0, 1.0]`.
14pub trait DrawProgressOps: DrawBackend {
15    /// Draw `bar`'s strip at the bottom of the view.
16    fn draw_progress_bar(&mut self, bar: ProgressBar) -> Result<(), Self::Error>;
17
18    /// Draw `gauge`: a faint `track` ring plus a `color` arc starting
19    /// at 12 o'clock and sweeping `value` of a full turn clockwise
20    /// (sysinfo `draw_gauge` lineage, generalized to `Fraction`
21    /// semantics).
22    ///
23    /// Multi-arc gauges (network up/down) and threshold markers
24    /// (temperature) stay consumer-side policy built on this primitive.
25    fn draw_gauge(&mut self, gauge: Gauge) -> Result<(), Self::Error>;
26}
27
28impl<T: DrawProgressOps + ?Sized> DrawProgressOps for &mut T {
29    fn draw_progress_bar(&mut self, bar: ProgressBar) -> Result<(), Self::Error> {
30        (**self).draw_progress_bar(bar)
31    }
32
33    fn draw_gauge(&mut self, gauge: Gauge) -> Result<(), Self::Error> {
34        (**self).draw_gauge(gauge)
35    }
36}
