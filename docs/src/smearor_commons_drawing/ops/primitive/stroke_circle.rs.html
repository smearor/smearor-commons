1//! The [`StrokeCircle`] draw spec.
2
3use smearor_commons_color::Alpha;
4use smearor_commons_color::Color;
5use smearor_commons_geometry::Circle;
6use smearor_commons_geometry::Pixels;
7
8/// A circle outline's draw specification: ring geometry, color,
9/// stroke width and alpha. The stroke is centered on the circle,
10/// matching cairo's `arc` + `stroke` semantics.
11#[derive(Debug, Clone, PartialEq)]
12pub struct StrokeCircle {
13    /// Circle whose outline is stroked.
14    pub circle: Circle<Pixels>,
15    /// Stroke color.
16    pub color: Color<f64>,
17    /// Stroke width in pixels, centered on the outline.
18    pub line_width: Pixels<f32>,
19    /// Alpha composited on top of the color's own alpha channel.
20    pub alpha: Alpha,
21}
22
23impl StrokeCircle {
24    /// Creates a circle stroke spec.
25    pub fn new(circle: Circle<Pixels>, color: impl Into<Color<f64>>, line_width: Pixels<f32>, alpha: impl Into<Alpha>) -> Self {
26        Self {
27            circle,
28            color: color.into(),
29            line_width,
30            alpha: alpha.into(),
31        }
32    }
33}
