1//! The [`DrawTarget`] trait: a full-vocabulary render target with a
2//! known pixel extent.
3
4use crate::ops::DrawOps;
5use smearor_commons_color::Color;
6use smearor_commons_geometry::Size;
7
8/// A render target that implements the full [`DrawOps`] vocabulary and
9/// reports its own pixel extent.
10///
11/// This is the bound [`crate::RenderContext`] places on the target it
12/// owns: its convenience methods need the target's size for layout
13/// (icon placement, progress bar position). Both buffer-side types
14/// implement it — `PixelBuffer` and `BufferViewMut` — and backend
15/// targets like `CairoTarget` carry their surface size for it.
16pub trait DrawTarget: DrawOps {
17    /// The target's pixel dimensions.
18    fn size(&self) -> Size<u32>;
19
20    /// The default color for text and icons in the given `is_active`
21    /// state. This is the single anchor for the crate's color policy:
22    /// the `Option<Color>` parameters of the capability traits resolve
23    /// `None` to this value. Backends may override it — a themed
24    /// target could derive its defaults from the UI theme.
25    fn default_text_color(&self, is_active: bool) -> Color<u8> {
26        if is_active {
27            Color::<u8>::new_rgba(100, 180, 255, 255)
28        } else {
29            Color::<u8>::new_rgba(240, 240, 240, 255)
30        }
31    }
32
33    /// The target's deferred error state, if the backend has one.
34    ///
35    /// Draw methods already propagate `Self::Error` per call; this is
36    /// the optional end-of-pass check for backends whose errors are
37    /// sticky — a failed cairo surface keeps reporting the same error
38    /// through `status()` while subsequent draws no-op. The raster
39    /// target is infallible and always reports `Ok`.
40    fn status(&self) -> Result<(), Self::Error> {
41        Ok(())
42    }
43}
44
45impl<T: DrawTarget + ?Sized> DrawTarget for &mut T {
46    fn size(&self) -> Size<u32> {
47        (**self).size()
48    }
49
50    fn default_text_color(&self, is_active: bool) -> Color<u8> {
51        (**self).default_text_color(is_active)
52    }
53
54    fn status(&self) -> Result<(), Self::Error> {
55        (**self).status()
56    }
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62    use smearor_commons_buffer::PixelBuffer;
63    use smearor_commons_buffer::Rgba;
64    use smearor_commons_geometry::Size;
65
66    #[test]
67    fn default_text_color_active_is_blue() {
68        let buf = PixelBuffer::<Rgba>::new(Size::new(4, 4));
69        let color = buf.default_text_color(true);
70        assert_eq!(color, Color::<u8>::new_rgba(100, 180, 255, 255));
71    }
72
73    #[test]
74    fn default_text_color_inactive_is_light_gray() {
75        let buf = PixelBuffer::<Rgba>::new(Size::new(4, 4));
76        let color = buf.default_text_color(false);
77        assert_eq!(color, Color::<u8>::new_rgba(240, 240, 240, 255));
78    }
