1//! The cairo render target: an owned [`cairo::Context`] handle plus a

3
4use crate::ops::DrawBackend;
5use cairo::Context;
6use smearor_commons_geometry::Size;
7
8/// A cairo render target: owns a [`cairo::Context`] handle plus the
9/// pixel extent it draws to.
10///
11/// Cloning a `Context` is a refcount bump — `CairoTarget` is a value
12/// type, deliberately not a borrow wrapper. GTK callbacks hand out
13/// `&Context`; [`new`](Self::new) clones that handle, so a target can be
14/// constructed inside `snapshot`/`set_draw_func` callbacks without
15/// ceremony, and [`RenderContext`](crate::RenderContext) can own it like
16/// `BufferViewMut`.
17///
18/// The extent is *declared*, not read back: not every surface type
19/// reports one (recording surfaces are unbounded), and `DrawTarget`
20/// needs a reliable `size()`.
21///
22/// Use [`context`](Self::context) for backend-specific operations that
23/// are not part of the shared vocabulary — `save`/`restore`,
24/// transforms, patterns, surface sources.
25pub struct CairoTarget {
26    pub(crate) ctx: Context,
27    size: Size<u32>,
28}
29
30impl CairoTarget {
31    /// Wraps `ctx`, declaring `size` as the drawable pixel extent.
32    /// Clones the handle (refcount bump only).
33    pub fn new(ctx: &Context, size: Size<u32>) -> Self {
34        Self { ctx: ctx.clone(), size }
35    }
36
37    /// The wrapped context, for backend-specific operations outside the
38    /// shared vocabulary (transforms, `save`/`restore`, patterns).
39    pub fn context(&self) -> &Context {
40        &self.ctx
41    }
42
43    /// The declared drawable pixel extent.
44    pub fn size(&self) -> Size<u32> {
45        self.size
46    }
47
48    /// The deferred cairo status of the wrapped context.
49    ///
50    /// cairo reports errors lazily: once a surface or context fails,
51    /// every subsequent operation is a no-op and `status()` keeps
52    /// reporting the sticky error. Draw methods already propagate
53    /// `Self::Error`; this is the optional end-of-pass check.
54    pub fn status(&self) -> Result<(), cairo::Error> {
55        self.ctx.status()
56    }
57}
58
59impl DrawBackend for CairoTarget {
60    /// cairo errors — deferred/sticky by cairo's own error model.
61    type Error = cairo::Error;
62}
