1//! The [`DrawProgressOps`] implementation for [`CairoTarget`].
2
3use crate::cairo::CairoTarget;
4use crate::ops::DrawProgressOps;
5use crate::ops::Gauge;
6use crate::ops::PROGRESS_BAR_BOTTOM_MARGIN;
7use crate::ops::ProgressBar;
8use smearor_commons_color::Color;
9use std::f64::consts::FRAC_PI_2;
10use std::f64::consts::TAU;
11
12impl DrawProgressOps for CairoTarget {
13    /// The same strip the raster backend draws — `bar.height` tall,
14    /// raised `PROGRESS_BAR_BOTTOM_MARGIN` off the bottom edge — as
15    /// two rectangles: the filled fraction in `bar.color`, the rest
16    /// darkened to a third.
17    fn draw_progress_bar(&mut self, bar: ProgressBar) -> Result<(), Self::Error> {
18        let ProgressBar { value, color, height } = bar;
19        let size = self.size();
20        let bar_height = height.as_f64().max(0.0);
21        let bar_y = (f64::from(size.y) - bar_height - f64::from(PROGRESS_BAR_BOTTOM_MARGIN)).max(0.0);
22        let fill_width = (f64::from(size.x) * value.clamped().as_f64()).min(f64::from(size.x));
23        let rgba: Color<u8> = color.into();
24
25        Color::<f64>::from(rgba).apply(&self.ctx);
26        self.ctx.rectangle(0.0, bar_y, fill_width, bar_height);
27        self.ctx.fill()?;
28
29        let rest_width = f64::from(size.x) - fill_width;
30        if rest_width > 0.0 {
31            let track = Color::<u8>::new_rgba(rgba.r / 3, rgba.g / 3, rgba.b / 3, 255);
32            Color::<f64>::from(track).apply(&self.ctx);
33            self.ctx.rectangle(fill_width, bar_y, rest_width, bar_height);
34            self.ctx.fill()?;
35        }
36        Ok(())
37    }
38
39    /// The gauge as two cairo arcs: the full `track` circle stroked,
40    /// then the `value` arc stroked on top from 12 o'clock clockwise —
41    /// cairo `arc` angles run clockwise in y-down user space, matching
42    /// the raster backend's `RotationRad` convention.
43    fn draw_gauge(&mut self, gauge: Gauge) -> Result<(), Self::Error> {
44        let Gauge {
45            center,
46            radius,
47            line_width,
48            value,
49            color,
50            track,
51        } = gauge;
52        let (center, radius, line_width) = (center.to_f64(), radius.as_f64(), line_width.as_f64());
53        if line_width <= 0.0 || radius <= line_width / 2.0 {
54            return Ok(());
55        }
56        self.ctx.set_line_width(line_width);
57
58        track.apply(&self.ctx);
59        self.ctx.arc(center.x, center.y, radius, 0.0, TAU);
60        self.ctx.stroke()?;
61
62        let sweep = value.clamped().as_f64() * TAU;
63        if sweep > 0.0 {
64            color.apply(&self.ctx);
65            self.ctx.arc(center.x, center.y, radius, -FRAC_PI_2, -FRAC_PI_2 + sweep);
66            self.ctx.stroke()?;
67        }
68        Ok(())
69    }
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75    use cairo::Context;
76    use cairo::Format;
77    use cairo::ImageSurface;
78    use smearor_commons_geometry::Fraction;
79    use smearor_commons_geometry::Pixels;
80    use smearor_commons_geometry::Size;
81    use smearor_commons_geometry::Vec2;
82
83    #[test]
84    fn draw_progress_bar_fills_fraction() {
85        let size = Size::new(10, 10);
86        let surface = ImageSurface::create(Format::ARgb32, 10, 10).unwrap();
87        let mut target = CairoTarget::new(&Context::new(&surface).unwrap(), size);
88        target
89            .draw_progress_bar(ProgressBar::new(Fraction::new(0.5), Color::<f64>::new(1.0, 0.0, 0.0), Pixels::new(4.0)))
90            .unwrap();
91        drop(target);
92
93        let mut surface = surface;
94        surface.flush();
95        let stride = surface.stride() as usize;
96        let data = surface.data().unwrap();
97        // Bar row y = 4 (10 - 4 - 2): left half filled red, right half
98        // dimmed to a third. ARgb32 memory order is B,G,R,A.
99        let filled = 4 * stride + 2 * 4;
100        assert_eq!(data[filled + 2], 255, "filled part should be the bar color");
101        let track = 4 * stride + 8 * 4;
102        assert_eq!(data[track + 2], 85, "unfilled part should be a third of the color");
103        assert_eq!(data[track + 3], 255, "track stays opaque");
104    }
105
106    #[test]
107    fn draw_gauge_strokes_track_and_arc() {
108        let size = Size::new(48, 48);
109        let surface = ImageSurface::create(Format::ARgb32, 48, 48).unwrap();
110        let mut target = CairoTarget::new(&Context::new(&surface).unwrap(), size);
111        target
112            .draw_gauge(Gauge::new(
113                Vec2::<Pixels, f32>::new(24.0, 24.0),
114                Pixels::new(16.0),
115                Pixels::new(4.0),
116                Fraction::new(0.25),
117                Color::<f64>::new(1.0, 0.0, 0.0),
118                Color::<f64>::new(0.3, 0.3, 0.3),
119            ))
120            .unwrap();
121        drop(target);
122
123        let mut surface = surface;
124        surface.flush();
125        let stride = surface.stride() as usize;
126        let data = surface.data().unwrap();
127        // 12 o'clock point on the ring (24, 8): quarter arc covers it → red.
128        let top = 8 * stride + 24 * 4;
129        assert_eq!(data[top + 2], 255, "arc should paint the top of the ring");
130        // 6 o'clock point on the ring (24, 40): outside the quarter →
131        // the gray track (0.3 → ~76 per channel, R == B).
132        let bottom = 40 * stride + 24 * 4;
133        assert!(data[bottom + 3] > 0, "track should paint the bottom of the ring");
134        assert_eq!(data[bottom + 2], data[bottom], "track is gray — equal red and blue channels");
135    }
