1//! The cairo backend: [`CairoTarget`] implements the shared capability
2//! vocabulary on a `cairo::Context`.
3//!
4//! Feature-gated behind `cairo` and headless-safe — `cairo-rs` alone
5//! pulls no display dependency, so this backend is testable without a
6//! compositor. Backend-specific extras stay reachable through
7//! [`CairoTarget::context`] instead of widening the shared traits.
8//!
9//! - `target`: the owning [`CairoTarget`] itself
10//! - `surface`: [`ViewSurface`] and readback — bridges buffer memory
11//!   and cairo `ImageSurface`s
12//! - `path`: generic cairo path building blocks (polyline, spline,
13//!   wave, band, rounded rect, tile fill)
14//! - `glyph_path`: crate-internal `ab_glyph` outline → cairo path
15//!   replay used by the icon and text capability impls
16//! - `<cap>_ops`: one file per capability trait — the cairo backend
17//!   has a single target type, so no per-target directory split
18//! - `error`: [`DrawingError`] for bridging failures outside cairo's
19//!   own status model

50#[cfg(test)]
51mod tests {
52    use super::*;
53    use crate::RenderContext;
54    use cairo::Context;
55    use cairo::Format;
56    use cairo::ImageSurface;
57    use smearor_commons_color::Color;
58    use smearor_commons_geometry::Fraction;
59    use smearor_commons_geometry::Pixels;
60
61    /// `RenderContext` owns the cairo target exactly like it owns
62    /// `BufferViewMut` — the generic facade works unchanged on both
63    /// backends.
64    #[test]
65    fn render_context_drives_cairo_target() {
66        let size = Size::new(32, 32);
67        let surface = ImageSurface::create(Format::ARgb32, 32, 32).unwrap();
68        let target = CairoTarget::new(&Context::new(&surface).unwrap(), size);
69        {
70            let mut ctx = RenderContext::new(target).background(Color::<f64>::new(1.0, 0.0, 0.0));
71            ctx.fill_background().unwrap();
72            ctx.draw_progress_bar(Fraction::new(0.5), Pixels::new(4.0)).unwrap();
73        }
74
75        let mut surface = surface;
76        surface.flush();
77        let data = surface.data().unwrap();
78        // ARgb32 memory order is B,G,R,A — a fully red fill.
79        assert_eq!(data[2], 255);
80        assert_eq!(data[3], 255);
81    }
