1//! Replays `ab_glyph` outlines as cairo vector paths — the cairo twin of
2//! the raster backend's coverage loop.
3//!
4//! `ab_glyph` keeps the raw curves private inside `OutlinedGlyph`, so
5//! the unscaled [`Outline`] comes from [`Font::outline`] and this module
6//! applies the same transform the rasterizer uses in
7//! `OutlinedGlyph::draw`: font-unit points map to screen pixels as
8//! `position + (x·horizontal, −y·vertical)` — the negated vertical
9//! factor converts font y-up coordinates into cairo's y-down space.
10
11use ab_glyph::Font;
12use ab_glyph::FontVec;
13use ab_glyph::Glyph;
14use ab_glyph::Outline;
15use ab_glyph::OutlineCurve;
16use ab_glyph::Point;
17use ab_glyph::PxScaleFactor;
18use ab_glyph::ScaleFont;
19use cairo::Context;
20use tracing::debug;
21
22/// Outlines `glyph` (id + scale + pen position) into the current path
23/// and fills it. Glyphs without outlines (spaces, missing codepoints)
24/// are skipped with a `debug!` note — matching the raster backend,
25/// which simply renders no coverage for them.
26pub(crate) fn fill_glyph(ctx: &Context, font: &FontVec, glyph: &Glyph) -> Result<(), cairo::Error> {
27    let Some(outline) = font.outline(glyph.id) else {
28        debug!("glyph {:?} has no outline, skipping", glyph.id);
29        return Ok(());
30    };
31    let scale_factor = font.as_scaled(glyph.scale).scale_factor();
32    glyph_outline_path(ctx, &outline, scale_factor, glyph.position);
33    ctx.fill()
34}
35
36/// Appends `outline`'s curves to the current cairo path, scaled by
37/// `scale_factor` and shifted to `position` (the glyph pen position, in
38/// surface pixels).
39///
40/// Quadratic Béziers are degree-elevated to cubics (`c1 = p0 + ⅔·(p1−p0)`,
41/// `c2 = p2 + ⅔·(p1−p2)`) because cairo has no quadratic segments. Each
42/// contour's first point is emitted via `move_to`; a curve whose start
43/// does not continue the previous endpoint starts a new contour
44/// (multi-contour glyphs like `i` or `ö`). Caller fills (or strokes)
45/// the path afterwards.
46pub(crate) fn glyph_outline_path(ctx: &Context, outline: &Outline, scale_factor: PxScaleFactor, position: Point) {
47    let to_screen = |point: Point| {
48        (
49            f64::from(position.x + point.x * scale_factor.horizontal),
50            f64::from(position.y - point.y * scale_factor.vertical),
51        )
52    };
53
54    let mut previous_end: Option<Point> = None;
55    for curve in &outline.curves {
56        let start = match *curve {
57            OutlineCurve::Line(p0, _) | OutlineCurve::Quad(p0, _, _) | OutlineCurve::Cubic(p0, _, _, _) => p0,
58        };
59        if previous_end != Some(start) {
60            if previous_end.is_some() {
61                ctx.close_path();
62            }
63            let (x, y) = to_screen(start);
64            ctx.move_to(x, y);
65        }
66        match *curve {
67            OutlineCurve::Line(_, p1) => {
68                let (x, y) = to_screen(p1);
69                ctx.line_to(x, y);
70                previous_end = Some(p1);
71            }
72            OutlineCurve::Quad(p0, p1, p2) => {
73                let c1 = Point {
74                    x: p0.x + (p1.x - p0.x) * (2.0 / 3.0),
75                    y: p0.y + (p1.y - p0.y) * (2.0 / 3.0),
76                };
77                let c2 = Point {
78                    x: p2.x + (p1.x - p2.x) * (2.0 / 3.0),
79                    y: p2.y + (p1.y - p2.y) * (2.0 / 3.0),
80                };
81                let (c1x, c1y) = to_screen(c1);
82                let (c2x, c2y) = to_screen(c2);
83                let (x, y) = to_screen(p2);
84                ctx.curve_to(c1x, c1y, c2x, c2y, x, y);
85                previous_end = Some(p2);
86            }
87            OutlineCurve::Cubic(_, p1, p2, p3) => {
88                let (c1x, c1y) = to_screen(p1);
89                let (c2x, c2y) = to_screen(p2);
90                let (x, y) = to_screen(p3);
91                ctx.curve_to(c1x, c1y, c2x, c2y, x, y);
92                previous_end = Some(p3);
93            }
94        }
95    }
96    ctx.close_path();
97}
