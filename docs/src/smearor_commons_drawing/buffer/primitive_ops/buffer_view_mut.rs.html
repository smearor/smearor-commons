28/// Blend `color` at `alpha` onto every pixel whose center lies in
29/// `bounds ∩ shape`, sampled at pixel centers like the other raster ops.
30fn blend_scanned<F>(view: &mut BufferViewMut<'_, Rgba>, bounds: Rect<Pixels, i32>, color: Color<f64>, alpha: impl Into<Alpha>, shape: F)
31where
32    F: Fn(&Vec2<Pixels, f64>) -> bool,
33{
34    let size = view.size();
35    let stride = view.stride().as_usize();
36    let pixels = view.data_mut();
37    let color: Color<u8> = color.into();
38    let alpha = alpha.into();
39    let (x0, y0) = (bounds.min.x.max(0), bounds.min.y.max(0));
40    let (x1, y1) = (bounds.max().x.min(size.x as i32), bounds.max().y.min(size.y as i32));
41
42    for y in y0..y1 {
43        for x in x0..x1 {
44            if shape(&Vec2::<Pixels>::new(x as f64 + 0.5, y as f64 + 0.5)) {
45                let idx = y as usize * stride + x as usize * Rgba::BYTES_PER_PIXEL;
46                blend_pixel(pixels, idx, color, alpha);
47            }
48        }
49    }
50}

52impl DrawPrimitiveOps for BufferViewMut<'_, Rgba> {
53    /// Fill the entire view with a solid background color.
54    ///
55    /// Packed views reinterpret the `&mut [u8]` data as `&mut [u32]` (RGBA
56    /// = 4 bytes = one `u32` in little-endian) and use `slice::fill()` for
57    /// SIMD-vectorized writes. Strided views fill each row's pixel bytes
58    /// and leave row padding untouched.
59    fn fill_background(&mut self, color: impl Into<Color<u8>>) -> Result<(), Self::Error> {
60        let color: Color<u8> = color.into();
61        let mut pixel = [0; Rgba::BYTES_PER_PIXEL];
62        Rgba::write_pixel(&mut pixel, color.r, color.g, color.b, color.a);
63        self.fill(&pixel);
64        Ok(())
65    }
66
67    fn fill_rect(&mut self, spec: FillRect) -> Result<(), Self::Error> {
68        let bounds = spec.rect.to_i32_ceil();
69        blend_scanned(self, bounds, spec.color, spec.alpha, |p| spec.rect.contains(p));
70        Ok(())
71    }
72
73    /// Stroke centered on the rect edge like cairo: the stroke region is
74    /// `rect ± line_width/2` minus the interior — implemented as the
75    /// expanded outline minus the shrunken one.
76    fn stroke_rect(&mut self, spec: StrokeRect) -> Result<(), Self::Error> {
77        if spec.line_width.as_raw() <= 0.0 {
78            return Ok(());
79        }
80        let half = Pixels::new(f64::from(spec.line_width.as_raw()) / 2.0);
81        let outer = spec.rect.inset(-half);
82        let inner = spec.rect.inset(half);
83        let bounds = outer.to_i32_ceil();
84        blend_scanned(self, bounds, spec.color, spec.alpha, |p| outer.contains(p) && !inner.contains(p));
85        Ok(())
86    }
87
88    fn fill_circle(&mut self, spec: FillCircle) -> Result<(), Self::Error> {
89        let bounds = spec.circle.bounding_box().to_i32_ceil();
90        blend_scanned(self, bounds, spec.color, spec.alpha, |p| spec.circle.contains(p));
91        Ok(())
92    }
93
94    /// Stroke centered on the radius like cairo: the stroke region is
95    /// every point within `line_width/2` of the circle.
96    fn stroke_circle(&mut self, spec: StrokeCircle) -> Result<(), Self::Error> {
97        let half = f64::from(spec.line_width.as_raw()) / 2.0;
98        if half <= 0.0 {
99            return Ok(());
100        }
101        let bounds = spec.circle.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
102        let center = spec.circle.center;
103        blend_scanned(self, bounds, spec.color, spec.alpha, |p| {
104            ((*p - center).magnitude().as_raw() - spec.circle.radius).abs() <= half
105        });
106        Ok(())
107    }
108
109    /// Stroke centered on each edge like cairo - the polygon's
110    /// `segments` closes the ring, so the seam strokes like any edge.
111    fn stroke_polygon(&mut self, spec: StrokePolygon) -> Result<(), Self::Error> {
112        let half = f64::from(spec.line_width.as_raw()) / 2.0;
113        if half <= 0.0 {
114            return Ok(());
115        }
116        for segment in spec.polygon.segments() {
117            let bounds = segment.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
118            blend_scanned(self, bounds, spec.color, spec.alpha, |p| segment.distance(p).as_raw() <= half);
119        }
120        Ok(())
121    }
122
123    fn fill_polygon(&mut self, spec: FillPolygon) -> Result<(), Self::Error> {
124        let Some(bounds) = spec.polygon.bounding_box() else {
125            return Ok(());
126        };
127        blend_scanned(self, bounds.to_i32_ceil(), spec.color, spec.alpha, |p| spec.polygon.contains(p));
128        Ok(())
129    }
130
131    /// Stroke centered on each segment like cairo: a segment's stroke
132    /// region is every point within `line_width/2` of it.
133    fn stroke_polyline(&mut self, spec: StrokePolyline) -> Result<(), Self::Error> {
134        let half = f64::from(spec.line_width.as_raw()) / 2.0;
135        if half <= 0.0 {
136            return Ok(());
137        }
138        for segment in spec.polyline.segments() {
139            let bounds = segment.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
140            blend_scanned(self, bounds, spec.color, spec.alpha, |p| segment.distance(p).as_raw() <= half);
141        }
142        match spec.cap {
143            Cap::Butt => {}
144            // Semicircles of radius `half` on the open endpoints -
145            // overlapping the band like the segment joints do.
146            Cap::Round => {
147                for endpoint in [spec.polyline.points().first(), spec.polyline.points().last()].into_iter().flatten() {
148                    let bounds = Circle::new(*endpoint, half).bounding_box().to_i32_ceil();
149                    blend_scanned(self, bounds, spec.color, spec.alpha, |p| p.distance(endpoint).as_raw() <= half);
150                }
151            }
152            // The `half`-long band beyond each open end, scanned
153            // through the extension segment's interior distance -
154            // a perpendicular band that stops square at `half`.
155            Cap::Square => {
156                for (segment, flip) in spec
157                    .polyline
158                    .segments()
159                    .next()
160                    .map(|s| (s, false))
161                    .into_iter()
162                    .chain(spec.polyline.segments().last().map(|s| (s, true)))
163                {
164                    let axis = Vec2::<Pixels>::new(1.0, 0.0).rotated(segment.direction());
165                    let cap = if flip {
166                        Segment::new(segment.to, segment.to + axis * half)
167                    } else {
168                        Segment::new(segment.from - axis * half, segment.from)
169                    };
170                    let bounds = cap.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
171                    blend_scanned(self, bounds, spec.color, spec.alpha, |p| cap.interior_distance(p).is_some_and(|d| d.as_raw() <= half));
172                }
173            }
174        }
175        Ok(())
176    }
177
178    /// The rim band `|dist − r| ≤ w/2` masked by `arc`'s interval -
179    /// caps are the same geometry as `stroke_polyline`'s: round ends
180    /// scan endpoint circles, square ends scan a tangent extension.
181    fn stroke_arc(&mut self, spec: StrokeArc) -> Result<(), Self::Error> {
182        let half = f64::from(spec.line_width.as_raw()) / 2.0;
183        if half <= 0.0 {
184            return Ok(());
185        }
186        let (center, radius) = (spec.circle.center, spec.circle.radius);
187        let bounds = spec.circle.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
188        blend_scanned(self, bounds, spec.color, spec.alpha, |p| {
189            (p.distance(&center).as_raw() - radius).abs() <= half && spec.arc.contains((*p - center).direction())
190        });
191        for (index, end) in [spec.arc.start, spec.arc.end()].iter().enumerate() {
192            let radial = Vec2::<Pixels>::new(1.0, 0.0).rotated(*end);
193            let endpoint = center + radial * radius;
194            match spec.cap {
195                Cap::Butt => {}
196                Cap::Round => {
197                    let cap_bounds = Circle::new(endpoint, half).bounding_box().to_i32_ceil();
198                    blend_scanned(self, cap_bounds, spec.color, spec.alpha, |p| p.distance(&endpoint).as_raw() <= half);
199                }
200                // The tangent continuation: forward past `arc.end`,
201                // backward past `arc.start`.
202                Cap::Square => {
203                    let tangent = radial.perpendicular() * if index == 0 { -half } else { half };
204                    let cap = Segment::new(endpoint, endpoint + tangent);
205                    let cap_bounds = cap.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
206                    blend_scanned(self, cap_bounds, spec.color, spec.alpha, |p| cap.interior_distance(p).is_some_and(|d| d.as_raw() <= half));
207                }
208            }
209        }
210        Ok(())
211    }
212
213    /// The sector boundary = the `w/2` bands on both rims inside
214    /// `arc`'s interval, plus the two radial edge segments.
215    fn stroke_sector(&mut self, spec: StrokeSector) -> Result<(), Self::Error> {
216        let half = f64::from(spec.line_width.as_raw()) / 2.0;
217        if half <= 0.0 {
218            return Ok(());
219        }
220        let (center, inner, outer) = (spec.annulus.center, spec.annulus.inner, spec.annulus.outer);
221        let bounds = spec.annulus.outer_circle().bounding_box().outset(Pixels::new(half)).to_i32_ceil();
222        blend_scanned(self, bounds, spec.color, spec.alpha, |p| {
223            spec.arc.contains((*p - center).direction())
224                && ((p.distance(&center).as_raw() - outer).abs() <= half || (p.distance(&center).as_raw() - inner).abs() <= half)
225        });
226        for direction in [spec.arc.start, spec.arc.end()] {
227            let radial = Vec2::<Pixels>::new(1.0, 0.0).rotated(direction);
228            let edge = Segment::new(center + radial * inner, center + radial * outer);
229            let edge_bounds = edge.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
230            blend_scanned(self, edge_bounds, spec.color, spec.alpha, |p| edge.distance(p).as_raw() <= half);
231        }
232        Ok(())
233    }
234
235    /// The sector is `annulus ∩ arc`, sampled like the other shapes.
236    fn fill_sector(&mut self, spec: FillSector) -> Result<(), Self::Error> {
237        let bounds = spec.annulus.outer_circle().bounding_box().to_i32_ceil();
238        let center = spec.annulus.center;
239        blend_scanned(self, bounds, spec.color, spec.alpha, |p| {
240            spec.annulus.contains(p) && spec.arc.contains((*p - center).direction())
241        });
242        Ok(())
243    }
244}

285        let mut data = vec![0u8; 32];

287        let color = Color::<u8>::new(255, 128, 64);

289

292        assert_eq!(&data[8..16], &[0u8; 8]);
293        assert_eq!(&data[16..24], &[bytes, bytes].concat()[..]);
294        assert_eq!(&data[24..32], &[0u8; 8]);
