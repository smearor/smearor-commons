21/// Blend `color` at `alpha` onto every pixel whose center lies in
22/// `bounds ∩ shape`, sampled at pixel centers like the other raster ops.
23fn blend_scanned<F>(view: &mut BufferViewMut<'_, Rgba>, bounds: Rect<Pixels, i32>, color: Color<f64>, alpha: impl Into<Alpha>, shape: F)
24where
25    F: Fn(&Vec2<Pixels, f64>) -> bool,
26{
27    let size = view.size();
28    let stride = view.stride().as_usize();
29    let pixels = view.data_mut();
30    let color: Color<u8> = color.into();
31    let alpha = alpha.into();
32    let (x0, y0) = (bounds.min.x.max(0), bounds.min.y.max(0));
33    let (x1, y1) = (bounds.max().x.min(size.x as i32), bounds.max().y.min(size.y as i32));
34
35    for y in y0..y1 {
36        for x in x0..x1 {
37            if shape(&Vec2::<Pixels>::new(x as f64 + 0.5, y as f64 + 0.5)) {
38                let idx = y as usize * stride + x as usize * Rgba::BYTES_PER_PIXEL;
39                blend_pixel(pixels, idx, color, alpha);
40            }
41        }
42    }
43}

45impl DrawPrimitiveOps for BufferViewMut<'_, Rgba> {
46    /// Fill the entire view with a solid background color.
47    ///
48    /// Packed views reinterpret the `&mut [u8]` data as `&mut [u32]` (RGBA
49    /// = 4 bytes = one `u32` in little-endian) and use `slice::fill()` for
50    /// SIMD-vectorized writes. Strided views fill each row's pixel bytes
51    /// and leave row padding untouched.
52    fn fill_background(&mut self, color: impl Into<Color<u8>>) -> Result<(), Self::Error> {
53        let color: Color<u8> = color.into();
54        let mut pixel = [0; Rgba::BYTES_PER_PIXEL];
55        Rgba::write_pixel(&mut pixel, color.r, color.g, color.b, color.a);
56        self.fill(&pixel);
57        Ok(())
58    }
59
60    fn fill_rect(&mut self, spec: FillRect) -> Result<(), Self::Error> {
61        let bounds = spec.rect.to_i32_ceil();
62        blend_scanned(self, bounds, spec.color, spec.alpha, |p| spec.rect.contains(p));
63        Ok(())
64    }
65
66    /// Stroke centered on the rect edge like cairo: the stroke region is
67    /// `rect ± line_width/2` minus the interior — implemented as the
68    /// expanded outline minus the shrunken one.
69    fn stroke_rect(&mut self, spec: StrokeRect) -> Result<(), Self::Error> {
70        if spec.line_width.as_raw() <= 0.0 {
71            return Ok(());
72        }
73        let half = Pixels::new(f64::from(spec.line_width.as_raw()) / 2.0);
74        let outer = spec.rect.inset(-half);
75        let inner = spec.rect.inset(half);
76        let bounds = outer.to_i32_ceil();
77        blend_scanned(self, bounds, spec.color, spec.alpha, |p| outer.contains(p) && !inner.contains(p));
78        Ok(())
79    }
80
81    fn fill_circle(&mut self, spec: FillCircle) -> Result<(), Self::Error> {
82        let bounds = spec.circle.bounding_box().to_i32_ceil();
83        blend_scanned(self, bounds, spec.color, spec.alpha, |p| spec.circle.contains(p));
84        Ok(())
85    }
86
87    fn fill_polygon(&mut self, spec: FillPolygon) -> Result<(), Self::Error> {
88        let Some(bounds) = spec.polygon.bounding_box() else {
89            return Ok(());
90        };
91        blend_scanned(self, bounds.to_i32_ceil(), spec.color, spec.alpha, |p| spec.polygon.contains(p));
92        Ok(())
93    }
94
95    /// Stroke centered on each segment like cairo: a segment's stroke
96    /// region is every point within `line_width/2` of it.
97    fn stroke_polyline(&mut self, spec: StrokePolyline) -> Result<(), Self::Error> {
98        let half = f64::from(spec.line_width.as_raw()) / 2.0;
99        if half <= 0.0 {
100            return Ok(());
101        }
102        for segment in spec.polyline.segments() {
103            let bounds = segment.bounding_box().outset(Pixels::new(half)).to_i32_ceil();
104            blend_scanned(self, bounds, spec.color, spec.alpha, |p| segment.distance(p).as_raw() <= half);
105        }
106        Ok(())
107    }
108
109    /// The sector is `annulus ∩ arc`, sampled like the other shapes.
110    fn fill_sector(&mut self, spec: FillSector) -> Result<(), Self::Error> {
111        let bounds = spec.annulus.outer_circle().bounding_box().to_i32_ceil();
112        let center = spec.annulus.center;
113        blend_scanned(self, bounds, spec.color, spec.alpha, |p| {
114            spec.annulus.contains(p) && spec.arc.contains((*p - center).direction())
115        });
116        Ok(())
117    }
118}

134    #[test]
135    fn fill_background_fills_all_pixels() {
136        let mut buf = PixelBuffer::new(Size::new(4, 3));
137        let color = Color::<f64>::new(10.0 / 255.0, 20.0 / 255.0, 30.0 / 255.0);
138        let expected: [u8; 4] = Color::<u8>::from(color).into();
139        buf.view_mut().fill_background(color).unwrap();
140
141        for px in buf.pixels().as_chunks::<4>().0.iter() {
142            assert_eq!(px, &expected);
143        }
144    }

155    #[test]
156    fn fill_background_strided_view_leaves_padding_untouched() {
157        // 2x2 pixels, stride 16 bytes (8 padding bytes per row):
158        // the fill must write only the 8 pixel bytes of each row.
159        let mut data = vec![0u8; 32];
160        let metadata = BufferMetadata::new(Pixels::new(2u32), Pixels::new(2u32), Stride::new(16));
161        let color = Color::<u8>::new(255, 128, 64);
162        BufferViewMut::<Rgba>::from_raw(&mut data, metadata).unwrap().fill_background(color).unwrap();
163
164        let bytes: [u8; 4] = color.into();
165        assert_eq!(&data[0..8], &[bytes, bytes].concat()[..]);
166        assert_eq!(&data[8..16], &[0u8; 8]);
167        assert_eq!(&data[16..24], &[bytes, bytes].concat()[..]);
168        assert_eq!(&data[24..32], &[0u8; 8]);
169    }
