8/// A right-angle rotation (`0`, `90`, `180`, `270` degrees) - the
9/// snapped orientations of displays, boards and rotated widgets.
10///
11/// The direction is **clockwise as seen by the viewer**: `Deg90`
12/// rotates the native frame's top edge to the viewer's right.
13/// Unlike [`RotationRad`] this is a discrete enum, not a continuous
14/// angle - use it where only the four snapped orientations exist.
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default, Serialize, Deserialize)]
16#[cfg_attr(feature = "async-graphql", derive(async_graphql::Enum))]
17pub enum QuarterTurn {
18    /// No rotation - native orientation.
19    #[default]
20    Deg0,
21    /// Quarter turn clockwise (90°) as seen by the viewer.
22    Deg90,
23    /// Half turn (180°) - upside down.
24    Deg180,
25    /// Three-quarter turn clockwise (270° = 90° counter-clockwise).
26    Deg270,
27}
28
29impl QuarterTurn {
30    /// All quarter turns in clockwise order.
31    pub const ALL: [Self; 4] = [Self::Deg0, Self::Deg90, Self::Deg180, Self::Deg270];
32
33    /// The quarter turn one step clockwise from `self`.
34    pub fn clockwise(self) -> Self {
35        match self {
36            Self::Deg0 => Self::Deg90,
37            Self::Deg90 => Self::Deg180,
38            Self::Deg180 => Self::Deg270,
39            Self::Deg270 => Self::Deg0,
40        }
41    }
42
43    /// The quarter turn one step counter-clockwise from `self`.
44    pub fn counter_clockwise(self) -> Self {
45        match self {
46            Self::Deg0 => Self::Deg270,
47            Self::Deg90 => Self::Deg0,
48            Self::Deg180 => Self::Deg90,
49            Self::Deg270 => Self::Deg180,
50        }
51    }
52
53    /// The turn undoing this one - `Deg0`/`Deg180` are their own
54    /// inverses, the two quarter turns swap.
55    pub fn inverse(self) -> Self {
56        match self {
57            Self::Deg0 | Self::Deg180 => self,
58            Self::Deg90 => Self::Deg270,
59            Self::Deg270 => Self::Deg90,
60        }
61    }
62
63    /// Whether the turn swaps width and height (`Deg90`/`Deg270`).
64    pub fn swaps_axes(self) -> bool {
65        matches!(self, Self::Deg90 | Self::Deg270)
66    }
67
68    /// The rotation angle in degrees (`0.0`, `90.0`, `180.0`, `270.0`).
69    pub fn degrees(self) -> f64 {
70        match self {
71            Self::Deg0 => 0.0,
72            Self::Deg90 => 90.0,
73            Self::Deg180 => 180.0,
74            Self::Deg270 => 270.0,
75        }
76    }
77
78    /// The equivalent continuous [`RotationRad`].
79    pub fn to_rotation_rad(self) -> RotationRad {
80        RotationRad::from_degrees(self.degrees())
81    }
82
83    /// The snapped quarter turn when `angle` is exactly `0`, `90`,
84    /// `180` or `270` degrees - `None` for every other direction.
85    /// The [`RotationRad`]-typed counterpart of
86    /// [`Self::from_degrees`].
87    pub fn from_rotation_rad(angle: RotationRad) -> Option<Self> {
88        Self::from_degrees(angle.to_degrees())
89    }
90
91    /// The snapped quarter turn for a degree value, or `None` when
92    /// `degrees` is not one of `0`, `90`, `180`, `270`.
93    pub fn from_degrees(degrees: f64) -> Option<Self> {
94        match degrees {
95            0.0 => Some(Self::Deg0),
96            90.0 => Some(Self::Deg90),
97            180.0 => Some(Self::Deg180),
98            270.0 => Some(Self::Deg270),
99            _ => None,
100        }
101    }
102
103    /// Rotates a position inside `bounds` into the turned orientation.
104    ///
105    /// The bounds swap when the turn [`Self::swaps_axes`] - the caller
106    /// uses the returned position against `(bounds.y, bounds.x)` in
107    /// that case.
108    pub fn apply<U>(self, pos: Vec2<U>, bounds: Vec2<U>) -> Vec2<U> {
109        match self {
110            Self::Deg0 => pos,
111            Self::Deg90 => Vec2::new(pos.y, bounds.x - pos.x),
112            Self::Deg180 => Vec2::new(bounds.x - pos.x, bounds.y - pos.y),
113            Self::Deg270 => Vec2::new(bounds.y - pos.y, pos.x),
114        }
115    }
116}

118impl fmt::Display for QuarterTurn {
119    /// The degrees spelling accepted by [`FromStr`].
120    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
121        write!(f, "{}", self.degrees() as u16)
122    }
123}
124
125impl FromStr for QuarterTurn {
126    type Err = String;
127
128    /// Accepts the quarter-turn values `0`, `90`, `180` or `270`.
129    fn from_str(s: &str) -> Result<Self, Self::Err> {
130        match s {
131            "0" => Ok(Self::Deg0),
132            "90" => Ok(Self::Deg90),
133            "180" => Ok(Self::Deg180),
134            "270" => Ok(Self::Deg270),
135            _ => Err(format!("invalid quarter turn '{s}' - use 0, 90, 180 or 270")),
136        }
137    }
138}

140#[cfg(test)]
141mod tests {
142    use super::QuarterTurn;
143    use crate::Millimeters;
144    use crate::Vec2;
145
146    /// All four turns rotate a position inside its bounds correctly.
147    #[test]
148    fn quarter_turns_rotate_inside_bounds() {
149        let bounds = Vec2::<Millimeters>::new(400.0, 300.0);
150        let pos = Vec2::<Millimeters>::new(100.0, 50.0);
151
152        let deg90 = QuarterTurn::Deg90.apply(pos, bounds);
153        assert!((deg90.x - 50.0).abs() < 1e-9, "x becomes y");
154        assert!((deg90.y - 300.0).abs() < 1e-9, "y becomes w - x");
155
156        let deg180 = QuarterTurn::Deg180.apply(pos, bounds);
157        assert!((deg180.x - 300.0).abs() < 1e-9);
158        assert!((deg180.y - 250.0).abs() < 1e-9);
159
160        let deg270 = QuarterTurn::Deg270.apply(pos, bounds);
161        assert!((deg270.x - 250.0).abs() < 1e-9, "x becomes h - y");
162        assert!((deg270.y - 100.0).abs() < 1e-9, "y becomes x");
163    }
164
165    /// `swaps_axes` is true exactly for the two odd turns.
166    #[test]
167    fn quarter_turns_swap_axes() {
168        assert!(!QuarterTurn::Deg0.swaps_axes());
169        assert!(QuarterTurn::Deg90.swaps_axes());
170        assert!(!QuarterTurn::Deg180.swaps_axes());
171        assert!(QuarterTurn::Deg270.swaps_axes());
172    }
173
174    /// `inverse` applied twice returns the original turn, and a turn
175    /// plus its inverse always completes a full circle (or none).
176    #[test]
177    fn inverse_is_involution() {
178        for turn in QuarterTurn::ALL {
179            assert_eq!(turn.inverse().inverse(), turn);
180            let sum = turn.degrees() + turn.inverse().degrees();
181            assert!(sum == 0.0 || sum == 360.0);
182        }
183    }
184
185    /// `clockwise`/`counter_clockwise` are mutual inverses over the cycle.
186    #[test]
187    fn clockwise_counter_clockwise_cycle() {
188        for turn in QuarterTurn::ALL {
189            assert_eq!(turn.clockwise().counter_clockwise(), turn);
190            assert_eq!(turn.counter_clockwise().clockwise(), turn);
191        }
192    }
193
194    /// `from_degrees` snaps only the four exact values.
195    #[test]
196    fn from_degrees_snaps_only_exact() {
197        assert_eq!(QuarterTurn::from_degrees(0.0), Some(QuarterTurn::Deg0));
198        assert_eq!(QuarterTurn::from_degrees(90.0), Some(QuarterTurn::Deg90));
199        assert_eq!(QuarterTurn::from_degrees(45.0), None);
200        assert_eq!(QuarterTurn::from_degrees(-90.0), None);
201    }
202
203    /// `to_rotation_rad` matches the degree value.
204    #[test]
205    fn to_rotation_rad_matches_degrees() {
206        assert!((QuarterTurn::Deg90.to_rotation_rad().to_degrees() - 90.0).abs() < 1e-9);
207    }
