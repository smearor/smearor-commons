5/// An aspect ratio - `width / height` of a [`crate::Size`]-like
6/// extent (`1.0` = square).
7///
8/// Wraps the plain `f64` so ratios cannot be confused with other
9/// dimensionless scalars such as [`crate::ScaleFactor`]. The value is
10/// strictly positive and finite by construction: [`AspectRatio::new`]
11/// rejects non-positive and non-finite inputs, so consumers like
12/// [`crate::Vec2::unrotated_fit`] need no defensive `aspect <= 0`
13/// guard - "invalid" is unrepresentable.
14#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Serialize)]
15#[serde(transparent)]
16pub struct AspectRatio(f64);
17
18impl AspectRatio {
19    /// Square aspect - width equals height (`1:1`).
20    pub const ONE: Self = Self(1.0);
21
22    /// Classic TV and monitor aspect `4:3` (~1.333).
23    pub const FOUR_THREE: Self = Self(4.0 / 3.0);
24
25    /// Photo aspect `3:2` - 35 mm film and DSLRs.
26    pub const THREE_TWO: Self = Self(3.0 / 2.0);
27
28    /// Monitor aspect `5:4` (1280x1024).
29    pub const FIVE_FOUR: Self = Self(5.0 / 4.0);
30
31    /// Widescreen aspect `16:9` - HDTV and common displays.
32    pub const SIXTEEN_NINE: Self = Self(16.0 / 9.0);
33
34    /// Widescreen aspect `16:10` - workstation displays.
35    pub const SIXTEEN_TEN: Self = Self(16.0 / 10.0);
36
37    /// Ultrawide aspect `21:9` (~2.333; panel-natively often `64:27`).
38    pub const TWENTY_ONE_NINE: Self = Self(21.0 / 9.0);
39
40    /// ISO 216 paper aspect `1:sqrt(2)` - DIN A series (~1.414). The
41    /// half/hold property: halving a sheet preserves the aspect.
42    pub const ISO_PAPER: Self = Self(std::f64::consts::SQRT_2);
43
44    /// Golden ratio `phi` = (1 + sqrt(5)) / 2 (~1.618).
45    pub const GOLDEN: Self = Self(std::f64::consts::GOLDEN_RATIO);
46
47    /// Creates an aspect ratio, rejecting non-positive and non-finite
48    /// ratios: a degenerate extent has no meaningful aspect.
49    pub fn new(ratio: f64) -> Option<Self> {
50        (ratio.is_finite() && ratio > 0.0).then_some(Self(ratio))
51    }
52
53    /// The ratio `width / height` (`1.0` = square).
54    pub const fn as_f64(self) -> f64 {
55        self.0
56    }
57
58    /// The ratio as `f32` for `f32`-based framework APIs (e.g. GTK4).
59    pub fn as_f32(self) -> f32 {
60        self.0 as f32
61    }
62
63    /// Whether the ratio is `1.0` - width equals height.
64    pub fn is_square(self) -> bool {
65        self.0 == 1.0
66    }
67
68    /// The reciprocal `height / width` - the aspect of the same extent
69    /// rotated a quarter turn (`a.inverse().inverse() == a`). The
70    /// named constants are landscape; `inverse()` yields their
71    /// portrait form.
72    pub const fn inverse(self) -> Self {
73        Self(1.0 / self.0)
74    }
75}
76
77impl Default for AspectRatio {
78    /// Square aspect.
79    fn default() -> Self {
80        Self::ONE
81    }
82}
83
84/// Deserialization validates: a stored ratio must satisfy the same
85/// `> 0` and finite invariant as [`AspectRatio::new`] - serde cannot
86/// smuggle in an invalid value.
87impl<'de> Deserialize<'de> for AspectRatio {
88    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
89    where
90        D: serde::Deserializer<'de>,
91    {
92        use serde::de::Error;
93        Self::new(f64::deserialize(deserializer)?).ok_or_else(|| D::Error::custom("aspect ratio must be positive and finite"))
94    }
95}

97impl From<AspectRatio> for f64 {
98    fn from(ratio: AspectRatio) -> Self {
99        ratio.0
100    }
101}

103/// No `From<f64>`: raw ratios must go through the validating
104/// [`AspectRatio::new`] instead.
105impl TryFrom<f64> for AspectRatio {
106    type Error = ();
107
108    fn try_from(ratio: f64) -> Result<Self, Self::Error> {
109        Self::new(ratio).ok_or(())
110    }
111}

113impl Display for AspectRatio {
114    /// Formats as `width:height` (`1.5` prints as `1.5:1`).
115    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
116        write!(f, "{}:1", self.0)
117    }

128#[cfg(test)]
129mod tests {
130    use super::AspectRatio;
131
132    /// Construction enforces the `> 0` and finite invariant.
133    #[test]
134    fn new_rejects_invalid_ratios() {
135        assert_eq!(AspectRatio::new(0.0), None);
136        assert_eq!(AspectRatio::new(-2.0), None);
137        assert_eq!(AspectRatio::new(f64::NAN), None);
138        assert_eq!(AspectRatio::new(f64::INFINITY), None);
139        assert_eq!(AspectRatio::new(2.5).map(AspectRatio::as_f64), Some(2.5));
140    }
141
142    /// `a.inverse().inverse()` lands back at `a`; `a * a.inverse()`
143    /// composes to the square aspect.
144    #[test]
145    fn inverse_flips_orientation() {
146        let wide = AspectRatio::new(5.25);
147        let narrow = wide.map(AspectRatio::inverse);
148        assert_eq!(narrow.map(AspectRatio::as_f64), Some(1.0 / 5.25));
149        assert_eq!(narrow.map(AspectRatio::inverse), wide);
150    }
151
152    /// The default aspect is square.
153    #[test]
154    fn default_is_square() {
155        assert_eq!(AspectRatio::default(), AspectRatio::ONE);
156        assert!(AspectRatio::ONE.is_square());
157    }
158
159    /// The named constants hold their nominal ratios; `inverse()`
160    /// yields the portrait form.
161    #[test]
162    fn named_ratios_match_their_fractions() {
163        assert_eq!(AspectRatio::FOUR_THREE.as_f64(), 4.0 / 3.0);
164        assert_eq!(AspectRatio::SIXTEEN_NINE.as_f64(), 16.0 / 9.0);
165        assert_eq!(AspectRatio::SIXTEEN_TEN.as_f64(), 1.6);
166        assert_eq!(AspectRatio::ISO_PAPER.as_f64(), std::f64::consts::SQRT_2);
167        assert_eq!(AspectRatio::SIXTEEN_NINE.inverse().as_f64(), 9.0 / 16.0);
168    }
169
170    /// Halving an ISO sheet along its long edge preserves the aspect:
171    /// `sqrt(2) = 2 / sqrt(2)`.
172    #[test]
173    fn iso_paper_halving_preserves_aspect() {
174        let a4 = AspectRatio::ISO_PAPER;
175        let a5 = a4.inverse().as_f64() * 2.0;
176        assert!((a5 - a4.as_f64()).abs() < 1e-9);
177    }
