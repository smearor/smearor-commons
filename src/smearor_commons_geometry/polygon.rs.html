11/// The miter cap for [`Polygon::offset`] corners - acute corners
12/// may extend at most this many edge distances.
13const MITER_CAP: f64 = 4.0;
14
15/// A closed ring of points in unit `U`: footprints, block
16/// outlines, park blobs - the last point connects back to the
17/// first. Derefs to the `[Vec2]` slice so every `&[Vec2]`
18/// geometry helper keeps working.
19#[derive(Debug, Clone, PartialEq)]
20pub struct Polygon<U>(pub Vec<Vec2<U>>);
21
22impl<U> Polygon<U> {
23    /// A polygon from its vertex ring.
24    pub fn new(points: Vec<Vec2<U>>) -> Self {
25        Self(points)
26    }
27
28    /// Removes and returns the vertex at `index`.
29    pub fn remove(&mut self, index: usize) -> Vec2<U> {
30        self.0.remove(index)
31    }
32}
33
34impl<U: Copy> Polygon<U> {
35    /// The ring's edge segments - the last vertex connects back
36    /// to the first.
37    pub fn segments(&self) -> impl Iterator<Item = Segment<U>> + '_ {
38        let points = &self.0;
39        (0..points.len()).map(|i| Segment::new(points[i], points[(i + 1) % points.len()]))
40    }
41
42    /// The ring's vertex centroid (uniform vertex average).
43    pub fn centroid(&self) -> Vec2<U> {
44        let count = self.0.len().max(1) as f64;
45        Vec2::new(self.0.iter().map(|p| p.x).sum::<f64>() / count, self.0.iter().map(|p| p.y).sum::<f64>() / count)
46    }
47
48    /// The axis-aligned bounding box of the vertex ring, or `None`
49    /// for an empty ring.
50    pub fn bounding_box(&self) -> Option<Rect<U>> {
51        let mut iter = self.0.iter();
52        let first = iter.next()?;
53        let (mut min, mut max) = (*first, *first);
54        for point in iter {
55            min = Vec2::new(min.x.min(point.x), min.y.min(point.y));
56            max = Vec2::new(max.x.max(point.x), max.y.max(point.y));
57        }
58        Some(min.rect(max - min))
59    }
60
61    /// Whether `point` lies strictly inside the ring (even-odd
62    /// ray cast).
63    pub fn contains(&self, point: &Vec2<U>) -> bool {
64        let mut inside = false;
65        for edge in self.segments() {
66            let (a, b) = (edge.from, edge.to);
67            let crosses_ray = (a.y > point.y) != (b.y > point.y);
68            if crosses_ray && point.x < a.x + (b.x - a.x) * (point.y - a.y) / (b.y - a.y) {
69                inside = !inside;
70            }
71        }
72        inside
73    }
74
75    /// Whether the ring is simple - no two non-adjacent edges
76    /// cross.
77    pub fn is_simple(&self) -> bool {
78        let edges: Vec<Segment<U>> = self.segments().collect();
79        let n = edges.len();
80        for i in 0..n {
81            for j in (i + 2)..n {
82                if i == 0 && j == n - 1 {
83                    continue;
84                }
85                if edges[i].crosses(&edges[j]) {
86                    return false;
87                }
88            }
89        }
90        true
91    }
92
93    /// Whether `segment` enters the ring: a strict edge crossing
94    /// or an endpoint inside. Tangent grazes do not count.
95    pub fn crossed(&self, segment: &Segment<U>) -> bool {
96        self.segments().any(|edge| segment.crosses(&edge)) || self.contains(&segment.from) || self.contains(&segment.to)
97    }
98
99    /// Whether `other` overlaps the ring: an edge crossing or a
100    /// vertex of either inside the other.
101    pub fn overlaps(&self, other: &[Vec2<U>]) -> bool {
102        let other = Polygon::new(other.to_vec());
103        self.segments().any(|edge| other.segments().any(|other_edge| edge.crosses(&other_edge)))
104            || self.iter().any(|p| other.contains(p))
105            || other.iter().any(|p| self.contains(p))
106    }
107}

109impl<U> Polygon<U>
110where
111    U: Copy + From<f64> + PartialOrd,
112    f64: From<U>,
113{
114    /// Pulls each edge of the ring (positive signed area,
115    /// interior left of each edge) `distances[i]` inward.
116    /// Corner vertices are the intersections of the two offset
117    /// edge lines, miter-capped. Not validated - the caller
118    /// checks the result (area, containment).
119    pub fn offset(&self, distances: &[f64]) -> Polygon<U> {
120        let n = self.0.len();
121        let mut lines = Vec::with_capacity(n);
122        for (i, edge) in self.segments().enumerate() {
123            let direction = edge.to - edge.from;
124            let unit = direction.normalized();
125            // Left normal points into the polygon (positive area).
126            lines.push(Line::new(edge.from + unit.perpendicular() * distances[i], unit));
127        }
128        Polygon::new(
129            (0..n)
130                .map(|i| {
131                    let previous = (i + n - 1) % n;
132                    let (l1, l2) = (lines[previous], lines[i]);
133                    let corner = self.0[i];
134                    let distance = distances[previous].max(distances[i]);
135                    // Near-parallel fallback: average the two
136                    // normals - the bounded bisector inset.
137                    let bisector = || {
138                        let normal = l1.direction.perpendicular() * distances[previous] + l2.direction.perpendicular() * distances[i];
139                        corner + normal * (distance / f64::from(normal.magnitude()).max(1e-9))
140                    };
141                    let Some(inset) = l1.intersect(&l2) else {
142                        return bisector();
143                    };
144                    let miter = inset - corner;
145                    if f64::from(miter.magnitude()) <= MITER_CAP * distance {
146                        return inset;
147                    }
148                    // Overshot miter: acute corners legitimately run
149                    // long in the bisector direction - cap them.
150                    // Degenerate corners (near-180° vertices, e.g.
151                    // a waypoint on a straight street) fly
152                    // off sideways - there the bounded bisector
153                    // inset is right.
154                    let bis = bisector();
155                    let miter_bisector = bis - corner;
156                    let aligned = miter.dot::<f64>(&miter_bisector) / (f64::from(miter.magnitude()) * f64::from(miter_bisector.magnitude())).max(1e-9);
157                    if aligned > 0.5 {
158                        corner + miter * (MITER_CAP * distance / f64::from(miter.magnitude()))
159                    } else {
160                        bis
161                    }
162                })
163                .collect(),
164        )
165    }
166
167    /// The distance from `point` to the nearest ring edge - how
168    /// far inside the polygon it sits; `deepest_point` maximizes
169    /// it.
170    pub fn clearance(&self, point: &Vec2<U>) -> U {
171        self.segments()
172            .map(|edge| edge.distance(point))
173            .fold(U::from(f64::INFINITY), |a, b| if a < b { a } else { b })
174    }
175}

177impl Polygon<Millimeters> {
178    /// The shoelace signed area in mm² (positive =
179    /// counter-clockwise in the coordinate frame).
180    pub fn signed_area(&self) -> SquareMillimeters {
181        self.segments()
182            .map(|edge| edge.from.cross::<SquareMillimeters>(&edge.to))
183            .sum::<SquareMillimeters>()
184            * 0.5
185    }
186
187    /// Whether `point` lies inside the ring or on its boundary -
188    /// [`Self::contains`] plus a tolerance for facade points
189    /// riding on an outline edge.
190    pub fn contains_or_touches(&self, point: &Vec2<Millimeters>) -> bool {
191        const EDGE_EPS: Millimeters = Millimeters::new(0.06);
192        if self.contains(point) {
193            return true;
194        }
195        self.segments().any(|edge| {
196            edge.interior_distance(point).is_some_and(|d| d < EDGE_EPS) || point.distance(&edge.from) < EDGE_EPS || point.distance(&edge.to) < EDGE_EPS
197        })
198    }
199
200    /// The chord of the ring through `point` along `axis`: the
201    /// segment between the nearest outline crossings bracketing
202    /// `point` on the line `point + axis * t` - `None` when
203    /// `point` sits outside.
204    pub fn chord(&self, point: &Vec2<Millimeters>, axis: &Vec2<Millimeters>) -> Option<Segment<Millimeters>> {
205        let epsilon = Millimeters::new(0.01);
206        let (t_lo, t_hi) =
207            self.segments()
208                .filter_map(|edge| edge.param_on_line(point, axis))
209                .fold((Millimeters::new(f64::MIN), Millimeters::INFINITY), |(lo, hi), t| {
210                    (
211                        if t < -epsilon { Millimeters::max(lo, t) } else { lo },
212                        if t > epsilon { Millimeters::min(hi, t) } else { hi },
213                    )
214                });
215        if !t_lo.as_f64().is_finite() || !t_hi.as_f64().is_finite() {
216            return None;
217        }
218        Some(Segment::new(*point + *axis * t_lo.as_f64(), *point + *axis * t_hi.as_f64()))
219    }
220
221    /// The interior point with the largest clearance to the
222    /// ring's edges (sampled on a 7x7 grid) - the spot a label
223    /// sits best.
224    pub fn deepest_point(&self) -> Option<Vec2<Millimeters>> {
225        self.deepest(7, &[])
226    }
227
228    /// Like [`Self::deepest_point`], but skips candidates inside
229    /// `avoid` polygons or within `AVOID_GAP` mm of their edges -
230    /// so a district caption never lands on top of a park name.
231    pub fn deepest_point_avoiding(&self, avoid: &[Polygon<Millimeters>]) -> Option<Vec2<Millimeters>> {
232        self.deepest(9, avoid)
233    }
234
235    /// The grid-sampled deepest interior point, optionally
236    /// blocking zones the caption must dodge.
237    fn deepest(&self, steps: usize, avoid: &[Polygon<Millimeters>]) -> Option<Vec2<Millimeters>> {
238        const AVOID_GAP: Millimeters = Millimeters::new(3.0);
239        let blocked = |point: &Vec2<Millimeters>| {
240            avoid
241                .iter()
242                .any(|polygon| polygon.contains(point) || polygon.segments().any(|edge| edge.interior_distance(point).is_some_and(|d| d < AVOID_GAP)))
243        };
244        let (min_x, max_x) = self.0.iter().fold((f64::MAX, f64::MIN), |(lo, hi), p| (lo.min(p.x), hi.max(p.x)));
245        let (min_y, max_y) = self.0.iter().fold((f64::MAX, f64::MIN), |(lo, hi), p| (lo.min(p.y), hi.max(p.y)));
246
247        let mut best: Option<(Millimeters, Vec2<Millimeters>)> = None;
248        for row in 1..steps {
249            for col in 1..steps {
250                let point = Vec2::new(min_x + (max_x - min_x) * row as f64 / steps as f64, min_y + (max_y - min_y) * col as f64 / steps as f64);
251                if !self.contains(&point) || blocked(&point) {
252                    continue;
253                }
254                let score = self.clearance(&point);
255                if best.is_none_or(|(best_score, _)| score > best_score) {
256                    best = Some((score, point));
257                }
258            }
259        }
260        best.map(|(_, point)| point)
261    }
262}

264impl<'a, U> IntoIterator for &'a Polygon<U> {
265    type Item = &'a Vec2<U>;
266    type IntoIter = std::slice::Iter<'a, Vec2<U>>;
267
268    fn into_iter(self) -> Self::IntoIter {
269        self.0.iter()
270    }
271}

273impl<U> Deref for Polygon<U> {

305        Polygon::new(vec![Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0), Vec2::new(10.0, 10.0), Vec2::new(0.0, 10.0)])
