13/// An open line of points in unit `U`: river courses, streets,
14/// decorative chords, edge routes - the last point does not connect
15/// back to the first. Derefs to the `[Vec2]` slice so every
16/// `&[Vec2]` geometry helper keeps working.
17///
18/// Serializes flat: `[x0, y0, x1, y1, ...]`.
19#[derive(Debug, Clone, Default, PartialEq)]
20pub struct Polyline<U>(pub Vec<Vec2<U>>);
21
22impl<U> Polyline<U> {
23    /// A polyline from its points.
24    pub fn new(points: Vec<Vec2<U>>) -> Self {
25        Self(points)
26    }
27
28    /// The polyline's points.
29    pub fn points(&self) -> &[Vec2<U>] {
30        &self.0
31    }
32}
33
34impl<U: Copy> Polyline<U> {
35    /// The polyline's point-to-point segments.
36    pub fn segments(&self) -> impl Iterator<Item = Segment<U>> + '_ {
37        self.0.windows(2).map(|pair| Segment::new(pair[0], pair[1]))
38    }
39
40    /// Whether `other` strictly crosses the polyline.
41    pub fn crosses(&self, other: &Polyline<U>) -> bool {
42        self.segments().any(|a| other.segments().any(|b| a.crosses(&b)))
43    }
44
45    /// Corner-cutting subdivision (Chaikin): each round replaces
46    /// every segment by points at 1/4 and 3/4, rounding corners
47    /// and roughly doubling the point count - denser spline
48    /// checkpoints and no miter spikes at sharp bends.
49    /// `pinned` lists point indices (of the original polyline)
50    /// that stay on the course exactly.
51    pub fn smoothed(&self, rounds: usize, pinned: &[usize]) -> Self {
52        let mut flags: Vec<bool> = self.0.iter().enumerate().map(|(index, _)| pinned.contains(&index)).collect();
53        let mut current = self.0.clone();
54        for _ in 0..rounds {
55            if current.len() < 3 {
56                break;
57            }
58            let mut next = Vec::with_capacity(current.len() * 2);
59            let mut next_flags = Vec::with_capacity(flags.len() * 2);
60            next.push(current[0]);
61            next_flags.push(flags[0]);
62            for (index, window) in current.windows(2).enumerate() {
63                let (from, to) = (window[0], window[1]);
64                next.push(Vec2::new(from.x * 0.75 + to.x * 0.25, from.y * 0.75 + to.y * 0.25));
65                next.push(Vec2::new(from.x * 0.25 + to.x * 0.75, from.y * 0.25 + to.y * 0.75));
66                next_flags.extend_from_slice(&[false, false]);
67                if flags.get(index + 1).copied().unwrap_or(false) {
68                    next.push(to);
69                    next_flags.push(true);
70                }
71            }
72            if let (Some(last), Some(flag)) = (current.last(), flags.last()) {
73                next.push(*last);
74                next_flags.push(*flag);
75            }
76            (current, flags) = (next, next_flags);
77        }
78        Self::new(current)
79    }
80}

82impl<U> Polyline<U>
83where
84    U: Copy + From<f64> + PartialOrd + Add<Output = U> + Mul<f64, Output = U>,
85    f64: From<U>,
86{
87    /// The axis parameter `t` where the line `center + axis * t`
88    /// meets this polyline - `None` when they do not intersect
89    /// within `reach`.
90    pub fn axis_param(&self, center: Vec2<U>, axis: Vec2<U>, reach: U) -> Option<U> {
91        let half = f64::from(reach);
92        let line = Segment::new(center - axis * half, center + axis * half);
93        self.segments()
94            .find_map(|segment| line.intersection(&segment).map(|point| U::from((point - center).dot::<f64>(&axis))))
95    }
96
97    /// The polyline smoothed `rounds` times except the points
98    /// within `within` of `anchors`, which stay pinned.
99    pub fn smoothed_anchored(&self, anchors: &[Vec2<U>], within: U, rounds: usize) -> Self {
100        self.smoothed(rounds, &self.indices_within(anchors, within))
101    }
102
103    /// The indices of points lying within `within` of any
104    /// `positions` - e.g. the waypoints pinned through a
105    /// `smoothed` call.
106    pub fn indices_within(&self, positions: &[Vec2<U>], within: U) -> Vec<usize> {
107        self.0
108            .iter()
109            .enumerate()
110            .filter(|(_, point)| positions.iter().any(|pos| pos.distance(point) < within))
111            .map(|(index, _)| index)
112            .collect()
113    }
114
115    /// The arc length along the polyline up to the line parameter
116    /// `at`: `at`'s integer part is a segment index, its fraction
117    /// the position along that segment. Zero on fewer than two
118    /// points.
119    pub fn arc_length(&self, at: f64) -> U {
120        if self.0.len() < 2 {
121            return U::from(0.0);
122        }
123        let index = (at.max(0.0).floor() as usize).min(self.0.len() - 1);
124        let full = self.0.windows(2).take(index).fold(U::from(0.0), |arc, pair| arc + pair[0].distance(&pair[1]));
125        let (a, b) = (self.0[index], self.0[(index + 1).min(self.0.len() - 1)]);
126        full + a.distance(&b) * at.max(0.0).fract()
127    }
128
129    /// The polyline shifted `distance` to the left of its
130    /// direction: bend vertices average the adjacent segment
131    /// normals, so the offset stays near-constant without miter
132    /// spikes.
133    pub fn offset(&self, distance: U) -> Self {
134        let normal = |from: Vec2<U>, to: Vec2<U>| {
135            let perpendicular = (to - from).normalized().perpendicular();
136            (perpendicular.x, perpendicular.y)
137        };
138        let distance = f64::from(distance);
139        let points: Vec<Vec2<U>> = self
140            .0
141            .iter()
142            .enumerate()
143            .map(|(index, point)| {
144                let (mut normal_x, mut normal_y) = (0.0, 0.0);
145                if index > 0 {
146                    let (x, y) = normal(self.0[index - 1], *point);
147                    normal_x += x;
148                    normal_y += y;
149                }
150                if index + 1 < self.0.len() {
151                    let (x, y) = normal(*point, self.0[index + 1]);
152                    normal_x += x;
153                    normal_y += y;
154                }
155                let norm = normal_x.hypot(normal_y).max(1e-6);
156                Vec2::new(point.x + normal_x / norm * distance, point.y + normal_y / norm * distance)
157            })
158            .collect();
159        Self::new(points)
160    }
161
162    /// The segment index and projection nearest to `point` -
163    /// `None` for an empty polyline.
164    pub fn nearest_segment(&self, point: &Vec2<U>) -> Option<(usize, Projection<U>)> {
165        self.segments()
166            .enumerate()
167            .filter_map(|(index, segment)| segment.nearest(point).map(|projection| (index, projection)))
168            .min_by(|(_, a), (_, b)| f64::from(point.distance(&a.point)).total_cmp(&f64::from(point.distance(&b.point))))
169    }
170}

207impl<U> Serialize for Polyline<U> {
208    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
209        let mut flat = Vec::with_capacity(self.0.len() * 2);
210        for p in &self.0 {
211            flat.push(p.x);
212            flat.push(p.y);
213        }
214        flat.serialize(serializer)
215    }
216}
217
218impl<'de, U> Deserialize<'de> for Polyline<U> {
219    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
220        let flat = <Vec<f64>>::deserialize(deserializer)?;
221        if flat.len() % 2 != 0 {
222            return Err(D::Error::custom("polyline must contain an even number of coordinates"));
223        }
224        let (pairs, _rest) = flat.as_chunks::<2>();
225        let points = pairs.iter().map(|xy| Vec2::new(xy[0], xy[1])).collect();
226        Ok(Self(points))
227    }

236    fn smoothing_keeps_endpoints_and_pinned_vertices() {
237        let polyline = Polyline::<Millimeters>::new(vec![Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0), Vec2::new(10.0, 10.0)]);
238        let smoothed = polyline.smoothed(2, &[1]);
239        assert_eq!(smoothed.first().copied(), Some(Vec2::new(0.0, 0.0)));
240        assert_eq!(smoothed.last().copied(), Some(Vec2::new(10.0, 10.0)));
241        assert!(smoothed.contains(&Vec2::new(10.0, 0.0)), "pinned vertex survives");
242    }
