8/// A sphere - the 3D analog of [`crate::Circle`] - tagged with its
9/// coordinate unit `U` and numeric component type `T`.
10///
11/// `T` defaults to `f64` for domain math. The type is a technical
12/// primitive without validation: `radius` should be non-negative,
13/// stricter bounds belong to consumer wrappers.
14#[derive(Debug, Serialize, Deserialize)]
15pub struct Sphere<U = (), T = f64> {
16    /// The center point.
17    pub center: Vec3<U, T>,
18    /// The radius.
19    pub radius: T,
20}
21
22impl<U, T: Copy> Copy for Sphere<U, T> {}

24impl<U, T: Clone> Clone for Sphere<U, T> {
25    fn clone(&self) -> Self {
26        Self::new(self.center.clone(), self.radius.clone())
27    }
28}

30impl<U, T: PartialEq> PartialEq for Sphere<U, T> {
31    fn eq(&self, other: &Self) -> bool {
32        self.center == other.center && self.radius == other.radius
33    }
34}
35
36impl<U, T: Eq> Eq for Sphere<U, T> {}

38impl<U, T: Hash> Hash for Sphere<U, T> {
39    fn hash<H: Hasher>(&self, state: &mut H) {
40        self.center.hash(state);
41        self.radius.hash(state);
42    }
43}

45impl<U, T> Sphere<U, T> {
46    /// Creates a sphere from its center and radius.
47    pub const fn new(center: Vec3<U, T>, radius: T) -> Self {
48        Self { center, radius }
49    }
50}

52impl<U> Sphere<U, f64> {
53    /// Whether `point` lies inside the ball - the surface counts
54    /// as inside (`distance <= radius`).
55    pub fn contains(&self, point: &Vec3<U>) -> bool {
56        self.center.distance(point) <= self.radius
57    }
58
59    /// The surface point at `latitude` (elevation `-90..90` degrees
60    /// above the equator plane `xz`) and `longitude` (azimuth in
61    /// degrees around `y`, counterclockwise seen from `+y`).
62    ///
63    /// `y` points up: `latitude = +90` is the north pole, `0`/`0` is
64    /// `+x` on the equator. The inverse is [`Self::angles_of`].
65    pub fn point_at(&self, latitude: Degrees, longitude: Degrees) -> Vec3<U> {
66        let lat = latitude.to_radians().as_f64();
67        let lon = longitude.to_radians().as_f64();
68        self.center + Vec3::<U>::new(lat.cos() * lon.cos(), lat.sin(), lat.cos() * lon.sin()) * self.radius
69    }
70
71    /// The `(latitude, longitude)` of a direction from the center -
72    /// the inverse of [`Self::point_at`]. Longitude is normalized to
73    /// `-180..180` degrees; the poles return longitude `0`.
74    pub fn angles_of(&self, point: &Vec3<U>) -> (Degrees, Degrees) {
75        let offset = *point - self.center;
76        let radius = offset.magnitude();
77        if radius <= f64::EPSILON {
78            return (Degrees::ZERO, Degrees::ZERO);
79        }
80        let lat = (offset.y / radius).clamp(-1.0, 1.0).asin().to_degrees();
81        let lon = if (offset.y / radius).abs() >= 1.0 - 1e-12 {
82            0.0
83        } else {
84            offset.z.atan2(offset.x).to_degrees()
85        };
86        (Degrees::new(lat), Degrees::new(lon))
87    }
88
89    /// The diameter.
90    pub fn diameter(&self) -> f64 {
91        self.radius * 2.0
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn point_at_equator_prime_meridian_is_x_axis() {
101        let sphere = Sphere::new(Vec3::<()>::new(0.0, 0.0, 0.0), 2.0);
102        let point = sphere.point_at(Degrees::ZERO, Degrees::ZERO);
103        assert!((point - Vec3::new(2.0, 0.0, 0.0)).magnitude() < 1e-12);
104    }
105
106    #[test]
107    fn point_at_north_and_south_poles() {
108        let sphere = Sphere::new(Vec3::<()>::new(1.0, 1.0, 1.0), 3.0);
109        let north = sphere.point_at(Degrees::new(90.0), Degrees::new(45.0));
110        let south = sphere.point_at(Degrees::new(-90.0), Degrees::new(200.0));
111        assert!((north - Vec3::new(1.0, 4.0, 1.0)).magnitude() < 1e-12);
112        assert!((south - Vec3::new(1.0, -2.0, 1.0)).magnitude() < 1e-12);
113    }
114
115    #[test]
116    fn angles_of_inverts_point_at() {
117        let sphere = Sphere::new(Vec3::<()>::new(0.0, 0.0, 0.0), 1.0);
118        let (lat, lon) = (Degrees::new(35.0), Degrees::new(-70.0));
119        let (back_lat, back_lon) = sphere.angles_of(&sphere.point_at(lat, lon));
120        assert!((back_lat - lat).abs().as_f64() < 1e-9);
121        assert!((back_lon - lon).abs().as_f64() < 1e-9);
122    }
123
124    #[test]
125    fn contains_boundary_and_outside() {
126        let sphere = Sphere::new(Vec3::<()>::new(0.0, 0.0, 0.0), 1.0);
127        assert!(sphere.contains(&Vec3::new(0.5, 0.0, 0.0)));
128        assert!(sphere.contains(&Vec3::new(1.0, 0.0, 0.0)));
129        assert!(!sphere.contains(&Vec3::new(1.0, 0.5, 0.0)));
130    }
