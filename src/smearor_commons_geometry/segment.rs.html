1//! `Segment` - a straight segment between two positions.
2
3use crate::Fraction;
4use crate::Projection;
5use crate::RotationRad;
6use crate::Vec2;
7
8/// A straight segment in unit `U` (`Vec2<Millimeters>` on a map,
9/// `Vec2<Pixels>` on screen) - the geometry an edge draws between
10/// its two endpoints.
11#[derive(Debug, Clone, Copy, PartialEq)]
12pub struct Segment<U> {
13    /// Segment start position.
14    pub from: Vec2<U>,
15    /// Segment end position.
16    pub to: Vec2<U>,
17}
18
19impl<U> Segment<U> {
20    /// A segment from `from` to `to`.
21    pub const fn new(from: Vec2<U>, to: Vec2<U>) -> Self {
22        Self { from, to }
23    }
24}
25
26impl<U: Copy> Segment<U> {
27    /// The segment's center point.
28    pub fn midpoint(&self) -> Vec2<U> {
29        Vec2::new((self.from.x + self.to.x) / 2.0, (self.from.y + self.to.y) / 2.0)
30    }
31
32    /// The segment's direction as an angle (atan2 of the
33    /// from→to vector).
34    pub fn direction(&self) -> RotationRad {
35        RotationRad::new((self.to.y - self.from.y).atan2(self.to.x - self.from.x))
36    }
37
38    /// Strict crossing - `true` when the interiors of both segments
39    /// intersect (shared endpoints are junctions, not crossings).
40    pub fn crosses(&self, other: &Self) -> bool {
41        let delta = |a: Vec2<U>, b: Vec2<U>| Vec2::<U>::new(b.x - a.x, b.y - a.y);
42        let orient = |p: Vec2<U>, q: Vec2<U>, r: Vec2<U>| delta(p, q).cross::<f64>(&delta(p, r));
43        orient(self.from, self.to, other.from) * orient(self.from, self.to, other.to) < 0.0
44            && orient(other.from, other.to, self.from) * orient(other.from, other.to, self.to) < 0.0
45    }
46
47    /// The unclamped projection of `point` onto the segment's
48    /// line - `None` only when the segment is degenerate.
49    fn project(&self, point: &Vec2<U>) -> Option<Projection<U>> {
50        let direction = self.to - self.from;
51        let len_sq = direction.dot::<f64>(&direction);
52        if len_sq == 0.0 {
53            return None;
54        }
55        let fraction = Fraction::new((*point - self.from).dot::<f64>(&direction) / len_sq);
56        Some(Projection {
57            fraction,
58            point: self.from + direction * fraction,
59        })
60    }
61
62    /// The projection of `point` onto the segment interior -
63    /// `None` when the projection falls on an endpoint or
64    /// outside the segment, or the segment is degenerate. Unlike
65    /// `interior_distance` the endpoints are excluded (a snapped
66    /// endpoint is no cut point).
67    pub fn interior_projection(&self, point: &Vec2<U>) -> Option<Projection<U>> {
68        let projection = self.project(point)?;
69        (Fraction::new(1e-6)..=Fraction::new(1.0 - 1e-6))
70            .contains(&projection.fraction)
71            .then_some(projection)
72    }
73
74    /// Perpendicular distance of `point` to the segment when the
75    /// closest point lies between the endpoints - `None` when the
76    /// projection falls beyond either end or the segment is
77    /// degenerate.
78    pub fn interior_distance(&self, point: &Vec2<U>) -> Option<U>
79    where
80        U: From<f64>,
81    {
82        let projection = self.project(point)?;
83        (Fraction::ZERO..=Fraction::ONE)
84            .contains(&projection.fraction)
85            .then(|| point.distance(&projection.point))
86    }
87
88    /// The nearest point on the segment to `point` plus its
89    /// `fraction` along `from -> to` (clamped to `[0, 1]`) -
90    /// `None` for a degenerate segment.
91    pub fn nearest(&self, point: &Vec2<U>) -> Option<Projection<U>> {
92        let fraction = self.project(point)?.fraction.clamped();
93        Some(Projection {
94            fraction,
95            point: self.from + (self.to - self.from) * fraction,
96        })
97    }
98
99    /// The distance from `point` to the segment - the
100    /// perpendicular distance when the projection lands inside,
101    /// the nearer endpoint distance otherwise.
102    pub fn distance(&self, point: &Vec2<U>) -> U
103    where
104        U: From<f64>,
105    {
106        self.nearest(point)
107            .map_or_else(|| point.distance(&self.from), |nearest| point.distance(&nearest.point))
108    }
109
110    /// The point where the two segments strictly cross - `None`
111    /// when they do not (a mere touch counts as a junction, not
112    /// a crossing).
113    pub fn crossing_point(&self, other: &Self) -> Option<Vec2<U>> {
114        self.crosses(other).then(|| self.intersection(other)).flatten()
115    }
116
117    /// The parameter `t` where the line `anchor + axis * t` meets
118    /// this segment - `None` when they are parallel or the meeting
119    /// lies off the segment.
120    pub fn param_on_line(&self, anchor: &Vec2<U>, axis: &Vec2<U>) -> Option<U>
121    where
122        U: From<f64>,
123    {
124        let edge_direction = self.to - self.from;
125        let denom = axis.cross::<f64>(&edge_direction);
126        if denom.abs() < 1e-9 {
127            return None;
128        }
129        let offset = self.from - *anchor;
130        let s = axis.cross::<f64>(&offset) / -denom;
131        if !(0.0..=1.0).contains(&s) {
132            return None;
133        }
134        Some(U::from(offset.cross::<f64>(&edge_direction) / denom))
135    }
136
137    /// The point where the two segments meet - `None` when they are
138    /// parallel, degenerate or the meeting lies outside either
139    /// segment.
140    pub fn intersection(&self, other: &Self) -> Option<Vec2<U>> {
141        let direction = Vec2::<U>::new(self.to.x - self.from.x, self.to.y - self.from.y);
142        let other_direction = Vec2::<U>::new(other.to.x - other.from.x, other.to.y - other.from.y);
143        let denom = direction.cross::<f64>(&other_direction);
144        if denom == 0.0 {
145            return None;
146        }
147        let offset = Vec2::<U>::new(other.from.x - self.from.x, other.from.y - self.from.y);
148        let t = offset.cross::<f64>(&other_direction) / denom;
149        let u = offset.cross::<f64>(&direction) / denom;
150        if !(0.0..=1.0).contains(&t) || !(0.0..=1.0).contains(&u) {
151            return None;
152        }
153        Some(Vec2::new(self.from.x + t * direction.x, self.from.y + t * direction.y))
154    }
155}

157#[cfg(test)]
158mod tests {
159    use super::*;
160    use crate::Millimeters;
161
162    /// `midpoint` averages both endpoints; `direction` is the
163    /// atan2 of the from→to vector.
164    #[test]
165    fn midpoint_and_direction() {
166        let segment = Segment::<Millimeters>::new(Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0));
167        assert_eq!(segment.midpoint(), Vec2::new(5.0, 0.0));
168        assert_eq!(segment.direction().as_f64(), 0.0);
169    }
170
171    /// `crosses` is strict: interior intersections count, shared
172    /// endpoints do not.
173    #[test]
174    fn crosses_interior_only() {
175        let horizontal = Segment::<Millimeters>::new(Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0));
176        let crossing = Segment::<Millimeters>::new(Vec2::new(5.0, -5.0), Vec2::new(5.0, 5.0));
177        let touching = Segment::<Millimeters>::new(Vec2::new(10.0, 0.0), Vec2::new(10.0, 5.0));
178        assert!(horizontal.crosses(&crossing));
179        assert!(!horizontal.crosses(&touching));
180        assert_eq!(horizontal.crossing_point(&crossing), Some(Vec2::new(5.0, 0.0)));
181        assert_eq!(horizontal.crossing_point(&touching), None);
182    }
183
184    /// `nearest` clamps the fraction to [0, 1]; `distance` falls
185    /// back to the nearer endpoint.
186    #[test]
187    fn nearest_clamps_to_endpoints() {
188        let segment = Segment::<Millimeters>::new(Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0));
189        let nearest = segment.nearest(&Vec2::new(15.0, 3.0));
190        assert!(nearest.is_some());
191        if let Some(nearest) = nearest {
192            assert_eq!(nearest.fraction, Fraction::ONE);
193            assert_eq!(nearest.point, Vec2::new(10.0, 0.0));
194        }
195        let distance = segment.distance(&Vec2::new(15.0, 4.0));
196        assert!((distance.as_f64() - 41.0_f64.sqrt()).abs() < 1e-9);
197        assert_eq!(segment.interior_distance(&Vec2::new(15.0, 4.0)), None);
198    }
199
200    /// `interior_projection` excludes the endpoints and degenerate
201    /// segments.
202    #[test]
203    fn interior_projection_excludes_endpoints() {
204        let segment = Segment::<Millimeters>::new(Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0));
205        assert!(segment.interior_projection(&Vec2::new(5.0, 3.0)).is_some());
206        assert!(segment.interior_projection(&Vec2::new(0.0, 3.0)).is_none());
207        let degenerate = Segment::<Millimeters>::new(Vec2::new(1.0, 1.0), Vec2::new(1.0, 1.0));
208        assert!(degenerate.interior_projection(&Vec2::new(1.0, 1.0)).is_none());
209        assert!(degenerate.nearest(&Vec2::new(1.0, 1.0)).is_none());
210    }
211
212    /// `intersection` returns the meeting point inside both
213    /// segments, `None` for parallel or off-segment meetings.
214    #[test]
215    fn intersection_inside_both() {
216        let horizontal = Segment::<Millimeters>::new(Vec2::new(0.0, 0.0), Vec2::new(10.0, 0.0));
217        let crossing = Segment::<Millimeters>::new(Vec2::new(5.0, -5.0), Vec2::new(5.0, 5.0));
218        let parallel = Segment::<Millimeters>::new(Vec2::new(0.0, 1.0), Vec2::new(10.0, 1.0));
219        assert_eq!(horizontal.intersection(&crossing), Some(Vec2::new(5.0, 0.0)));
220        assert_eq!(horizontal.intersection(&parallel), None);
221    }
222
223    /// `param_on_line` gives the `t` where `anchor + axis * t`
224    /// meets the segment.
225    #[test]
226    fn param_on_line_meets_inside() {
227        let segment = Segment::<Millimeters>::new(Vec2::new(0.0, 5.0), Vec2::new(10.0, 5.0));
228        let t = segment.param_on_line(&Vec2::new(5.0, 0.0), &Vec2::new(0.0, 1.0));
229        assert_eq!(t, Some(Millimeters::new(5.0)));
230        let miss = segment.param_on_line(&Vec2::new(15.0, 0.0), &Vec2::new(0.0, 1.0));
231        assert_eq!(miss, None);
232    }
