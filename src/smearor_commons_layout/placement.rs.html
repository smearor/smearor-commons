6/// The technical target of an arrangement: where an item's center
7/// lands, how it is rotated and how large its footprint is.
8///
9/// Consumers map their domain items onto `Placement`s and back -
10/// the type carries no semantics about what the item is. Generic
11/// over the coordinate unit `U` (and, like [`Vec2`], the numeric
12/// component type `T`, default `f64`) so pixel, millimeter and
13/// erased coordinate spaces share the same algorithms.
14#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
15pub struct Placement<U = (), T = f64> {
16    /// Target center point inside the arranged bounds.
17    pub center: Vec2<U, T>,
18    /// Reading orientation, snapped to a quarter turn.
19    pub rotation: QuarterTurn,
20    /// Footprint extent (rotated bounding box of the item),
21    /// passed through unchanged by the arrangement.
22    pub size: Vec2<U, T>,
23}
24
25impl<U, T: Copy> Placement<U, T> {
26    /// Creates a placement at `center` facing `rotation` with
27    /// footprint `size`.
28    pub fn new(center: Vec2<U, T>, rotation: QuarterTurn, size: Vec2<U, T>) -> Self {
29        Self { center, rotation, size }
30    }
31}
