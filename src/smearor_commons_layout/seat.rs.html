11/// A seat on a tabletop surface: which [`TableSide`] it faces plus
12/// where along that edge it sits.
13///
14/// A seat is more than a side - users can sit anywhere along an edge
15/// (two on one long side, off-center), so the offset carries the
16/// position along the edge axis. `offset` `None` means "assigned to
17/// the side, a strategy may distribute the item along it";
18/// `Some(fraction)` pins the item to that edge position
19/// (`HALF` = centered). Which *user* sits there stays consumer domain.
20#[derive(Debug, Default, Clone, Copy, PartialEq, Serialize, Deserialize)]
21pub struct Seat {
22    /// The table edge the seat faces.
23    pub side: TableSide,
24    /// Position along the edge axis (`0.0` = edge start, `1.0` = edge
25    /// end, `0.5` = centered). `None` leaves distribution along the
26    /// side to the strategy.
27    pub offset: Option<Fraction>,
28}
29
30impl Seat {
31    /// A seat assigned to `side` without a pinned position - a
32    /// strategy distributes the item along the edge.
33    pub const fn new(side: TableSide) -> Self {
34        Self { side, offset: None }
35    }
36
37    /// A seat pinned to `offset` along the edge of `side`.
38    pub const fn at(side: TableSide, offset: Fraction) -> Self {
39        Self { side, offset: Some(offset) }
40    }
41
42    /// A seat centered on `side`'s edge.
43    pub const fn center(side: TableSide) -> Self {
44        Self::at(side, Fraction::HALF)
45    }
46
47    /// The seat one step clockwise (`side` steps, `offset` keeps).
48    pub fn clockwise(&self) -> Self {
49        Self {
50            side: self.side.clockwise(),
51            ..*self
52        }
53    }
54
55    /// The seat one step counter-clockwise.
56    pub fn counter_clockwise(&self) -> Self {
57        Self {
58            side: self.side.counter_clockwise(),
59            ..*self
60        }
61    }
62
63    /// The centered seat nearest a (free) rotation.
64    pub fn nearest(rotation: RotationRad) -> Self {
65        Self::center(TableSide::nearest(rotation))
66    }

68    /// The seat's position along the table perimeter, clockwise in
69    /// edge-length units `0.0..4.0`: the bottom edge spans `0..1`
70    /// (right-to-left, `offset` decreases), the left edge `1..2`
71    /// (bottom-to-top), the top edge `2..3` (left-to-right), the right
72    /// edge `3..4` (top-to-bottom) - continuous at the corners, so
73    /// seats on neighboring sides compare directly. An unpinned
74    /// `offset` resolves to `HALF`, like [`Self::placement`].
75    ///
76    /// The coordinate orders seats for "pass to the next occupied
77    /// seat" and lets consumers position seat indicators around the
78    /// table.
79    pub fn perimeter_position(&self) -> f64 {
80        let offset = self.offset.unwrap_or(Fraction::HALF).as_f64();
81        match self.side {
82            TableSide::Bottom => 1.0 - offset,
83            TableSide::Left => 2.0 - offset,
84            TableSide::Top => 2.0 + offset,
85            TableSide::Right => 3.0 + offset,
86        }
87    }
88
89    /// The occupied seat one step clockwise along the table perimeter:
90    /// the roster seat with the smallest positive distance in
91    /// [`Self::perimeter_position`]. Empty sides are skipped, several
92    /// seats on one side are visited in order, and a seat at this
93    /// seat's own position does not count.
94    ///
95    /// Returns `None` when `roster` is empty or holds no position
96    /// other than this seat's.
97    pub fn next_in(&self, roster: &[Seat]) -> Option<Seat> {
98        self.step_in(roster, true)
99    }
100
101    /// The occupied seat one step counter-clockwise - the mirror of
102    /// [`Self::next_in`].
103    pub fn previous_in(&self, roster: &[Seat]) -> Option<Seat> {
104        self.step_in(roster, false)
105    }
106
107    /// Shared walk of `next_in`/`previous_in`: picks the roster seat
108    /// with the smallest strictly-positive perimeter distance in the
109    /// chosen direction.
110    fn step_in(&self, roster: &[Seat], clockwise: bool) -> Option<Seat> {
111        let here = self.perimeter_position();
112        let mut best: Option<(Seat, f64)> = None;
113        for seat in roster {
114            let position = seat.perimeter_position();
115            let distance = if clockwise { position - here } else { here - position }.rem_euclid(4.0);
116            if distance <= f64::EPSILON {
117                continue;
118            }
119            if best.is_none_or(|(_, shortest)| distance < shortest) {
120                best = Some((*seat, distance));
121            }
122        }
123        best.map(|(seat, _)| seat)
124    }

126    /// Places `footprint` in front of this seat: `margin` (scalar in
127    /// the space of `bounds`) off the edge, positioned along the
128    /// edge at `offset` (unpinned offsets resolve to `HALF`),
129    /// rotated so a user sitting there reads it correctly.
130    ///
131    /// `footprint` is the item's extent *as displayed at the seat's
132    /// rotation* - compute it with [`crate::footprint_for`] when the
133    /// placement changes orientation.
134    pub fn placement<U>(&self, bounds: Rect<U>, footprint: Vec2<U>, margin: f64) -> Placement<U> {
135        let offset = self.offset.unwrap_or(Fraction::HALF).as_f64();
136        let center = match self.side {
137            TableSide::Bottom => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + bounds.size.y - margin - footprint.y / 2.0),
138            TableSide::Left => Vec2::new(bounds.min.x + margin + footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
139            TableSide::Top => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + margin + footprint.y / 2.0),
140            TableSide::Right => Vec2::new(bounds.min.x + bounds.size.x - margin - footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
141        };
142        Placement::new(center, Rotation::QuarterTurn(self.side.to_quarter_turn()), footprint)
143    }

145    /// Places the item in the best free spot nearest this seat: like
146    /// [`Self::placement`], but `taken` items are treated as physical
147    /// obstacles and the item slips into the closest free lateral gap
148    /// instead of stacking on them.
149    ///
150    /// `taken` holds each occupying item as `(center, footprint)` -
151    /// its live center and displayed extent; *every* item near the
152    /// seat counts, regardless of any seat assignment. Spots are
153    /// evaluated as lanes perpendicular to the edge: lane 0 is the
154    /// edge lane at `margin` inset (the preferred position), deeper
155    /// lanes stack behind it. Within a lane the free center with the
156    /// smallest lateral distance wins, so slots freed by removed items
157    /// are reused and a bigger item only takes a middle slot when it
158    /// actually fits between its neighbors with `gap` clearance.
159    /// Candidates are scored by seat-space distance
160    /// `hypot(offset, ROW_JUMP_WEIGHT * depth)`, so a card may leave
161    /// the edge lane when the queue there reaches notably farther
162    /// from the seat than a spot behind the first row.
163    pub fn fanned_placement<U>(&self, bounds: Rect<U>, footprint: Vec2<U>, margin: f64, taken: &[(Vec2<U>, Vec2<U>)], gap: f64) -> Placement<U> {
164        /// Perpendicular depth counts this much when scoring lanes -
165        /// staying on the edge is preferred unless the queue there is
166        /// notably farther from the seat.
167        const ROW_JUMP_WEIGHT: f64 = 2.0;
168
169        let mut placement = self.placement(bounds, footprint, margin);
170        let right: Vec2<U> = self.side.right_direction();
171        let inward = Vec2::new(right.y, -right.x);
172        let anchor = placement.center;
173        let own_t = self.tangential_extent(footprint) / 2.0;
174        let own_p = self.perpendicular_extent(footprint) / 2.0;
175        let anchor_inset = margin + own_p;
176
177        let (mut best_offset, mut best_depth, mut best_score) = (0.0, 0.0, f64::MAX);
178        let mut lane_inset = anchor_inset;
179        loop {
180            let mut blocked = Vec::new();
181            let mut inner_edge = lane_inset;
182            for &(center, item_footprint) in taken {
183                let item_inset = self.inset_of(center, bounds);
184                let item_p = self.perpendicular_extent(item_footprint) / 2.0;
185                if (item_inset - lane_inset).abs() >= item_p + own_p + gap {
186                    continue;
187                }
188                inner_edge = inner_edge.max(item_inset + item_p);
189                let delta = center - anchor;
190                let offset = delta.x * right.x + delta.y * right.y;
191                let item_t = self.tangential_extent(item_footprint) / 2.0;
192                blocked.push((offset - item_t - gap - own_t, offset + item_t + gap + own_t));
193            }
194            let depth = lane_inset - anchor_inset;
195            let offset = nearest_free_slot(&mut blocked);
196            let weighted = ROW_JUMP_WEIGHT * depth;
197            let score = (offset * offset + weighted * weighted).sqrt();
198            if score < best_score {
199                (best_offset, best_depth, best_score) = (offset, depth, score);
200            }
201            // An empty lane has no inner edge to stack the next lane
202            // behind; deeper lanes only score worse once their depth
203            // alone already beats the best candidate.
204            if blocked.is_empty() || weighted >= best_score {
205                break;
206            }
207            lane_inset = inner_edge + gap + own_p;
208        }
209        placement.center += right * best_offset + inward * best_depth;
210        placement
211    }

213    /// The item's extent along this seat's edge axis - x for the
214    /// bottom/top edges, y for the left/right edges.
215    fn tangential_extent<U>(&self, footprint: Vec2<U>) -> f64 {
216        match self.side {
217            TableSide::Bottom | TableSide::Top => footprint.x,
218            TableSide::Left | TableSide::Right => footprint.y,
219        }
220    }
221
222    /// The item's extent perpendicular to this seat's edge - y for the
223    /// bottom/top edges, x for the left/right edges.
224    fn perpendicular_extent<U>(&self, footprint: Vec2<U>) -> f64 {
225        match self.side {
226            TableSide::Bottom | TableSide::Top => footprint.y,
227            TableSide::Left | TableSide::Right => footprint.x,
228        }
229    }
230
231    /// The distance of a center point inward from this seat's edge.
232    fn inset_of<U>(&self, center: Vec2<U>, bounds: Rect<U>) -> f64 {
233        match self.side {
234            TableSide::Bottom => bounds.min.y + bounds.size.y - center.y,
235            TableSide::Top => center.y - bounds.min.y,
236            TableSide::Left => center.x - bounds.min.x,
237            TableSide::Right => bounds.min.x + bounds.size.x - center.x,
238        }
239    }
240}

242/// The center coordinate on a 1D lane nearest `0` where `blocked`
243/// intervals (in newcomer-center space, already expanded by gap and
244/// own half-extent) leave room. Interval boundaries count as free -
245/// landing exactly on a boundary means exactly `gap` clearance.
246/// Prefers the positive direction on ties.
247fn nearest_free_slot(blocked: &mut Vec<(f64, f64)>) -> f64 {
248    blocked.sort_by(|left, right| left.0.partial_cmp(&right.0).unwrap_or(std::cmp::Ordering::Equal));
249    let mut merged: Vec<(f64, f64)> = Vec::with_capacity(blocked.len());
250    for &(left, right) in blocked.iter() {
251        match merged.last_mut() {
252            Some((_, merged_right)) if left <= *merged_right => *merged_right = merged_right.max(right),
253            _ => merged.push((left, right)),
254        }
255    }
256    let is_free = |candidate: f64| merged.iter().all(|&(left, right)| candidate <= left || candidate >= right);
257    std::iter::once(0.0)
258        .chain(merged.iter().flat_map(|&(left, right)| [left, right]))
259        .filter(|&candidate| is_free(candidate))
260        .min_by(|left, right| {
261            left.abs()
262                .partial_cmp(&right.abs())
263                .unwrap_or(std::cmp::Ordering::Equal)
264                .then(right.partial_cmp(left).unwrap_or(std::cmp::Ordering::Equal))
265        })
266        .unwrap_or(0.0)
267}
