11/// A seat on a tabletop surface: which [`TableSide`] it faces plus
12/// where along that edge it sits.
13///
14/// A seat is more than a side - users can sit anywhere along an edge
15/// (two on one long side, off-center), so the offset carries the
16/// position along the edge axis. `offset` `None` means "assigned to
17/// the side, a strategy may distribute the item along it";
18/// `Some(fraction)` pins the item to that edge position
19/// (`HALF` = centered). Which *user* sits there stays consumer domain.
20#[derive(Debug, Default, Clone, Copy, PartialEq, Serialize, Deserialize)]
21pub struct Seat {
22    /// The table edge the seat faces.
23    pub side: TableSide,
24    /// Position along the edge axis (`0.0` = edge start, `1.0` = edge
25    /// end, `0.5` = centered). `None` leaves distribution along the
26    /// side to the strategy.
27    pub offset: Option<Fraction>,
28}
29
30impl Seat {
31    /// A seat assigned to `side` without a pinned position - a
32    /// strategy distributes the item along the edge.
33    pub const fn new(side: TableSide) -> Self {
34        Self { side, offset: None }
35    }
36
37    /// A seat pinned to `offset` along the edge of `side`.
38    pub const fn at(side: TableSide, offset: Fraction) -> Self {
39        Self { side, offset: Some(offset) }
40    }
41
42    /// A seat centered on `side`'s edge.
43    pub const fn center(side: TableSide) -> Self {
44        Self::at(side, Fraction::HALF)
45    }
46
47    /// The seat one step clockwise (`side` steps, `offset` keeps).
48    pub fn clockwise(&self) -> Self {
49        Self {
50            side: self.side.clockwise(),
51            ..*self
52        }
53    }
54
55    /// The seat one step counter-clockwise.
56    pub fn counter_clockwise(&self) -> Self {
57        Self {
58            side: self.side.counter_clockwise(),
59            ..*self
60        }
61    }
62
63    /// The centered seat nearest a (free) rotation.
64    pub fn nearest(rotation: RotationRad) -> Self {
65        Self::center(TableSide::nearest(rotation))
66    }
67
68    /// Places `footprint` in front of this seat: `margin` (scalar in
69    /// the space of `bounds`) off the edge, positioned along the
70    /// edge at `offset` (unpinned offsets resolve to `HALF`),
71    /// rotated so a user sitting there reads it correctly.
72    ///
73    /// `footprint` is the item's extent *as displayed at the seat's
74    /// rotation* - compute it with [`crate::footprint_for`] when the
75    /// placement changes orientation.
76    pub fn placement<U>(&self, bounds: Rect<U>, footprint: Vec2<U>, margin: f64) -> Placement<U> {
77        let offset = self.offset.unwrap_or(Fraction::HALF).as_f64();
78        let center = match self.side {
79            TableSide::Bottom => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + bounds.size.y - margin - footprint.y / 2.0),
80            TableSide::Left => Vec2::new(bounds.min.x + margin + footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
81            TableSide::Top => Vec2::new(bounds.min.x + bounds.size.x * offset, bounds.min.y + margin + footprint.y / 2.0),
82            TableSide::Right => Vec2::new(bounds.min.x + bounds.size.x - margin - footprint.x / 2.0, bounds.min.y + bounds.size.y * offset),
83        };
84        Placement::new(center, Rotation::QuarterTurn(self.side.to_quarter_turn()), footprint)
85    }
86}

100    fn bounds() -> Rect<Pixels> {
101        Rect {
102            min: Vec2::new(0.0, 0.0),
103            size: Vec2::new(400.0, 300.0),
104        }
105    }
106
107    /// `placement` puts the footprint against the correct edge,
108    /// `margin` off it, centered on the edge axis, rotated for
109    /// the side.
110    #[test]
111    fn placement_attaches_to_each_edge() {
112        let footprint = Vec2::new(100.0, 50.0);
113        let cases = [
114            (TableSide::Bottom, 200.0, 300.0 - 24.0 - 25.0, QuarterTurn::Deg0),
115            (TableSide::Left, 24.0 + 50.0, 150.0, QuarterTurn::Deg90),
116            (TableSide::Top, 200.0, 24.0 + 25.0, QuarterTurn::Deg180),
117            (TableSide::Right, 400.0 - 24.0 - 50.0, 150.0, QuarterTurn::Deg270),
118        ];
119        for (side, x, y, turn) in cases {
120            let placement = Seat::center(side).placement(bounds(), footprint, 24.0);
121            assert!((placement.center.x - x).abs() < 1e-9, "{side:?} x");
122            assert!((placement.center.y - y).abs() < 1e-9, "{side:?} y");
123            assert_eq!(placement.rotation, Rotation::QuarterTurn(turn));
124        }
125    }
126
127    /// A pinned offset positions the footprint along the edge axis.
128    #[test]
129    fn offset_moves_along_the_edge() {
130        let footprint = Vec2::new(50.0, 50.0);
131        let quarter = Seat::at(TableSide::Bottom, Fraction::new(0.25)).placement(bounds(), footprint, 0.0);
132        assert!((quarter.center.x - 100.0).abs() < 1e-9);
133        let top = Seat::at(TableSide::Left, Fraction::new(0.75)).placement(bounds(), footprint, 0.0);
134        assert!((top.center.y - 225.0).abs() < 1e-9);
135    }
136
137    /// Stepping keeps the offset while the side moves.
138    #[test]
139    fn clockwise_keeps_offset() {
140        let seat = Seat::at(TableSide::Bottom, Fraction::new(0.25));
141        assert_eq!(seat.clockwise(), Seat::at(TableSide::Left, Fraction::new(0.25)));
142        assert_eq!(seat.counter_clockwise(), Seat::at(TableSide::Right, Fraction::new(0.25)));
143    }
144
145    /// `nearest` yields the centered seat of the nearest side.
146    #[test]
147    fn nearest_is_centered() {
148        assert_eq!(Seat::nearest(RotationRad::from_degrees(100.0)), Seat::center(TableSide::Left));
149    }
150
151    /// Serde round-trips a seat with and without a pinned offset.
152    #[test]
153    fn serde_round_trip() {
154        for seat in [
155            Seat::new(TableSide::Top),
156            Seat::center(TableSide::Bottom),
157            Seat::at(TableSide::Left, Fraction::new(0.3)),
158        ] {
159            let json = serde_json::to_string(&seat).expect("serialize");
160            assert_eq!(serde_json::from_str::<Seat>(&json).expect("deserialize"), seat);
161        }
162    }
