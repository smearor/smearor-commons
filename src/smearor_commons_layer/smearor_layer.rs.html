6/// The four layer-shell layers - the stacked order of surfaces from
7/// desktop background to overlay.
8///
9/// The vocabulary shared by Smearor consumers: CLI arguments parse it
10/// via [`FromStr`], configuration files deserialize it via `serde`
11/// and [`From<&str>`] provides the lenient config-bridge conversion
12/// that falls back to [`Top`](Self::Top) on unknown values.
13#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
14pub enum SmearorLayer {
15    /// The bottom-most layer, beneath regular windows.
16    #[serde(alias = "background", alias = "Background", alias = "BACKGROUND")]
17    Background,
18    /// Above the background, beneath regular windows.
19    #[serde(alias = "bottom", alias = "Bottom", alias = "BOTTOM")]
20    Bottom,
21    /// Above regular windows - the default.
22    #[serde(alias = "top", alias = "Top", alias = "TOP")]
23    #[default]
24    Top,
25    /// The top-most layer, above everything else.
26    #[serde(alias = "overlay", alias = "Overlay", alias = "OVERLAY")]
27    Overlay,
28}
29
30impl FromStr for SmearorLayer {
31    type Err = ParseLayerError;
32
33    /// Parses a layer name case-insensitively; unknown values are
34    /// rejected - use [`From<&str>`] for the lenient fallback.
35    fn from_str(s: &str) -> Result<Self, Self::Err> {
36        match s.to_lowercase().as_str() {
37            "background" => Ok(Self::Background),
38            "bottom" => Ok(Self::Bottom),
39            "top" => Ok(Self::Top),
40            "overlay" => Ok(Self::Overlay),
41            _ => Err(ParseLayerError::new(s)),
42        }
43    }
44}
45
46impl From<&str> for SmearorLayer {
47    /// The lenient counterpart of [`FromStr`]: unknown values fall
48    /// back to [`Top`](Self::Top). Used where a configuration value
49    /// feeds a CLI argument and must not fail.
50    fn from(s: &str) -> Self {
51        s.parse().unwrap_or_default()
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::SmearorLayer;
58
59    /// Layer names parse case-insensitively in all four variants.
60    #[test]
61    fn parses_all_layers_case_insensitively() {
62        assert_eq!("background".parse(), Ok(SmearorLayer::Background));
63        assert_eq!("Bottom".parse(), Ok(SmearorLayer::Bottom));
64        assert_eq!("TOP".parse(), Ok(SmearorLayer::Top));
65        assert_eq!("oVeRlAy".parse(), Ok(SmearorLayer::Overlay));
66    }
67
68    /// `FromStr` rejects unknown values while `From<&str>` falls
69    /// back to `Top`.
70    #[test]
71    fn unknown_values_error_or_fall_back() {
72        assert!("wallpaper".parse::<SmearorLayer>().is_err());
73        assert_eq!(SmearorLayer::from("wallpaper"), SmearorLayer::Top);
74    }
75
76    /// Serde aliases accept the same case variants as `FromStr`.
77    #[test]
78    fn serde_aliases_deserialize() {
79        assert_eq!(serde_json::from_str::<SmearorLayer>("\"Overlay\"").ok(), Some(SmearorLayer::Overlay));
80        assert_eq!(serde_json::to_string(&SmearorLayer::Bottom).ok().as_deref(), Some("\"Bottom\""));
81    }
