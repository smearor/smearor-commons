5/// Errors from buffer construction and view validation.
6#[derive(Debug, Error)]
7pub enum BufferError {
8    /// The data length does not match the extent's packed size.
9    #[error("buffer length {actual} does not match {width}x{height} x {bytes_per_pixel} = {expected} bytes")]
10    LengthMismatch {
11        /// The expected packed length (`width * height * bytes_per_pixel`).
12        expected: usize,
13        /// The actual data length.
14        actual: usize,
15        /// The declared width.
16        width: Pixels<u32>,
17        /// The declared height.
18        height: Pixels<u32>,
19        /// The format's bytes per pixel.
20        bytes_per_pixel: usize,
21    },
22
23    /// The stride is smaller than a packed pixel row.
24    #[error("stride {stride} is smaller than the packed row width {row_bytes}")]
25    StrideTooSmall {
26        /// The declared stride in bytes per row.
27        stride: Stride,
28        /// The packed row width (`width * bytes_per_pixel`).
29        row_bytes: usize,
30    },
31
32    /// The data is too short for the declared metadata.
33    #[error("data length {actual} is smaller than the required {required} bytes")]
34    DataTooShort {
35        /// The minimal length covering every pixel row.
36        required: usize,
37        /// The actual data length.
38        actual: usize,
39    },
40}
