1//! GTK bridge (`gtk4` feature): `PixelBuffer` ⇄ `gdk::Texture`.
2//!
3//! - [`PixelBuffer::to_memory_texture`] uploads buffer pixels as a
4//!   [`MemoryTexture`](gtk4::gdk::MemoryTexture) — display them via
5//!   `gtk4::Image::from_paintable` or any paintable consumer.
6//! - [`PixelBuffer::from_texture`] downloads a texture's pixels back
7//!   into a `PixelBuffer<Rgba>`, converting through the `PixelFormat`
8//!   channel interface (premultiplied → straight alpha included).
9//! - [`PixelBuffer::decode_image`] decodes embedded image bytes
10//!   (PNG, JPEG, …) through `gdk::Texture::from_bytes` — the
11//!   generalized chess `decode_texture` helper.
12//! - `From<PixelFormatId> for MemoryFormat` and the fallible inverse
13//!   carry the byte-exact format mapping next to the type it
14//!   describes.

43    /// Builds a texture over `rgba` bytes through a
44    /// `PixelBuffer<Rgba>` — the upload path under test.
45    fn rgba_texture(width: u32, height: u32, rgba: &[u8]) -> Texture {
46        let mut buffer = PixelBuffer::<Rgba>::new(Size::new(width, height));
47        buffer.pixels_mut().copy_from_slice(rgba);
48        buffer.to_memory_texture().upcast()
49    }

51    #[test]
52    fn memory_format_mapping_is_total() {
53        for (format, expected) in [
54            (PixelFormatId::Rgba, MemoryFormat::R8g8b8a8),
55            (PixelFormatId::Bgra, MemoryFormat::B8g8r8a8),
56            (PixelFormatId::RgbaPremultiplied, MemoryFormat::R8g8b8a8Premultiplied),
57            (PixelFormatId::BgraPremultiplied, MemoryFormat::B8g8r8a8Premultiplied),
58            (PixelFormatId::Rgb, MemoryFormat::R8g8b8),
59            (PixelFormatId::Bgr, MemoryFormat::B8g8r8),
60            (PixelFormatId::Rgbx, MemoryFormat::R8g8b8x8),
61            (PixelFormatId::Bgrx, MemoryFormat::B8g8r8x8),
62            (PixelFormatId::Gray, MemoryFormat::G8),
63        ] {
64            assert_eq!(MemoryFormat::from(format), expected);
65            assert_eq!(PixelFormatId::try_from(expected), Ok(format));
66        }
67    }

75    #[test]
76    fn buffer_texture_roundtrip_keeps_pixels() {
77        let expected: [u8; 16] = [10, 20, 30, 255, 40, 50, 60, 128, 70, 80, 90, 0, 100, 110, 120, 255];
78        let texture = rgba_texture(2, 2, &expected);
79        let buffer = PixelBuffer::<Rgba>::from_texture(&texture);
80        // Download goes through premultiplied BGRA — a=0 pixels lose
81        // their color channels; ±1 tolerance for the alpha round-trip.
82        for (s, d) in expected.as_chunks::<4>().0.iter().zip(buffer.pixels().as_chunks::<4>().0.iter()) {
83            if s[3] == 0 {
84                assert_eq!(d, &[0, 0, 0, 0]);
85            } else {
86                for (x, y) in s.iter().zip(d) {
87                    assert!((i32::from(*x) - i32::from(*y)).abs() <= 1, "{s:?} != {d:?}");
88                }
89            }
90        }
91    }

93    #[test]
94    fn premultiplied_source_unpremultiplies() {
95        let mut buffer = PixelBuffer::<BgraPremultiplied>::new(Size::new(1, 1));
96        BgraPremultiplied::write_pixel(&mut buffer.pixels_mut()[..4], 200, 100, 50, 128);
97        let texture: Texture = buffer.to_memory_texture().upcast();
98        let back = PixelBuffer::<Rgba>::from_texture(&texture);
99        let [r, g, b, a] = Rgba::read_pixel(back.pixels());
100        assert_eq!(a, 128);
101        assert!((r as i32 - 200).abs() <= 1 && (g as i32 - 100).abs() <= 1 && (b as i32 - 50).abs() <= 1);
102    }

104    #[test]
105    fn decode_image_decodes_a_png() {
106        let buffer = PixelBuffer::<Rgba>::decode_image(TEST_PNG).unwrap();
107        assert_eq!(buffer.size(), Size::new(2, 1));
108        let [r0, g0, b0, _] = Rgba::read_pixel(&buffer.pixels()[..4]);
109        let [r1, g1, b1, _] = Rgba::read_pixel(&buffer.pixels()[4..8]);
110        assert!(r0 > 200 && g0 < 60 && b0 < 60, "left pixel red: {r0},{g0},{b0}");
111        assert!(r1 < 60 && g1 < 60 && b1 > 200, "right pixel blue: {r1},{g1},{b1}");
112    }
