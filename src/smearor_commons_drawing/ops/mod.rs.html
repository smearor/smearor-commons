1//! The drawing capability traits and the [`DrawOps`] umbrella trait.
2//!
3//! The rendering vocabulary is split into per-capability traits so that
4//! backends can implement them incrementally and consumers can bound on
5//! exactly the capabilities they need:
6//!
7//! - [`DrawBackend`]: the per-target error type shared by all capability traits
8//! - [`DrawPrimitiveOps`]: surface fills and shape primitives
9//! - [`DrawIconOps`]: icon and glyph rendering
10//! - [`DrawTextOps`]: label and centered text rendering
11//! - [`DrawProgressOps`]: progress and gauge indicators
12//!
13//! [`DrawOps`] bundles all of them as a supertrait with a blanket impl —
14//! use it as a bound when a consumer needs the full vocabulary. Method
15//! calls still require the declaring trait in scope; importing
16//! [`crate::prelude`] brings in all capability traits at once.
17//!
18//! Every capability trait is also implemented for `&mut T` where `T`
19//! implements it, so mutable references compose transparently — e.g.
20//! `&mut PixelBuffer` can serve as a [`DrawTarget`].
21
22mod backend;
23mod icon;
24mod primitive;
25mod progress;
26mod target;
27mod text;

45/// Umbrella trait bundling all drawing capability traits.
46///
47/// Blanket-implemented for every type that implements all capability
48/// traits; useful as a bound (`T: DrawOps`) when a consumer needs the
49/// full vocabulary. It declares no methods itself — import the
50/// individual capability traits (or [`crate::prelude`]) to call them.
51pub trait DrawOps: DrawIconOps + DrawPrimitiveOps + DrawProgressOps + DrawTextOps {}
52
53impl<T> DrawOps for T where T: DrawIconOps + DrawPrimitiveOps + DrawProgressOps + DrawTextOps {}
