1//! The [`DrawPrimitiveOps`] implementation for `CairoTarget`.
2//!
3//! One impl for the single cairo target type — unlike the buffer
4//! backend there is no second packed-view target to delegate to.
5

19impl DrawPrimitiveOps for CairoTarget {
20    /// Fill the entire surface with `color`, preserving its alpha like
21    /// the buffer `fill`: `Operator::Source` replaces destination pixels
22    /// instead of compositing over them. `restore` runs even when
23    /// `paint` fails so the context state stays balanced.
24    fn fill_background(&mut self, color: impl Into<Color<u8>>) -> Result<(), Self::Error> {
25        let color: Color<u8> = color.into();
26        color.apply_alpha(&self.ctx, Alpha::from(color.a));
27        self.ctx.save()?;
28        self.ctx.set_operator(cairo::Operator::Source);
29        let result = self.ctx.paint();
30        self.ctx.restore()?;
31        result
32    }
33
34    fn fill_rect(&mut self, spec: FillRect) -> Result<(), Self::Error> {
35        spec.color.apply_alpha(&self.ctx, spec.alpha);
36        self.ctx.rectangle(spec.rect.min.x, spec.rect.min.y, spec.rect.width(), spec.rect.height());
37        self.ctx.fill()
38    }
39
40    /// Stroke centered on the rect edge — cairo already centers strokes
41    /// on the path, so the rectangle is emitted at `spec.rect` itself (the
42    /// buffer impl needs the `inset(±w/2)` ring for its pixel scan).
43    fn stroke_rect(&mut self, spec: StrokeRect) -> Result<(), Self::Error> {
44        if spec.line_width.as_raw() <= 0.0 {
45            return Ok(());
46        }
47        spec.color.apply_alpha(&self.ctx, spec.alpha);
48        self.ctx.rectangle(spec.rect.min.x, spec.rect.min.y, spec.rect.width(), spec.rect.height());
49        self.ctx.set_line_width(f64::from(spec.line_width.as_raw()));
50        self.ctx.stroke()
51    }
52
53    fn fill_circle(&mut self, spec: FillCircle) -> Result<(), Self::Error> {
54        spec.color.apply_alpha(&self.ctx, spec.alpha);
55        self.ctx.arc(spec.circle.center.x, spec.circle.center.y, spec.circle.radius, 0.0, TAU);
56        self.ctx.fill()
57    }
58
59    fn fill_polygon(&mut self, spec: FillPolygon) -> Result<(), Self::Error> {
60        spec.color.apply_alpha(&self.ctx, spec.alpha);
61        polyline_path(&self.ctx, &spec.polygon);
62        self.ctx.close_path();
63        self.ctx.fill()
64    }
65
66    /// Stroke centered on the segments like cairo — the open path
67    /// gets `line_width` symmetric around the vertex polyline.
68    fn stroke_polyline(&mut self, spec: StrokePolyline) -> Result<(), Self::Error> {
69        if spec.line_width.as_raw() <= 0.0 {
70            return Ok(());
71        }
72        spec.color.apply_alpha(&self.ctx, spec.alpha);
73        polyline_path(&self.ctx, spec.polyline.points());
74        self.ctx.set_line_width(f64::from(spec.line_width.as_raw()));
75        self.ctx.stroke()
76    }
77
78    /// The annulus sector as outer arc swept forward (`arc`) and
79    /// inner arc swept back — `arc_negative` handles the wrap across
80    /// the full turn through the un-normalized end direction.
81    fn fill_sector(&mut self, spec: FillSector) -> Result<(), Self::Error> {
82        spec.color.apply_alpha(&self.ctx, spec.alpha);
83        let center = spec.annulus.center;
84        let (start, end) = (spec.arc.start.as_f64(), spec.arc.start.as_f64() + spec.arc.width.as_f64());
85        self.ctx.arc(center.x, center.y, spec.annulus.outer, start, end);
86        self.ctx.arc_negative(center.x, center.y, spec.annulus.inner, end, start);
87        self.ctx.close_path();
88        self.ctx.fill()
89    }
90}

104    /// Render through a `CairoTarget` onto a fresh `ImageSurface` and
105    /// return the raw surface bytes (ARgb32 = BGRA, premultiplied).
106    fn render(size: Size<u32>, draw: impl FnOnce(&mut CairoTarget) -> Result<(), cairo::Error>) -> Vec<u8> {
107        let mut surface = ImageSurface::create(Format::ARgb32, size.x as i32, size.y as i32).expect("image surface");
108        {
109            let ctx = Context::new(&surface).expect("context");
110            let mut target = CairoTarget::new(&ctx, size);
111            draw(&mut target).expect("draw operation");
112        }
113        surface.flush();
114        surface.data().expect("surface data").to_vec()
115    }
116
117    fn px(data: &[u8], size: Size<u32>, x: u32, y: u32) -> [u8; 4] {
118        let idx = (y * size.x + x) as usize * 4;
119        data[idx..idx + 4].try_into().expect("pixel")
120    }
121
122    #[test]
123    fn fill_background_paints_all_pixels() {
124        let size = Size::new(8, 8);
125        let data = render(size, |t| t.fill_background(Color::<u8>::new_rgba(255, 0, 0, 255)));
126        // ARgb32 stores BGRA premultiplied: red = [B=0, G=0, R=255, A=255]
127        assert_eq!(px(&data, size, 0, 0), [0, 0, 255, 255]);
128        assert_eq!(px(&data, size, 7, 7), [0, 0, 255, 255]);
129    }
130
131    #[test]
132    fn fill_background_preserves_alpha() {
133        let size = Size::new(4, 4);
134        let data = render(size, |t| t.fill_background(Color::<u8>::new_rgba(255, 0, 0, 128)));
135        let pixel = px(&data, size, 1, 1);
136        assert_eq!(pixel[3], 128, "alpha channel should be replaced, not composited");
137        assert!((pixel[2] as i32 - 128).abs() <= 1, "premultiplied red should be ~128, got {}", pixel[2]);
138    }
139
140    #[test]
141    fn fill_rect_covers_interior_only() {
142        let size = Size::new(16, 16);
143        let rect = Rect::<Pixels>::new(Vec2::new(4.0, 4.0), Vec2::new(8.0, 8.0));
144        let data = render(size, |t| t.fill_rect(FillRect::new(rect, Color::<f64>::new(0.0, 1.0, 0.0), Alpha::OPAQUE)));
145        assert_eq!(px(&data, size, 8, 8), [0, 255, 0, 255], "interior pixel");
146        assert_eq!(px(&data, size, 0, 0), [0, 0, 0, 0], "outside corner");
147        assert_eq!(px(&data, size, 15, 15), [0, 0, 0, 0], "outside edge");
148    }

150    #[test]
151    fn stroke_rect_covers_edges_only() {
152        let size = Size::new(16, 16);
153        let rect = Rect::<Pixels>::new(Vec2::new(4.0, 4.0), Vec2::new(8.0, 8.0));
154        let data = render(size, |t| {
155            t.stroke_rect(StrokeRect::new(rect, Color::<f64>::new(1.0, 1.0, 1.0), Pixels::new(2.0), Alpha::OPAQUE))
156        });
157        assert_eq!(px(&data, size, 4, 8)[3], 255, "left edge should be stroked");
158        assert_eq!(px(&data, size, 8, 8), [0, 0, 0, 0], "center should be empty");
159        assert_eq!(px(&data, size, 0, 0), [0, 0, 0, 0], "outside should be empty");
160    }

172    #[test]
173    fn fill_circle_covers_center_only() {
174        let size = Size::new(16, 16);
175        let circle = Circle::<Pixels>::new(Vec2::new(8.0, 8.0), 5.0);

178        assert_eq!(px(&data, size, 0, 0), [0, 0, 0, 0], "corner outside");
