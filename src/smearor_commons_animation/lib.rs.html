1//! Timed animations for the Smearor ecosystem.
2//!
3//! Pure `std` core: an [`TimedAnimation`] pairs a [`Trajectory`] - the
4//! shape of the animated output - with a [`Duration`] and an
5//! [`EasingFunction`]. Trajectories compose: [`Span`] interpolates
6//! between two values of any [`Interpolate`] type, [`ThreePhaseSpan`]
7//! eases out, holds and eases back, and `(S1, S2)` tuples animate
8//! composite values on one clock. The `gtk4` feature gate adds the
9//! frame-clock driver that pumps an animation on a widget.
10//!
11//! `serde` is unconditional; `schemars` schema derives live behind
12//! their feature gate.
13

39/// An eased, timed rotation animation - the shortest-arc lerp
40/// between two [`RotationRad`] values.
41pub type RotationAnimation = TimedAnimation<Span<RotationRad>>;
42
43/// An eased, timed zoom animation - the plain lerp between two
44/// [`ScaleFactor`] values.
45pub type ZoomAnimation = TimedAnimation<Span<ScaleFactor>>;
46
47/// A pulse animation - zoom out, hold, zoom back in via
48/// [`ThreePhaseSpan`].
49pub type PulseAnimation = TimedAnimation<ThreePhaseSpan<ScaleFactor>>;
50
51/// An eased, timed animation of any vector value - position, size
52/// or offset, in whatever coordinate space `U` tags.
53pub type Vec2Animation<U> = TimedAnimation<Span<Vec2<U>>>;
54
55/// A rotation-zoom composite - shortest-arc rotation plus a
56/// three-phase zoom (out, hold, back in) on one clock.
57pub type RotationZoomAnimation = TimedAnimation<(Span<RotationRad>, ThreePhaseSpan<ScaleFactor>)>;
