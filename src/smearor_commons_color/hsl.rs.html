10/// An HSL color: hue in degrees (`0-360`), saturation and lightness
11/// in `0.0-1.0`.
12///
13/// `Hsl` is the sibling of [`crate::Hsv`] with the symmetric
14/// lightness axis: `l = 0.5` is the pure hue, `0` black and `1`
15/// white. The `From` conversions perform the standard sRGB-space
16/// HSL round trip; a saturation of `0` has no hue and round-trips as
17/// `h = 0`.
18#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
19#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
20pub struct Hsl {
21    /// Hue in degrees (`0-360`, wraps around).
22    pub h: Hue,
23    /// Saturation (`0.0` gray to `1.0` full chroma).
24    pub s: Saturation,
25    /// Lightness (`0.0` black, `0.5` pure hue, `1.0` white).
26    pub l: Lightness,
27}
28
29impl Hsl {
30    /// Creates an HSL color from its channels.
31    pub fn new(h: impl Into<Hue>, s: impl Into<Saturation>, l: impl Into<Lightness>) -> Self {
32        Self {
33            h: h.into(),
34            s: s.into(),
35            l: l.into(),
36        }
37    }
38}

40impl Display for Hsl {
41    /// `hsl(h, s%, l%)` - saturation and lightness in percent.
42    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
43        write!(f, "hsl({:.1}, {:.1}%, {:.1}%)", self.h.as_f64(), self.s.as_f64() * 100.0, self.l.as_f64() * 100.0)
44    }
45}

47/// The sRGB value of an HSL color (sector decomposition, alpha
48/// opaque).
49impl From<Hsl> for Color<f64> {
50    fn from(hsl: Hsl) -> Self {
51        let (h, s, l) = (hsl.h.as_f64(), hsl.s.as_f64(), hsl.l.as_f64());
52        let c = (1.0 - (2.0 * l - 1.0).abs()) * s;
53        let x = c * (1.0 - ((h / 60.0).rem_euclid(2.0) - 1.0).abs());
54        let m = l - c / 2.0;
55        let (r, g, b) = match (h / 60.0) as u32 {
56            0 => (c, x, 0.0),
57            1 => (x, c, 0.0),
58            2 => (0.0, c, x),
59            3 => (0.0, x, c),
60            4 => (x, 0.0, c),
61            _ => (c, 0.0, x),
62        };
63        Color::<f64>::new(r + m, g + m, b + m)
64    }
65}

81/// The HSL decomposition of an sRGB color: `h` from max-channel
82/// sector, `l = (max + min) / 2`, `s = delta / (1 - |2l - 1|)`.
83/// Achromatic colors yield `h = 0, s = 0`.
84impl From<Color<f64>> for Hsl {
85    fn from(color: Color<f64>) -> Self {
86        let max = color.r.max(color.g).max(color.b);
87        let min = color.r.min(color.g).min(color.b);
88        let delta = max - min;
89        let l = (max + min) / 2.0;
90        let h = if delta <= 0.0 {
91            0.0
92        } else if max == color.r {
93            60.0 * ((color.g - color.b) / delta).rem_euclid(6.0)
94        } else if max == color.g {
95            60.0 * ((color.b - color.r) / delta + 2.0)
96        } else {
97            60.0 * ((color.r - color.g) / delta + 4.0)
98        };
99        let s = if delta <= 0.0 { 0.0 } else { delta / (1.0 - (2.0 * l - 1.0).abs()) };
100        Self::new(h, s, l)
101    }
102}

118/// HSV to HSL: `l = v * (1 - s_v / 2)`, `s_l = (v - l) / min(l, 1 - l)`.
119impl From<crate::Hsv> for Hsl {
120    fn from(hsv: crate::Hsv) -> Self {
121        let (s_v, v) = (hsv.s.as_f64(), hsv.v.as_f64());
122        let l = v * (1.0 - s_v / 2.0);
123        let s_l = if l > 0.0 && l < 1.0 { (v - l) / l.min(1.0 - l) } else { 0.0 };
124        Self::new(hsv.h, s_l, l)
125    }
126}
127
128/// HSL to HSV: `v = l + s_l * min(l, 1 - l)`, `s_v = 2 * (1 - l / v)`.
129impl From<Hsl> for crate::Hsv {
130    fn from(hsl: Hsl) -> Self {
131        let (s_l, l) = (hsl.s.as_f64(), hsl.l.as_f64());
132        let v = l + s_l * l.min(1.0 - l);
133        let s_v = if v > 0.0 { 2.0 * (1.0 - l / v) } else { 0.0 };
134        Self::new(hsl.h, s_v, v)
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    #[test]
143    fn primaries_have_full_saturation_at_half_lightness() {
144        let red = Hsl::from(Color::<u8>::new(255, 0, 0));
145        assert_eq!(red.h.as_f64(), 0.0);
146        assert_eq!(red.s.as_f64(), 1.0);
147        assert_eq!(red.l.as_f64(), 0.5);
148        assert_eq!(Hsl::from(Color::<u8>::new(0, 0, 255)).h.as_f64(), 240.0);
149    }
150
151    #[test]
152    fn achromatic_has_no_hue() {
153        let hsl = Hsl::from(Color::<u8>::new(128, 128, 128));
154        assert_eq!(hsl.s, Saturation::MIN);
155        assert!((hsl.l.as_f64() - 128.0 / 255.0).abs() < 1e-9);
156    }
157
158    #[test]
159    fn round_trip_keeps_the_color() {
160        for color in [Color::<u8>::new(255, 0, 0), Color::<u8>::new(18, 130, 200), Color::<u8>::new(200, 40, 180)] {
161            let back = Color::<u8>::from(Hsl::from(color));
162            assert!((i32::from(back.r) - i32::from(color.r)).abs() <= 1);
163            assert!((i32::from(back.g) - i32::from(color.g)).abs() <= 1);
164            assert!((i32::from(back.b) - i32::from(color.b)).abs() <= 1);
165        }
166    }
167
168    #[test]
169    fn hsv_hsl_round_trip() {
170        let hsv = crate::Hsv::new(210.0, 0.7, 0.6);
171        let back = crate::Hsv::from(Hsl::from(hsv));
172        assert!((back.h.as_f64() - hsv.h.as_f64()).abs() < 1e-9);
173        assert!((back.s.as_f64() - hsv.s.as_f64()).abs() < 1e-9);
174        assert!((back.v.as_f64() - hsv.v.as_f64()).abs() < 1e-9);
175    }
