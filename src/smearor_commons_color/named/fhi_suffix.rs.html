10/// The Pantone Fashion, Home + Interiors (FHI) material suffix - the
11/// code of a Pantone FHI color name (`Pantone 19-4057 TCX True Blue`)
12/// naming the material the physical swatch is produced on.
13///
14/// FHI colors are material-independent: the same code and name exist
15/// on every substrate, so the suffix qualifies a swatch rather than
16/// the color itself - unlike [`crate::PantoneSuffix`], which is part
17/// of the PMS color name.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
19#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
20pub enum PantoneFhiSuffix {
21    /// `TCX` - Textile Cotton eXtended, the cotton fabric guide.
22    #[cfg_attr(feature = "schemars", schemars(rename = "TCX"))]
23    Cotton,
24    /// `TPX` - Textile Paper eXtended, the discontinued paper guide
25    /// (superseded by `TPG`).
26    #[cfg_attr(feature = "schemars", schemars(rename = "TPX"))]
27    Paper,
28    /// `TPG` - Textile Paper Green, the current paper guide.
29    #[cfg_attr(feature = "schemars", schemars(rename = "TPG"))]
30    PaperGreen,
31}
32
33impl PantoneFhiSuffix {
34    /// The suffix code: `"TCX"`, `"TPX"` or `"TPG"`.
35    pub const fn as_str(&self) -> &'static str {
36        match self {
37            Self::Cotton => "TCX",
38            Self::Paper => "TPX",
39            Self::PaperGreen => "TPG",
40        }
41    }
42}
43
44impl FromStr for PantoneFhiSuffix {
45    type Err = ParseColorError;
46
47    /// Parses the suffix code or the spelled-out variant, case-insensitively.
48    fn from_str(s: &str) -> Result<Self, Self::Err> {
49        match s.to_ascii_lowercase().as_str() {
50            "tcx" | "cotton" => Ok(Self::Cotton),
51            "tpx" | "paper" => Ok(Self::Paper),
52            "tpg" | "paper green" => Ok(Self::PaperGreen),
53            _ => Err(ParseColorError::UnknownPantoneFhiSuffix { input: s.into() }),
54        }
55    }
56}
57
58impl Display for PantoneFhiSuffix {
59    /// Formats the suffix code (`TCX`, `TPX` or `TPG`).
60    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
61        f.write_str(self.as_str())
62    }
63}

65/// The wire format is the suffix code.
66impl Serialize for PantoneFhiSuffix {
67    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
68        serializer.serialize_str(self.as_str())
69    }
70}

72/// Deserialization accepts the code or the spelled-out variant.
73impl<'de> Deserialize<'de> for PantoneFhiSuffix {
74    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {

76    }

91    #[test]
92    fn as_str_and_display_produce_the_code() {
93        assert_eq!(PantoneFhiSuffix::Cotton.as_str(), "TCX");
94        assert_eq!(PantoneFhiSuffix::Paper.to_string(), "TPX");
95        assert_eq!(PantoneFhiSuffix::PaperGreen.to_string(), "TPG");
96    }
97
98    #[test]
99    fn from_str_accepts_code_and_word_case_insensitively() {
100        assert_eq!("TCX".parse::<PantoneFhiSuffix>().unwrap(), PantoneFhiSuffix::Cotton);
101        assert_eq!("cotton".parse::<PantoneFhiSuffix>().unwrap(), PantoneFhiSuffix::Cotton);
102        assert_eq!("TPX".parse::<PantoneFhiSuffix>().unwrap(), PantoneFhiSuffix::Paper);
103        assert_eq!("tpg".parse::<PantoneFhiSuffix>().unwrap(), PantoneFhiSuffix::PaperGreen);
104        assert_eq!("Paper Green".parse::<PantoneFhiSuffix>().unwrap(), PantoneFhiSuffix::PaperGreen);
105        assert_eq!("x".parse::<PantoneFhiSuffix>().unwrap_err(), ParseColorError::UnknownPantoneFhiSuffix { input: "x".into() });
106    }

108    #[test]
