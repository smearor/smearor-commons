1//! The Natural Colour System (NCS) - the Swedish appearance-based
2//! color notation (SIS, 1979; ISO/CIE standard).
3//!
4//! Every NCS color is a resemblance recipe against the six
5//! elementary percepts - white, black and the four chromatic hues
6//! yellow (`Y`), red (`R`), green (`G`), blue (`B`):
7//!
8//! - [`NcsHue`] is the position on the hue circle (`0` = `Y`,
9//!   `100` = `R`, `200` = `B`, `300` = `G`), where the two digits
10//!   after a base letter give the resemblance toward the next
11//!   elementary hue: `Y90R` is yellow that resembles red to 90%.
12//! - [`Ncs`] is the full `S sccc-HUE` specification: blackness `s`,
13//!   chromaticness `c` and the implicit whiteness `w` with
14//!   `s + c + w = 100`.
15//!
16//! The modeled colors are a *documented approximation*: the official
17//! NCS atlas values are proprietary, so hue resemblance is mapped
18//! onto CIELAB hue angles between the four elementary anchors and
19//! blackness/chromaticness onto `L*`/`C*` ranges. The geometry and
20//! the notation are the real NCS ones.
21

30/// Approximate CIELAB hue angles of the elementary hues in circle
31/// order `Y`, `R`, `B`, `G` in degrees - a documented
32/// approximation, see the module documentation.
33const ELEMENTARY_ANGLES: [f64; 4] = [90.0, 25.0, 255.0, 155.0];
34
35/// A position on the NCS hue circle (`0` = `Y`, `100` = `R`,
36/// `200` = `B`, `300` = `G`).
37#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Serialize, Deserialize)]
38#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
39#[serde(transparent)]
40pub struct NcsHue(f64);
41
42impl NcsHue {
43    /// Creates a hue position, wrapped into `0..400`.
44    pub fn new(position: f64) -> Self {
45        Self(position.rem_euclid(400.0))
46    }
47
48    /// The numeric circle position (`0..400`).
49    pub const fn as_f64(self) -> f64 {
50        self.0
51    }
52
53    /// The approximate CIELAB hue angle, interpolated along the
54    /// shortest arc between the elementary anchors.
55    pub fn lab_hue(&self) -> Hue {
56        let index = (self.0 / 100.0).floor() as usize % 4;
57        let t = self.0.rem_euclid(100.0) / 100.0;
58        let a0 = ELEMENTARY_ANGLES[index];
59        let a1 = ELEMENTARY_ANGLES[(index + 1) % 4];
60        let span = (a1 - a0 + 180.0).rem_euclid(360.0) - 180.0;
61        Hue::new(a0 + span * t)
62    }
63
64    /// The approximate sRGB value of an NCS cell of this hue: the
65    /// elementary anchor hue at `c%` chroma, lightened by whiteness
66    /// and darkened by blackness. `blackness` and `chromaticness`
67    /// are percents (`0..=100`).
68    pub fn color(&self, blackness: f64, chromaticness: f64) -> Color<u8> {
69        let lightness = 95.0 - 0.7 * blackness - 0.3 * chromaticness;
70        let (sin, cos) = self.lab_hue().as_f64().to_radians().sin_cos();
71        let chroma = 0.9 * chromaticness;
72        Color::from(Lab::new(lightness, chroma * cos, chroma * sin))
73    }
74}
75
76impl Display for NcsHue {
77    /// The hue notation, e.g. `"Y"`, `"Y90R"`, `"B10Y"`.
78    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
79        let sector = (self.0 / 100.0).floor() as usize % 4;
80        let pct = self.0.rem_euclid(100.0).round() as i32;
81        let (from, to) = (["Y", "R", "B", "G"][sector], ["R", "B", "G", "Y"][sector]);
82        if pct == 0 { write!(f, "{from}") } else { write!(f, "{from}{pct}{to}") }
83    }
84}

100/// A complete NCS color specification `S sccc-HUE`: a hue plus
101/// blackness `s` and chromaticness `c` percents, with the implicit
102/// whiteness `w = 100 - s - c`.
103#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
104#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
105pub struct Ncs {
106    /// The position on the NCS hue circle.
107    pub hue: NcsHue,
108    /// The blackness `s` in percent (`0..=100`).
109    pub blackness: f64,
110    /// The chromaticness `c` in percent (`0..=100`,
111    /// `s + c <= 100` for a valid NCS color).
112    pub chromaticness: f64,
113}
114
115impl Ncs {
116    /// Creates an NCS color; `blackness + chromaticness` is clamped
117    /// to `100` by scaling both down proportionally.
118    pub fn new(hue: NcsHue, blackness: f64, chromaticness: f64) -> Self {
119        let sum = blackness + chromaticness;
120        if sum > 100.0 {
121            Self {
122                hue,
123                blackness: blackness * 100.0 / sum,
124                chromaticness: chromaticness * 100.0 / sum,
125            }
126        } else {
127            Self { hue, blackness, chromaticness }
128        }
129    }
130
131    /// The whiteness `w = 100 - s - c` in percent.
132    pub fn whiteness(&self) -> f64 {
133        100.0 - self.blackness - self.chromaticness
134    }
135
136    /// The approximate sRGB value (see [`NcsHue::color`] for the
137    /// documented approximation).
138    pub fn color(&self) -> Color<u8> {
139        self.hue.color(self.blackness, self.chromaticness)
140    }
141}
142
143impl Display for Ncs {
144    /// The standard notation, e.g. `"S 2030-Y90R"`.
145    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
146        write!(f, "S {:02}{:02}-{}", self.blackness.round() as i32, self.chromaticness.round() as i32, self.hue)
147    }
148}

150#[cfg(test)]
151mod tests {
152    use super::*;
153
154    #[test]
155    fn the_position_wraps_around_the_circle() {
156        assert_eq!(NcsHue::new(450.0).as_f64(), 50.0);
157        assert_eq!(NcsHue::new(-10.0).as_f64(), 390.0);
158    }
159
160    #[test]
161    fn the_elementary_positions_have_their_letter_notation() {
162        assert_eq!(NcsHue::new(0.0).to_string(), "Y");
163        assert_eq!(NcsHue::new(100.0).to_string(), "R");
164        assert_eq!(NcsHue::new(200.0).to_string(), "B");
165        assert_eq!(NcsHue::new(300.0).to_string(), "G");
166    }
167
168    #[test]
169    fn the_resemblance_is_the_sector_percentage() {
170        assert_eq!(NcsHue::new(90.0).to_string(), "Y90R");
171        assert_eq!(NcsHue::new(310.0).to_string(), "G10Y");
172    }
173
174    #[test]
175    fn the_lab_hue_interpolates_along_the_shortest_arc() {
176        // Y->R descends 90 -> 25, so the middle is ~57.5, not 237.5.
177        assert!((NcsHue::new(50.0).lab_hue().as_f64() - 57.5).abs() < 1e-6);
178        // R->B descends 25 -> 255 through the wrap (the violets).
179        let hue = NcsHue::new(150.0).lab_hue().as_f64();
180        assert!(hue > 300.0 || hue < 20.0);
181        // G->Y descends 155 -> 90, middle is ~122.5.
182        assert!((NcsHue::new(350.0).lab_hue().as_f64() - 122.5).abs() < 1e-6);
183    }
184
185    #[test]
186    fn the_notation_follows_the_s_sccc_hue_form() {
187        assert_eq!(Ncs::new(NcsHue::new(90.0), 20.0, 30.0).to_string(), "S 2030-Y90R");
188    }
189
190    #[test]
191    fn the_parts_sum_to_one_hundred() {
192        let ncs = Ncs::new(NcsHue::new(150.0), 20.0, 30.0);
193        assert!((ncs.blackness + ncs.chromaticness + ncs.whiteness() - 100.0).abs() < 1e-9);
194    }
195
196    #[test]
197    fn blackness_is_dark_and_whiteness_is_light() {
198        let hue = NcsHue::new(0.0);
199        let dark = hue.color(100.0, 0.0);
200        let light = hue.color(0.0, 0.0);
201        assert!(dark.r < light.r && dark.g < light.g && dark.b < light.b);
202    }
203}
