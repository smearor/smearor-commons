1/// RGB channel weights for a perceived-luminance standard.
2///
3/// The luminance of a color is the weighted sum `r*w_r + g*w_g + b*w_b`.
4/// The weights of a standard sum to 1.0; [`crate::Color::luminance`]
5/// accepts any custom triple, the result is clamped to `0.0-1.0`
6/// regardless.
7#[derive(Debug, Clone, Copy, PartialEq)]
8pub struct LuminanceWeights {
9    /// Weight of the red channel.
10    pub r: f64,
11    /// Weight of the green channel.
12    pub g: f64,
13    /// Weight of the blue channel.
14    pub b: f64,
15}
16
17impl LuminanceWeights {
18    /// ITU-R BT.709 weights (HDTV, sRGB default).
19    pub const REC_709: Self = Self {
20        r: 0.2126,
21        g: 0.7152,
22        b: 0.0722,
23    };
24
25    /// ITU-R BT.601 weights (SDTV, JPEG/MPEG luma).
26    pub const REC_601: Self = Self { r: 0.299, g: 0.587, b: 0.114 };
27}
28
29#[cfg(test)]
30mod tests {
31    use super::*;
32
33    #[test]
34    fn standard_weights_sum_to_one() {
35        for weights in [LuminanceWeights::REC_709, LuminanceWeights::REC_601] {
36            assert!((weights.r + weights.g + weights.b - 1.0).abs() < 1e-9);
37        }
38    }
