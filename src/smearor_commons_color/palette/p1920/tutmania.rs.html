6palette_definition! {
7    /// The "Tutmania" palette - 1920s chapter of "Pantone: The
8    /// Twentieth Century in Color" (Eiseman & Recker, 2011).
9    pub enum TutmaniaColor {
10        /// PANTONE 16-1836 TCX `Rich Gold`. The caption reads 16-1836; the official code of `Rich Gold` is 16-0836.
11        RichGold => "rich-gold",
12        /// PANTONE 19-1540 TCX `Burnt Henna`.
13        BurntHenna => "burnt-henna",
14        /// PANTONE 19-4245 TCX `Imperial Blue`.
15        ImperialBlue => "imperial-blue",
16        /// PANTONE 15-5209 TCX `Aqua Haze`.
17        AquaHaze => "aqua-haze",
18        /// PANTONE 14-0121 TCX `Nile Green`.
19        NileGreen => "nile-green",
20        /// PANTONE 14-0936 TCX `Sahara Sun`.
21        SaharaSun => "sahara-sun",
22        /// PANTONE 16-0110 TCX `Desert Sage`.
23        DesertSage => "desert-sage"
24    }
25}
26
27/// The sRGB values of the [`TutmaniaColor`] slots.
28pub static TUTMANIA: StaticPalette<TutmaniaColor, u8> = static_palette! {
29    TutmaniaColor::RichGold => PantoneFhiNamedColor::RichGold.to_color(),
30    TutmaniaColor::BurntHenna => PantoneFhiNamedColor::BurntHenna.to_color(),
31    TutmaniaColor::ImperialBlue => PantoneFhiNamedColor::ImperialBlue.to_color(),
32    TutmaniaColor::AquaHaze => PantoneFhiNamedColor::AquaHaze.to_color(),
33    TutmaniaColor::NileGreen => PantoneFhiNamedColor::NileGreen.to_color(),
34    TutmaniaColor::SaharaSun => PantoneFhiNamedColor::SaharaSun.to_color(),
35    TutmaniaColor::DesertSage => PantoneFhiNamedColor::DesertSage.to_color(),
36};
