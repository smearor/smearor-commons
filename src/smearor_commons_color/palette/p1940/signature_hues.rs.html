8    /// The signature hues of the 1940s - the lead swatch of each
9    /// chapter palette of "Pantone: The Twentieth Century in Color"
10    /// (Eiseman & Recker, 2011), in chapter order.
11    pub enum SignatureHues1940Color {
12        /// The lead swatch of the "Fantasia" palette - PANTONE 17-1663 TCX `Bittersweet`.
13        Fantasia => "fantasia",
14        /// The lead swatch of the "Edward Hopper" palette - Pantone 7477 C.
15        EdwardHopper => "edward-hopper",
16        /// The lead swatch of the "World War II" palette - PANTONE 16-1334 TCX `Tan`.
17        WorldWar2 => "world-war-2",
18        /// The lead swatch of the "Carefree and Casual" palette - PANTONE 18-5621 TCX `Fir`.
19        CarefreeAndCasual => "carefree-and-casual",
20        /// The lead swatch of the "The American Dream" palette - PANTONE 14-1230 TCX `Apricot Wash`.
21        AmericanDream => "american-dream",
22        /// The lead swatch of the "Hit Parade" palette - Pantone 3005 C.
23        HitParade => "hit-parade",
24        /// The lead swatch of the "Film Noir" palette - PANTONE 13-0002 TCX `White Sand`.
25        FilmNoir => "film-noir",
26        /// The lead swatch of the "La Mode" palette - PANTONE 19-4006 TCX `Caviar`.
27        LaMode => "la-mode"
28    }
29}
