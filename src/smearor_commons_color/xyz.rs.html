5/// A CIE 1931 XYZ tristimulus value.
6///
7/// The sRGB conversions assume the D65 white point used by the sRGB
8/// specification (IEC 61966-2-1). Values outside the sRGB gamut are
9/// clamped channel-wise when converting back to [`Color`].
10#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
11pub struct Xyz {
12    /// The X tristimulus value (`Y = 100.0` for the white point).
13    pub x: f64,
14    /// The Y tristimulus value - the luminance.
15    pub y: f64,
16    /// The Z tristimulus value.
17    pub z: f64,
18}
19
20impl Xyz {
21    /// The CIE D65 (daylight, ~6504K) white point at `Y = 100`.
22    pub const WHITE_D65: Self = Self {
23        x: 95.047,
24        y: 100.0,
25        z: 108.883,
26    };
27
28    /// The CIE D50 (horizon light, ~5003K) white point at `Y = 100` -
29    /// the standard illuminant of print and graphic arts (ICC PCS,
30    /// ISO 12647 / FOGRA characterization data).
31    pub const WHITE_D50: Self = Self { x: 96.42, y: 100.0, z: 82.49 };
32
33    /// The CIE illuminant C (average daylight, ~6774K) white point at
34    /// `Y = 100` - the white point of the Munsell renotation data.
35    pub const WHITE_C: Self = Self {
36        x: 98.074,
37        y: 100.0,
38        z: 118.232,
39    };
40
41    /// Creates a tristimulus value from the components.
42    pub const fn new(x: f64, y: f64, z: f64) -> Self {
43        Self { x, y, z }
44    }
45
46    /// Creates a tristimulus value from chromaticity coordinates and a
47    /// luminance (`xyY` notation).
48    ///
49    /// `y_chroma` must be positive; for `y_chroma <= 0.0` (no defined
50    /// chromaticity, e.g. black) the result is black.
51    pub fn from_xyy(x: f64, y_chroma: f64, luminance: f64) -> Self {
52        if y_chroma <= 0.0 || luminance <= 0.0 {
53            return Self::new(0.0, 0.0, 0.0);
54        }
55        Self {
56            x: x * luminance / y_chroma,
57            y: luminance,
58            z: (1.0 - x - y_chroma) * luminance / y_chroma,
59        }
60    }
61
62    /// The chromaticity coordinates of the value (`x`, `y`, `Y`).
63    pub fn to_xyy(&self) -> (f64, f64, f64) {
64        let sum = self.x + self.y + self.z;
65        if sum <= 0.0 {
66            return (0.0, 0.0, self.y);
67        }
68        (self.x / sum, self.y / sum, self.y)
69    }
70
71    /// Adapts the tristimulus value from one white point to another
72    /// using the Bradford chromatic adaptation transform.
73    pub fn adapted(&self, source_white: Self, target_white: Self) -> Self {
74        const BRADFORD: [[f64; 3]; 3] = [[0.8951, 0.2664, -0.1614], [-0.7502, 1.7135, 0.0367], [0.0389, -0.0685, 1.0296]];
75        const BRADFORD_INV: [[f64; 3]; 3] = [
76            [0.9869929, -0.1470543, 0.1599627],
77            [0.4323053, 0.5183603, 0.0492912],
78            [-0.0085287, 0.0400428, 0.9684867],
79        ];
80        let (sx, sy, sz) = (mat_mul(BRADFORD, self), mat_mul(BRADFORD, &source_white), mat_mul(BRADFORD, &target_white));
81        let scale = Self::new(
82            if sy.x != 0.0 { sz.x / sy.x } else { 1.0 },
83            if sy.y != 0.0 { sz.y / sy.y } else { 1.0 },
84            if sy.z != 0.0 { sz.z / sy.z } else { 1.0 },
85        );
86        let scaled = Self::new(sx.x * scale.x, sx.y * scale.y, sx.z * scale.z);
87        mat_mul(BRADFORD_INV, &scaled)
88    }
89}

91/// Multiplies a 3×3 matrix with a tristimulus value.
92fn mat_mul(m: [[f64; 3]; 3], v: &Xyz) -> Xyz {
93    Xyz::new(
94        m[0][0] * v.x + m[0][1] * v.y + m[0][2] * v.z,
95        m[1][0] * v.x + m[1][1] * v.y + m[1][2] * v.z,
96        m[2][0] * v.x + m[2][1] * v.y + m[2][2] * v.z,
97    )
98}

100/// Linearizes one sRGB channel (IEC 61966-2-1).
101fn srgb_to_linear(channel: f64) -> f64 {
102    if channel <= 0.04045 {
103        channel / 12.92
104    } else {
105        ((channel + 0.055) / 1.055).powf(2.4)
106    }
107}
108
109/// Compands one linear channel back to sRGB.
110fn linear_to_srgb(channel: f64) -> f64 {
111    if channel <= 0.0031308 {
112        12.92 * channel
113    } else {
114        1.055 * channel.powf(1.0 / 2.4) - 0.055
115    }
116}

118impl From<Color<f64>> for Xyz {
119    /// Converts a normalized sRGB color to CIE XYZ (D65).
120    fn from(color: Color<f64>) -> Self {
121        let (r, g, b) = (srgb_to_linear(color.r), srgb_to_linear(color.g), srgb_to_linear(color.b));
122        Self::new(
123            41.23908 * r + 35.75843 * g + 18.04808 * b,
124            21.26390 * r + 71.51687 * g + 7.21923 * b,
125            1.93308 * r + 11.91948 * g + 95.05322 * b,
126        )
127    }
128}

130impl From<Color<u8>> for Xyz {
131    /// Converts an sRGB color to CIE XYZ (D65).
132    fn from(color: Color<u8>) -> Self {
133        Self::from(Color::<f64>::new(color.r as f64 / 255.0, color.g as f64 / 255.0, color.b as f64 / 255.0))
134    }
135}

137impl From<Xyz> for Color<f64> {
138    /// Converts a CIE XYZ (D65) value to normalized sRGB.
139    ///
140    /// Out-of-gamut results are clamped channel-wise.
141    fn from(xyz: Xyz) -> Self {
142        let r = 0.032404542 * xyz.x - 0.015371385 * xyz.y - 0.004985314 * xyz.z;
143        let g = -0.009692660 * xyz.x + 0.018760108 * xyz.y + 0.000415560 * xyz.z;
144        let b = 0.000556434 * xyz.x - 0.002040259 * xyz.y + 0.010572252 * xyz.z;
145        Self::new(linear_to_srgb(r).clamp(0.0, 1.0), linear_to_srgb(g).clamp(0.0, 1.0), linear_to_srgb(b).clamp(0.0, 1.0))
146    }
147}

149impl From<Xyz> for Color<u8> {
150    /// Converts a CIE XYZ (D65) value to sRGB, clamped to the gamut.
151    fn from(xyz: Xyz) -> Self {
152        let normalized = Color::<f64>::from(xyz);
153        Self::new_rgba(
154            (normalized.r * 255.0).round() as u8,
155            (normalized.g * 255.0).round() as u8,
156            (normalized.b * 255.0).round() as u8,
157            255,
158        )
159    }
160}

162#[cfg(test)]
163mod tests {
164    use super::*;
165
166    #[test]
167    fn srgb_white_is_the_d65_white_point() {
168        let xyz = Xyz::from(Color::<u8>::new(255, 255, 255));
169        assert!((xyz.x - 95.047).abs() < 0.1);
170        assert!((xyz.y - 100.0).abs() < 0.1);
171        assert!((xyz.z - 108.883).abs() < 0.1);
172    }
173
174    #[test]
175    fn xyz_round_trip_is_stable() {
176        let color = Color::<u8>::new(200, 100, 50);
177        let round_tripped = Color::<u8>::from(Xyz::from(color));
178        assert!((round_tripped.r as i32 - 200).abs() <= 1);
179        assert!((round_tripped.g as i32 - 100).abs() <= 1);
180        assert!((round_tripped.b as i32 - 50).abs() <= 1);
181    }
182
183    #[test]
184    fn out_of_gamut_values_are_clamped() {
185        let color = Color::<f64>::from(Xyz::new(0.0, 100.0, 0.0));
186        assert!(color.r >= 0.0 && color.g <= 1.0 && color.b >= 0.0);
187    }
188
189    #[test]
190    fn xyy_round_trip_is_stable() {
191        let (x, y, yy) = Xyz::new(20.0, 30.0, 50.0).to_xyy();
192        let xyz = Xyz::from_xyy(x, y, yy);
193        assert!((xyz.x - 20.0).abs() < 0.001 && (xyz.y - 30.0).abs() < 0.001 && (xyz.z - 50.0).abs() < 0.001);
194    }
195
196    #[test]
197    fn bradford_adaptation_maps_white_to_white() {
198        let adapted = Xyz::WHITE_C.adapted(Xyz::WHITE_C, Xyz::WHITE_D65);
199        assert!((adapted.x - Xyz::WHITE_D65.x).abs() < 0.05);
200        assert!((adapted.y - Xyz::WHITE_D65.y).abs() < 0.05);
201        assert!((adapted.z - Xyz::WHITE_D65.z).abs() < 0.05);
202    }
